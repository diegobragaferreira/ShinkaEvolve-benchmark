    def _initialize_grid_points(self) -> np.ndarray:
        """Create grid-based initial points with better distribution."""
        n_per_side = int(np.ceil(np.sqrt(self.n_points)))
        # Use a more uniform spacing with padding for better coverage
        x = np.linspace(0.05, 0.95, n_per_side)
        y = np.linspace(0.05, 0.95, n_per_side)
        xx, yy = np.meshgrid(x, y)
        points = np.column_stack([xx.ravel(), yy.ravel()])[:self.n_points]
        return points

    def _initialize_golden_spiral(self) -> np.ndarray:
        """Create points following a golden spiral pattern for good spread."""
        points = []
        phi = (1 + np.sqrt(5)) / 2  # Golden ratio

        for i in range(self.n_points):
            theta = i * 2 * np.pi / phi
            r = i / np.sqrt(self.n_points)
            x = 0.5 + r * np.cos(theta) * 0.4
            y = 0.5 + r * np.sin(theta) * 0.4
            points.append([x, y])

        return np.array(points)

    def _initialize_voronoi_sampling(self) -> np.ndarray:
        """Create initial points using Voronoi-based sampling for even distribution."""
        # Start with a regular grid then perturb strategically
        points = []
        n_per_side = int(np.ceil(np.sqrt(self.n_points)))
        x = np.linspace(0.1, 0.9, n_per_side)
        y = np.linspace(0.1, 0.9, n_per_side)
        xx, yy = np.meshgrid(x, y)
        grid_points = np.column_stack([xx.ravel(), yy.ravel()])[:self.n_points]

        # Add slight perturbations to avoid perfect grid symmetries
        for i in range(self.n_points):
            # Apply different noise patterns to break symmetries
            noise_x = np.random.normal(0, 0.01, 1)[0] + 0.001 * np.sin(i * 0.5)
            noise_y = np.random.normal(0, 0.01, 1)[0] + 0.001 * np.cos(i * 0.5)
            grid_points[i] += [noise_x, noise_y]

        # Clip to ensure bounds
        points = np.clip(grid_points, 0, 1)
        return points
