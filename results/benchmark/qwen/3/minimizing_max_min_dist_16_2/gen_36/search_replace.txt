    def optimize_with_simulated_annealing():

        # Initialize with improved hexagonal grid (more optimized arrangement)
        points = np.zeros((16, 2))

        # Create a more structured hexagonal pattern that maximizes initial spread
        row_spacing = 0.3
        col_spacing = 0.3 * np.sqrt(3) / 2.0

        idx = 0
        for row in range(4):
            for col in range(4):
                # Create a staggered pattern with better distribution

                idx += 1

            points[:, 0] = (points[:, 0] - x_min) / (x_max - x_min) * 0.9 + 0.05
        if y_max > y_min:

        np.random.seed(42)
        # Apply different perturbation magnitudes to different points to break symmetry
        for i in range(len(points)):
            # Apply varying perturbations based on position
            perturbation_magnitude = 0.01 + (i % 3) * 0.005
