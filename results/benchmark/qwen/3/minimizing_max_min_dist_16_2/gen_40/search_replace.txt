    def initialize_points(self) -> np.ndarray:
        """Initialize points using a structured approach."""
        # Create a more sophisticated hexagonal grid layout
        # This provides better initial distribution than simple grids

        # For 16 points, create a hexagonal arrangement with proper spacing
        points = []

        # Hexagonal packing parameters
        sqrt3 = np.sqrt(3)
        spacing_x = 1.0
        spacing_y = sqrt3 / 2.0

        # Arrange points in approximately hexagonal pattern
        # Use 4 rows with alternating column offsets
        for i in range(4):
            row_offset = i * spacing_y
            for j in range(4):
                col_offset = j * spacing_x + (i % 2) * spacing_x / 2.0
                points.append([col_offset, row_offset])

        # Convert to numpy array
        points = np.array(points[:self.n_points])

        # Normalize to [0,1] range
        if np.max(points) > 0:
            points = points / np.max(points)

        # Scale to fit within [0,1] bounds properly
        x_range = np.max(points[:, 0]) - np.min(points[:, 0])
        y_range = np.max(points[:, 1]) - np.min(points[:, 1])

        if x_range > 0:
            points[:, 0] = (points[:, 0] - np.min(points[:, 0])) / x_range
        if y_range > 0:
            points[:, 1] = (points[:, 1] - np.min(points[:, 1])) / y_range

        # Ensure bounds
        points = np.clip(points, 0, 1)

        # Add small random perturbations to break symmetry
        noise = np.random.normal(0, 0.005, points.shape)
        points += noise

        # Clip again after noise addition
        points = np.clip(points, 0, 1)

        return points
