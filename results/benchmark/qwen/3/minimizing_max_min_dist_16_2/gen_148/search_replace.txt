    def create_fibonacci_spiral() -> np.ndarray:
        """Create points arranged in a Fibonacci spiral pattern"""
        points = []
        golden_ratio = (1 + np.sqrt(5)) / 2

        for i in range(16):
            angle = i * 2 * np.pi / golden_ratio
            radius = np.sqrt(i / 15.0)  # Normalize to [0,1]
            x = 0.5 + radius * np.cos(angle) * 0.4
            y = 0.5 + radius * np.sin(angle) * 0.4
            points.append([x, y])

        points = np.array(points)
        # Add small noise to break perfect symmetry
        points += np.random.normal(0, 0.01, points.shape)
        points = np.clip(points, 0, 1)
        return points

    def create_irregular_square_grid() -> np.ndarray:
        """Create a square grid with irregular spacing to break symmetry"""
        points = []
        # Create a 4x4 grid with some irregularity
        for i in range(4):
            for j in range(4):
                # Add some randomness to the grid positions
                x = j/3.0 + np.random.normal(0, 0.02)
                y = i/3.0 + np.random.normal(0, 0.02)
                points.append([x, y])

        points = np.array(points)
        points = np.clip(points, 0, 1)
        return points
