        # For hexagonal packing, area per point = (3*sqrt(3)/2)*r^2 where r is the radius
        # So r^2 = 0.0625 / (3*sqrt(3)/2) = 0.0625 * 2 / (3*sqrt(3)) = 0.125 / (3*sqrt(3))
        area_per_point = 1.0 / 16.0
