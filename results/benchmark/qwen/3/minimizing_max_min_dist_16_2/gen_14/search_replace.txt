    def _generate_hexagon_points(self):
        """Generate points in a more sophisticated hexagonal pattern."""
        # Create a hexagonal lattice pattern
        points = []
        center_x, center_y = 0.5, 0.5
        radius = 0.3  # Adjusted radius for better coverage

        # Generate hexagonal grid points
        # Use hexagonal packing with appropriate spacing
        hex_radius = radius / np.sqrt(3)  # Correct spacing for hexagonal packing

        # Generate points in hexagonal pattern
        rows = 3
        cols = 3
        y_spacing = hex_radius * np.sqrt(3)
        x_spacing = hex_radius * 2

        for i in range(rows):
            for j in range(cols):
                if len(points) < self.n_points:
                    x = center_x - (cols-1)*x_spacing/2 + j*x_spacing
                    y = center_y - (rows-1)*y_spacing/2 + i*y_spacing
                    # Add some randomness to avoid perfect symmetry
                    x += np.random.normal(0, hex_radius * 0.1)
                    y += np.random.normal(0, hex_radius * 0.1)
                    # Clip to ensure within bounds
                    x = np.clip(x, 0, 1)
                    y = np.clip(y, 0, 1)
                    points.append([x, y])

        # Fill remaining points with random distribution
        while len(points) < self.n_points:
            points.append([np.random.rand(), np.random.rand()])

        return np.array(points[:self.n_points])
