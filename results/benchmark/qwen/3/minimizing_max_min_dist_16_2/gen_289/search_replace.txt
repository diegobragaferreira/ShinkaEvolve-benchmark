        # For 16 points, we want to arrange them in approximately 4 rows of 4 points each
        # But with proper hexagonal offsets

        points = []
        rows = 4
        cols = 4

        # Proper hexagonal lattice spacing
        # In a perfect hexagonal packing, horizontal spacing = 1, vertical spacing = sqrt(3)/2
        spacing_x = 1.0
        spacing_y = np.sqrt(3) / 2.0

        # Create hexagonal pattern with proper mathematical spacing
        for i in range(rows):
            for j in range(cols):
                # Position in hexagonal grid
                x = j * spacing_x + (i % 2) * spacing_x / 2
                y = i * spacing_y

                # Apply enhanced symmetry breaking using prime-based perturbations
                # This ensures no simple rotational or translational symmetries
                prime_i = (i * 7) % 11
                prime_j = (j * 13) % 17
