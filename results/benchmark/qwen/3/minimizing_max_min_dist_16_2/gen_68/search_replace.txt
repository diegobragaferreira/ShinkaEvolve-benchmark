    def initialize_points(self) -> np.ndarray:
        """Initialize points using a more sophisticated hexagonal grid approach."""
        # Create a more mathematically optimized hexagonal arrangement
        # For 16 points, arrange in a pattern closer to the optimal spherical distribution

        points = []
        sqrt3 = np.sqrt(3)

        # Use a 4x4 hexagonal pattern but adjust spacing for better distribution
        # Calculate ideal spacing based on area considerations for 16 points in unit square
        ideal_spacing = 1.0 / np.sqrt(16 / np.pi)  # Approximate spacing for uniform density
        spacing_x = ideal_spacing * 1.2  # Slightly increased to avoid overcrowding
        spacing_y = spacing_x * sqrt3 / 2.0

        # Create hexagonal grid with better visual and mathematical distribution
        for i in range(4):
            row_offset = i * spacing_y
            for j in range(4):
                col_offset = j * spacing_x + (i % 2) * spacing_x / 2.0
                points.append([col_offset, row_offset])

        # Convert to numpy array and truncate to required number of points
        points = np.array(points[:self.n_points])

        # Better normalization to maintain aspect ratio while fitting in [0,1]^2
        if len(points) > 0:
            x_min, x_max = np.min(points[:, 0]), np.max(points[:, 0])
            y_min, y_max = np.min(points[:, 1]), np.max(points[:, 1])

            # If we have significant spread, normalize to use most of the space
            if x_max > x_min and y_max > y_min:
                # Scale to fit nicely in [0.1, 0.9] range to leave margins
                scale_x = 0.8 / (x_max - x_min) if (x_max - x_min) > 1e-10 else 1.0
                scale_y = 0.8 / (y_max - y_min) if (y_max - y_min) > 1e-10 else 1.0

                # Apply scaling and centering
                points[:, 0] = (points[:, 0] - x_min) * scale_x + 0.1
                points[:, 1] = (points[:, 1] - y_min) * scale_y + 0.1
            elif x_max > x_min:
                points[:, 0] = (points[:, 0] - x_min) / (x_max - x_min) * 0.8 + 0.1
                points[:, 1] = 0.5  # Center vertically
            elif y_max > y_min:
                points[:, 0] = 0.5  # Center horizontally
                points[:, 1] = (points[:, 1] - y_min) / (y_max - y_min) * 0.8 + 0.1
            else:
                # All points same - center them
                points[:, 0] = 0.5
                points[:, 1] = 0.5

        # Ensure bounds
        points = np.clip(points, 0, 1)

        # Use different noise levels with more mathematical basis
        np.random.seed(self.seed)  # Ensure reproducibility

        noise_magnitudes = np.linspace(0.015, 0.005, self.n_points)

        for i in range(self.n_points):

            row = i // 4

            noise_magnitude = base_magnitude * position_factor

            noise = np.random.normal(0, noise_magnitude, self.dimension)
