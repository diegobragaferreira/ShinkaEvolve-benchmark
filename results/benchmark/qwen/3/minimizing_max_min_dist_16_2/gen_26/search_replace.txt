        return violations

    def hexagonal_grid_init():
        points = np.zeros((16, 2))
        # Arrange in hexagonal pattern (roughly)
        rows = 4

        spacing_x = 1.0 / (cols - 1)
        spacing_y = 1.0 / (rows - 1)

                if i * cols + j < 16:  # Only take first 16 points

                    y = i * spacing_y
                    # Add slight offset to break symmetry
                    if i % 2 == 1:
