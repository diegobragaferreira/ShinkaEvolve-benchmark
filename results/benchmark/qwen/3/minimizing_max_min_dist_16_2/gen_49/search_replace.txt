    # Strategy 1: Enhanced hexagonal grid with adaptive perturbations
    def hexagonal_grid_init():
        points = np.zeros((16, 2))
        # Create a more refined 4x4 hexagonal grid
        rows = 4
        cols = 4
        spacing_x = 1.0 / (cols - 1) if cols > 1 else 1.0
        spacing_y = 1.0 / (rows - 1) if rows > 1 else 1.0

        for i in range(rows):
            for j in range(cols):
                if i * cols + j < 16:  # Only take first 16 points
                    x = j * spacing_x
                    y = i * spacing_y
                    # Add hexagonal offset to create true hexagonal pattern
                    if i % 2 == 1:
                        x += spacing_x * 0.5
                    points[i * cols + j] = [x, y]

        # Apply adaptive perturbations that are stronger in the center
        center = np.mean(points, axis=0)
        distances_from_center = np.sqrt(np.sum((points - center)**2, axis=1))
        max_distance = np.max(distances_from_center)

        # Normalize distances and apply adaptive perturbations
        normalized_distances = distances_from_center / max_distance if max_distance > 0 else np.zeros_like(distances_from_center)
        perturbation_magnitude = 0.03 * (1 - normalized_distances)  # Stronger perturbations near center

        np.random.seed(42)
        perturbations = np.random.normal(0, 0.01, points.shape)
        # Scale perturbations adaptively
        perturbations *= perturbation_magnitude.reshape(-1, 1)

        points += perturbations

        # Apply symmetry breaking through rotation
        angle_offset = np.pi / 12  # Small rotation to break symmetry
        cos_a = np.cos(angle_offset)
        sin_a = np.sin(angle_offset)
        rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])

        # Apply rotation to half of the points
        for i in range(0, 16, 2):
            points[i] = rotation_matrix @ (points[i] - center) + center

        return points
