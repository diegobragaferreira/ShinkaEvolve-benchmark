    # This creates a more uniform distribution than simple grids

    scale_factor_y = 0.95

def initialize_points_hexagonal_lattice():
    """Initialize points using a more sophisticated hexagonal lattice with better spacing."""
    # Create a hexagonal lattice with 16 points arranged in a pattern closer to optimal distribution
    # Using a 4x4 hexagonal grid with improved spacing and symmetry breaking

    points = []
    sqrt3 = math.sqrt(3)

    # Create a hexagonal pattern that's more evenly distributed
    rows = 4
    cols = 4

    # More precise hexagonal packing with better distribution
    for i in range(rows):
        for j in range(cols):
            # Offset every other row for proper hexagonal arrangement
            x = j + (i % 2) * 0.5
            y = i * sqrt3 / 2

            # Add slight variation to create better distribution
            if i == 0 or i == rows-1:
                x += (j % 2) * 0.1  # Slight offset for edge rows
            if j == 0 or j == cols-1:
                y += (i % 2) * 0.1  # Slight offset for edge columns

            points.append([x, y])

    # Convert to numpy array
    points = np.array(points)

    # Normalize to fit nicely in [0,1] square while preserving relative spacing
    x_range = np.max(points[:, 0]) - np.min(points[:, 0])
    y_range = np.max(points[:, 1]) - np.min(points[:, 1])

    if x_range > 0:
        points[:, 0] = (points[:, 0] - np.min(points[:, 0])) / x_range
    if y_range > 0:
        points[:, 1] = (points[:, 1] - np.min(points[:, 1])) / y_range

    # Scale to fit within [0.05, 0.95] to maintain interior spacing
    points[:, 0] = 0.05 + 0.9 * points[:, 0]
    points[:, 1] = 0.05 + 0.9 * points[:, 1]

    # Apply more sophisticated perturbations to break symmetry
    np.random.seed(42)

    # Different noise patterns for different rows/columns to enhance diversity
    noise = np.zeros_like(points)
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            # Row-based noise variation
            noise[idx, 0] = np.random.normal(0, 0.008, 1)[0] * (1.0 + 0.3 * (i % 3))
            noise[idx, 1] = np.random.normal(0, 0.008, 1)[0] * (1.0 + 0.3 * (j % 3))

    points += noise

    # Clamp to [0,1] bounds
    points = np.clip(points, 0, 1)

    return points
