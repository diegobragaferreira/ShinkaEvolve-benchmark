    def _generate_hexagonal_grid(self):

        points = []

        # For 16 points, we want a 4x4 triangular lattice (hexagonal packing)
        # We'll use a more mathematically precise approach

        # Define the triangular lattice parameters
        # In an ideal hexagonal packing, points form equilateral triangles
        side_length = 1.0  # Normalized to unit square
        spacing_x = side_length
        spacing_y = side_length * np.sqrt(3) / 2

        # Generate points in triangular lattice
        rows = 4
        cols = 4

        # Create triangular lattice with proper spacing
        for i in range(rows):
            for j in range(cols):
                if len(points) < self.n_points:
                    # Standard triangular lattice positioning
                    x = j * spacing_x + (i % 2) * spacing_x / 2
                    y = i * spacing_y

                    # Apply enhanced symmetry breaking with better mathematical basis
                    # Use more systematic perturbations that preserve hexagonal structure
                    if i + j < self.n_points:

                        perturbation_factor = 0.01

                        y += y_pert + y_unique

        points = np.array(points[:self.n_points])

        # Normalize to [0,1] bounds properly
        if len(points) > 0:
            # Find bounding box
