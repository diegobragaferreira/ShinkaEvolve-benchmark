def generate_multi_scale_geometric_initial(n_steps):

def generate_optimized_step_function(n_steps):
    """Generate an optimized step function specifically designed to maximize C₂"""
    # Create a step function with carefully arranged heights that encourage flat convolution profiles
    f_values = np.zeros(n_steps)

    # Divide the domain into sections with different height patterns
    # This creates strategic peaks and valleys that promote uniform convolution behavior

    # Determine step boundaries
    sections = 5
    section_size = n_steps // sections

    # Create alternating high/low regions to encourage uniform convolution
    for i in range(sections):
        start_idx = i * section_size
        end_idx = start_idx + section_size if i < sections - 1 else n_steps

        # Alternate between high and low values
        if i % 2 == 0:
            # High region
            f_values[start_idx:end_idx] = 1.0
        else:
            # Low region
            f_values[start_idx:end_idx] = 0.3

    # Add some smoothing at boundaries to avoid sharp transitions
    for i in range(1, n_steps-1):
        if f_values[i-1] != f_values[i+1]:
            f_values[i] = (f_values[i-1] + f_values[i+1]) / 2.0

    # Add a central peak to create more complex convolution structure
    center = n_steps // 2
    peak_width = n_steps // 8
    for i in range(max(0, center - peak_width), min(n_steps, center + peak_width)):
        distance_from_center = abs(i - center)
        f_values[i] += 0.5 * (1.0 - distance_from_center / peak_width)

    # Ensure non-negativity
    f_values = np.maximum(f_values, 0)

    # Normalize area to maintain reasonable scale
    total_area = np.sum(f_values) * (0.5 / n_steps)
    if total_area > 0:
        f_values = f_values / total_area * 2.0

    return f_values.tolist()

def multi_resolution_search(n_steps, time_limit=85):
    """Search at multiple resolutions to find good starting points"""
    best_solution = None
    best_c2 = -np.inf

    # Try different resolutions
    resolutions = [100, 150, 200, 250]

    for res in resolutions:
        if time.time() - start_time > time_limit:
            break

        try:
            # Generate initial function at this resolution
            if res <= 150:
                f_values = generate_multi_scale_geometric_initial(res)
            else:
                # Use patterned construction for larger resolutions
                f_values = generate_multi_scale_geometric_initial(res)
                # Add controlled noise for exploration
                noise_level = 0.1
                for i in range(len(f_values)):
                    if np.random.random() < 0.2:
                        f_values[i] *= (1 + np.random.normal(0, noise_level))

            # Ensure non-negativity
            f_values = [max(0, x) for x in f_values]

            # Normalize for numerical stability
            total = sum(f_values)
            if total > 0:
                f_values = [x / total * 5.0 for x in f_values]

            # Local refinement at this resolution
            refined_f, refined_c2 = adaptive_local_search(f_values, max_iterations=10, time_limit=time_limit)

            if refined_c2 > best_c2:
                best_c2 = refined_c2
                # Interpolate back to target resolution if needed
                if res != n_steps:
                    # Create target resolution function
                    target_f = np.interp(np.linspace(0, len(refined_f)-1, n_steps),
                                       np.arange(len(refined_f)), refined_f)
                    best_solution = target_f.tolist()
                else:
                    best_solution = refined_f

        except Exception as e:
            continue

    # If no good solution found, return a default pattern
    if best_solution is None:
        best_solution = generate_multi_scale_geometric_initial(n_steps)
