def compute_autoconvolution_norms(f_values):
    """
    Compute the autoconvolution g = f*f and return its L2, L1, and L-infinity norms.
    Uses piecewise linear integration for L2 norm.
    """
    n = len(f_values)
    if n == 0:
        return 0.0, 0.0, 0.0

    # Use direct convolution for better reliability with small arrays
    # and avoid FFT artifacts. For our case, direct computation is actually
    # more reliable and often faster for moderate sized inputs.
    g = np.zeros(2 * n - 1)

    # Compute autoconvolution manually for better control and numerical stability
    for i in range(n):
        for j in range(n):
            g[i + j] += f_values[i] * f_values[j]

    # Compute norms
    # L2 norm squared
    l2_norm_squared = 0.0
    if len(g) >= 2:
        # Use more accurate integration formula for L2 norm
        # Trapezoidal rule for the integral of g^2
        # For intervals with heights y1, y2, the contribution to integral
        # of y^2 is h*(y1^2 + y1*y2 + y2^2)/3
        h = 1.0  # Step size
        for i in range(len(g) - 1):
            y1 = g[i]
            y2 = g[i+1]
            # More accurate piecewise quadratic integration for g^2
            l2_norm_squared += (h/3.0) * (y1*y1 + y1*y2 + y2*y2)

    # L1 norm - integrate |g| using trapezoidal rule
    l1_norm = 0.0
    if len(g) >= 2:
        h = 1.0
        for i in range(len(g) - 1):
            y1 = np.abs(g[i])
            y2 = np.abs(g[i+1])
            l1_norm += (h/2.0) * (y1 + y2)
    else:
        l1_norm = np.abs(g[0]) if len(g) > 0 else 0.0
    # Normalize by number of intervals (not +1 as before)
    l1_norm /= len(g)

    # L-infinity norm
    l_inf_norm = np.max(np.abs(g))

    return l2_norm_squared, l1_norm, l_inf_norm
