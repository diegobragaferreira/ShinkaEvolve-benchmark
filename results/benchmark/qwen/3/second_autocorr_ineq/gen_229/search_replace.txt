class AutoconvolutionCalculator:
    """Efficient computation of autoconvolution norms"""

    @staticmethod
    def compute_autoconvolution_norms(f_values: List[float]) -> Tuple[float, float, float]:
        """
        Compute the three norms needed for C2 calculation:
        ||g||₂², ||g||₁, ||g||∞ where g = f*f
        Using piecewise linear integration as specified in mathematical formulation
        """
        # Convert to numpy array for efficient computation
        f = np.array(f_values)
        n = len(f)

        # Manually compute autoconvolution g = f * f
        # For step functions, we compute the convolution sum directly
        g = np.zeros(2 * n - 1)
        for i in range(n):
            for j in range(n):
                g[i + j] += f[i] * f[j]

        # Extract the valid convolution part (middle portion)
        center_idx = len(g) // 2
        half_len = len(f)
        g = g[center_idx - half_len + 1:center_idx + half_len]

        # Compute norms using trapezoidal-like integration
        # For ||g||₂², we use piecewise linear integration formula:
        # For consecutive points y0, y1 with width dx: contribution = (dx/3)(y0² + y0*y1 + y1²)
        dx = 1.0 / n  # Assuming unit domain [-1/4, 1/4] with n steps

        # Compute ||g||₂² using trapezoidal-like integration
        g2_sq = 0.0
        for i in range(len(g) - 1):
            y0, y1 = g[i], g[i+1]
            g2_sq += (dx/3) * (y0**2 + y0*y1 + y1**2)

        # Compute ||g||₁ as sum of absolute values with dx scaling
        g1 = np.sum(np.abs(g)) * dx

        # Compute ||g||∞ as maximum absolute value
        g_inf = np.max(np.abs(g))

        return g2_sq, g1, g_inf
