def adaptive_evolutionary_optimization():
    """Main optimization routine using adaptive evolutionary strategy"""
    n_steps = 1000  # Using larger number of steps for better resolution
    pop_size = 50

    # Generate diverse initial population using Latin Hypercube Sampling
    # Create a Latin Hypercube sampler
    sampler = qmc.LatinHypercube(d=n_steps, seed=42)

    # Generate samples in [0,1]^n_steps hypercube
    samples = sampler.random(n=pop_size)

    # Transform samples to appropriate range and create individuals
    population = []
    for i in range(pop_size):
        # Create individual from LHS sample
        individual = samples[i].tolist()

        # Apply some diversity by mixing with geometric pattern occasionally
        if np.random.random() < 0.3:
            geometric_pattern = construct_geometric_initial_function(n_steps)
            # Blend the LHS sample with geometric pattern
            alpha = np.random.random()
            individual = [alpha * individual[j] + (1 - alpha) * geometric_pattern[j]
                         for j in range(n_steps)]

        # Ensure non-negativity and reasonable scaling
        individual = [max(0, x) for x in individual]
        # Normalize to avoid extremely large peaks
        total = sum(individual)
        if total > 0:
            individual = [x / total * 10 for x in individual]

        population.append(individual)

def mutate_individual(individual, mutation_rate, generation=None, max_generations=None):
    """Mutate an individual with given probability and adaptive step size"""
    for i in range(len(individual)):
        if np.random.random() < mutation_rate:
            # Adaptive mutation strength based on generation and parameter value
            if generation is not None and max_generations is not None:
                # Decrease mutation strength over generations
                decay_factor = 1.0 - (generation / max_generations)
                base_strength = 0.1 * individual[i] if individual[i] > 0 else 0.1
                mutation_strength = base_strength * decay_factor
            else:
                mutation_strength = 0.1 * individual[i] if individual[i] > 0 else 0.1

            individual[i] = max(0, individual[i] + np.random.normal(0, mutation_strength))
