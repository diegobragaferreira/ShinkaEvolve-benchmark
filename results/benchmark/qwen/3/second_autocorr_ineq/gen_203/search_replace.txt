<DESCRIPTION>
Adding Latin Hypercube Sampling (LHS) for better initialization diversity. This improves the multi-start strategy by ensuring more systematic exploration of the search space compared to simple random sampling. LHS will create more diverse and representative starting points that maintain the beneficial geometric structure while exploring different regions of the optimization landscape, which should help discover higher-value local optima that uniform random sampling might miss.

def generate_lhs_initial_functions(n_functions, n_steps, seed=42):
    """Generate multiple initial functions using Latin Hypercube Sampling"""
    # Set up LHS sampler
    sampler = qmc.LatinHypercube(d=n_steps, seed=seed)

    # Generate samples in [0,1]^n_steps
    samples = sampler.random(n=n_functions)

    # Transform to appropriate range
    lhs_functions = []

    for sample in samples:
        # Create function using the LHS sample
        # Use the sample as weights for a combination of base patterns
        f_values = np.zeros(n_steps)

        # Base pattern with LHS-generated weights
        x = np.linspace(-0.25, 0.25, n_steps)

        # Create multiple Gaussian bumps with varying parameters based on sample values
        for i in range(min(5, n_steps//10)):  # Limit to reasonable number of bumps
            # Use sample values to determine bump properties
            height = 0.5 + 0.5 * sample[i % len(sample)]  # Height between 0.5-1.0
            width = 0.02 + 0.08 * sample[(i+1) % len(sample)]  # Width between 0.02-0.1
            center = -0.2 + 0.4 * sample[(i+2) % len(sample)]  # Center between -0.2 and 0.2

            bump = height * np.exp(-0.5 * ((x - center) / width)**2)
            f_values += bump

        # Ensure non-negativity
        f_values = np.maximum(f_values, 0)

        # Normalize
        total_area = np.sum(f_values) * (0.5 / n_steps)
        if total_area > 0:
            f_values = f_values / total_area * 2.0

        lhs_functions.append(f_values.tolist())

    return lhs_functions
