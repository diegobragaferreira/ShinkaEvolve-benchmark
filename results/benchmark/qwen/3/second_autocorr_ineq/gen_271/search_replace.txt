def generate_lhs_initial_function(length, num_bumps=8):
    """Generate initial function using Latin Hypercube Sampling for better distribution"""
    # Create multi-scale Gaussian bumps with LHS positioning
    individual = np.zeros(length)

    # Use Latin Hypercube Sampling to get well-distributed bump parameters
    sampler = qmc.LatinHypercube(d=3, seed=42)  # 3 parameters: center, width, height
    samples = sampler.random(n=num_bumps)

    # Transform samples to appropriate ranges
    centers = qmc.scale(samples[:, 0], [0, length], [0, length]).astype(int)
    widths = qmc.scale(samples[:, 1], [1, length//20], [1, length//20]).astype(int)
    heights = qmc.scale(samples[:, 2], [0.5, 1.5], [0.5, 1.5])

    # Add Gaussian bumps at LHS-dispersed locations
    for center, width, height in zip(centers, widths, heights):
        # Ensure center is within bounds
        center = max(0, min(length-1, center))
        # Fill in the Gaussian shape
        for i in range(length):
            distance = abs(i - center)
            if distance < width * 3:  # Only fill within reasonable range
                gaussian_val = height * math.exp(-0.5 * (distance / width)**2)
                individual[i] += gaussian_val

    # Apply some additional randomization to avoid perfect symmetry
    for i in range(length):
        if np.random.random() < 0.5:
            individual[i] = max(0.0, individual[i] + np.random.normal(0, 0.1))

    return individual.tolist()
