import numpy as np
from scipy.optimize import differential_evolution, minimize
from numba import njit
import time

@njit
def compute_autoconvolution_norms(f_values):
    """Compute autoconvolution and norms efficiently with numba JIT"""
    n = len(f_values)
    # Compute autoconvolution g = f * f
    g = np.zeros(2*n - 1)
    for i in range(n):
        for j in range(n):
            g[i + j] += f_values[i] * f_values[j]

    # Compute norms
    g_squared = g * g
    norm_g2_squared = np.sum(g_squared)

    # Approximate L1 norm using trapezoidal rule
    norm_g1_approx = np.sum(np.abs(g)) / (len(g) + 1)

    # Infinity norm
    norm_ginf = np.max(np.abs(g))

    return norm_g2_squared, norm_g1_approx, norm_ginf

@njit
def calculate_c2(f_values):
    """Calculate C2 value for given step function values"""
    norm_g2_squared, norm_g1_approx, norm_ginf = compute_autoconvolution_norms(f_values)

    # Avoid division by zero
    if norm_g1_approx < 1e-15 or norm_ginf < 1e-15:
        return 0.0

    c2 = norm_g2_squared / (norm_g1_approx * norm_ginf)
    return c2

def construct_function() -> list[float]:
    """Function to construct step-function with high C2 value using adaptive optimization."""
    
    # Set random seed for reproducibility
    np.random.seed(42)
    
    # Start timing
    start_time = time.time()
    
    # Initialize parameters
    best_c2 = 0.0
    best_f_values = None
    
    # Try multiple configurations to find the optimal
    for attempt in range(5):  # Run multiple attempts
        # Randomly determine function length
        n = np.random.randint(1000, 10000)
        
        # Create initial guess with more sophisticated pattern
        # Start with a pattern that tends to produce good results
        if attempt % 3 == 0:
            # Create a symmetric pattern with peaks
            f_values = np.ones(n) * 0.5
            # Add some variation in the middle
            mid_point = n // 2
            f_values[mid_point-10:mid_point+10] = 1.0
        elif attempt % 3 == 1:
            # Create a more uniform distribution
            f_values = np.random.random(n) * 0.8 + 0.2
        else:
            # Create alternating pattern
            f_values = np.ones(n) * 0.3
            f_values[::2] = 0.8
            
        # Clip negative values to zero
        f_values = np.clip(f_values, 0, None)
        
        # Apply optimization using differential evolution
        try:
            # Create bounds for each parameter
            bounds = [(0.0, 2.0) for _ in range(n)]
            
            # Perform optimization
            result = differential_evolution(
                lambda x: -calculate_c2(x),  # Minimize negative C2 (maximize C2)
                bounds,
                maxiter=50,
                popsize=15,
                seed=42,
                disp=False
            )
            
            if result.success:
                optimized_values = np.clip(result.x, 0, None)
                current_c2 = calculate_c2(optimized_values)
                
                if current_c2 > best_c2:
                    best_c2 = current_c2
                    best_f_values = optimized_values.tolist()
                    
        except Exception:
            # If optimization fails, use the original values
            current_c2 = calculate_c2(f_values)
            if current_c2 > best_c2:
                best_c2 = current_c2
                best_f_values = f_values.tolist()
    
    # If still no good solution found, fall back to a well-known good pattern
    if best_f_values is None:
        n = 2000
        # Create a specific good pattern based on prior knowledge
        f_values = np.ones(n) * 0.5
        # Add peaks at center
        mid_point = n // 2
        f_values[mid_point-50:mid_point+50] = 1.0
        best_f_values = f_values.tolist()
    
    # Ensure we don't exceed time limit
    elapsed_time = time.time() - start_time
    if elapsed_time > 85:  # Leave buffer for final processing
        # Re-run with smaller size if we're running out of time
        n_small = min(1000, len(best_f_values))
        best_f_values = best_f_values[:n_small]
    
    return best_f_values

# EVOLVE-BLOCK-END

if __name__ == "__main__":
    f_values = construct_function()
    print(f"Function: {f_values}")
