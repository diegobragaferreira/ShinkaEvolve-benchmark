def coordinate_wise_refinement(initial_params: np.ndarray, size: int, max_iter: int = 30) -> list:
    """
    Perform coordinate-wise refinement on the solution with adaptive step sizes.
    """
    current_params = initial_params.copy()
    current_value = -smooth_objective(current_params, size)

    # Track improvement for early termination
    last_improvement = 0
    consecutive_no_improvement = 0

    # Adaptive step size tracking
    step_sizes = np.full(len(current_params), 0.05)  # Initial step sizes for each parameter

    for iteration in range(max_iter):
        improved = False
        total_improvements = 0

        # Try improving each coordinate individually
        # Shuffle indices for more diverse exploration
        indices = list(range(len(current_params)))
        np.random.shuffle(indices)

        for i in indices:
            original_value = current_params[i]

            # Adaptive step size: start large, decrease over iterations
            # and also adjust based on recent performance
            adaptive_step = step_sizes[i] * (1.0 - iteration / max_iter * 0.5)

            # Try both positive and negative changes with adaptive step
            deltas = [adaptive_step * original_value, -adaptive_step * original_value]

            for delta in deltas:
                if abs(delta) < 1e-10:
                    continue

                new_params = current_params.copy()
                new_params[i] = max(0, original_value + delta)

                # Project to valid space
                new_params = create_discrete_projection(new_params)

                new_value = -smooth_objective(new_params, size)

                if new_value > current_value:
                    current_params = new_params.copy()
                    current_value = new_value
                    improved = True
                    total_improvements += 1
                    last_improvement = iteration

                    # Update step size: increase if successful
                    step_sizes[i] = min(step_sizes[i] * 1.2, 0.5)  # Cap at 0.5
                    consecutive_no_improvement = 0
                    break
                else:
                    # Decrease step size if unsuccessful to avoid overshooting
                    step_sizes[i] = max(step_sizes[i] * 0.8, 0.001)

            if improved:
                break

        if not improved:
            consecutive_no_improvement += 1
            # Further reduce step sizes if no improvements
            step_sizes *= 0.9
        else:
            # Reset step sizes for parameters that were just improved
            step_sizes = np.where(np.isin(np.arange(len(step_sizes)), [i for i in indices if improved]),
                                np.minimum(step_sizes * 1.1, 0.5), step_sizes)

        # Early termination if no improvement for many iterations
        if consecutive_no_improvement > 5:
            break

    return current_params.tolist()
