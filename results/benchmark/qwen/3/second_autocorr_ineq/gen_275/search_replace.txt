def generate_spectral_initial_function(n: int) -> List[float]:
    """
    Create initial function based on spectral properties that should yield good C2 values.
    """
    # Create a function that should yield relatively flat convolution
    # This starts with a low-pass filtered approach

    # Base: sine wave with controlled harmonics
    x = np.linspace(0, 2*np.pi, n)

    # Primary component (fundamental)
    fundamental = 0.5 + 0.3 * np.sin(x)

    # Add some mid-frequency components to avoid overly simple patterns
    harmonic1 = 0.1 * np.sin(3*x)
    harmonic2 = 0.1 * np.cos(5*x)

    # Combine
    combined = fundamental + harmonic1 + harmonic2

    # Add some localized peaks for better convolution mixing

    combined = np.maximum(combined, 0)

def generate_multi_scale_gaussian_initial_function(n: int) -> List[float]:
    """
    Create initial function using multi-scale Gaussian patterns for better C2 values.
    This approach creates a hierarchy of Gaussian bumps with different scales and positions,
    following successful patterns from top-performing programs.
    """
    # Create a multi-scale Gaussian pattern
    f_values = np.zeros(n)

    # Define multiple scales for the Gaussian bumps
    scales = [0.05, 0.1, 0.15, 0.2]  # Different widths for different scales
    num_bumps_per_scale = max(1, n // 50)  # Number of bumps per scale

    # Generate Gaussian bumps at different scales
    for scale_idx, scale in enumerate(scales):
        # Determine number of bumps for this scale
        current_num_bumps = min(num_bumps_per_scale, n // 10)

        # Place bumps at regular intervals for this scale
        for bump_idx in range(current_num_bumps):
            # Position the bump
            position = (bump_idx + 0.5) * (n / (current_num_bumps + 1))

            # Create Gaussian bump
            x = np.arange(n)
            height = 0.5 + 0.3 * random.random()  # Random height
            center = int(position)

            # Ensure center is within bounds
            if 0 <= center < n:
                gaussian = height * np.exp(-0.5 * ((x - center) / (scale * n)) ** 2)
                f_values += gaussian

    # Add some additional randomness to avoid overly structured patterns
    noise_level = 0.05
    noise = np.random.normal(0, noise_level, n)
    f_values += noise

    # Ensure non-negativity
    f_values = np.maximum(f_values, 0)

    # Normalize to reasonable values
    if np.max(f_values) > 0:
        f_values = f_values / (np.max(f_values) + 1e-10) * 2.0

    return f_values.tolist()
