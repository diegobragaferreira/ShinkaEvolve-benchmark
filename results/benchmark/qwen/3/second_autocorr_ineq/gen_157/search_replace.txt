def construct_multiscale_gaussian_initial_function(n_steps: int = 200) -> List[float]:
    """
    Construct step function using multi-scale Gaussian patterns for better convolution properties.
    This creates a more principled geometric foundation that systematically varies scales.
    """
    f_values = np.zeros(n_steps)

    # Create multiple Gaussian bumps with varying scales and positions
    # This follows a systematic progression from large-scale to fine-scale features

    # Parameters for multi-scale Gaussian construction
    num_bumps = min(10, n_steps // 10)  # Number of bumps depends on resolution
    bump_positions = np.linspace(0.1, 0.9, num_bumps)  # Spread bumps across domain
    bump_scales = np.logspace(-1, 0, num_bumps, base=10)  # Varying scales (from coarse to fine)
    bump_heights = np.logspace(0, 1, num_bumps, base=10)  # Varying heights

    # Normalize heights to avoid extreme values
    bump_heights = bump_heights / np.sum(bump_heights) * 5.0

    # Create each Gaussian bump
    x = np.linspace(0, 1, n_steps)
    for i in range(num_bumps):
        # Position bump according to logarithmic spacing
        position = bump_positions[i]
        scale = bump_scales[i] * 0.1  # Scale factor to control spread
        height = bump_heights[i]

        # Create Gaussian bump
        bump = height * np.exp(-((x - position)**2) / (2 * scale**2))
        f_values += bump

    # Add some additional smooth variation
    smooth_variation = 0.3 * np.sin(4 * np.pi * x) + 0.7
    f_values = f_values * smooth_variation

    # Ensure non-negativity and normalize
    f_values = np.clip(f_values, 0, None)

    # Normalize to reasonable range
    total = np.sum(f_values)
    if total > 0:
        f_values = f_values / total * 10

    # Add some controlled noise for diversification
    noise = np.random.normal(0, 0.01, n_steps)
    f_values = np.clip(f_values + noise, 0, None)

    return f_values.tolist()

def gradient_free_construct_function() -> List[float]:
    """Construct step function using gradient-free optimization with Nevergrad"""
    # Set seed for reproducibility
    random.seed(42)
    np.random.seed(42)

    # Use a smarter initialization
    initial_guess = construct_multiscale_gaussian_initial_function(200)

    # Define the optimization problem with more robust handling
    def objective(x):
        # Clip negative values to prevent numerical issues
        x = np.maximum(x, 0)
        # Calculate C2 (we want to maximize it, so minimize negative)
        c2_value = calculate_c2(x.tolist())
        # Return negative because Nevergrad minimizes
        return -c2_value if c2_value > 0 else 1e10

    # Create a more suitable optimizer for this problem
    # CMA-ES is often better for continuous black-box optimization
    try:
        optimizer = ng.optimizers.CMA(
            dimension=len(initial_guess),
            budget=500,  # Limit evaluations to stay within time constraints
            num_workers=1
        )
    except:
        # Fallback to OnePlusOne if CMA is not available
        optimizer = ng.optimizers.OnePlusOne(
            dimension=len(initial_guess),
            budget=500,
            num_workers=1
        )

    # Optimize with error handling
    try:
        recommendation = optimizer.minimize(objective, verbosity=0)
        result = recommendation.args[0]
    except:
        # If optimization fails, return the initial guess
        result = np.array(initial_guess)

    # Ensure non-negativity and clip extreme values
    result = np.clip(result, 0, 100)

    # Return as list
    return result.tolist()
