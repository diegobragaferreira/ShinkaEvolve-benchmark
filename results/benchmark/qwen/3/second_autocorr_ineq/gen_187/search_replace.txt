def construct_geometric_initial_function(n: int) -> List[float]:
    """
    Create initial function with geometric patterns that are likely to produce good C2 values
    """
    # Create a pattern that balances high peaks with spread-out structure
    f_vals = []

    # Base pattern: alternating high/low with some smooth transitions
    for i in range(n):
        # Create a base geometric structure
        x = i / (n - 1) if n > 1 else 0.5
        # Add some smooth variation around a geometric base
        base_val = 0.5 + 0.5 * np.sin(4 * np.pi * x)
        # Add some randomization but keep values positive
        noise = 0.1 * (np.random.random() - 0.5)
        val = max(0, base_val + noise)
        f_vals.append(val)

    # Occasionally add a few strong peaks for convolution enhancement
    num_peaks = max(1, n // 20)
    for _ in range(num_peaks):
        peak_pos = random.randint(0, n-1)
        f_vals[peak_pos] = max(f_vals[peak_pos], 2.0 * random.random())

    return f_vals

def construct_lhs_initial_function(n: int) -> List[float]:
    """
    Create initial function using Latin Hypercube Sampling for better diversity
    """
    # Create Latin Hypercube Sampler
    sampler = qmc.LatinHypercube(d=1, seed=42)

    # Generate samples in [0, 1]
    samples = sampler.random(n)

    # Transform to appropriate range and add variation
    f_vals = []

    # Create a base pattern with some structure
    for i in range(n):
        # Use LHS sample as base with some sinusoidal modulation
        x = samples[i][0]

        # Create structured pattern with LHS influence
        base_val = 0.3 + 0.7 * np.sin(2 * np.pi * x + np.pi/4) * np.cos(4 * np.pi * x)

        # Add some random component but keep positive
        noise = 0.1 * (np.random.random() - 0.5)
        val = max(0.01, base_val + noise)

        f_vals.append(val)

    # Occasionally add stronger peaks for convolution enhancement
    num_peaks = max(1, n // 25)
    for _ in range(num_peaks):
        peak_pos = random.randint(0, n-1)
        f_vals[peak_pos] = max(f_vals[peak_pos], 1.5 + 0.5 * random.random())

    return f_vals
