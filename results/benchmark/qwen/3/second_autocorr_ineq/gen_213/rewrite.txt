import numpy as np

@jit(nopython=True)
def compute_autoconvolution_fast(f_vals):
    """Fast numba-based autoconvolution computation"""
    n = len(f_vals)
    g = np.zeros(2*n - 1)
    
    for i in range(n):
        for j in range(n):
            g[i + j] += f_vals[i] * f_vals[j]
    
    return g

@jit(nopython=True) 
def compute_convolution_norms_fast(g_vals):
    """Fast computation of convolution norms"""
    n = len(g_vals)
    l2_sq = 0.0
    l1 = 0.0
    l_inf = 0.0
    
    for i in range(n):
        val = g_vals[i]
        l2_sq += val * val
        l1 += abs(val)
        if abs(val) > l_inf:
            l_inf = abs(val)
    
    return l2_sq, l1, l_inf

def compute_autoconvolution_norms(f_values: List[float]) -> Tuple[float, float, float]:
    """
    Compute the three norms needed for C2 calculation with proper numerical handling
    """
    n = len(f_values)
    if n == 0:
        return 0.0, 0.0, 0.0
    
    # Ensure non-negative values
    f_array = np.array(f_values)
    f_array = np.maximum(f_array, 0.0)
    
    # Compute autoconvolution
    g_vals = compute_autoconvolution_fast(f_array)
    
    # Compute norms
    l2_sq, l1, l_inf = compute_convolution_norms_fast(g_vals)
    
    # Handle edge cases
    if l1 <= 1e-15:
        l1 = 1e-15
    if l_inf <= 1e-15:
        l_inf = 1e-15
        
    return l2_sq, l1, l_inf

def calculate_c2(f_values: List[float]) -> float:
    """
    Calculate the C2 constant from the step function values.
    """
    try:
        g_norm_2_sq, g_norm_1, g_norm_inf = compute_autoconvolution_norms(f_values)
        c2 = g_norm_2_sq / (g_norm_1 * g_norm_inf)
        return c2
    except:
        return 0.0

def construct_harmonic_initial_function(n: int) -> List[float]:
    """
    Create initial function based on harmonic superposition principles
    designed to maximize convolution properties
    """
    # Create a function with specific frequency characteristics
    # that should lead to good C2 values through constructive interference
    f_vals = []
    
    # Base frequency: use a fundamental frequency that gives good convolution behavior
    base_freq = 8
    
    # Generate a function with primary harmonic components
    for i in range(n):
        x = i / (n - 1) if n > 1 else 0.5
        
        # Construct function with multiple harmonics for good convolution properties
        # Primary fundamental with amplitude modulation
        fundamental = 0.5 * (1 + np.sin(2 * np.pi * base_freq * x))
        
        # Add second harmonic with different phase
        second_harm = 0.3 * np.sin(4 * np.pi * base_freq * x)
        
        # Add third harmonic
        third_harm = 0.2 * np.sin(6 * np.pi * base_freq * x)
        
        # Add some randomness but ensure positivity and reasonable values
        noise_amp = 0.1 * (0.5 - np.random.random())
        
        val = max(0.0, fundamental + second_harm + third_harm + noise_amp)
        f_vals.append(val)
    
    # Introduce some localized peaks for convolution enhancement
    # These help create a better balance between ||g||₂² and ||g||₁·||g||∞
    for _ in range(max(1, n // 20)):
        peak_pos = random.randint(0, n-1)
        # Make it stronger for convolution effect
        f_vals[peak_pos] = max(f_vals[peak_pos], 2.0 * random.random())
    
    return f_vals

def analyze_convolution_structure(f_values: List[float]) -> dict:
    """
    Analyze the convolution structure to understand key characteristics
    """
    g_norm_2_sq, g_norm_1, g_norm_inf = compute_autoconvolution_norms(f_values)
    
    # Compute the normalized convolution to understand its shape
    n = len(f_values)
    dx = 0.5 / n
    
    # Get the full convolution
    f_array = np.array(f_values)
    g_full = compute_autoconvolution_fast(f_array)
    
    # Extract central part (corresponding to [-1/4, 1/4])
    central_start = (len(g_full) - n) // 2
    g_centered = g_full[central_start:central_start+n]
    
    # Analyze the profile
    g_abs = np.abs(g_centered)
    
    # Compute average height and variance
    avg_height = np.mean(g_abs)
    var_height = np.var(g_abs)
    
    # Find peak location and magnitude
    peak_idx = np.argmax(g_abs)
    peak_magnitude = g_abs[peak_idx]
    
    # Compute energy concentration (how much of the L2 norm is concentrated in peaks)
    total_energy = np.sum(g_abs**2) * dx
    peak_energy = g_abs[peak_idx]**2 * dx
    energy_concentration = peak_energy / total_energy if total_energy > 0 else 0
    
    return {
        'avg_height': avg_height,
        'variance': var_height,
        'peak_magnitude': peak_magnitude,
        'peak_position': peak_idx,
        'energy_concentration': energy_concentration,
        'total_l2_squared': g_norm_2_sq,
        'total_l1': g_norm_1,
        'inf_norm': g_norm_inf
    }

def harmonic_guided_local_search(start_point: List[float], iterations: int = 100) -> Tuple[List[float], float]:
    """
    Local search guided by harmonic analysis of the convolution structure
    """
    current_solution = np.array(start_point)
    current_c2 = calculate_c2(start_point)
    
    best_solution = current_solution.copy()
    best_c2 = current_c2
    
    n = len(current_solution)
    
    # Analyze the current convolution for guidance
    analysis = analyze_convolution_structure(start_point)
    
    # Try to improve by adjusting in direction that increases L2^2 while decreasing ||g||₁·||g||∞
    for iter_num in range(iterations):
        # Slightly perturb the solution in a way that might improve the ratio
        # Focus on modifying areas that influence the energy distribution
        perturbed = current_solution.copy()
        
        # Select a subset of positions to modify
        num_changes = max(1, n // 20)
        change_indices = random.sample(range(n), num_changes)
        
        # For each selected index, modify based on convolution analysis
        for idx in change_indices:
            # Modify based on local information
            current_val = perturbed[idx]
            
            # Determine perturbation based on current structure
            # If we have a sharp peak, might want to flatten it slightly
            # If we have flat regions, might want to boost them a bit
            if analysis['peak_magnitude'] > analysis['avg_height'] * 1.5:
                # Reduce peak height slightly to increase L2 norm efficiency
                delta = -0.01 * current_val
            else:
                # Boost moderate values to increase energy concentration
                delta = 0.01 * current_val
                
            perturbed[idx] = max(0.0, current_val + delta)
            
            # Add some randomness to escape local minima
            if random.random() < 0.3:
                noise = random.gauss(0, 0.005 * current_val if current_val > 0 else 0.005)
                perturbed[idx] = max(0.0, perturbed[idx] + noise)
        
        # Evaluate new solution
        try:
            new_c2 = calculate_c2(perturbed.tolist())
            if new_c2 > current_c2:
                current_solution = perturbed
                current_c2 = new_c2
                if new_c2 > best_c2:
                    best_c2 = new_c2
                    best_solution = current_solution.copy()
        except:
            # If evaluation fails, revert to current
            continue
    
    return best_solution.tolist(), best_c2

def multi_scale_harmonic_optimization(max_starts: int = 15) -> Tuple[List[float], float]:
    """
    Multi-scale optimization using harmonic principles
    """
    best_overall_c2 = 0.0
    best_overall_solution = None
    
    # Try different harmonic configurations
    for start_iter in range(max_starts):
        # Vary the number of steps to explore different resolutions
        n = random.randint(200, 1000)
        
        # Create harmonic-initialization with different parameters
        if start_iter % 3 == 0:
            # Use basic harmonic pattern
            initial_solution = construct_harmonic_initial_function(n)
        elif start_iter % 3 == 1:
            # Use more complex harmonic with different frequencies
            initial_solution = construct_harmonic_initial_function(n)
            # Add some structured modification
            for i in range(n):
                if i % 10 == 0:
                    initial_solution[i] = max(0.0, initial_solution[i] * 1.2)
        else:
            # Use enhanced harmonic with peak modulation
            initial_solution = construct_harmonic_initial_function(n)
            # Enhance mid-range values
            for i in range(n):
                if 0.3 < i/n < 0.7:
                    initial_solution[i] = max(0.0, initial_solution[i] * 1.1)
        
        # First local search
        solution, c2 = harmonic_guided_local_search(initial_solution, 50)
        
        # If not good, try more aggressive search
        if c2 < 0.8 * best_overall_c2 and best_overall_c2 > 0:
            solution, c2 = harmonic_guided_local_search(solution, 100)
        
        if c2 > best_overall_c2:
            best_overall_c2 = c2
            best_overall_solution = solution.copy()
    
    return best_overall_solution, best_overall_c2

    """
    Function to construct step-function with high C2 value.
    Uses harmonic-based optimization framework.
    """
    start_time = time.time()
    
    # Set seeds for reproducibility
