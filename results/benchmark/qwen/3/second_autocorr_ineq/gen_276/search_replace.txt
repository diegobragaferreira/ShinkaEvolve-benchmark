def generate_sinc_initial_function(n):
    """
    Generate initial function based on sinc-like pattern which tends to create
    well-behaved convolutions with minimal peakiness
    """
    x = np.linspace(-0.25, 0.25, n)

    # Create a pattern with low-frequency oscillations that build up to a smooth profile
    # Sinc-like behavior works well for reducing extreme peaks in convolution
    pattern = np.sinc(2 * x) * 0.5 + 0.5

    # Add some structured variation
    # Create a smooth base that's roughly increasing toward center and decreasing
    center = n // 2
    base = np.zeros(n)
    for i in range(n):
        dist = abs(i - center)
        base[i] = max(0, 1 - dist / (n/2))

    # Blend the patterns
    final = 0.6 * pattern + 0.4 * base

    # Ensure non-negativity
    final = np.maximum(final, 0)

    # Normalize to reasonable scale
    total = np.sum(final)
    if total > 0:
        final = final / total * 5.0

    return final.tolist()

def generate_multiscale_gaussian_initial_function(n):
    """
    Generate initial function using multi-scale Gaussian pattern construction.

    This creates a structured pattern with multiple Gaussian bumps at different scales
    to encourage good convolution behavior across various spatial frequencies.
    Similar to successful approaches in prior implementations.
    """
    # Create base function with multi-scale Gaussian components
    pattern = np.zeros(n)

    # Define multiple scales of Gaussian bumps
    scales = [n // 20, n // 15, n // 10, n // 8]
    centers = [n // 4, n // 2, 3 * n // 4]

    # Create Gaussian bumps at different scales and positions
    for scale in scales:
        for center in centers:
            # Create Gaussian bump
            x = np.arange(n)
            gauss = np.exp(-0.5 * ((x - center) / scale) ** 2)
            pattern += gauss * np.random.uniform(0.5, 1.5)  # Random amplitude

    # Add some additional structured variation
    # Create a base pattern that encourages uniformity in convolution
    base_pattern = np.sin(np.linspace(0, 4*np.pi, n)) * 0.3 + 0.7
    pattern += base_pattern * 0.5

    # Ensure non-negativity and normalize
    pattern = np.maximum(pattern, 0)

    # Normalize to control the overall magnitude
    total = np.sum(pattern)
    if total > 0:
        pattern = pattern / total * 5.0
