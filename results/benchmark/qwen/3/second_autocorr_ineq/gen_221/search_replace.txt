def evolve_generation(population: list, fitness_scores: list, generation: int) -> list:
    """
    Evolve one generation with adaptive parameters.
    """
    new_population = []
    pop_size = len(population)

    # Elitism: keep best individual
    best_idx = np.argmax(fitness_scores)
    new_population.append(population[best_idx].copy())

    # Calculate population diversity
    fitness_array = np.array(fitness_scores)
    diversity = np.std(fitness_array) / (np.mean(fitness_array) + 1e-10)

    # Adaptive mutation strength based on generation and diversity
    # Start with high mutation for exploration, decrease with generation
    # But also increase if diversity is low (indicating premature convergence)
    if generation < 10:
        base_mutation = 0.3  # High exploration
    elif generation < 20:
        base_mutation = 0.15  # Balanced
    else:
        base_mutation = 0.05  # High exploitation

    # Increase mutation if diversity is low (premature convergence risk)
    if diversity < 0.1:
        mutation_strength = base_mutation * 2.0
    else:
        mutation_strength = base_mutation

    # Generate rest through selection, crossover, and mutation
    for _ in range(pop_size - 1):
        # Selection
        parent1 = tournament_selection(population, fitness_scores)
        parent2 = tournament_selection(population, fitness_scores)

        # Simple crossover (average of two parents)
        child = (parent1 + parent2) / 2.0

        # Mutation
        child = evolve_individual(child, mutation_strength)

        new_population.append(child)

    return new_population
