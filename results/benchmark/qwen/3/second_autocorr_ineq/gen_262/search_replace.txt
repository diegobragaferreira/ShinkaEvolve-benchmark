    # g[k] = integral f(x) * f(y) where x+y = k*dx - domain_length/2
    # We'll compute this as a discrete approximation

    g_size = 2 * n_steps - 1
    g = np.zeros(g_size)

    # Compute autoconvolution - JIT compiled loop
    for i in range(n_steps):
        for j in range(n_steps):
            k = i + j
            if 0 <= k < g_size:
                # Trapezoidal-like integration contribution
                # For two adjacent pieces with heights f[i] and f[j],
                # the convolution contributes to g[k]

    # Compute norms
    # ||g||₂² = sum(g_i² * dx) - but we're doing piecewise linear integration

    g1 = np.sum(np.abs(g)) * dx

    # ||g||∞ = max(|g_i|)
    ginf = np.max(np.abs(g))

@njit
def compute_convolution_norms_numba(f_values, domain_length=0.5):
    """
    Compute the three norms needed for C2 calculation using the provided step function.
    JIT compiled version for improved performance.
    """
    n_steps = len(f_values)
    if n_steps == 0:
        return 0.0, 0.0, 0.0

    # Step size
    dx = domain_length / n_steps

    # Compute autoconvolution using more efficient method for piecewise linear integration
    # For each output index k, we need to compute the integral of f(x)*f(y) where x+y=k*dx
    # This involves computing overlaps of step functions
    g_size = 2 * n_steps - 1
    g = np.zeros(g_size)

    # Efficient computation for autoconvolution using step function overlap
    # Each f[i] contributes to g[k] for all k where i+j=k, i,j in [0,n_steps-1]
    for i in range(n_steps):
        for j in range(n_steps):
            k = i + j
            if 0 <= k < g_size:
                g[k] += f_values[i] * f_values[j]

    # For the L2 norm, compute ∫ g(x)² dx using trapezoidal-like piecewise integration:
    # For consecutive values g[i], g[i+1], we approximate ∫ g(x)² dx on that segment
    # Using: (dx/3)(g[i]² + g[i]g[i+1] + g[i+1]²)
    g2_sq = 0.0
    for i in range(len(g)-1):
        g2_sq += (dx/3) * (g[i]**2 + g[i]*g[i+1] + g[i+1]**2)

    # L1 norm: ∫ |g(x)| dx ≈ sum(|g[i]| * dx)
    g1 = np.sum(np.abs(g)) * dx

    # L-infinity norm: max(|g[i]|)
    ginf = np.max(np.abs(g))

    return g2_sq, g1, ginf
