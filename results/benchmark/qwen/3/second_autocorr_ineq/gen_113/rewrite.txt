import numpy as np
from numba import njit
from scipy import signal
import random
from typing import List

@njit
def compute_autoconvolution_norms(f_vals):
    """
    Compute the autoconvolution g = f*f and return its norms.
    Uses fast numba-compiled operations.
    """
    n = len(f_vals)
    # Autoconvolution using direct computation
    g = np.zeros(2*n - 1)

    # Compute convolution manually for efficiency
    for i in range(n):
        for j in range(n):
            g[i + j] += f_vals[i] * f_vals[j]

    # Compute norms
    g_squared = g * g
    norm_g2_squared = np.sum(g_squared)
    norm_g1 = np.sum(np.abs(g))
    norm_g_inf = np.max(np.abs(g))

    return norm_g2_squared, norm_g1, norm_g_inf

@njit
def calculate_c2(f_vals):
    """
    Calculate C2 value for given step function values.
    """
    norm_g2_squared, norm_g1, norm_g_inf = compute_autoconvolution_norms(f_vals)

    # Avoid division by zero
    if norm_g1 < 1e-15 or norm_g_inf < 1e-15:
        return 0.0

    c2 = norm_g2_squared / (norm_g1 * norm_g_inf)
    return c2

def construct_function() -> list[float]:
    """
    Sparse Convex Evolution Optimizer for maximizing C2.
    Uses a novel approach combining sparse signal recovery,
    multi-scale pattern analysis, and adaptive refinement.
    """
    # Set seeds for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    # Stage 1: Multi-scale initialization with sparse structures
    # Generate candidate functions with varying sparsity patterns
    min_steps = 300
    max_steps = 2000
    n_steps = random.randint(min_steps, max_steps)
    
    # Create a sparse, structured foundation using wavelet-inspired approach
    # This captures the self-similar nature often optimal for C2
    base_pattern = np.zeros(n_steps)
    
    # Place significant weight at center (most important for convolution)
    center = n_steps // 2
    base_pattern[center] = 5.0
    
    # Add sparse spikes at strategic locations (creating beneficial overlap in convolution)
    spike_positions = [center // 2, center, 3 * center // 2]
    for pos in spike_positions:
        if 0 <= pos < n_steps:
            base_pattern[pos] = 3.0
    
    # Apply geometric decay from center to form a basis for the function
    for i in range(center):
        base_pattern[center - i] *= 0.9**(i/10)
        base_pattern[center + i] *= 0.9**(i/10)
    
    # Add some random sparse components for exploration
    for _ in range(n_steps // 50):  # Add roughly 2% of all positions
        idx = random.randint(0, n_steps - 1)
        base_pattern[idx] += random.uniform(0.1, 1.0)
    
    # Normalize to prevent overflow
    if np.sum(base_pattern) > 0:
        base_pattern = base_pattern / np.sum(base_pattern) * 100
    
    # Stage 2: Spectral filtering and adaptive thresholding
    # Apply a form of soft-thresholding in frequency domain (approximate)
    filtered_pattern = base_pattern.copy()
    
    # Apply moving average with variable window to smooth while preserving peaks
    window_sizes = [3, 5, 7]
    for window_size in window_sizes:
        if window_size < n_steps:
            # Simple moving average smoothing
            smoothed = np.convolve(filtered_pattern, np.ones(window_size)/window_size, mode='same')
            filtered_pattern = (filtered_pattern + smoothed) / 2
    
    # Stage 3: Iterative refinement with adaptive pruning
    refined_pattern = filtered_pattern.copy()
    
    # Adaptive pruning and enhancement
    iterations = 5
    for iteration in range(iterations):
        # Apply thresholding based on magnitude
        threshold = np.percentile(refined_pattern, 30)  # Remove bottom 30%
        pruned = np.where(refined_pattern > threshold, refined_pattern, 0)
        
        # Enhance remaining components
        pruned = pruned * 1.1  # Slight amplification
        
        # Ensure non-negativity after modifications
        pruned = np.maximum(pruned, 0)
        
        # Normalize
        if np.sum(pruned) > 0:
            pruned = pruned / np.sum(pruned) * 100
            
        refined_pattern = pruned.copy()
    
    # Stage 4: Local optimization refinement
    initial_vals = refined_pattern.tolist()
    
    # Try to improve using simulated annealing inspired approach
    best_vals = initial_vals.copy()
    best_c2 = calculate_c2(best_vals)
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.95
    min_temp = 0.01
    max_iterations = 100
    
    for iteration in range(max_iterations):
        if temp < min_temp:
            break
            
        # Create neighbor solution via small perturbation
        neighbor = best_vals.copy()
        # Perturb a few random elements
        num_perturb = max(1, len(neighbor) // 50)  # About 2% of elements
        for _ in range(num_perturb):
            idx = random.randint(0, len(neighbor) - 1)
            # Add small random change
            delta = random.uniform(-0.5, 0.5)
            neighbor[idx] = max(0, neighbor[idx] + delta)
        
        # Evaluate neighbor
        neighbor_c2 = calculate_c2(neighbor)
        
        # Accept or reject based on temperature
        if neighbor_c2 > best_c2 or random.random() < np.exp((neighbor_c2 - best_c2) / temp):
            best_vals = neighbor
            best_c2 = neighbor_c2
            
        temp *= cooling_rate
    
    # Final normalization to ensure stability
    if np.sum(best_vals) > 0:
        best_vals = [val / np.sum(best_vals) * 100 for val in best_vals]
    
    return best_vals
