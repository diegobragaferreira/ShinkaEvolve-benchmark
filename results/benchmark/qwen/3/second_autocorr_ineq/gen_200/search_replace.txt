def construct_geometric_initial_function(n: int) -> List[float]:
    """
    Create initial function with multi-scale Gaussian patterns for better C2 values.
    This implements a more sophisticated approach inspired by multi-scale Gaussian constructions.
    """
    # Create a multi-scale Gaussian pattern that builds complexity gradually
    f_vals = np.zeros(n)

    # Generate multiple Gaussian bumps at different scales
    # This creates a more structured and mathematically well-behaved initial function

    # Scale 1: Large-scale smooth variation
    x = np.linspace(0, 1, n)
    scale1_amp = 0.6
    scale1_width = 0.3
    f_vals += scale1_amp * np.exp(-((x - 0.5)**2) / (2 * scale1_width**2))

    # Scale 2: Medium-scale structure
    scale2_amp = 0.3
    scale2_width = 0.1
    f_vals += scale2_amp * np.exp(-((x - 0.3)**2) / (2 * scale2_width**2)) + \
              scale2_amp * np.exp(-((x - 0.7)**2) / (2 * scale2_width**2))

    # Scale 3: Fine-scale detail
    scale3_amp = 0.1
    scale3_width = 0.05
    f_vals += scale3_amp * np.exp(-((x - 0.2)**2) / (2 * scale3_width**2)) + \
              scale3_amp * np.exp(-((x - 0.4)**2) / (2 * scale3_width**2)) + \
              scale3_amp * np.exp(-((x - 0.6)**2) / (2 * scale3_width**2)) + \
              scale3_amp * np.exp(-((x - 0.8)**2) / (2 * scale3_width**2))

    # Add some sinusoidal modulation for additional complexity
    f_vals += 0.1 * np.sin(12 * np.pi * x)

    # Ensure non-negative values and normalize
    f_vals = np.clip(f_vals, 0, None)

    # Normalize to reasonable range to avoid extreme values
    if np.max(f_vals) > 0:
        f_vals = f_vals / np.max(f_vals) * 1.0

    # Add small random noise to break symmetry and avoid local minima
    noise_amplitude = 0.02
    noise = np.random.normal(0, noise_amplitude, n)
    f_vals = np.clip(f_vals + noise, 0, None)

    # Convert to list and return
    return f_vals.tolist()
