@jit(nopython=True)
def compute_autoconvolution_norms_fast(f_values: np.ndarray) -> Tuple[float, float, float]:
    """
    Fast computation of autoconvolution norms using numba-compiled loop
    """
    n = len(f_values)
    # Pre-allocate convolution result
    g = np.zeros(2*n - 1)

    # Manual convolution computation for speed
    for i in range(n):
        for j in range(n):
            g[i + j] += f_values[i] * f_values[j]

    # Extract valid convolution (center part)
    center = len(g) // 2
    g_valid = g[center - n + 1:center + n]

    # Compute norms
    norm_l2_sq = 0.0
    norm_l1 = 0.0
    norm_linf = 0.0

    for val in g_valid:
        norm_l2_sq += val * val
        norm_l1 += abs(val)
        if abs(val) > norm_linf:
            norm_linf = abs(val)

    return norm_l2_sq, norm_l1, norm_linf

@jit(nopython=True)
def calculate_c2_fast(f_values: np.ndarray) -> float:
    """Fast calculation of C2 with numerical stability"""
    norm_l2_sq, norm_l1, norm_linf = compute_autoconvolution_norms_fast(f_values)

    # Avoid division by zero
    if norm_l1 <= 1e-12 or norm_linf <= 1e-12:
        return 0.0

    c2 = norm_l2_sq / (norm_l1 * norm_linf)
    return c2

def mutate_individual_numba(individual: List[float], mutation_rate: float,
                     mutation_strength: float = 0.1) -> List[float]:
    """Mutate an individual with given mutation rate and strength"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            # Add Gaussian noise with bounded result
            mutated[i] = max(0, mutated[i] + np.random.normal(0, mutation_strength))
    return mutated

@jit(nopython=True)
def crossover_numba(parent1: List[float], parent2: List[float],
             crossover_rate: float = 0.8) -> List[float]:
    """Single-point crossover between two parents"""
    if np.random.random() > crossover_rate:
        return parent1.copy()

    point = np.random.randint(1, len(parent1) - 1)
    child = parent1[:point] + parent2[point:]
    return child

<DESCRIPTION>
This change introduces adaptive mutation rates that decrease gradually throughout the evolutionary process. The idea is to explore broadly in early generations and exploit good solutions in later generations. The mutation rate starts at 0.15 and decreases to 0.05 over time, while mutation strength decreases from 0.15 to 0.05. This helps avoid getting stuck in local optima while focusing on improving already good solutions in later iterations.
</DESCRIPTION>

    # Decrease mutation rate from 0.15 to 0.05
    mutation_rate = 0.15 - (0.1 * (gen / max_generations))

improved_initialization_strategies
</NAME>

<DESCRIPTION>
This change adds improved initialization strategies that create more favorable starting conditions for the evolutionary algorithm. Instead of just random initialization, we now include:
1. A "flat_top" strategy that creates a nearly flat profile with a small central peak
2. A "double_peak" strategy that creates two prominent peaks separated by a valley
3. An "alternating" strategy that creates a pattern of high-low-high-low values

These strategies are designed to produce step functions that are likely to generate better convolution profiles early in the evolution, potentially leading to faster convergence to higher C2 values.
</DESCRIPTION>

def generate_individual(size: int, method: str = 'uniform') -> List[float]:
    """Generate a single individual with specified initialization method"""
    if method == 'uniform':
        return [random.uniform(0, 1) for _ in range(size)]
    elif method == 'gaussian':
        return [max(0, random.gauss(0.5, 0.3)) for _ in range(size)]
    elif method == 'spike':
        # Create a spike-like pattern
        individual = [0.0] * size
        mid = size // 2
        individual[mid] = random.uniform(0.5, 1.0)
        return individual
    elif method == 'flat_top':
        # Create a flat profile with a central peak
        individual = [0.3] * size
        center = size // 2
        individual[center] = random.uniform(0.7, 1.0)
        return individual
    elif method == 'double_peak':
        # Create two peaks with a valley in between
        individual = [0.0] * size
        left_peak = size // 3
        right_peak = 2 * size // 3
        individual[left_peak] = random.uniform(0.7, 1.0)
        individual[right_peak] = random.uniform(0.7, 1.0)
        return individual
    elif method == 'alternating':
        # Create alternating high/low pattern
        individual = []
        for i in range(size):
            if i % 2 == 0:
                individual.append(random.uniform(0.6, 1.0))
            else:
                individual.append(random.uniform(0.0, 0.3))
        return individual
    else:
        return [random.uniform(0, 1) for _ in range(size)]
