def initialize_smart_parameters(size: int) -> np.ndarray:
    """
    Initialize step function parameters with multi-scale Gaussian pattern.
    This creates a hierarchical structure with varying scales for better exploration.
    """
    # Create multi-scale Gaussian pattern for structured initialization
    pattern = np.zeros(size)

    # Define multiple scales for Gaussian bumps (from coarse to fine)
    scales = [size/8, size/16, size/32, size/64]  # Different scale factors
    amplitudes = [1.0, 0.7, 0.4, 0.2]  # Decreasing amplitudes for hierarchy

    # Place Gaussian bumps at strategic positions
    for i, (scale, amp) in enumerate(zip(scales, amplitudes)):
        if scale > 1:  # Only create bumps with meaningful scale
            # Position bumps in a structured pattern across the domain
            position = size // 4 + (i * size // 8) % (size // 2) if i < 4 else size // 2
            # Generate Gaussian bump
            indices = np.arange(size)
            gaussian = amp * np.exp(-0.5 * (indices - position)**2 / scale**2)
            pattern += gaussian

    # Add some harmonic structure for complexity
    t = np.linspace(-2, 2, size)
    harmonic_pattern = (
        0.8 +
        0.3 * np.sin(2 * np.pi * t) +
        0.2 * np.cos(4 * np.pi * t) +
        0.1 * np.sin(6 * np.pi * t)
    )

    # Combine all components and ensure non-negativity
    pattern = np.maximum(pattern + harmonic_pattern, 0.0)

    # Normalize to reasonable magnitude
    if np.sum(pattern) > 0:
        pattern = pattern / np.sum(pattern) * 100

    return pattern
