@njit
def compute_convolution_norms_numba(f_values, domain_length=0.5):
    """
    Fast computation of autoconvolution norms using numba-compiled loop
    """
    n_steps = len(f_values)
    if n_steps == 0:
        return 0.0, 0.0, 0.0

    # Step size
    dx = domain_length / n_steps

    # Manual convolution computation for speed
    g = np.zeros(2*n_steps - 1)
    for i in range(n_steps):
        for j in range(n_steps):
            g[i + j] += f_values[i] * f_values[j]

    # Extract valid convolution (center part)
    center = len(g) // 2
    g_valid = g[center - n_steps + 1:center + n_steps]

    # Compute norms using piecewise linear integration
    # For ||g||₂²: use trapezoidal-like formula
    g2_sq = 0.0
    for i in range(len(g_valid)-1):
        g2_sq += (dx/3) * (g_valid[i]**2 + g_valid[i]*g_valid[i+1] + g_valid[i+1]**2)

    # ||g||₁ = sum(|g_i| * dx)
    g1 = np.sum(np.abs(g_valid)) * dx

    # ||g||∞ = max(|g_i|)
    ginf = np.max(np.abs(g_valid))

    return g2_sq, g1, ginf

@njit
def compute_c2_numba(f_values):
    """Fast calculation of C2 with numerical stability"""
    g2_sq, g1, ginf = compute_convolution_norms_numba(f_values)

    # Avoid division by zero
    if g1 <= 1e-12 or ginf <= 1e-12:
        return 0.0

    c2 = g2_sq / (g1 * ginf)
    return c2

@njit
def compute_convolution_norms_numba(f_values, domain_length=0.5):
    """
    Fast computation of autoconvolution norms using numba-compiled loop
    Uses FFT for larger arrays to improve performance
    """
    n_steps = len(f_values)
    if n_steps == 0:
        return 0.0, 0.0, 0.0

    # Step size
    dx = domain_length / n_steps

    # For small arrays, use direct convolution - for larger arrays, use FFT
    if n_steps < 100:
        # Manual convolution computation for speed
        g = np.zeros(2*n_steps - 1)
        for i in range(n_steps):
            for j in range(n_steps):
                g[i + j] += f_values[i] * f_values[j]

        # Extract valid convolution (center part)
        center = len(g) // 2
        g_valid = g[center - n_steps + 1:center + n_steps]
    else:
        # For larger arrays, use FFT-based convolution for efficiency
        # Pad to appropriate size for FFT
        padded_size = 2 * n_steps - 1
        f_padded = np.zeros(padded_size)
        f_padded[:n_steps] = f_values

        # FFT-based convolution: g = f * f
        f_fft = np.fft.fft(f_padded)
        g_fft = f_fft * f_fft.conj()  # Pointwise multiplication in frequency domain
        g = np.fft.ifft(g_fft).real  # Inverse FFT to get convolution result

        # Extract the valid convolution part (center portion)
        g = g[n_steps-1:2*n_steps-1]

    # Compute norms using piecewise linear integration
    # For ||g||₂²: use trapezoidal-like formula
    g2_sq = 0.0
    for i in range(len(g)-1):
        g2_sq += (dx/3) * (g[i]**2 + g[i]*g[i+1] + g[i+1]**2)

    # ||g||₁ = sum(|g_i| * dx)
    g1 = np.sum(np.abs(g)) * dx

    # ||g||∞ = max(|g_i|)
    ginf = np.max(np.abs(g))

    return g2_sq, g1, ginf

    if g1 <= 1e-12 or ginf <= 1e-12:

<DESCRIPTION>
I'm introducing a hybrid optimization approach that combines the strengths of evolutionary algorithms with local optimization techniques. The idea is to run the evolutionary algorithm first to find promising regions of the search space, then apply a more precise local optimization (using scipy's differential evolution) to refine the best solutions found. This approach leverages the global exploration power of EAs with the fine-tuning capabilities of local optimizers, which should help improve the final C2 score.

The implementation adds a new hybrid optimization function that takes the best solution from the evolutionary algorithm and applies differential evolution to it for further improvement.

def adaptive_evolutionary_optimization(n_steps: int = 1000):
    """Main optimization routine using adaptive evolutionary strategy"""

    # Generate diverse initial population
    pop_size = 50
    population = []
    for _ in range(pop_size):
        # Mix of multi-scale Gaussian and random patterns
        if np.random.random() < 0.7:
            individual = construct_multiscale_gaussian_initial_function(n_steps)
        else:
            individual = np.random.random(n_steps).tolist()

        # Ensure non-negativity and reasonable scaling
        individual = [max(0, x) for x in individual]
        # Normalize to avoid extremely large peaks
        total = sum(individual)
        if total > 0:
            individual = [x / total * 10 for x in individual]

        population.append(individual)

    # Track best solution
    best_individual = None
    best_c2 = -1
    max_generations = 25  # Limit to keep within time constraints

    # Adaptive parameters
    mutation_rate = 0.3
    crossover_rate = 0.8
    elite_count = 5

    # Evolution loop
    for generation in range(max_generations):
        # Evaluate population
        fitness_scores = []
        for individual in population:
            c2 = calculate_c2(individual)
            fitness_scores.append(c2)
            if c2 > best_c2:
                best_c2 = c2
                best_individual = individual.copy()

        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]

        # Create new population
        new_population = []

        # Elitism: keep best individuals
        for i in range(elite_count):
            new_population.append(sorted_population[i].copy())

        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection for parents
            parent1 = tournament_selection(sorted_population, sorted_fitness, 3)
            parent2 = tournament_selection(sorted_population, sorted_fitness, 3)

            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = uniform_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            # Mutation
            mutate_individual(child1, mutation_rate)
            mutate_individual(child2, mutation_rate)

            new_population.extend([child1, child2])

        # Trim to exact population size
        population = new_population[:pop_size]

        # Adapt mutation rate: decrease over time to fine-tune
        mutation_rate = max(0.05, mutation_rate * 0.95)

    return best_individual

def hybrid_optimization(initial_solution: List[float], max_evaluations: int = 300) -> List[float]:
    """
    Hybrid optimization that refines a solution using differential evolution
    """
    def objective(x):
        # Clip negative values to prevent numerical issues
        x = np.maximum(x, 0)
        # Calculate C2 (we want to maximize it, so minimize negative)
        c2_value = calculate_c2(x.tolist())
        # Return negative because differential evolution minimizes
        return -c2_value if c2_value > 0 else 1e10

    try:
        # Use differential evolution for local refinement
        result = differential_evolution(
            objective,
            bounds=[(0, 100) for _ in range(len(initial_solution))],
            maxiter=max_evaluations // 10,
            popsize=15,
            seed=42,
            disp=False,
            polish=True
        )
        refined_solution = result.x
    except:
        # If optimization fails, return the original solution
        refined_solution = np.array(initial_solution)

    # Ensure non-negativity and clip extreme values
    refined_solution = np.clip(refined_solution, 0, 100)

    return refined_solution.tolist()

def construct_function() -> List[float]:
    """Function to construct step-function with high C2 value."""
    np.random.seed(42)  # For reproducibility

    start_time = time.time()

    # Try evolutionary optimization first with higher resolution
    try:
        result_evo = adaptive_evolutionary_optimization(1500)
        elapsed = time.time() - start_time
        if elapsed < 80:  # Leave margin for refinement
            # Apply hybrid refinement
            try:
                result_refined = hybrid_optimization(result_evo, max_evaluations=300)
                elapsed = time.time() - start_time
                if elapsed < 85:
                    return result_refined
            except Exception as e:
                print(f"Hybrid refinement failed: {e}")
                pass
            return result_evo
    except Exception as e:
        print(f"Evolutionary optimization failed: {e}")
