class PopulationManager:
    """Handles population creation, selection, and evolution"""

    @staticmethod
    def generate_geometric_initial_function(size: int) -> List[float]:
        """
        Generate an initial function using geometric patterns that encourage good C2 values.
        Creates a function with multiple scales and controlled variations to promote flat convolution.
        """
        # Create a multi-scale geometric pattern
        f_values = np.zeros(size)

        # Scale 1: Main body with smooth transitions
        x = np.linspace(0, 1, size)

        # Create a bell-shaped pattern with multiple peaks to encourage uniform convolution
        # This pattern helps avoid sharp peaks that would decrease C2
        pattern1 = 0.3 * np.exp(-((x - 0.3)**2) / (2 * 0.1**2)) + \
                   0.4 * np.exp(-((x - 0.7)**2) / (2 * 0.1**2)) + \
                   0.3 * np.exp(-((x - 0.5)**2) / (2 * 0.15**2))

        # Add some sinusoidal component for additional variation
        pattern2 = 0.1 * np.sin(6 * np.pi * x) + 0.05 * np.cos(12 * np.pi * x)

        # Combine patterns
        combined = pattern1 + pattern2

        # Normalize and clip to [0, 1]
        combined = np.clip(combined, 0, 1)

        # Ensure sum is reasonable to prevent extreme values
        f_values = combined

        return f_values.tolist()

    def generate_individual(size: int) -> List[float]:
