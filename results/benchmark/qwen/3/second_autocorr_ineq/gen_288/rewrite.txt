# Core computational functions with numba acceleration
@jit(nopython=True)
def compute_autoconvolution_fast(f_vals):
    """Fast numba-based autoconvolution computation"""
    n = len(f_vals)
    g = np.zeros(2*n - 1)
    
    for i in range(n):
        for j in range(n):
            g[i + j] += f_vals[i] * f_vals[j]
    
    return g

@jit(nopython=True) 
def compute_convolution_norms_fast(g_vals):
    """Fast computation of convolution norms"""
    n = len(g_vals)
    l2_sq = 0.0
    l1 = 0.0
    l_inf = 0.0
    
    for i in range(n):
        val = g_vals[i]
        l2_sq += val * val
        l1 += abs(val)
        if abs(val) > l_inf:
            l_inf = abs(val)
    
    return l2_sq, l1, l_inf

def compute_autoconvolution_norms(f_values: List[float]) -> Tuple[float, float, float]:
    """
    Compute the three norms needed for C2 calculation with proper numerical handling
    """
    n = len(f_values)
    if n == 0:
        return 0.0, 0.0, 0.0
    
    # Ensure non-negative values
    f_array = np.array(f_values)
    f_array = np.maximum(f_array, 0.0)
    
    # Compute autoconvolution
    g_vals = compute_autoconvolution_fast(f_array)
    
    # Compute norms
    l2_sq, l1, l_inf = compute_convolution_norms_fast(g_vals)
    
    # Handle edge cases
    if l1 <= 1e-15:
        l1 = 1e-15
    if l_inf <= 1e-15:
        l_inf = 1e-15
        
    return l2_sq, l1, l_inf

def calculate_c2(f_values: List[float]) -> float:
    """
    Calculate the C2 constant from the step function values.
    """
    try:
        g_norm_2_sq, g_norm_1, g_norm_inf = compute_autoconvolution_norms(f_values)
        c2 = g_norm_2_sq / (g_norm_1 * g_norm_inf)
        return c2
    except:
        return 0.0

class FrequencyGuidedEvolution:
    """
    A novel evolutionary algorithm that operates directly in frequency space
    to optimize step functions for maximum C2 value.
    """
    
    def __init__(self):
        self.n_samples = 200  # Number of samples for frequency representation
        self.max_generations = 40
        self.population_size = 30
        
    def generate_spectral_individual(self, n_coeffs: int) -> np.ndarray:
        """
        Generate a random individual in spectral domain
        """
        # Create random spectral coefficients with controlled magnitude
        coeffs = np.random.normal(0, 0.5, n_coeffs) + 1j * np.random.normal(0, 0.5, n_coeffs)
        
        # Apply some frequency filtering to encourage desirable convolution properties
        # Low frequencies dominate to promote smooth convolution
        freqs = fftfreq(n_coeffs, d=1.0/n_coeffs)
        filter_weights = np.exp(-0.5 * (freqs / (n_coeffs/10))**2)
        coeffs *= filter_weights
        
        return coeffs
    
    def spectral_to_spatial(self, spectral_coeffs: np.ndarray, n_points: int) -> np.ndarray:
        """
        Convert spectral representation to spatial domain (step function)
        """
        # Use inverse FFT to get real-valued spatial function
        spatial = np.real(ifft(spectral_coeffs, n=n_points))
        
        # Ensure non-negative values (clip negative parts)
        spatial = np.maximum(spatial, 0.0)
        
        # Normalize to reasonable scale
        if np.max(spatial) > 1e-10:
            spatial = spatial / np.max(spatial) * 2.0
            
        return spatial
    
    def evaluate_spectral_individual(self, spectral_coeffs: np.ndarray) -> Tuple[float, float, float]:
        """
        Evaluate a spectral individual by converting to spatial and computing norms
        """
        spatial = self.spectral_to_spatial(spectral_coeffs, self.n_samples)
        return compute_autoconvolution_norms(spatial.tolist())
    
    def compute_spectral_fitness(self, spectral_coeffs: np.ndarray) -> float:
        """
        Compute fitness from spectral representation for C2 maximization
        """
        try:
            g_norm_2_sq, g_norm_1, g_norm_inf = self.evaluate_spectral_individual(spectral_coeffs)
            if g_norm_1 <= 1e-15 or g_norm_inf <= 1e-15:
                return 0.0
            c2 = g_norm_2_sq / (g_norm_1 * g_norm_inf)
            return c2
        except:
            return 0.0
    
    def mutate_spectral(self, spectral_coeffs: np.ndarray, generation: int) -> np.ndarray:
        """
        Mutate spectral coefficients with adaptive parameters
        """
        mutated = spectral_coeffs.copy()
        n_coeffs = len(mutated)
        
        # Adaptive mutation rate
        mutation_rate = 0.2 * (1.0 - generation/self.max_generations)
        
        # Mutate each coefficient
        for i in range(n_coeffs):
            if random.random() < mutation_rate:
                # Apply mutation to both real and imaginary parts
                magnitude_change = random.gauss(0, 0.3)
                angle_change = random.gauss(0, 0.2)
                
                # Apply changes
                current_real = mutated[i].real
                current_imag = mutated[i].imag
                
                new_real = current_real + magnitude_change * (1.0 + 0.5 * random.random())
                new_imag = current_imag + magnitude_change * (1.0 + 0.5 * random.random())
                
                # Keep some structure by applying angle rotation
                new_magnitude = np.sqrt(new_real**2 + new_imag**2)
                new_angle = np.arctan2(new_imag, new_real) + angle_change
                
                mutated[i] = new_magnitude * np.cos(new_angle) + 1j * new_magnitude * np.sin(new_angle)
        
        return mutated
    
    def crossover_spectral(self, parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
        """
        Perform crossover on spectral coefficients
        """
        child = parent1.copy()
        n_coeffs = len(child)
        
        # Uniform crossover
        for i in range(n_coeffs):
            if random.random() < 0.5:
                child[i] = parent2[i]
                
        return child
    
    def evolve(self) -> Tuple[np.ndarray, float]:
        """
        Run the frequency-guided evolution process
        """
        # Initialize population
        population = []
        fitness_scores = []
        
        for i in range(self.population_size):
            # Generate initial spectral individuals
            spectral_ind = self.generate_spectral_individual(self.n_samples)
            population.append(spectral_ind)
            fitness = self.compute_spectral_fitness(spectral_ind)
            fitness_scores.append(fitness)
        
        # Track best solution
        best_fitness = max(fitness_scores)
        best_spectral = population[fitness_scores.index(best_fitness)].copy()
        
        # Evolution loop
        for gen in range(self.max_generations):
            # Sort by fitness (descending)
            sorted_indices = sorted(range(len(fitness_scores)), 
                                  key=lambda i: fitness_scores[i], reverse=True)
            sorted_population = [population[i] for i in sorted_indices]
            sorted_fitness = [fitness_scores[i] for i in sorted_indices]
            
            # Elitism: keep top 2
            elite = sorted_population[:2]
            
            # Generate new population
            new_population = elite.copy()
            
            # Create offspring through crossover and mutation
            while len(new_population) < self.population_size:
                # Selection: tournament
                tournament_size = 3
                tournament_indices = random.sample(range(len(sorted_population)), tournament_size)
                tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
                parent_idx = tournament_indices[np.argmax(tournament_fitness)]
                parent1 = sorted_population[parent_idx].copy()
                
                # Another parent
                parent2 = random.choice(sorted_population)
                
                # Crossover
                child = self.crossover_spectral(parent1, parent2)
                
                # Mutation
                child = self.mutate_spectral(child, gen)
                
                new_population.append(child)
            
            population = new_population[:self.population_size]
            
            # Evaluate new population
            fitness_scores = [self.compute_spectral_fitness(ind) for ind in population]
            
            # Update best
            current_max_fitness = max(fitness_scores)
            if current_max_fitness > best_fitness:
                best_fitness = current_max_fitness
                best_spectral = population[fitness_scores.index(best_fitness)].copy()
        
        return best_spectral, best_fitness
    
    def refine_solution(self, best_spectral: np.ndarray) -> Tuple[List[float], float]:
        """
        Refine the best solution using coordinate-wise local search
        """
        # Convert to spatial domain
        spatial = self.spectral_to_spatial(best_spectral, self.n_samples)
        
        # Local refinement through coordinate-wise improvements
        old_c2 = calculate_c2(spatial.tolist())
        refined_spatial = spatial.copy()
        
        # Try small adjustments to each position
        for i in range(self.n_samples):
            old_val = refined_spatial[i]
            
            # Try a few different step sizes and directions
            step_sizes = [0.01, 0.05, 0.1]
            for step in step_sizes:
                for direction in [1, -1]:
                    test_spatial = refined_spatial.copy()
                    new_val = old_val + direction * step
                    test_spatial[i] = max(0, new_val)
                    new_c2 = calculate_c2(test_spatial.tolist())
                    
                    if new_c2 > old_c2:
                        refined_spatial = test_spatial
                        old_c2 = new_c2
        
        return refined_spatial.tolist(), old_c2

def construct_function() -> List[float]:
    """
    Function to construct step-function with high C2 value using frequency-guided evolution.
    
    This approach differs fundamentally from previous methods by:
    1. Operating in frequency domain rather than spatial domain
    2. Evolving spectral coefficients that directly influence convolution properties
    3. Using mathematical insight to guide optimization toward favorable autoconvolution characteristics
    """
    start_time = time.time()
    
    # Set seeds for reproducibility
    random.seed(42)
    np.random.seed(42)
    
    try:
        # Initialize the frequency-guided evolutionary optimizer
        optimizer = FrequencyGuidedEvolution()
        
        # Run evolution
        best_spectral, best_fitness = optimizer.evolve()
        
        # Refine the solution
        final_solution, final_c2 = optimizer.refine_solution(best_spectral)
        
        # Ensure we have a reasonable solution
        if final_c2 < 0.1:
            # Fallback to a simple geometric pattern if optimization failed
            n = 200
            x = np.linspace(0, 1, n)
            pattern = 0.5 + 0.3 * np.sin(4 * np.pi * x)
            pattern = np.maximum(pattern, 0)
            final_solution = pattern.tolist()
            final_c2 = calculate_c2(final_solution)
        
        end_time = time.time()
        
        return final_solution
        
    except Exception as e:
        # Final fallback to geometric pattern
        print(f"Fallback due to error: {e}")
        n = 200
        x = np.linspace(0, 1, n)
        pattern = 0.5 + 0.3 * np.sin(4 * np.pi * x)
        pattern = np.maximum(pattern, 0)
        return pattern.tolist()
