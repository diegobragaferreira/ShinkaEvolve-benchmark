    def create_lhs_individual():
        """Create individual using Latin Hypercube Sampling for better space coverage."""
        n_steps = random.randint(150, 800)

        # Generate Latin Hypercube Sample for each dimension (step)
        individual = []
        for i in range(n_steps):
            # Create uniform random sample in [0,1] and map to appropriate range
            # Using inverse transform sampling to get values that are more evenly distributed
            lhs_sample = (i + random.random()) / n_steps  # LHS value
            # Map to range [0.1, 1.0] with some curvature to encourage variety
            # This creates a more structured but diverse pattern
            value = 0.1 + 0.9 * (lhs_sample ** 0.7)  # Slightly skewed distribution
            individual.append(value)

        return creator.Individual(individual)
