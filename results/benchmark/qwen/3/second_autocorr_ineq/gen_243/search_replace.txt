This change aims to improve the computational efficiency of the core convolution operation, allowing more iterations or larger problem sizes within the time constraint.

def compute_autoconvolution_fft(f_vals):
    """Use FFT-based convolution for improved performance on large arrays"""
    # Convert to numpy array
    f = np.array(f_vals)

    # For efficiency, use scipy's fftconvolve for large arrays
    # When array size is large enough, FFT convolution is much faster
    if len(f) > 1000:
        # Use FFT-based convolution
        g = signal.fftconvolve(f, f, mode='full')

        # Keep only the central portion (valid convolution)
        center_idx = len(g) // 2
        half_len = len(f)
        g = g[center_idx - half_len + 1:center_idx + half_len]
    else:
        # For smaller arrays, use the original method for better precision
        g = signal.convolve(f, f, mode='full')
        center_idx = len(g) // 2
        half_len = len(f)
        g = g[center_idx - half_len + 1:center_idx + half_len]

    return g
