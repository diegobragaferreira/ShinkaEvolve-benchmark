                          use_differential_evolution: bool = True) -> List[float]:

    try:
        if not use_differential_evolution:
            # Enhanced gradient-based local search with better finite difference estimation
            solution_array = np.array(solution)
            best_solution = solution_array.copy()
            best_c2 = calculate_c2(best_solution.tolist())

            # Gradient estimation using central differences for better accuracy
            max_iterations = min(max_evals, 50)  # Limit iterations for performance

            for iteration in range(max_iterations):
                # Estimate gradient using central differences with adaptive epsilon
                gradient = np.zeros_like(best_solution)
                epsilon_base = 1e-4

                for i in range(len(best_solution)):
                    # Perturb forward and backward
                    eps = epsilon_base * max(1e-6, best_solution[i])
                    if eps == 0:
                        eps = 1e-6

                    # Forward difference
                    f_plus = best_solution.copy()
                    f_plus[i] = max(0, best_solution[i] + eps)
                    c2_plus = calculate_c2(f_plus.tolist())

                    # Backward difference
                    f_minus = best_solution.copy()
                    f_minus[i] = max(0, best_solution[i] - eps)
                    c2_minus = calculate_c2(f_minus.tolist())

                    gradient[i] = (c2_plus - c2_minus) / (2 * eps)

                # Adaptive step size based on gradient magnitude
                grad_norm = np.linalg.norm(gradient)
                if grad_norm > 1e-12:  # Avoid division by near-zero
