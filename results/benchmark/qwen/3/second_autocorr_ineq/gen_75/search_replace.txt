<DESCRIPTION>
Adding Numba JIT compilation to critical functions will significantly improve computational performance by compiling Python code to machine code at runtime. This is particularly important for the autoconvolution computation and fitness calculation which are called repeatedly during evolution. The key functions to compile are the autoconvolution norm computations and the C2 calculation, which currently involve nested loops and repeated array operations that benefit greatly from JIT compilation.

class AutoconvolutionResult:

class StepFunction:
    """Represents a step function with optimized operations."""

    def __init__(self, heights: List[float]):
        self.heights = np.array(heights, dtype=np.float64)
        self.length = len(heights)

    def get_heights(self) -> np.ndarray:
        return self.heights.copy()

    def get_length(self) -> int:
        return self.length

class AutoconvolutionEvaluator:
    """Handles computation of autoconvolution norms with optimized numerical methods."""

    @staticmethod
    def compute_autoconvolution_norms(step_func: StepFunction) -> AutoconvolutionResult:
        """
        Compute the L2, L1, and L-infinity norms of the autoconvolution of f.

        Args:
            step_func: StepFunction object containing step heights

        Returns:
            AutoconvolutionResult containing the three norms
        """
        heights = step_func.get_heights()
        length = step_func.get_length()

        if length == 0:
            return AutoconvolutionResult(0.0, 0.0, 0.0)

        # Compute autoconvolution g = f * f (discrete convolution)
        # Using numpy's convolution - it computes cross-correlation with flipped kernel
        g = np.convolve(heights, heights, mode='full')

        # Compute norms using optimized piecewise integration
        # ||g||₂² - integrate g² using trapezoidal rule approximation
        g_squared = g * g
        norm2_sq = AutoconvolutionEvaluator._compute_trapz_norm2(g_squared)

        # ||g||₁ - integrate |g| using trapezoidal rule
        g_abs = np.abs(g)
        norm1 = AutoconvolutionEvaluator._compute_trapz_norm1(g_abs)

        # ||g||∞ - infinity norm (maximum absolute value)
        norm_inf = np.max(np.abs(g)) if len(g) > 0 else 0.0

        return AutoconvolutionResult(norm2_sq, norm1, norm_inf)

    @staticmethod
    def _compute_trapz_norm2(g_values: np.ndarray) -> float:
        """Compute L2 norm squared using piecewise linear integration."""
        if len(g_values) < 2:
            return g_values[0] * g_values[0] if len(g_values) > 0 else 0.0

        # Simplified trapezoidal integration for g²
        h = 1.0 / (len(g_values) - 1)  # Normalized spacing
        total = 0.0

        # For piecewise linear integration of g², use (y₁² + y₁y₂ + y₂²)/3 * h per segment
        for i in range(len(g_values) - 1):
            y1, y2 = g_values[i], g_values[i+1]
            total += (y1*y1 + y1*y2 + y2*y2) * h / 3.0

        return total

    @staticmethod
    def _compute_trapz_norm1(g_values: np.ndarray) -> float:
        """Compute L1 norm using trapezoidal rule."""
        if len(g_values) < 2:
            return g_values[0] if len(g_values) > 0 else 0.0

        h = 1.0 / (len(g_values) - 1)  # Normalized spacing
        total = 0.0

        # Trapezoidal rule for ∫|g|
        for i in range(len(g_values) - 1):
            y1, y2 = g_values[i], g_values[i+1]
            total += (y1 + y2) * h / 2.0

        return total

class FitnessCalculator:
    """Computes fitness (C2 values) with proper error handling."""

    @staticmethod

            C₂ value

            return 0.0

import numpy as np
from typing import List, Tuple, Optional
import time
from dataclasses import dataclass
from functools import lru_cache
from numba import jit, njit

@njit
def compute_autoconvolution_norms_numba(heights: np.ndarray) -> Tuple[float, float, float]:
    """
    Compute the L2, L1, and L-infinity norms of the autoconvolution of f.
    Numba JIT compiled version for performance.

    Args:
        heights: Array of step heights

    Returns:
        Tuple of (||g||₂², ||g||₁, ||g||∞)
    """
    if len(heights) == 0:
        return 0.0, 0.0, 0.0

    # Compute autoconvolution g = f * f (discrete convolution)
    # Manual implementation for better control and JIT compatibility
    g_len = 2 * len(heights) - 1
    g = np.zeros(g_len, dtype=np.float64)

    # Direct convolution computation
    for i in range(len(heights)):
        for j in range(len(heights)):
            g[i + j] += heights[i] * heights[j]

    # Compute norms using manual loop for JIT compatibility
    # ||g||₂² - integrate g² using trapezoidal rule approximation
    g_squared = g * g
    trapz_sum = 0.0

    # Use trapezoidal integration for ||g||₂²
    if len(g) >= 2:
        h = 1.0 / (len(g) - 1)  # Normalized spacing
        for i in range(len(g) - 1):
            y1, y2 = g_squared[i], g_squared[i+1]
            trapz_sum += (y1 + y1*y2 + y2) * h / 3.0
    else:
        trapz_sum = g_squared[0] if len(g_squared) > 0 else 0.0

    # ||g||₁ - integrate |g| using trapezoidal rule
    g_abs = np.abs(g)
    trapz_l1_sum = 0.0

    if len(g) >= 2:
        h = 1.0 / (len(g) - 1)  # Normalized spacing
        for i in range(len(g) - 1):
            y1, y2 = g_abs[i], g_abs[i+1]
            trapz_l1_sum += (y1 + y2) * h / 2.0
    else:
        trapz_l1_sum = g_abs[0] if len(g_abs) > 0 else 0.0

    # ||g||∞ - infinity norm (maximum absolute value)
    g_max = np.max(np.abs(g)) if len(g) > 0 else 0.0

    return trapz_sum, trapz_l1_sum, g_max
