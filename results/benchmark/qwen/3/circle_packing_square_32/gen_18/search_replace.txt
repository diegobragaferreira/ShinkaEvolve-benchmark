import numpy as np

# You can define functions outside the main function below.
# Remember that any function used in parallel computation must be defined globally and not locally.

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 32
    circles = np.zeros((n, 3))

    # Hexagonal grid initialization
    # Arrange circles in a hexagonal pattern to ensure they don't overlap initially
    rows = 6
    cols = 6
    spacing_x = 1.0 / cols
    spacing_y = 1.0 / rows
    radius = min(spacing_x, spacing_y) * 0.4  # Initial radius slightly less than spacing

    # Create hexagonal grid
    circle_count = 0
    for i in range(rows):
        for j in range(cols):
            if circle_count >= n:
                break
            x = (j + 0.5) * spacing_x
            y = (i + 0.5) * spacing_y

            # Offset every other row for hexagonal pattern
            if i % 2 == 1:
                x += spacing_x * 0.5

            # Ensure circle stays within unit square
            x = max(radius, min(1-radius, x))
            y = max(radius, min(1-radius, y))

            circles[circle_count] = [x, y, radius]
            circle_count += 1

            if circle_count >= n:
                break

    # Fill remaining circles with smaller radii if needed
    for i in range(circle_count, n):
        circles[i] = [0.5, 0.5, 0.01]

    return circles
