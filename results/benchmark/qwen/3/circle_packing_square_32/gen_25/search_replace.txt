import numpy as np

# You can define functions outside the main function below.
# Remember that any function used in parallel computation must be defined globally and not locally.

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 32
    circles = np.zeros((n, 3))

    # Hexagonal grid initialization
    # Determine grid dimensions to fit 32 circles
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Create hexagonal grid
    # For hexagonal packing, horizontal spacing is 2*r, vertical spacing is sqrt(3)*r
    # We'll start with a reasonable initial radius and adjust
    hex_radius = 0.05  # Initial guess

    # Generate hexagonal grid points
    grid_points = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5 * (i % 2)) * 2 * hex_radius
            y = i * np.sqrt(3) * hex_radius
            if x <= 1 - hex_radius and y <= 1 - hex_radius:
                grid_points.append([x, y])
            if len(grid_points) >= n:
                break
        if len(grid_points) >= n:
            break

    # Adjust number of points to exactly 32
    while len(grid_points) < n:
        # Add more points in the corners if needed
        grid_points.append([0.1 + len(grid_points) * 0.05, 0.1])

    # Fill the circles array with positions and initial radii
    for i in range(min(len(grid_points), n)):
        circles[i, 0] = grid_points[i][0]  # x coordinate
        circles[i, 1] = grid_points[i][1]  # y coordinate
        circles[i, 2] = hex_radius         # initial radius

    # Ensure all circles fit within unit square
    for i in range(n):
        # Adjust for boundary constraints
        circles[i, 0] = max(circles[i, 2], min(1 - circles[i, 2], circles[i, 0]))
        circles[i, 1] = max(circles[i, 2], min(1 - circles[i, 2], circles[i, 1]))

    return circles
