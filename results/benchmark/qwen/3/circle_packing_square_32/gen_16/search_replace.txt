import numpy as np

# You can define functions outside the main function below.
# Remember that any function used in parallel computation must be defined globally and not locally.

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 32
    circles = np.zeros((n, 3))

    # Hexagonal grid initialization
    # Calculate grid parameters for 32 circles
    rows = 6
    cols = 6
    grid_size = min(rows, cols)

    # Create hexagonal grid with proper spacing
    hex_radius = 0.15  # Initial guess for radius

    # Generate hexagonal grid points
    y_positions = []
    x_positions = []

    # Hexagonal arrangement with offset rows
    for i in range(rows):
        y = hex_radius + i * hex_radius * 2 * 0.866  # sqrt(3)/2 spacing
        if y + hex_radius > 1.0:
            break
        for j in range(cols):
            if i % 2 == 0:
                x = hex_radius + j * hex_radius * 2
            else:
                x = hex_radius + j * hex_radius * 2 + hex_radius
            if x + hex_radius <= 1.0:
                x_positions.append(x)
                y_positions.append(y)

    # Take first 32 points or fill remaining with random
    actual_count = min(len(x_positions), n)
    for i in range(actual_count):
        circles[i, 0] = x_positions[i]
        circles[i, 1] = y_positions[i]
        circles[i, 2] = hex_radius

    # Fill remaining circles with random positions (smaller radius)
    np.random.seed(42)  # For reproducibility
    for i in range(actual_count, n):
        circles[i, 0] = np.random.uniform(hex_radius, 1 - hex_radius)
        circles[i, 1] = np.random.uniform(hex_radius, 1 - hex_radius)
        circles[i, 2] = hex_radius * 0.5

    return circles
