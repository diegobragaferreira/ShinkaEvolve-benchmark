    # Hexagonal lattice initialization
    def initialize_hexagonal_lattice(pop_size):
        population = []
        for _ in range(pop_size):
            # Calculate hexagonal grid parameters
            rows = 6
            cols = 6
            if rows * cols < n_circles:
                rows = 5
                cols = 7
            if rows * cols < n_circles:
                rows = 4
                cols = 8

            # Create hexagonal grid
            circles = []
            hex_radius = 0.1  # Initial guess for hexagonal packing

            # For simplicity, we'll create a rectangular grid with slight offset to simulate hexagonal packing
            row_spacing = 1.0 / rows
            col_spacing = 1.0 / cols

            for i in range(rows):
                for j in range(cols):
                    if len(circles) >= n_circles:
                        break
                    # Offset every other row to create hexagonal pattern
                    offset = (i % 2) * col_spacing / 2
                    x = offset + (j + 0.5) * col_spacing
                    y = (i + 0.5) * row_spacing
                    # Adjust radius to fit within bounds
                    max_radius = min(x, 1-x, y, 1-y)
                    r = min(hex_radius, max_radius * 0.9)  # Leave some padding
                    if r > 0.001:
                        circles.append((x, y, r))

            # If we don't have enough circles, fill with random ones
            while len(circles) < n_circles:
                x = random.uniform(0.05, 0.95)
                y = random.uniform(0.05, 0.95)
                r = min(x, 1-x, y, 1-y) * 0.4
                if r > 0.001:
                    circles.append((x, y, r))

            # Trim to exactly n_circles
            circles = circles[:n_circles]

            population.append(creator.Individual([item for circle in circles for item in circle]))
        return population
