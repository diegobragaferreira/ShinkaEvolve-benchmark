    """Initialize population with well-distributed circle configurations using Voronoi + Poisson hybrid"""
    population = []

    # Use hybrid approach: Voronoi-inspired initial points with Poisson refinement
    # First, generate points using a more sophisticated method
    if n_circles <= 50:
        # For smaller number of circles, use Voronoi-based initialization with better spacing
        points = generate_voronoi_grid(n_circles * 2, 1.0, 1.0)  # Generate more points for variety

        # Apply Poisson disk refinement to ensure good distribution
        if len(points) >= n_circles:
            # Select points that are well-separated
            selected_points = []
            used_indices = set()

            # Start with a random point
            start_idx = np.random.randint(len(points))
            selected_points.append(points[start_idx])
            used_indices.add(start_idx)

            # Greedily select remaining points ensuring minimum distance
            tree = cKDTree(points)

            while len(selected_points) < n_circles:
                # Find point furthest from existing selected points
                max_min_dist = -1
                best_candidate = None
                best_idx = -1

                # Sample candidate points
                sample_indices = np.random.choice(len(points), min(20, len(points)), replace=False)
                for idx in sample_indices:
                    if idx in used_indices:
                        continue
                    point = points[idx]

                    # Calculate minimum distance to all selected points
                    if selected_points:
                        distances = tree.query(selected_points, k=1, p=np.inf)[0]
                        min_dist = np.min(distances)
                    else:
                        min_dist = float('inf')

                    if min_dist > max_min_dist:
                        max_min_dist = min_dist
                        best_candidate = point
                        best_idx = idx

                if best_candidate is not None:
                    selected_points.append(best_candidate)
                    used_indices.add(best_idx)
                else:
                    # If we can't find a good candidate, use remaining random points
                    remaining_indices = [i for i in range(len(points)) if i not in used_indices]
                    if remaining_indices:
                        idx = remaining_indices[0]
                        selected_points.append(points[idx])
                        used_indices.add(idx)
                    else:
                        break

            initial_points = np.array(selected_points[:n_circles])
        else:
            # Fall back to basic approach
            initial_points = points[:n_circles]
    else:
        # For larger numbers, fall back to standard Poisson sampling
        initial_points = poisson_disk_sampling(1.0, 1.0, POISSON_DISK_MIN_DISTANCE)

    # Ensure we have exactly n_circles points
    if len(initial_points) < n_circles:

        extra_points[:, 1] = np.clip(extra_points[:, 1], BOUNDARY_MARGIN, 1 - BOUNDARY_MARGIN)
        initial_points = np.vstack([initial_points, extra_points])

    for _ in range(pop_size):
        # Initialize circles with selected points
        circles = np.zeros((n_circles, 3))

        # Distribute circles with varying initial radii based on point density
        # Points in denser regions should get smaller initial radii
        if len(initial_points) > 1:

            avg_distances = np.mean(distances[:, 1:], axis=1)  # Exclude self-distance

            normalized_densities = 1.0 / (avg_distances + 1e-8)
