def create_spatial_grid(circles: np.ndarray) -> dict:
    """Create a spatial grid for efficient overlap checking."""
    grid = {}

    for i, (x, y, r) in enumerate(circles):
        # Determine which grid cells this circle touches

    if grid is None:
        grid = create_spatial_grid(circles)

    for cell, circle_indices in grid.items():
        if len(circle_indices) > 1:

            for i in range(len(circle_indices)):
                idx1 = circle_indices[i]

    hex_spacing = hex_radius * 2

            if len(points) >= n_points:

    grid_density = min(20, int(np.ceil(np.sqrt(n_points * 2))) + 5)
    spacing_x = 0.9 / grid_density
    spacing_y = 0.9 / grid_density

    for i in range(grid_density):
        for j in range(grid_density):
            x = 0.05 + (j + 0.5) * spacing_x + np.random.uniform(-spacing_x*0.1, spacing_x*0.1)
            y = 0.05 + (i + 0.5) * spacing_y + np.random.uniform(-spacing_y*0.1, spacing_y*0.1)
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                candidate_points.append([x, y])

    if len(candidate_points) >= n_points * 2:

            else:
                # Fall back to original sampling method
                points = [candidate_points[i % len(candidate_points)] for i in range(n_points)]
