import numpy as np
from scipy.spatial import Voronoi, distance
from scipy.optimize import minimize
from shapely.geometry import Point, Polygon
from shapely.ops import unary_union
import math

def circle_packing26() -> np.ndarray:
    """
    Places 26 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (26,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    np.random.seed(42)  # For reproducibility
    
    n = 26
    
    # Step 1: Generate initial candidate points using Voronoi + Poisson sampling
    # Create initial points for Voronoi
    initial_points = []
    
    # Generate points in a grid pattern with some randomness
    grid_size = 5
    for i in range(grid_size):
        for j in range(grid_size):
            x = (i + 0.5) / grid_size + np.random.uniform(-0.02, 0.02)
            y = (j + 0.5) / grid_size + np.random.uniform(-0.02, 0.02)
            if 0 <= x <= 1 and 0 <= y <= 1:
                initial_points.append([x, y])
    
    # Add some random points
    for _ in range(6):
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        initial_points.append([x, y])
    
    initial_points = np.array(initial_points)
    
    # Perform Voronoi tessellation
    vor = Voronoi(initial_points)
    
    # Sample points from Voronoi cells using Poisson disk sampling approach
    candidate_points = []
    
    # Get Voronoi vertices and regions
    points = []
    for region in vor.point_region:
        if region >= 0:
            # Get points in this region
            region_points = vor.regions[region]
            if len(region_points) > 0:
                # Sample a point inside this region
                try:
                    # Approximate the centroid of the polygon
                    region_coords = [vor.vertices[i] for i in region_points if i >= 0]
                    if len(region_coords) > 2:
                        region_coords = np.array(region_coords)
                        centroid_x = np.mean(region_coords[:, 0])
                        centroid_y = np.mean(region_coords[:, 1])
                        
                        # Only add if inside the unit square
                        if 0 <= centroid_x <= 1 and 0 <= centroid_y <= 1:
                            candidate_points.append([centroid_x, centroid_y])
                except:
                    pass
    
    # If we don't have enough points, add some randomly
    while len(candidate_points) < n:
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        candidate_points.append([x, y])
    
    # Select n points, preferring those in the center for better packing
    selected_points = []
    remaining_points = candidate_points[:min(n, len(candidate_points))]
    
    # Sort by distance from center to prioritize central positions
    center = np.array([0.5, 0.5])
    distances = [np.sqrt((p[0]-center[0])**2 + (p[1]-center[1])**2) for p in remaining_points]
    sorted_indices = np.argsort(distances)
    
    # Take the first n points
    for i in range(min(n, len(sorted_indices))):
        selected_points.append(remaining_points[sorted_indices[i]])
    
    # If we still don't have n points, add random ones
    while len(selected_points) < n:
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        selected_points.append([x, y])
    
    selected_points = selected_points[:n]
    
    # Step 2: Initialize with reasonable radii
    circles = np.zeros((n, 3))
    for i, (x, y) in enumerate(selected_points):
        circles[i] = [x, y, min(x, 1-x, y, 1-y)]
    
    # Step 3: Optimize using constrained optimization
    def objective(radii):
        return -np.sum(radii)  # Negative because we minimize
    
    def constraint_containment(i, x, y, r):
        # Circle must be fully contained in unit square
        return min(x-r, 1-x-r, y-r, 1-y-r)
    
    def constraint_overlap(i, j, x_i, y_i, r_i, x_j, y_j, r_j):
        # Distance between centers >= sum of radii
        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        return dist - (r_i + r_j)
    
    # Flatten the initial solution: [x0, y0, r0, x1, y1, r1, ...]
    initial_solution = []
    for i in range(n):
        initial_solution.extend([circles[i][0], circles[i][1], circles[i][2]])
    
    # Set up bounds for each variable (x, y, r)
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])  # x, y in [0,1], r in [0, 0.5]
    
    # Define constraints
    def create_constraint_func(i, j):
        def constraint_func(solution):
            x_i, y_i, r_i = solution[3*i], solution[3*i+1], solution[3*i+2]
            x_j, y_j, r_j = solution[3*j], solution[3*j+1], solution[3*j+2]
            return constraint_overlap(i, j, x_i, y_i, r_i, x_j, y_j, r_j)
        return constraint_func
    
    # Create constraint list
    constraints = []
    
    # Add containment constraints
    for i in range(n):
        def containment_constraint_func(i):
            def constraint_func(solution):
                x, y, r = solution[3*i], solution[3*i+1], solution[3*i+2]
                return constraint_containment(i, x, y, r)
            return constraint_func
        
        constraints.append({'type': 'ineq', 'fun': containment_constraint_func(i)})
    
    # Add overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': create_constraint_func(i, j)})
    
    # Perform optimization
    try:
        result = minimize(
            objective,
            initial_solution,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-6, 'eps': 1e-6}
        )
        
        if result.success:
            # Extract final solution
            optimized_circles = []
            for i in range(n):
                x = max(0.001, min(0.999, result.x[3*i]))
                y = max(0.001, min(0.999, result.x[3*i+1]))
                r = max(0.001, min(0.5, result.x[3*i+2]))
                optimized_circles.append([x, y, r])
            
            # Final check and adjustment
            final_circles = []
            for i in range(n):
                x, y, r = optimized_circles[i]
                # Ensure containment
                r = min(r, x, 1-x, y, 1-y)
                final_circles.append([x, y, r])
                
            circles = np.array(final_circles)
        else:
            # Use the initial solution if optimization fails
            pass
            
    except Exception as e:
        # If optimization fails, return initial solution
        pass
    
    # Final validation
    def validate_solution(circles_arr):
        # Check containment
        for i in range(len(circles_arr)):
            x, y, r = circles_arr[i]
            if not (r <= x <= 1-r and r <= y <= 1-r):
                return False
        
        # Check overlap
        for i in range(len(circles_arr)):
            for j in range(i+1, len(circles_arr)):
                x1, y1, r1 = circles_arr[i]
                x2, y2, r2 = circles_arr[j]
                dist_sq = (x1-x2)**2 + (y1-y2)**2
                if dist_sq < (r1+r2)**2:
                    return False
        return True
    
    # Validate and adjust if needed
    if not validate_solution(circles):
        # Reset to better initial solution if validation fails
        return circles
    
    return circles
