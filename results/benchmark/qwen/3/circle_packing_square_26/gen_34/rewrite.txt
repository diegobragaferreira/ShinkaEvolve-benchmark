import numpy as np
import random
from deap import base, creator, tools, algorithms
import math
from scipy.spatial import cKDTree
import time

# Set random seed for reproducibility
random.seed(42)
np.random.seed(42)

def circle_packing26() -> np.ndarray:
    """
    Places 26 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (26,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Problem parameters
    n_circles = 26
    max_generations = 100
    population_size = 50
    crossover_prob = 0.7
    mutation_prob = 0.2
    
    # Define the fitness and individual classes
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)
    
    def evaluate(individual):
        """Evaluate the fitness of an individual (set of circles)."""
        # Convert individual to circles array
        circles = np.array(individual).reshape(-1, 3)
        
        # Check if all circles are within bounds
        total_radius = 0
        penalty = 0
        
        # Calculate total radius first
        for i in range(n_circles):
            x, y, r = circles[i]
            total_radius += r
            
        # Check boundary constraints
        for i in range(n_circles):
            x, y, r = circles[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                # Apply penalty based on how much it violates boundaries
                boundary_penalty = 0
                if x - r < 0:
                    boundary_penalty += (r - x)**2
                if x + r > 1:
                    boundary_penalty += (x + r - 1)**2
                if y - r < 0:
                    boundary_penalty += (r - y)**2
                if y + r > 1:
                    boundary_penalty += (y + r - 1)**2
                penalty += boundary_penalty * 1000
        
        # Check overlap constraints using spatial indexing for efficiency
        # Create KDTree for fast neighbor search
        points = [(circles[i][0], circles[i][1]) for i in range(n_circles)]
        tree = cKDTree(points)
        
        # Find neighbors within distance 2*r (potential overlaps)
        pairs = tree.query_pairs(2.0, output_type='ndarray')
        
        for i, j in zip(pairs[0], pairs[1]):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            distance = math.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if distance < (r1 + r2):
                # Apply penalty proportional to overlap amount
                overlap = (r1 + r2) - distance
                penalty += overlap * 50000
                
        # Return fitness (total radius minus penalty)
        return (total_radius - penalty),
    
    def mutate(individual):
        """Mutate an individual."""
        # Adaptive mutation rate decreases over time
        gen = getattr(mutate, 'generation', 0)
        adapt_rate = 0.15 * (1 - gen/max_generations) + 0.01
        mutate.generation = gen + 1
        
        for i in range(len(individual)):
            if random.random() < adapt_rate:
                # Mutate either x, y, or r coordinate
                if i % 3 == 0:  # x coordinate
                    individual[i] = max(0, min(1, individual[i] + random.gauss(0, 0.02)))
                elif i % 3 == 1:  # y coordinate
                    individual[i] = max(0, min(1, individual[i] + random.gauss(0, 0.02)))
                else:  # radius
                    individual[i] = max(0.001, min(0.5, individual[i] + random.gauss(0, 0.01)))
        return individual,
    
    def initialize_population():
        """Initialize population with Voronoi-based seed points."""
        population = []
        # Generate Voronoi-like distribution of points
        # Use grid with some randomness for better distribution
        grid_size = int(math.ceil(math.sqrt(n_circles)))
        spacing = 1.0 / (grid_size + 1)
        
        points = []
        for i in range(grid_size):
            for j in range(grid_size):
                if len(points) < n_circles:
                    x = (i + 1) * spacing + random.uniform(-spacing/4, spacing/4)
                    y = (j + 1) * spacing + random.uniform(-spacing/4, spacing/4)
                    points.append([x, y])
        
        # Ensure we have exactly n_circles points
        while len(points) < n_circles:
            x = random.random()
            y = random.random()
            points.append([x, y])
            
        # Create initial individuals
        for _ in range(population_size):
            individual = []
            for i in range(n_circles):
                x, y = points[i]
                # Add small random perturbation
                x += random.uniform(-0.01, 0.01)
                y += random.uniform(-0.01, 0.01)
                # Assign radius based on density - smaller circles in dense regions
                r = min(0.1, max(0.01, random.random() * 0.05))
                individual.extend([x, y, r])
            population.append(individual)
            
        return population
    
    # Initialize toolbox
    toolbox = base.Toolbox()
    toolbox.register("individual", tools.initRepeat, creator.Individual, 
                     lambda: random.uniform(0, 1), n_circles*3)
    toolbox.register("population", initialize_population)
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxUniform, indpb=0.5)
    toolbox.register("mutate", mutate)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Run evolution
    population = toolbox.population()
    
    # Statistics logging
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("min", np.min)
    stats.register("max", np.max)
    
    # Evolution algorithm
    hof = tools.ParetoFront()
    
    # Run with more generations but keep it under time limit
    final_pop, logbook = algorithms.eaSimple(
        population, toolbox, cxpb=crossover_prob, mutpb=mutation_prob,
        ngen=max_generations, stats=stats, halloffame=hof, verbose=False
    )
    
    # Get the best individual
    best_individual = hof[0]
    best_circles = np.array(best_individual).reshape(-1, 3)
    
    # Final refinement using local optimization for best solution
    refined_best = refine_solution(best_circles.copy())
    
    return refined_best

def refine_solution(circles):
    """Apply local refinement to improve the solution."""
    # Simple local optimization: try to increase radii while respecting constraints
    max_attempts = 50
    n_circles = len(circles)
    
    # Build spatial index for efficient overlap checking
    points = [(circles[i][0], circles[i][1]) for i in range(n_circles)]
    
    for attempt in range(max_attempts):
        improved = False
        for i in range(n_circles):
            # Try to slightly increase radius
            x, y, r = circles[i]
            old_r = r
            
            # Attempt to increase radius
            new_r = min(r * 1.05, 0.5)
            
            # Check if this change is valid
            valid = True
            
            # Check boundary constraints
            if x - new_r < 0 or x + new_r > 1 or y - new_r < 0 or y + new_r > 1:
                valid = False
            else:
                # Check overlap constraints
                for j in range(n_circles):
                    if i != j:
                        x2, y2, r2 = circles[j]
                        distance = math.sqrt((x - x2)**2 + (y - y2)**2)
                        if distance < (new_r + r2):
                            valid = False
                            break
            
            if valid and new_r > old_r:
                circles[i] = [x, y, new_r]
                improved = True
                
        if not improved:
            break
            
    # If we still have room, try another approach
    # Use a greedy method to find good radii
    if attempt < max_attempts - 1:
        # Re-evaluate and make small adjustments
        for i in range(n_circles):
            x, y, r = circles[i]
            # Slightly adjust to reduce conflicts
            circles[i] = [max(0.001, min(0.999, x)), 
                         max(0.001, min(0.999, y)), 
                         max(0.001, min(0.5, r))]
    
    return circles
