    def _check_overlap_constraints(self, individual: np.ndarray) -> float:
        """Check overlap constraints efficiently using optimized spatial queries"""
        penalty = 0.0

        if len(individual) <= 1:
            return penalty

        # For better performance with many circles, we'll use a two-phase approach
        # Phase 1: Use a spatial grid for coarse filtering
        grid_size = self.grid_size
        grid_cells = {}

        # Assign each circle to grid cells
        for i in range(len(individual)):
            x, y, r = individual[i]
            # Determine grid cell bounds for this circle
            min_gx = int((x - r) / grid_size)
            max_gx = int((x + r) / grid_size)
            min_gy = int((y - r) / grid_size)
            max_gy = int((y + r) / grid_size)

            # Add to all relevant grid cells
            for gx in range(min_gx, max_gx + 1):
                for gy in range(min_gy, max_gy + 1):
                    key = (gx, gy)
                    if key not in grid_cells:
                        grid_cells[key] = []
                    grid_cells[key].append(i)

        # Phase 2: Check overlaps within and between cells
        checked_pairs = set()

        for cell_key, circle_indices in grid_cells.items():
            if len(circle_indices) < 2:
                continue

            # Check all pairs in this cell
            for i in range(len(circle_indices)):
                idx1 = circle_indices[i]
                x1, y1, r1 = individual[idx1]

                for j in range(i + 1, len(circle_indices)):
                    idx2 = circle_indices[j]

                    # Skip if already checked (avoid duplicate work)
                    if (idx1, idx2) in checked_pairs or (idx2, idx1) in checked_pairs:
                        continue

                    checked_pairs.add((idx1, idx2))
                    x2, y2, r2 = individual[idx2]

                    # Fast distance check using squared distance
                    dx = x2 - x1
                    dy = y2 - y1
                    distance_squared = dx*dx + dy*dy

                    # Check if circles are close enough to potentially overlap
                    combined_radius = r1 + r2
                    if distance_squared < combined_radius * combined_radius:
                        actual_distance = math.sqrt(distance_squared)
                        if actual_distance < combined_radius:
                            overlap = combined_radius - actual_distance
                            # Apply penalty scaled with overlap severity
                            penalty += overlap * 1000 * (1.0 + overlap * 0.01)

        return penalty

    def _adjust_for_overlaps(self, individual: np.ndarray):

        # Iterative overlap resolution
        max_iterations = 100
        for iteration in range(max_iterations):
            any_changes = False

            for i in range(len(individual)):
                for j in range(i+1, len(individual)):
                    x1, y1, r1 = individual[i]
                    x2, y2, r2 = individual[j]

                    dx = x2 - x1
                    dy = y2 - y1
                    distance = math.sqrt(dx*dx + dy*dy)

                    # If circles overlap
                    if distance < r1 + r2:
                        overlap = (r1 + r2) - distance
                        if distance > 0:
                            # Push them apart along the line connecting centers
                            push_x = dx / distance * overlap * 0.5
                            push_y = dy / distance * overlap * 0.5

                            individual[i, 1] -= push_y

    def run_evolution(self) -> np.ndarray:
        """Run the evolutionary optimization process"""
        # Initialize population
        population = []
        for _ in range(self.population_size):
            individual = self.initialize_individual()
            population.append(individual)

        # Evolutionary loop
        best_fitness_history = []
        for generation in range(self.generations):
            # Evaluate fitness
            fitness_scores = []
            for individual in population:
                fitness = self.evaluate_fitness(individual)
                fitness_scores.append(fitness)

            # Track best fitness
            best_fitness = max(fitness_scores)
            best_fitness_history.append(best_fitness)

            # Select top individuals (elitism)
            sorted_indices = np.argsort(fitness_scores)[::-1]
            elite = [population[i] for i in sorted_indices[:self.elite_size]]

            # Generate new population
            new_population = elite.copy()

            # Adaptive mutation rate: decrease over time
            adaptive_mutation_rate = self.mutation_rate * (1 - generation / self.generations)
            if adaptive_mutation_rate < 0.01:
                adaptive_mutation_rate = 0.01

            # Fill rest of population through crossover and mutation
            while len(new_population) < self.population_size:
                parent1 = self.tournament_selection(population, fitness_scores)
                parent2 = self.tournament_selection(population, fitness_scores)

                if random.random() < 0.8:  # Crossover probability
                    child1, child2 = self.crossover(parent1, parent2)
                else:
                    child1, child2 = parent1.copy(), parent2.copy()

                # Apply mutation with adaptive rate
                if random.random() < adaptive_mutation_rate:
                    self.mutate_individual(child1, adaptive_mutation_rate)
                if random.random() < adaptive_mutation_rate:
                    self.mutate_individual(child2, adaptive_mutation_rate)

                new_population.extend([child1, child2])

            # Trim to exact population size
            population = new_population[:self.population_size]

            # Print progress
            if generation % 50 == 0:
                print(f"Generation {generation}: Best fitness = {best_fitness:.6f}")

        # Return best solution
        final_fitness_scores = [self.evaluate_fitness(ind) for ind in population]
        best_index = np.argmax(final_fitness_scores)
        return population[best_index]
