import numpy as np
from scipy.spatial import cKDTree
import random
from deap import base, creator, tools, algorithms
import time

# Set random seeds for reproducibility
np.random.seed(42)
random.seed(42)

def circle_packing26() -> np.ndarray:
    """
    Places 26 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (26,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    
    # Constants
    N_CIRCLES = 26
    MAX_RADIUS = 0.5
    UNIT_SIZE = 1.0
    
    def is_valid_position(x, y, r, existing_circles):
        """Check if a circle position is valid (within bounds and non-overlapping)"""
        # Check boundary constraints
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return False
            
        # Check overlap constraints with existing circles
        for cx, cy, cr in existing_circles:
            distance = np.sqrt((x - cx)**2 + (y - cy)**2)
            if distance < r + cr:
                return False
                
        return True
    
    def create_initial_population(pop_size):
        """Create initial population with heuristic placement"""
        population = []
        for _ in range(pop_size):
            circles = []
            # Try to place circles using a greedy approach with some randomness
            for i in range(N_CIRCLES):
                max_attempts = 1000
                placed = False
                attempts = 0
                
                while not placed and attempts < max_attempts:
                    # Random placement with bias toward center for better packing
                    x = np.random.uniform(0.1, 0.9)
                    y = np.random.uniform(0.1, 0.9)
                    
                    # Calculate maximum possible radius for this position
                    min_radius = min(x, 1-x, y, 1-y)
                    
                    # Try different radius values
                    for r_guess in np.linspace(min_radius/2, min_radius/4, 5):
                        if is_valid_position(x, y, r_guess, circles):
                            circles.append([x, y, r_guess])
                            placed = True
                            break
                    
                    attempts += 1
                    
                # If we couldn't place a circle, try with smaller radius
                if not placed and len(circles) > 0:
                    best_radius = 0
                    for attempt in range(100):
                        x = np.random.uniform(0.1, 0.9)
                        y = np.random.uniform(0.1, 0.9)
                        min_radius = min(x, 1-x, y, 1-y)
                        if min_radius > best_radius:
                            best_radius = min_radius
                    if best_radius > 0:
                        circles.append([0.5, 0.5, best_radius/2])
            
            # Ensure we have exactly N_CIRCLES
            while len(circles) < N_CIRCLES:
                circles.append([0.5, 0.5, 0.01])
                
            population.append(circles[:N_CIRCLES])  # Trim to exact size
            
        return population
    
    def evaluate_individual(individual):
        """Evaluate fitness of an individual (sum of radii)"""
        total_radius = sum(circle[2] for circle in individual)
        return (total_radius,)
    
    def mutate_individual(individual, indpb=0.1):
        """Mutate an individual by slightly adjusting positions and radii"""
        mutated = []
        for circle in individual:
            x, y, r = circle
            if np.random.random() < indpb:
                # Mutate position
                x = max(0.01, min(0.99, x + np.random.normal(0, 0.02)))
                y = max(0.01, min(0.99, y + np.random.normal(0, 0.02)))
            if np.random.random() < indpb:
                # Mutate radius (with bounds)
                r = max(0.001, min(0.49, r + np.random.normal(0, 0.01)))
            mutated.append([x, y, r])
        
        # Recheck validity and adjust if needed
        valid_circles = []
        for x, y, r in mutated:
            if is_valid_position(x, y, r, valid_circles):
                valid_circles.append([x, y, r])
            else:
                # Try to find a valid position
                best_r = min(x, 1-x, y, 1-y) * 0.9
                if best_r > 0.001:
                    valid_circles.append([x, y, best_r * 0.8])
                else:
                    valid_circles.append([0.5, 0.5, 0.01])
        
        # Ensure exactly N_CIRCLES
        while len(valid_circles) < N_CIRCLES:
            valid_circles.append([0.5, 0.5, 0.01])
        return valid_circles[:N_CIRCLES]
    
    def crossover_individuals(parent1, parent2):
        """Crossover two individuals by mixing their properties"""
        child1 = []
        child2 = []
        
        for i in range(len(parent1)):
            # Alternate genes (position and radius) between parents
            if np.random.random() < 0.5:
                child1.append(parent1[i][:])
                child2.append(parent2[i][:])
            else:
                child1.append(parent2[i][:])
                child2.append(parent1[i][:])
        
        return child1, child2
    
    # Setup DEAP
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)
    
    toolbox = base.Toolbox()
    toolbox.register("population", create_initial_population, pop_size=30)
    toolbox.register("evaluate", evaluate_individual)
    toolbox.register("mate", crossover_individuals)
    toolbox.register("mutate", mutate_individual)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Create initial population
    population = toolbox.population()
    
    # Evaluate initial population
    fitnesses = list(map(toolbox.evaluate, population))
    for ind, fit in zip(population, fitnesses):
        ind.fitness.values = fit
    
    # Evolution parameters
    CXPB = 0.7  # Crossover probability
    MUTPB = 0.3  # Mutation probability
    NGEN = 50   # Number of generations
    POPSIZE = 30
    
    # Main evolution loop
    start_time = time.time()
    best_fitness = 0
    best_individual = None
    
    for gen in range(NGEN):
        # Select the next generation individuals
        offspring = toolbox.select(population, len(population))
        offspring = list(map(toolbox.clone, offspring))
        
        # Apply crossover and mutation
        for child1, child2 in zip(offspring[::2], offspring[1::2]):
            if np.random.random() < CXPB:
                child1, child2 = toolbox.mate(child1, child2)
                del child1.fitness.values
                del child2.fitness.values
        
        for mutant in offspring:
            if np.random.random() < MUTPB:
                mutant = toolbox.mutate(mutant)
                del mutant.fitness.values
        
        # Evaluate the individuals with an invalid fitness
        invalid_ind = [ind for ind in offspring if not ind.fitness.valid]
        fitnesses = map(toolbox.evaluate, invalid_ind)
        for ind, fit in zip(invalid_ind, fitnesses):
            ind.fitness.values = fit
        
        # Replace population
        population[:] = offspring
        
        # Track best individual
        current_best = max(population, key=lambda x: x.fitness.values[0])
        current_fitness = current_best.fitness.values[0]
        if current_fitness > best_fitness:
            best_fitness = current_fitness
            best_individual = current_best[:]
        
        # Early stopping if we're getting close to the target
        if best_fitness > 2.6:
            break
            
        if time.time() - start_time > 55:  # Leave 5 seconds for cleanup
            break
    
    # Final refinement
    if best_individual is None:
        # Fallback to heuristic solution
        # Place circles in a grid-like pattern with diminishing sizes
        best_individual = []
        grid_size = int(np.ceil(np.sqrt(N_CIRCLES)))
        spacing = 1.0 / (grid_size + 1)
        
        count = 0
        for i in range(grid_size):
            for j in range(grid_size):
                if count >= N_CIRCLES:
                    break
                x = (i + 1) * spacing
                y = (j + 1) * spacing
                # Radius based on distance to edges and position
                min_edge_dist = min(x, 1-x, y, 1-y)
                max_radius = min_edge_dist * 0.8
                if max_radius > 0:
                    best_individual.append([x, y, max_radius])
                else:
                    best_individual.append([0.5, 0.5, 0.01])
                count += 1
                if count >= N_CIRCLES:
                    break
    
    # Ensure we have exactly N_CIRCLES
    while len(best_individual) < N_CIRCLES:
        best_individual.append([0.5, 0.5, 0.01])
    
    result = np.array(best_individual[:N_CIRCLES])
    
    # Final validation and adjustment
    final_circles = []
    for i, (x, y, r) in enumerate(result):
        # Ensure the circle is valid
        if r <= 0:
            r = 0.01
            
        # Adjust for boundaries if needed
        edge_dist = min(x, 1-x, y, 1-y)
        if edge_dist < r:
            r = edge_dist * 0.9
            if r <= 0:
                r = 0.01
                
        final_circles.append([x, y, r])
    
    return np.array(final_circles)
