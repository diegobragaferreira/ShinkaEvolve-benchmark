def check_overlap(circles: np.ndarray) -> bool:
    """Check if any circles overlap using efficient spatial indexing."""
    n = len(circles)
    if n < 2:
        return True

    # Use cKDTree for efficient nearest neighbor search
    positions = circles[:, :2]
    radii = circles[:, 2]

    try:
        # Build spatial tree once
        tree = cKDTree(positions)

        # Query pairs within sum of radii distance
        pairs = tree.query_pairs(r=2.0, output_type='ndarray')

        # Check each potential overlapping pair
        for i, j in pairs:
            if i < j:  # Avoid duplicate checking
                dist = np.linalg.norm(positions[i] - positions[j])
                if dist < radii[i] + radii[j]:
                    return False

    except Exception:
        # Fallback to original method if tree fails
        # Calculate pairwise distances
        distances = cdist(positions, positions)

        # Check for overlaps
        for i in range(n):
            for j in range(i+1, n):
                dist = distances[i, j]
                if dist < radii[i] + radii[j]:
                    return False

    return True

def check_overlap(circles: np.ndarray) -> bool:
    """Check if any circles overlap using efficient spatial indexing."""
    n = len(circles)
    if n < 2:
        return True

    # Use spatial grid for O(n) complexity instead of O(n^2)
    GRID_SIZE = 20
    grid = {}

    # Create spatial grid
    for i, (x, y, r) in enumerate(circles):
        # Determine which grid cells this circle touches
        min_x_cell = max(0, int((x - r) * GRID_SIZE))
        max_x_cell = min(GRID_SIZE - 1, int((x + r) * GRID_SIZE))
        min_y_cell = max(0, int((y - r) * GRID_SIZE))
        max_y_cell = min(GRID_SIZE - 1, int((y + r) * GRID_SIZE))

        for gx in range(min_x_cell, max_x_cell + 1):
            for gy in range(min_y_cell, max_y_cell + 1):
                if (gx, gy) not in grid:
                    grid[(gx, gy)] = []
                grid[(gx, gy)].append(i)

    # Check overlaps within each grid cell
    for cell, circle_indices in grid.items():
        # Only check pairs within the same grid cell
        for i in range(len(circle_indices)):
            idx1 = circle_indices[i]
            x1, y1, r1 = circles[idx1]

            for j in range(i + 1, len(circle_indices)):
                idx2 = circle_indices[j]
                x2, y2, r2 = circles[idx2]

                # Calculate distance between circle centers
                dx = x1 - x2
                dy = y1 - y2
                distance_squared = dx*dx + dy*dy

                # Check if circles overlap
                if distance_squared < (r1 + r2)**2:
                    return False

    return True

def mutate(circles: np.ndarray, generation: int = 0, max_generations: int = 1000,
           population_fitnesses: List[float] = None) -> np.ndarray:
    """Mutate a circle configuration with adaptive mutation rate and smart mutation strategy."""
    # Adaptive mutation rate with better scaling
    mutation_rate_start = 0.25
    mutation_rate_end = 0.01
    mutation_rate = mutation_rate_end + (mutation_rate_start - mutation_rate_end) * (
        1 / (1 + np.exp(15 * (generation / max_generations - 0.5)))
    )

    # If we have population fitness info, adjust mutation rate based on diversity
    if population_fitnesses is not None and len(population_fitnesses) > 1:
        # Calculate population variance to detect stagnation
        fitness_variance = np.var(population_fitnesses)
        if fitness_variance < 1e-6:  # Low diversity - increase mutation rate
            mutation_rate *= 1.5

    mutated = circles.copy()
    n = len(mutated)

    for i in range(n):
        if random.random() < mutation_rate:
            # Choose mutation type based on position in evolution and individual fitness
            # Higher chance of position mutations early, radius mutations later
            if generation < max_generations * 0.4:
                # Early stages: prioritize position changes with higher probability
                choice = random.choices([0, 1, 2], weights=[0.6, 0.6, 0.2])[0]
                strength = 0.06
            elif generation < max_generations * 0.7:
                # Mid stages: balanced
                choice = random.choices([0, 1, 2], weights=[0.4, 0.4, 0.4])[0]
                strength = 0.03
            else:
                # Late stages: prioritize radius fine-tuning
                choice = random.choices([0, 1, 2], weights=[0.2, 0.2, 0.8])[0]
                strength = 0.015

            if choice == 0:  # Mutate x position
                mutated[i, 0] = np.clip(mutated[i, 0] + np.random.normal(0, strength),
                                      mutated[i, 2], 1 - mutated[i, 2])
            elif choice == 1:  # Mutate y position
                mutated[i, 1] = np.clip(mutated[i, 1] + np.random.normal(0, strength),
                                      mutated[i, 2], 1 - mutated[i, 2])
            else:  # Mutate radius
                # Use log-normal for radius mutations to avoid extreme values
                log_factor = np.random.normal(0, 0.15)
                mutated[i, 2] = np.clip(mutated[i, 2] * np.exp(log_factor), 0.001, 0.4)

    return mutated
