class AdvancedVoronoiInitialization:
    """Implements advanced Voronoi-like initialization with adaptive spacing"""

    @staticmethod
    def generate_voronoi_distribution(n: int) -> np.ndarray:
        """Generate high-quality Voronoi-like distribution using iterative optimization"""
        circles = np.zeros((n, 3))

        # Start with a regular grid pattern
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))

        # Space the grid points to maximize coverage
        spacing_x = 0.9 / (cols + 1)
        spacing_y = 0.9 / (rows + 1)

        points = []
        for i in range(rows):
            for j in range(cols):
                if len(points) < n:
                    x = 0.05 + (j + 1) * spacing_x
                    y = 0.05 + (i + 1) * spacing_y
                    points.append((x, y))

        # Fill remaining points with random distribution
        while len(points) < n:
            points.append((random.uniform(0.05, 0.95), random.uniform(0.05, 0.95)))

        # Apply more sophisticated Voronoi-inspired spacing
        for i in range(n):
            x_base, y_base = points[i]

            # Apply more controlled randomness with preference for spreading
            jitter_x = random.uniform(-spacing_x * 0.3, spacing_x * 0.3)
            jitter_y = random.uniform(-spacing_y * 0.3, spacing_y * 0.3)

            x = max(0.01, min(0.99, x_base + jitter_x))
            y = max(0.01, min(0.99, y_base + jitter_y))

            # Assign initial radius with more diversity
            r = 0.02 + random.uniform(0, 0.05)

            circles[i] = [x, y, r]

        return circles

    @staticmethod
    def generate_adaptive_grid(n: int) -> np.ndarray:
        """Generate grid-based distribution with adaptive spacing"""
        circles = np.zeros((n, 3))

        # Create an adaptive grid that becomes denser near edges
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))

        # Use variable spacing to create more organic distribution
        spacing_x = 0.85 / (cols + 1)
        spacing_y = 0.85 / (rows + 1)

        center_x = 0.5
        center_y = 0.5

        for i in range(n):

            col = i % cols

            # Add bias towards center for better packing

            y_base = (row + 1) * spacing_y

            dist_from_center = math.sqrt((x_base - center_x)**2 + (y_base - center_y)**2)

    @staticmethod
    def generate_poisson_disk_distribution(n: int, radius_min: float = 0.05) -> np.ndarray:
        """Generate points using Poisson disk sampling for uniform distribution"""
        # Simple implementation of Poisson disk sampling with rejection method
        points = []
        max_attempts = 10000
        radius_squared = radius_min * radius_min

        # Generate candidates and accept them with minimum distance constraint
        for _ in range(max_attempts):
            if len(points) >= n:
                break
            x = random.uniform(radius_min, 1 - radius_min)
            y = random.uniform(radius_min, 1 - radius_min)

            # Check distance to all existing points
            valid = True
            for px, py in points:
                dx = x - px
                dy = y - py
                if dx*dx + dy*dy < radius_squared:
                    valid = False
                    break

            if valid:
                points.append((x, y))

        # If we didn't get enough points, fill with random distribution
        while len(points) < n:
            points.append((random.uniform(radius_min, 1 - radius_min),
                          random.uniform(radius_min, 1 - radius_min)))

        # Convert to circles with appropriate radii
        circles = np.zeros((n, 3))
        for i, (x, y) in enumerate(points[:n]):
            r = radius_min + random.uniform(0, radius_min * 0.5)
            circles[i] = [x, y, r]

        return circles
