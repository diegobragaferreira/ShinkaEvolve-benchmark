The current implementation uses a custom spatial hashing approach for overlap detection, but this is inefficient and error-prone. I'll replace it with a more robust and efficient approach using scipy's cKDTree for neighbor lookups, which will significantly improve the performance of overlap detection while maintaining correctness.

This change addresses the bottleneck in overlap checking that's preventing us from reaching the benchmark score. The cKDTree approach is much faster for our use case and provides better scalability.

        """Calculate penalty based on constraint violations with detailed information"""
        penalty = 0.0
        boundary_violations = 0.0
        overlap_violations = 0.0
        overlap_count = 0

        n = len(circles)

        # Check containment penalties (more precise calculation)
        for circle in circles:
            x, y, r = circle
            if not self.is_valid_position(x, y, r):
                # Calculate violation amounts
                left_violation = max(0, r - x)
                right_violation = max(0, r - (1 - x))
                bottom_violation = max(0, r - y)
                top_violation = max(0, r - (1 - y))
                boundary_violations += (left_violation + right_violation +
                                       bottom_violation + top_violation)

        # Use spatial indexing with cKDTree for efficient overlap detection
        valid_circles = [c for c in circles if c[2] > 0]
        if len(valid_circles) > 1:
            # Convert to numpy array for efficient processing
            positions = np.array([[c[0], c[1]] for c in valid_circles])
            radii = np.array([c[2] for c in valid_circles])

            # Build KDTree for fast neighbor queries
            tree = cKDTree(positions)

            # Find all pairs within 2 * max_radius to reduce unnecessary checks
            max_radius = np.max(radii) if len(radii) > 0 else 0.0
            pairs = tree.query_pairs(2 * max_radius, p=np.inf)

            # Check actual overlaps among close pairs
            for i, j in pairs:
                if i != j:
                    c1 = valid_circles[i]
                    c2 = valid_circles[j]
                    distance = np.sqrt((c1[0] - c2[0])**2 + (c1[1] - c2[1])**2)
                    if distance < (c1[2] + c2[2]):
                        overlap_count += 1
                        # Calculate overlap amount
                        overlap = (c1[2] + c2[2]) - distance
                        overlap_violations += overlap

                     OVERLAP_PENALTY_BASE * overlap_violations

        return penalty, boundary_violations, overlap_violations
