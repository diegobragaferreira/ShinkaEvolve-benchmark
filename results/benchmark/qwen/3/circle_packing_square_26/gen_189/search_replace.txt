I'll implement a multi-scale spatial indexing approach to improve the efficiency of overlap detection. The current implementation uses a single cKDTree with a fixed query radius, but this can be improved by using adaptive grid-based indexing that becomes more refined as the algorithm progresses. This approach leverages the fact that early generations need faster but less precise overlap detection, while later generations benefit from more accurate but computationally expensive checks.

The key insight is to use a hierarchical spatial indexing system where:
1. Early generations use coarse grids for fast overlap detection
2. Later generations switch to finer grids for more precise checking
3. This reduces unnecessary computation while maintaining accuracy

    """Check if all circles are within bounds and non-overlapping using spatial indexing."""
    n = len(circles)

    # Check boundary constraints
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return False

    # Use spatial indexing for efficient overlap checking
    if n > 1:
        try:
            positions = [(x, y) for x, y, r in circles]
            tree = cKDTree(positions)

def is_valid(circles: np.ndarray, generation: int = 0) -> bool:
    """Check if all circles are within bounds and non-overlapping using spatial indexing."""
    n = len(circles)

    # Check boundary constraints
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return False

    # Use spatial indexing for efficient overlap checking
    if n > 1:
        # Determine grid scale based on generation
        if generation < GRID_TRANSITION_GEN[0]:
            grid_size = GRID_SCALES[0]  # Coarse grid
        elif generation < GRID_TRANSITION_GEN[1]:
            grid_size = GRID_SCALES[1]  # Medium grid
        else:
            grid_size = GRID_SCALES[2]  # Fine grid

        try:
            # Use spatial grid indexing for faster overlap detection
            positions = [(x, y) for x, y, r in circles]

            # Create spatial grid structure for overlap checking
            grid = {}
            for i, (x, y) in enumerate(positions):
                # Determine grid cell for this point
                grid_x = int(x / grid_size)
                grid_y = int(y / grid_size)
                if (grid_x, grid_y) not in grid:
                    grid[(grid_x, grid_y)] = []
                grid[(grid_x, grid_y)].append(i)

            # Check for overlaps in neighboring cells
            for (gx, gy), indices in grid.items():
                # Check against neighbors in 3x3 grid
                for dx in [-1, 0, 1]:
                    for dy in [-1, 0, 1]:
                        neighbor_key = (gx + dx, gy + dy)
                        if neighbor_key in grid:
                            for i in indices:
                                for j in grid[neighbor_key]:
                                    if i < j:  # Avoid duplicate checks
                                        x1, y1, r1 = circles[i]
                                        x2, y2, r2 = circles[j]
                                        distance = math.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                                        if distance < r1 + r2:
                                            return False
        except Exception:
            # Fallback to brute force if anything goes wrong
            for i in range(n):
                for j in range(i+1, n):
                    x1, y1, r1 = circles[i]
                    x2, y2, r2 = circles[j]
                    distance = math.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                    if distance < r1 + r2:
                        return False

    return True
