def generate_voronoi_based_points(n_points: int) -> np.ndarray:
    """Generate well-distributed points using Voronoi-inspired approach with Poisson disk sampling."""
    # Generate points using a more systematic Voronoi-based method
    points = []

    # Use a hexagonal lattice with Poisson disk distribution for better uniformity
    # This creates a more natural distribution that avoids clustering

    # Create a base hexagonal grid with sufficient spacing
    hex_radius = 0.8 / np.sqrt(n_points)  # Adjust based on desired number of points
    hex_spacing = hex_radius * 2

    # Create points in a hexagonal pattern
    rows = int(np.ceil(np.sqrt(n_points)))
    cols = int(np.ceil(n_points / rows))

    for i in range(rows):
        for j in range(cols):
            if len(points) >= n_points:
                break

            # Hexagonal offset pattern
            x_offset = j * hex_spacing + (i % 2) * hex_spacing * 0.5
            y_offset = i * hex_spacing * np.sqrt(3) / 2

            # Add jitter to avoid perfect grid
            x = 0.1 + x_offset + np.random.uniform(-hex_spacing*0.2, hex_spacing*0.2)
            y = 0.1 + y_offset + np.random.uniform(-hex_spacing*0.2, hex_spacing*0.2)

            # Ensure within bounds
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            points.append([x, y])

        # Add points around the edges to improve distribution
        x = np.random.uniform(0.05, 0.95)

        points.append([x, y])

    return np.array(points[:n_points])
