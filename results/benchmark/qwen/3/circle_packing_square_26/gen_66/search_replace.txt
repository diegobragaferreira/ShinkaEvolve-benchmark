        grid_cells = defaultdict(list)

        # Precompute grid dimensions

        grid_height = int(1.0 / self.grid_size) + 1

        # Place circles in grid cells
        for i in range(len(individual)):
            x, y, r = individual[i]
            # Get grid cell indices
            gx_min = max(0, int((x - r) / self.grid_size))
            gx_max = min(grid_width, int((x + r) / self.grid_size) + 1)
            gy_min = max(0, int((y - r) / self.grid_size))

            for gx in range(gx_min, gx_max):

    def _check_overlap_constraints(self, individual: np.ndarray) -> float:
        """Check overlap constraints efficiently using KDTree"""
        penalty = 0.0

        if len(individual) <= 1:
            return penalty

        # Build KDTree for efficient neighbor queries
        tree = cKDTree(individual[:, :2])

        # Query for neighbors within distance of (r1 + r2) for each circle
        # We iterate through each point and query neighbors
        for i in range(len(individual)):
            x1, y1, r1 = individual[i]

            # Find all neighbors within distance 2*(r1 + max_radius) to optimize search
            # In practice, we'll search for neighbors within r1 + max possible radius
            # This ensures we catch all potential overlaps
            max_radius = np.max(individual[:, 2])
            neighbors = tree.query_ball_point([x1, y1], 2 * (r1 + max_radius), p=2)

            # Check overlaps with neighbors
            for j in neighbors:
                if i >= j:  # Skip self-comparison and duplicate pairs
                    continue

                x2, y2, r2 = individual[j]

                # Fast distance check using squared distance
                dx = x2 - x1
                dy = y2 - y1
                distance_squared = dx*dx + dy*dy

                # Check if circles are close enough to potentially overlap
                if distance_squared < (r1 + r2)**2:
                    actual_distance = math.sqrt(distance_squared)
                    if actual_distance < r1 + r2:
                        overlap = (r1 + r2) - actual_distance
                        # Scale penalty based on overlap severity
                        penalty += overlap * 1000 * (1 + overlap * 0.1)

        return penalty
