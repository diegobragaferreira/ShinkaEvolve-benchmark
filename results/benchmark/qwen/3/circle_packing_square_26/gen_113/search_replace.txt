        for _ in range(self.population_size):
            circles = np.zeros((n, 3))

            # Create better distributed points using hexagonal grid with randomness
            rows = int(np.ceil(np.sqrt(n)))
            cols = int(np.ceil(n / rows))

            # Ensure sufficient grid dimensions
            while rows * cols < n:
                rows += 1

            # Calculate spacing with padding
            padding = 0.1
            width = 1 - 2 * padding
            height = 1 - 2 * padding

            if cols > 0:
                spacing_x = width / cols
