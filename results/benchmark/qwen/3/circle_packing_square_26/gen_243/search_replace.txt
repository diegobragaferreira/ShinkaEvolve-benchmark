def generate_initial_voronoi_points(n_circles: int, max_iter: int = 1000) -> np.ndarray:
    """Generate initial points using Voronoi-based optimization for better distribution"""
    # Start with random points
    points = np.random.rand(n_circles, 2) * 0.8 + 0.1  # Keep away from edges

    # Apply simple Voronoi-like repulsion to spread points
    for iteration in range(max_iter):
        # Calculate pairwise distances
        distances = cdist(points, points)

        # Set diagonal to large value to skip self-comparisons
        np.fill_diagonal(distances, np.inf)

        # Find minimum distances for each point
        min_distances = np.min(distances, axis=1)

        # Calculate forces from neighbors
        forces = np.zeros_like(points)
        for i in range(n_circles):
            for j in range(n_circles):
                if i != j:
                    dx = points[i, 0] - points[j, 0]
                    dy = points[i, 1] - points[j, 1]
                    dist = max(1e-6, np.sqrt(dx*dx + dy*dy))

                    # Repulsive force (inverse square law)
                    force_magnitude = 1.0 / (dist * dist) if dist > 0 else 0
                    forces[i, 0] += force_magnitude * dx / dist
                    forces[i, 1] += force_magnitude * dy / dist

        # Apply forces with damping
        points += forces * 0.001

        # Keep within bounds
        points[:, 0] = np.clip(points[:, 0], 0.01, 0.99)
        points[:, 1] = np.clip(points[:, 1], 0.01, 0.99)

        # Check for convergence (very small movements)
        if np.max(np.abs(forces)) < 1e-8:
            break

    return points

def calculate_initial_radii(points: np.ndarray, n_circles: int) -> np.ndarray:
    """Calculate initial radii based on distance to nearest neighbors"""
    if n_circles <= 1:
        return np.full(n_circles, 0.05)

    # Compute pairwise distances
    distances = cdist(points, points, 'euclidean')

    # Set diagonal to large value to ignore self-distance
    np.fill_diagonal(distances, np.inf)

    # Find minimum distances to neighbors
    min_distances = np.min(distances, axis=1)

    # Calculate initial radii - inversely proportional to neighbor distances
    # but bounded by boundary constraints
    initial_radii = np.minimum(
        min_distances / 3.0,  # Half of min distance to nearest neighbor
        np.minimum(
            points[:, 0],  # Distance to left boundary
            np.minimum(
                1 - points[:, 0],  # Distance to right boundary
                np.minimum(
                    points[:, 1],  # Distance to bottom boundary
                    1 - points[:, 1]   # Distance to top boundary
                )
            )
        )
    )

    # Ensure minimum radius
    initial_radii = np.maximum(initial_radii, 0.005)

    # Add some randomness to break symmetry
    initial_radii *= (1 + np.random.normal(0, 0.1, n_circles))
    initial_radii = np.maximum(initial_radii, 0.005)

def calculate_initial_radii(points: np.ndarray, n_circles: int) -> np.ndarray:
    """Calculate initial radii using Voronoi-based approach for better distribution"""
    if n_circles <= 1:
        return np.full(n_circles, 0.05)

    # Compute pairwise distances
    distances = cdist(points, points, 'euclidean')

    # Set diagonal to large value to ignore self-distance
    np.fill_diagonal(distances, np.inf)

    # Find minimum distances to neighbors
    min_distances = np.min(distances, axis=1)

    # Calculate initial radii based on Voronoi cell areas - larger distances mean room for bigger radii
    # Use a more sophisticated approach:
    # - radii proportional to minimum neighbor distance (with safety factor)
    # - bounded by boundary constraints
    # - adjusted for local density
    ideal_radii = min_distances * 0.4  # Scale down to ensure non-overlap

    # Apply boundary constraints
    boundary_radii = np.minimum(
        points[:, 0],  # Distance to left boundary
        np.minimum(
            1 - points[:, 0],  # Distance to right boundary
            np.minimum(
                points[:, 1],  # Distance to bottom boundary
                1 - points[:, 1]   # Distance to top boundary
            )
        )
    )

    # Take minimum of ideal and boundary radii to ensure feasibility
    initial_radii = np.minimum(ideal_radii, boundary_radii * 0.9)

    # Ensure minimum radius
    initial_radii = np.maximum(initial_radii, 0.005)

    # Add some randomness to break symmetry and make solutions more diverse
    # Use log-normal distribution for more realistic spread
    initial_radii *= np.random.lognormal(0, 0.2, n_circles)
    initial_radii = np.maximum(initial_radii, 0.005)

    # Ensure we don't exceed reasonable limits
    initial_radii = np.minimum(initial_radii, 0.4)

    return initial_radii
