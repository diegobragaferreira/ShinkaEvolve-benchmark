    @staticmethod
    def mutate_individual(individual: np.ndarray, mutation_rate: float, generation: int, total_generations: int):
        """Mutate an individual with adaptive parameters"""
        # More aggressive mutation in early generations, less in later
        adaptive_mutation_rate = mutation_rate * (1 - generation / total_generations)**2

        for i in range(len(individual)):

                individual[i, 0] += np.random.normal(0, 0.015)

                # Keep within bounds
                individual[i, 0] = np.clip(individual[i, 0], 0.01, 0.99)
                individual[i, 1] = np.clip(individual[i, 1], 0.01, 0.99)

                # Mutate radius
                individual[i, 2] += np.random.normal(0, 0.008)
                individual[i, 2] = max(0.001, individual[i, 2])

    @staticmethod
    def differential_evolution_mutation(population: List[np.ndarray], target_idx: int,
                                      mutation_factor: float = 0.8) -> np.ndarray:
        """Apply differential evolution mutation to create a mutant vector"""
        # Select three random individuals different from target
        indices = list(range(len(population)))
        indices.remove(target_idx)
        a, b, c = random.sample(indices, 3)

        # Create mutant vector using DE/rand/1/bin strategy
        mutant = population[a].copy()

        # Apply differential mutation
        for i in range(len(mutant)):
            mutant[i, 0] = population[a][i, 0] + mutation_factor * (population[b][i, 0] - population[c][i, 0])
            mutant[i, 1] = population[a][i, 1] + mutation_factor * (population[b][i, 1] - population[c][i, 1])
            mutant[i, 2] = population[a][i, 2] + mutation_factor * (population[b][i, 2] - population[c][i, 2])

            # Ensure bounds
            mutant[i, 0] = np.clip(mutant[i, 0], 0.01, 0.99)
            mutant[i, 1] = np.clip(mutant[i, 1], 0.01, 0.99)
            mutant[i, 2] = max(0.001, mutant[i, 2])

        return mutant
