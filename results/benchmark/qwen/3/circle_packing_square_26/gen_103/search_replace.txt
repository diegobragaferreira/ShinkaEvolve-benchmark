<DESCRIPTION>
I'll implement progressive constraint relaxation to improve the optimization performance. The idea is to gradually tighten constraint penalties during evolution rather than applying them uniformly from the beginning. This allows for more aggressive exploration in early generations when the population is diverse, and enforces stricter constraints as the algorithm converges toward optimal solutions.

This approach helps avoid premature convergence and allows the algorithm to explore a wider search space initially before focusing on fine-tuning valid solutions.

    """Evaluates fitness with adaptive penalty system"""

    def __init__(self, boundary_weight: float = 1000.0, overlap_weight: float = 100000.0):
        self.boundary_weight = boundary_weight
        self.overlap_weight = overlap_weight
        self.penalty_weights = [boundary_weight, overlap_weight]

    def evaluate(self, circles: np.ndarray, spatial_indexer: Optional[SpatialIndexer] = None) -> float:
        """Evaluate fitness with constraint penalties"""
        # Check constraints
        if not ConstraintValidator.validate_containment(circles):
            penalty = self._compute_boundary_penalty(circles)
            return -penalty

        if not ConstraintValidator.validate_overlap(circles, spatial_indexer):
            penalty = self._compute_overlap_penalty(circles)
            return -penalty

        # Valid solution - return sum of radii
        return float(np.sum(circles[:, 2]))

    def _compute_boundary_penalty(self, circles: np.ndarray) -> float:
        """Compute penalty based on boundary violations"""
        penalty = 0.0

        for x, y, r in circles:
            # Calculate boundary violations
            if x - r < 0:
                penalty += abs(x - r) * self.boundary_weight
            elif x + r > 1:

            if y - r < 0:
                penalty += abs(y - r) * self.boundary_weight
            elif y + r > 1:
                penalty += abs(y + r - 1) * self.boundary_weight

        return penalty

        """Compute penalty based on overlap violations"""

class FitnessEvaluator:
    """Evaluates fitness with adaptive penalty system"""

    def __init__(self, boundary_weight: float = 1000.0, overlap_weight: float = 100000.0):
        self.boundary_weight = boundary_weight
        self.overlap_weight = overlap_weight

    def evaluate(self, circles: np.ndarray, spatial_indexer: Optional[SpatialIndexer] = None,
                generation: int = 0, total_generations: int = 100) -> float:
        """Evaluate fitness with constraint penalties that adapt based on generation progress"""
        # Progressive penalty scaling
        penalty_scale = 1.0 + (generation / total_generations) * 5.0

        # Check constraints
        if not ConstraintValidator.validate_containment(circles):
            penalty = self._compute_boundary_penalty(circles, penalty_scale)
            return -penalty

        if not ConstraintValidator.validate_overlap(circles, spatial_indexer):
            penalty = self._compute_overlap_penalty(circles, penalty_scale)
            return -penalty

        # Valid solution - return sum of radii
        return float(np.sum(circles[:, 2]))

    def _compute_boundary_penalty(self, circles: np.ndarray, penalty_scale: float = 1.0) -> float:
        """Compute penalty based on boundary violations"""
        penalty = 0.0

        for x, y, r in circles:
            # Calculate boundary violations
            if x - r < 0:
                penalty += (abs(x - r) ** 2) * self.boundary_weight * penalty_scale
            elif x + r > 1:
                penalty += (abs(x + r - 1) ** 2) * self.boundary_weight * penalty_scale
            if y - r < 0:
                penalty += (abs(y - r) ** 2) * self.boundary_weight * penalty_scale
            elif y + r > 1:
                penalty += (abs(y + r - 1) ** 2) * self.boundary_weight * penalty_scale
