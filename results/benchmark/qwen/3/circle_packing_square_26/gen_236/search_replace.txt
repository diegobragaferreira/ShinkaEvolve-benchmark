def compute_penalty(circles: np.ndarray, generation: int = 0, total_generations: int = 100,
                   population_fitness: float = 0.0, prev_fitness: float = 0.0) -> float:
    """Compute penalty based on constraint violations with adaptive scaling."""
    penalty = 0.0

    # Adaptive penalty scaling based on multiple factors
    # 1. Generation progress (linear as before)
    progress_factor = 1.0 + (generation / total_generations) * 5.0

    # 2. Progress toward benchmark
    benchmark_target = 2.6358627564136983
    current_best = population_fitness
    target_progress = min(1.0, current_best / benchmark_target)

    # 3. Population stability (how much fitness has changed)
    fitness_change = abs(population_fitness - prev_fitness) if prev_fitness > 0 else 0.0
    stability_factor = 1.0 + min(1.0, fitness_change * 10.0)

    # Combine factors for dynamic penalty scaling
    penalty_scale = progress_factor * (1.0 + target_progress * 2.0) * stability_factor

    # Check containment violations with scaled penalties
    for x, y, r in circles:
        # Boundary violations
        if x - r < 0:
            penalty += (abs(x - r) ** 2) * 10000 * penalty_scale
        elif x + r > 1:
            penalty += (abs(x + r - 1) ** 2) * 10000 * penalty_scale
        if y - r < 0:
            penalty += (abs(y - r) ** 2) * 10000 * penalty_scale
        elif y + r > 1:
            penalty += (abs(y + r - 1) ** 2) * 10000 * penalty_scale

    # Check overlap violations with scaled penalties
    if check_overlap_efficient(circles):
        penalty += 10000000.0 * penalty_scale

    return penalty
