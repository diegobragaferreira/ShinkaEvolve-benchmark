class SpatialGrid:
    """Efficient spatial grid for fast overlap detection"""

    def __init__(self, resolution: int = 20):
        self.resolution = resolution
        self.grid = {}
        self.cell_size = 1.0 / resolution
        self.neighbors_cache = {}

    def _get_cell_coords(self, x: float, y: float) -> Tuple[int, int]:
        """Get grid cell coordinates for a point"""
        return (int(x / self.cell_size), int(y / self.cell_size))

    def clear(self):
        """Clear the spatial grid"""
        self.grid.clear()
        self.neighbors_cache.clear()

    def add_circle(self, idx: int, x: float, y: float, radius: float):
        """Add a circle to the spatial grid"""
        cell_coords = self._get_cell_coords(x, y)
        if cell_coords not in self.grid:
            self.grid[cell_coords] = []
        self.grid[cell_coords].append((idx, x, y, radius))

    def get_neighbors(self, x: float, y: float, radius: float, use_cache: bool = True) -> List[Tuple[int, float, float, float]]:
        """Get all circles in neighboring cells that could potentially overlap"""

class MultiScaleSpatialGrid:
    """Multi-scale spatial grid for efficient overlap detection"""

    def __init__(self):
        # Multiple resolutions for different circle sizes
        self.resolutions = [10, 20, 40]
        self.grids = {}
        self.cell_sizes = {}
        self.neighbors_cache = {}

        # Precompute cell sizes for all resolutions
        for res in self.resolutions:
            self.cell_sizes[res] = 1.0 / res
            self.grids[res] = {}

    def _get_cell_coords(self, x: float, y: float, resolution: int) -> Tuple[int, int]:
        """Get grid cell coordinates for a point with specific resolution"""
        cell_size = self.cell_sizes[resolution]
        return (int(x / cell_size), int(y / cell_size))

    def clear(self):
        """Clear all grids and cache"""
        for grid in self.grids.values():
            grid.clear()
        self.neighbors_cache.clear()

    def add_circle(self, idx: int, x: float, y: float, radius: float):
        """Add a circle to all relevant grids"""
        for resolution in self.resolutions:
            cell_coords = self._get_cell_coords(x, y, resolution)
            if cell_coords not in self.grids[resolution]:
                self.grids[resolution][cell_coords] = []
            self.grids[resolution][cell_coords].append((idx, x, y, radius))

    def get_best_resolution(self, radius: float) -> int:
        """Choose appropriate resolution based on circle radius"""
        if radius < 0.02:
