        """Check overlap constraints efficiently using optimized spatial queries"""
        penalty = 0.0

        if len(individual) <= 1:
            return penalty

        # Phase 1: Use a spatial grid for coarse filtering
        grid_size = self.grid_size
        grid_cells = {}

        # Assign each circle to grid cells
        for i in range(len(individual)):
            x, y, r = individual[i]
            # Determine grid cell bounds for this circle
            min_gx = int((x - r) / grid_size)
            max_gx = int((x + r) / grid_size)
            min_gy = int((y - r) / grid_size)
            max_gy = int((y + r) / grid_size)

                        grid_cells[key] = []
                    grid_cells[key].append(i)

        checked_pairs = set()
