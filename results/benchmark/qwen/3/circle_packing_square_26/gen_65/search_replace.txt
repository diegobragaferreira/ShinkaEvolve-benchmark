def initialize_population(pop_size: int, n_circles: int) -> List[np.ndarray]:
    """Initialize population with improved Voronoi-based distribution"""
    population = []

    # Generate initial points using a more sophisticated approach:
    # 1. Generate points using a Poisson disk sampling approach for uniformity
    # 2. Then apply Voronoi cell centroids for better spread

    # Generate initial seed points using a systematic approach
    n_points = max(n_circles, 36)  # Use more points for better Voronoi coverage

    # Create a more uniform grid with some randomness
    grid_size = int(np.ceil(np.sqrt(n_points)))
    x_coords = np.linspace(0.05, 0.95, grid_size)
    y_coords = np.linspace(0.05, 0.95, grid_size)

    # Create grid points
    grid_points = []
    for x in x_coords:
        for y in y_coords:
            if len(grid_points) < n_points:
                grid_points.append([x, y])

    # Add some random points to ensure good distribution
    for _ in range(n_points - len(grid_points)):
        grid_points.append([random.uniform(0.05, 0.95), random.uniform(0.05, 0.95)])

    # Create initial population with better distributed points
    for _ in range(pop_size):
        # Select n_circles points from our generated points
        selected_points = random.sample(grid_points, n_circles)

        individual = np.zeros((n_circles, 3))

        # Assign positions with slight perturbation and appropriate radii
        for i in range(n_circles):
            x, y = selected_points[i]

            # Perturb slightly
            perturbation = random.uniform(-0.02, 0.02)
            individual[i, 0] = max(0.05, min(0.95, x + perturbation))
            individual[i, 1] = max(0.05, min(0.95, y + perturbation))

            # Assign radius with consideration for position
            # Circles near edges get smaller radii to avoid boundary issues
            margin = min(individual[i, 0], individual[i, 1], 1 - individual[i, 0], 1 - individual[i, 1])
            # Base radius inversely proportional to distance to edge
            base_radius = min(0.15, margin / 2.0)
            # Add some randomness to radius
            individual[i, 2] = max(0.01, base_radius * random.uniform(0.7, 1.3))

        population.append(individual)

    return population
