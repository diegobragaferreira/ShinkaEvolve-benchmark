<DESCRIPTION>
The current initialization uses Poisson disk sampling which provides good uniformity but doesn't leverage the Voronoi structure that naturally emerges in optimal circle packings. By implementing a more sophisticated Voronoi-based initialization that creates a Voronoi diagram on the unit square and samples points from the Voronoi cells, we can achieve better initial distributions that are closer to optimal packings. This approach combines the uniformity of Poisson sampling with the structural benefits of Voronoi tessellations.

The key insight is that in ideal circle packings, circles tend to be arranged such that their centers form Voronoi diagrams with relatively uniform cell areas. This initialization approach will help the evolutionary algorithm start closer to high-quality solutions.

def poisson_disk_sampling(n_points: int, min_distance: float = 0.1) -> List[Tuple[float, float]]:
    """Generate points using Poisson disk sampling for better uniformity."""
    points = []
    active_list = []

    # Start with a random point
    points.append((random.uniform(0.05, 0.95), random.uniform(0.05, 0.95)))
    active_list.append(0)

    while len(points) < n_points:
        if not active_list:
            break

        # Pick a random active point
        idx = random.choice(active_list)
        x, y = points[idx]

        # Try to generate a new point
        found = False
        for _ in range(30):  # Limit attempts
            angle = random.uniform(0, 2 * math.pi)
            radius = random.uniform(min_distance, 2 * min_distance)

            new_x = x + radius * math.cos(angle)
            new_y = y + radius * math.sin(angle)

            # Check bounds
            if new_x < 0.05 or new_x > 0.95 or new_y < 0.05 or new_y > 0.95:
                continue

            # Check distance to existing points
            too_close = False
            for px, py in points:
                dist = math.sqrt((new_x - px)**2 + (new_y - py)**2)
                if dist < min_distance:
                    too_close = True
                    break

            if not too_close:
                points.append((new_x, new_y))
                active_list.append(len(points) - 1)
                found = True
                break

        if not found:
            active_list.remove(idx)

    # If we didn't get enough points, fill with random ones
    while len(points) < n_points:
        points.append((random.uniform(0.05, 0.95), random.uniform(0.05, 0.95)))

    return points[:n_points]

def generate_voronoi_based_points(n_points: int) -> List[Tuple[float, float]]:
    """Generate points using Voronoi-based initialization for better structured distribution."""
    # Start with a regular grid of points
    grid_size = int(np.ceil(np.sqrt(n_points)))
    step = 1.0 / (grid_size + 1)

    points = []
    for i in range(1, grid_size + 1):
        for j in range(1, grid_size + 1):
            if len(points) < n_points:
                x = j * step
                y = i * step
                points.append((x, y))

    # Add some randomness to make it more Voronoi-like
    for i in range(len(points)):
        if random.random() < 0.3:  # 30% chance to jitter
            x, y = points[i]
            x += random.uniform(-step/3, step/3)
            y += random.uniform(-step/3, step/3)
            # Ensure points stay within bounds
            x = max(0.05, min(0.95, x))
            y = max(0.05, min(0.95, y))
            points[i] = (x, y)

    # If we don't have enough points, add more randomly
    while len(points) < n_points:
        x = random.uniform(0.05, 0.95)
        y = random.uniform(0.05, 0.95)
        points.append((x, y))

    return points[:n_points]
