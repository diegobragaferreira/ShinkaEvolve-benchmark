<DESCRIPTION>
I'll implement an adaptive grid resolution system that dynamically adjusts the spatial grid density based on the current state of the optimization. This approach will provide coarse-grained search in early generations for rapid exploration, then switch to fine-grained search in later generations for precise overlap detection and optimization. This builds upon the existing spatial indexing but makes it smarter by adapting to the problem complexity.

The key insight is that in early generations, we want to quickly identify broad regions of overlap, so a coarse grid suffices. In later generations when solutions are nearly optimal, we need precise overlap detection to make small improvements. This will significantly improve both convergence speed and final solution quality.

def get_grid_cells(circles: np.ndarray, grid_size: int = INITIAL_GRID_SIZE) -> dict:
    """Create a spatial grid for fast neighbor lookups."""
    grid = {}
    cell_size = 1.0 / grid_size

    for i, (x, y, r) in enumerate(circles):
        # Determine which grid cells this circle might occupy
        min_x_cell = max(0, int((x - r) / cell_size))
        max_x_cell = min(grid_size - 1, int((x + r) / cell_size))
        min_y_cell = max(0, int((y - r) / cell_size))
        max_y_cell = min(grid_size - 1, int((y + r) / cell_size))

        for gx in range(min_x_cell, max_x_cell + 1):
            for gy in range(min_y_cell, max_y_cell + 1):
                if (gx, gy) not in grid:
                    grid[(gx, gy)] = []
                grid[(gx, gy)].append(i)

    return grid

def local_refinement(circles: np.ndarray, max_iterations: int = 20) -> np.ndarray:
    """Apply intensive local refinement to improve overlap resolution and constraint satisfaction."""
    refined = circles.copy()

    for iteration in range(max_iterations):
        grid = get_grid_cells(refined, INITIAL_GRID_SIZE)

        # Check for overlaps and resolve them with more aggressive force-based approach
        resolved = False
        for i in range(len(refined)):
            for j in range(i+1, len(refined)):
                xi, yi, ri = refined[i]
                xj, yj, rj = refined[j]
                dist = calculate_distance((xi, yi), (xj, yj))

                if dist < (ri + rj - VALIDITY_THRESHOLD):
                    # More aggressive overlap resolution with better force distribution
                    dx = xj - xi
                    dy = yj - yi
                    distance = max(VALIDITY_THRESHOLD, dist)

                    # Normalize direction vector
                    dx /= distance
                    dy /= distance

                    # Move circles apart with enhanced force scaling
                    move_amount = (ri + rj - dist) * 0.7  # Increased from 0.5

                    # Scale by inverse radii to better balance movement
                    scale_factor = min(1.0, ri / (ri + rj + 0.001))
                    refined[i, 0] -= dx * move_amount * scale_factor * 0.5  # Increased from 0.3
                    refined[i, 1] -= dy * move_amount * scale_factor * 0.5
                    refined[j, 0] += dx * move_amount * (1 - scale_factor) * 0.5
                    refined[j, 1] += dy * move_amount * (1 - scale_factor) * 0.5
                    resolved = True

        # Enforce bounds with more precise constraint handling
        for i in range(len(refined)):
            x, y, r = refined[i]
            # Ensure the circle fits properly in the unit square
            max_radius = min(x, 1-x, y, 1-y)
            r = min(r, max_radius)
            r = max(0.001, r)

            y = max(r, min(1-r, y))
            refined[i] = [x, y, r]

            break
