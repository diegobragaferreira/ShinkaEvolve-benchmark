class SymmetryAwareMutation:
    """Custom mutation class that respects hexagon symmetry relationships."""

    def __init__(self):
        self.mutation_rate = 0.1
        self.max_displacement = 0.3
        self.rotation_variance = 5.0

    def mutate_symmetrically(self, individual, generation=0, max_generations=100):
        """Apply mutation that preserves symmetry relationships."""
        mutated = individual.copy()

        # Adaptively reduce mutation rates as optimization progresses
        current_mut_rate = self.mutation_rate * (0.9 ** generation)
        current_max_disp = self.max_displacement * (0.9 ** generation)

        # Apply mutations to 12 hexagons
        for i in range(12):
            if np.random.random() < current_mut_rate:
                # Get position and angle for this hexagon
                pos_idx = i * 3
                x, y, angle = mutated[pos_idx], mutated[pos_idx+1], mutated[pos_idx+2]

                # Mutate x and y coordinates with adaptive random displacements
                dx = np.random.normal(0, current_max_disp * 0.5)
                dy = np.random.normal(0, current_max_disp * 0.5)

                # Apply similar changes to related hexagons to maintain some symmetry
                # For hexagonal pattern, we can consider rotations and reflections
                d_angle = np.random.normal(0, self.rotation_variance * (0.9 ** generation))

                mutated[pos_idx] += dx
                mutated[pos_idx+1] += dy
                mutated[pos_idx+2] += d_angle

                # Clip to reasonable bounds to prevent extreme positions
                mutated[pos_idx] = np.clip(mutated[pos_idx], -8, 8)
                mutated[pos_idx+1] = np.clip(mutated[pos_idx+1], -8, 8)
                mutated[pos_idx+2] = mutated[pos_idx+2] % 360

        return mutated
