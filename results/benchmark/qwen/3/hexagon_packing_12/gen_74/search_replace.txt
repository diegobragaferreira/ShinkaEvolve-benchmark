import numpy as np


def hexagon_packing_12():
    """
    Constructs a packing of 12 disjoint unit regular hexagons inside a larger regular hexagon, maximizing 1/outer_hex_side_length.
    Returns
        inner_hex_data: np.ndarray of shape (12,3), where each row is of the form (x, y, angle_degrees) containing the (x,y) coordinates and angle_degree of the respective inner hexagon.
        outer_hex_data: np.ndarray of shape (3,) of form (x,y,angle_degree) containing the (x,y) coordinates and angle_degree of the outer hexagon.
        outer_hex_side_length: float representing the side length of the outer hexagon.
    """
    # Hexagon side length
    side_length = 1.0

    # Calculate hexagon height (distance between parallel sides)
    hex_height = side_length * np.sqrt(3)

    # Define positions in a hexagonal pattern
    # Center hexagon + 6 surrounding hexagons + 5 additional ones
    positions = []

    # Central hexagon
    positions.append((0.0, 0.0, 0.0))

    # First ring (6 hexagons around center)
    for i in range(6):
        angle = i * np.pi / 3
        x = hex_height * np.cos(angle)
        y = hex_height * np.sin(angle)
        positions.append((x, y, 0.0))

    # Second ring (additional hexagons)
    # We'll place them at positions that maintain good packing efficiency
    # Top hexagon (above center)
    positions.append((0.0, hex_height * 2, 0.0))

    # Bottom hexagon (below center)
    positions.append((0.0, -hex_height * 2, 0.0))

    # Left and right hexagons (at same vertical level as center)
    positions.append((-hex_height * 2, 0.0, 0.0))
    positions.append((hex_height * 2, 0.0, 0.0))

    # Diagonal positions to fill out to 12 total
    positions.append((-hex_height * 1.5, hex_height * 1.5, 0.0))
    positions.append((hex_height * 1.5, hex_height * 1.5, 0.0))
    positions.append((-hex_height * 1.5, -hex_height * 1.5, 0.0))
    positions.append((hex_height * 1.5, -hex_height * 1.5, 0.0))

    # Convert to numpy array
    inner_hex_data = np.array(positions[:12])

    # Determine outer hexagon size needed
    max_distance = 0.0

    # Calculate maximum distance from center to any vertex of any inner hexagon
    # For a unit hexagon, distance from center to vertex is 1
    # But we need to account for actual placement
    for i in range(12):
        x, y, _ = inner_hex_data[i]
        # Distance from origin to the center of this hexagon
        dist_to_center = np.sqrt(x*x + y*y)
        # Add the radius of the hexagon (distance from center to vertex)
        # For a unit hexagon, this is 1
        max_distance = max(max_distance, dist_to_center + 1.0)

    # Set outer hexagon side length to accommodate all inner hexagons
    # Add a small buffer for numerical precision
    outer_hex_side_length = max_distance * 1.05

    # Outer hexagon centered at origin
    outer_hex_data = np.array([0.0, 0.0, 0.0])

    return inner_hex_data, outer_hex_data, outer_hex_side_length
