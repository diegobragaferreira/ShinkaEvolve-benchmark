def optimize_with_multi_start_evolution():
    """Perform multi-start evolutionary optimization to improve convergence and avoid local optima."""
    max_time = 170  # Leave some buffer for final processing
    population_size = 20
    generations = 30
    elite_size = 4

    # Generate diverse initial population
    population = []

    # Add symmetric configurations
    for _ in range(4):
        population.append(generate_initial_symmetric_config())

    # Add random configurations
    for _ in range(4):
        population.append(generate_random_config())

    # Add specialized configurations
    # Configuration 1: Tight cluster arrangement
    config1 = generate_initial_symmetric_config().copy()
    for i in range(12):
        # Reduce spacing between hexagons to encourage tighter packing
        config1[i*3] *= 0.8
        config1[i*3 + 1] *= 0.8
    population.append(config1)

    # Configuration 2: More spread out arrangement
    config2 = generate_initial_symmetric_config().copy()
    for i in range(12):
        # Increase spacing to test wider arrangements
        config2[i*3] *= 1.2
        config2[i*3 + 1] *= 1.2
    population.append(config2)

    best_result = None
    best_objective = float('inf')

    # Evolutionary optimization
    for gen in range(generations):
        if time.time() - start_time > max_time:
            break

        print(f"Evolution generation {gen + 1}/{generations}")

        # Evaluate fitness of entire population
        fitness_scores = []
        for individual in population:
            objective_value = objective_function(individual)
            fitness_scores.append(objective_value)

        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]

        # Update best overall
        if fitness_scores[0] < best_objective:
            best_objective = fitness_scores[0]
            best_result = population[0].copy()

        # Early termination if we're approaching the benchmark
        if fitness_scores[0] < -0.24:  # Near the target
            break

        # Create new population through selection and reproduction
        new_population = population[:elite_size]  # Elitism

        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]

            # Select parent
            parent = population[winner_index]

            # Create offspring through mutation
            offspring = mutate_config(parent)

            # Apply bounds checking
            for i in range(len(offspring) - 1):  # Exclude outer side length
                if i % 3 == 0:  # x position
                    offspring[i] = np.clip(offspring[i], -10, 10)
                elif i % 3 == 1:  # y position
                    offspring[i] = np.clip(offspring[i], -10, 10)
                elif i % 3 == 2:  # angle
                    offspring[i] = offspring[i] % 360

            # Ensure outer side length is valid
            offspring[-1] = max(1.0, offspring[-1])

            new_population.append(offspring)

        population = new_population

    # Final refinement of best solution
    if best_result is not None:
        try:
            # Apply final optimization to the best solution found
            bounds_final = [(None, None)] * 36
            bounds_final.extend([(1.0, 15.0)])

            result_final = minimize(
                objective_function,
                best_result,
                method='L-BFGSB',
                bounds=bounds_final,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8},
                tol=1e-8
            )

            if result_final.success:
                best_result = result_final.x
        except Exception:
            pass  # Keep the previous best result if optimization fails

    return best_result
