def hexagon_packing_optimized():
    """Main optimized hexagon packing function using multi-start approach"""
    start_time = time.time()

    # Multiple initial configurations to try
    initial_configs = [
        generate_kagome_lattice_config(),      # Kagome lattice
        generate_hcp_lattice_config(),        # HCP lattice
        generate_triangular_arrangement(),    # Triangular arrangement
        generate_deterministic_initial_solution()  # Original deterministic
    ]

    best_config = None
    best_fitness = -float('inf')
    best_outer_radius = float('inf')

    # Try each initial configuration
    for i, initial_config in enumerate(initial_configs):
        try:
            # Apply constraint solving to improve the configuration
            refined_config = solve_constraint_equilibrium(initial_config)

            # Apply additional optimization if time allows
            if time.time() - start_time < MAX_EVAL_TIME - 10:
                final_config = solve_constraint_equilibrium(refined_config)
            else:
                final_config = refined_config

            # Evaluate the final configuration
            valid, msg = validate_solution_complete(final_config, [0, 0, 0])

            if valid:
                outer_radius = compute_outer_hexagon_radius(final_config)
                fitness = 1.0 / outer_radius if outer_radius > 0 else 0.0

                    best_config = final_config
                    best_outer_radius = outer_radius
