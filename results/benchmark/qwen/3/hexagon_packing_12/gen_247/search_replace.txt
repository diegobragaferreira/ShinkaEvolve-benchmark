class SymmetryAwareMutation:
    """Handles mutation strategies that preserve geometric symmetries"""

    @staticmethod
    def mutate_symmetrically(hex_data: np.ndarray, mutation_strength: float = 0.2, stage: int = 1) -> np.ndarray:
        """Apply advanced symmetric mutation to maintain hexagonal properties"""
        mutated_data = hex_data.copy()

        # Different mutation strategies based on optimization stage
        if stage == 1:  # Coarse stage - aggressive mutation for exploration
            mutation_factor = mutation_strength * 2.0
        elif stage == 2:  # Fine stage - moderate mutation
            mutation_factor = mutation_strength * 1.0
        else:  # Final stage - conservative mutation
            mutation_factor = mutation_strength * 0.5

        # Group hexagons based on their radial positions for better symmetry handling
        # Hexagon positions are grouped in rings: center (1), first ring (6), second ring (5)
        # This grouping allows us to apply symmetry operations more effectively

        # Mutate center hexagon
        mutated_data[0][0] += random.uniform(-mutation_factor, mutation_factor)
        mutated_data[0][1] += random.uniform(-mutation_factor, mutation_factor)

        # Mutate first ring hexagons in groups of three to maintain approximate rotational symmetry
        # Group 1: Top, top-right, bottom-right
        group1_indices = [1, 3, 5]  # Top, top-right, bottom-right
        group1_mutations = [random.uniform(-mutation_factor, mutation_factor) for _ in range(2)]

        for i, idx in enumerate(group1_indices):
            if i < len(group1_mutations):
                mutated_data[idx][0] += group1_mutations[i]
                mutated_data[idx][1] += group1_mutations[i]

        # Group 2: Bottom, bottom-left, top-left (opposite positions)
        group2_indices = [4, 6, 2]  # Bottom, bottom-left, top-left
        group2_mutations = [random.uniform(-mutation_factor, mutation_factor) for _ in range(2)]

        for i, idx in enumerate(group2_indices):
            if i < len(group2_mutations):
                mutated_data[idx][0] += group2_mutations[i]
                mutated_data[idx][1] += group2_mutations[i]

        # Mutate second ring hexagons with symmetric constraints
        # For second ring, we apply symmetric operations to maintain structural integrity
        second_ring_indices = [7, 8, 9, 10, 11]  # Far positions
        for i, idx in enumerate(second_ring_indices):
            # Apply a more controlled mutation respecting hexagonal symmetry
            if i % 2 == 0:  # Even indices get one type of variation
                mutated_data[idx][0] += random.uniform(-mutation_factor * 0.5, mutation_factor * 0.5)
                mutated_data[idx][1] += random.uniform(-mutation_factor * 0.5, mutation_factor * 0.5)

        # Mirror positions across x-axis (symmetry group 1)
