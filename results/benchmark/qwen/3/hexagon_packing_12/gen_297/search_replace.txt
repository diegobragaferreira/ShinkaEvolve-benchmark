def mutate_symmetrically(config, mutation_strength=0.1):
    """Apply symmetric mutation to maintain structural properties while exploring solution space"""
    # For 12 hexagons arranged in a hexagonal pattern, we can leverage rotational symmetry
    # The first 6 hexagons are typically arranged in a ring around the center
    # We'll apply mutations that preserve this symmetric structure

    mutated_config = config.copy()

    # Apply different mutation strategies based on position in the configuration
    for i in range(len(mutated_config)):
        if i == 0:  # Center hexagon - keep fixed or apply minor perturbation
            # Center hexagon is usually fixed, but allow small movement if beneficial
            mutated_config[i][0] += np.random.normal(0, mutation_strength * 0.1)
            mutated_config[i][1] += np.random.normal(0, mutation_strength * 0.1)
        elif i <= 6:  # First ring - preserve angular relationships
            # Apply rotation-invariant perturbations
            angle = np.arctan2(mutated_config[i][1], mutated_config[i][0])
            radius = np.sqrt(mutated_config[i][0]**2 + mutated_config[i][1]**2)

            # Mutate radius and angle independently
            radius *= (1 + np.random.normal(0, mutation_strength * 0.2))
            angle += np.random.normal(0, mutation_strength * 0.3)

            mutated_config[i][0] = radius * np.cos(angle)
            mutated_config[i][1] = radius * np.sin(angle)
        else:  # Second ring - maintain symmetric pattern
            # Apply similar mutation but with different strength
            mutated_config[i][0] += np.random.normal(0, mutation_strength * 0.5)
            mutated_config[i][1] += np.random.normal(0, mutation_strength * 0.5)

    return mutated_config
