        """Enhanced symmetry-aware mutation that better preserves hexagonal lattice symmetries"""
        mutated = individual.copy()

        # Exponential decay for mutation strength based on generation
        decay_factor = 0.95
        current_mutation = mutation_strength * (decay_factor ** generation)

        # Stage-specific scaling
        if stage == 1:  # Exploration - more aggressive
            current_mutation *= 2.0
        elif stage == 2:  # Exploitation - moderate
            current_mutation *= 1.0
        else:  # Refinement - conservative
            current_mutation *= 0.5

        # Group hexagons by radial distance for coordinated mutation
        # Central hexagon (index 0)
        # First ring (indices 1-6) - 6 positions forming a hexagon
        # Second ring (indices 7-11) - 5 positions roughly forming a larger hexagon

        # Mutate central hexagon
        mutated[0][0] += random.uniform(-current_mutation, current_mutation)
        mutated[0][1] += random.uniform(-current_mutation, current_mutation)

        # Mutate first ring with rotational symmetry preservation
        # First ring has 6 positions arranged in hexagonal pattern
        first_ring_indices = list(range(1, 7))

        # For better symmetry preservation, mutate in groups of 2 that are 180 degrees apart
        for i in range(0, len(first_ring_indices), 2):

            idx2 = first_ring_indices[i+1] if i+1 < len(first_ring_indices) else first_ring_indices[0]

            dx = random.uniform(-current_mutation, current_mutation) * 0.7  # Slightly reduce for coordination

            mutated[idx1][1] += dy
            mutated[idx2][0] += dx
            mutated[idx2][1] += dy

            # Apply coordinated rotation changes with some randomness
