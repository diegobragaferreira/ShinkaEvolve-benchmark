def mutate_symmetrically(individual, mut_pb=0.3, mut_strength=0.2):
    """
    Mutate an individual while preserving symmetry properties.
    For hexagon packing, this maintains the ring structure.
    """
    # Copy individual to avoid modifying original
    mutated = individual.copy()

    # Define symmetric groups
    # Group 0: Central hexagon (index 0)
    # Group 1: First ring (indices 1-6)
    # Group 2: Second ring (indices 7-11)

    # Mutate central hexagon
    if random.random() < mut_pb:
        mutated[0, 0] += random.uniform(-mut_strength, mut_strength)
        mutated[0, 1] += random.uniform(-mut_strength, mut_strength)
        mutated[0, 2] += random.uniform(-mut_strength, mut_strength)

    # Mutate first ring (6 hexagons) - preserve rotational symmetry within the ring
    if random.random() < mut_pb:
        # Get current ring positions to understand structure
        ring1_positions = mutated[1:7, :2]
        ring1_angles = mutated[1:7, 2]

        # Calculate average position and radius of first ring
        avg_x = np.mean(ring1_positions[:, 0])
        avg_y = np.mean(ring1_positions[:, 1])
        avg_radius = np.mean(np.sqrt((ring1_positions[:, 0] - avg_x)**2 + (ring1_positions[:, 1] - avg_y)**2))

        # Apply radial and angular perturbations to maintain ring structure
        radial_delta = random.uniform(-mut_strength * 0.5, mut_strength * 0.5)
        angular_delta = random.uniform(-mut_strength * 0.3, mut_strength * 0.3)

        # Update all ring hexagons with coordinated changes
        for i in range(1, 7):
            # Get original angle relative to center
            rel_x = ring1_positions[i-1, 0] - avg_x
            rel_y = ring1_positions[i-1, 1] - avg_y
            original_angle = np.arctan2(rel_y, rel_x)

            # Apply perturbations
            new_radius = max(0.1, avg_radius + radial_delta)
            new_angle = original_angle + angular_delta

            # Update position maintaining ring structure
            mutated[i, 0] = avg_x + new_radius * np.cos(new_angle)
            mutated[i, 1] = avg_y + new_radius * np.sin(new_angle)

            # Apply small angle variation
            mutated[i, 2] += random.uniform(-mut_strength * 0.2, mut_strength * 0.2)

    # Mutate second ring (6 hexagons) - preserve rotational symmetry within the ring
    if random.random() < mut_pb:
        # Get current ring positions to understand structure
        ring2_positions = mutated[7:12, :2]
        ring2_angles = mutated[7:12, 2]

        avg_x = np.mean(ring2_positions[:, 0])

        avg_radius = np.mean(np.sqrt((ring2_positions[:, 0] - avg_x)**2 + (ring2_positions[:, 1] - avg_y)**2))

        # Apply radial and angular perturbations to maintain ring structure
        radial_delta = random.uniform(-mut_strength * 0.5, mut_strength * 0.5)
        angular_delta = random.uniform(-mut_strength * 0.3, mut_strength * 0.3)

        # Update all ring hexagons with coordinated changes
        for i in range(7, 12):
            # Get original angle relative to center
            rel_x = ring2_positions[i-7, 0] - avg_x
            rel_y = ring2_positions[i-7, 1] - avg_y
            original_angle = np.arctan2(rel_y, rel_x)
