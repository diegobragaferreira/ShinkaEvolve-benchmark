class SymmetryAwareMutation:
    """Enhanced symmetry-aware mutation that leverages hexagonal lattice structure."""

    def __init__(self):
        self.initial_mut_rate = 0.1
        self.initial_max_dist = 0.5
        self.initial_angle_std = 10.0
        self.mutation_decay_factor = 0.9
        # Define hexagon rings for structured mutation
        self.ring_indices = [
            [0],               # Center hexagon
            list(range(1, 7)), # First ring (6 hexagons)
            list(range(7, 12)) # Second ring (5 hexagons, will be adjusted)
        ]

    def _get_ring_info(self, positions):
        """Analyze positions to determine which ring each hexagon belongs to."""
        ring_assignment = [0] * 12  # Default to center

        # Determine ring membership based on distance from center
        center_pos = positions[0, :2]  # Center hexagon position

        for i in range(1, 12):
            pos = positions[i, :2]
            distance = np.sqrt((pos[0] - center_pos[0])**2 + (pos[1] - center_pos[1])**2)
            # Assign to ring based on distance ranges
            if 1.5 <= distance <= 2.5:  # First ring
                ring_assignment[i] = 1
            elif distance > 3.0:  # Second ring
                ring_assignment[i] = 2

        return ring_assignment

    def _mutate_ring_positions(self, positions, ring_idx, current_max_dist, current_angle_std):
        """Mutate positions within a specified ring while preserving ring structure."""
        if ring_idx == 0:  # Center hexagon
            # Mutate center position
            positions[0, 0] += np.random.normal(0, current_max_dist * 0.3)
            positions[0, 1] += np.random.normal(0, current_max_dist * 0.3)
        elif ring_idx == 1:  # First ring (6 hexagons)
            # Get average position of ring 1 for symmetry preservation
            ring_positions = positions[1:7, :2]
            avg_x = np.mean(ring_positions[:, 0])
            avg_y = np.mean(ring_positions[:, 1])

            for i in range(1, 7):
                # Apply coordinated displacement to maintain ring structure
                dx = np.random.normal(0, current_max_dist * 0.5)

                positions[i, 0] += dx
                positions[i, 1] += dy

                # Apply slight rotational consistency

                positions[i, 0] = avg_x + (positions[i, 0] - avg_x) * np.cos(angle_to_center * 0.1) - \
                                 (positions[i, 1] - avg_y) * np.sin(angle_to_center * 0.1)
                positions[i, 1] = avg_y + (positions[i, 0] - avg_x) * np.sin(angle_to_center * 0.1) + \
                                 (positions[i, 1] - avg_y) * np.cos(angle_to_center * 0.1)
        elif ring_idx == 2:  # Second ring (5 hexagons)
            # Get average position of ring 2 for symmetry preservation
            ring_positions = positions[7:12, :2]
            avg_x = np.mean(ring_positions[:, 0])

            # Mutate all ring 2 hexagons with coordinated movement
            for i in range(7, 12):

                dx = np.random.normal(0, current_max_dist * 0.4)

    def mutate_symmetrically(self, individual, generation=0, max_generations=100):
