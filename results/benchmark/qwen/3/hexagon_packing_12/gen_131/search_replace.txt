def create_symmetric_initial_guess():
    """Create a highly symmetric initial guess for the 12-hexagon packing"""
    # Following a known optimal symmetric structure
    # This is based on mathematical analysis of optimal packings
    sqrt3 = np.sqrt(3)

    # Base ring of 6 hexagons around center
    base_ring = np.array([
        [0.0, 0.0, 0],         # Center
        [0.0, 2.0, 0],         # Top
        [sqrt3, 1.0, 0],       # Top right
        [sqrt3, -1.0, 0],      # Bottom right
        [0.0, -2.0, 0],        # Bottom
        [-sqrt3, -1.0, 0],     # Bottom left
        [-sqrt3, 1.0, 0],      # Top left
    ], dtype=float)

    # Outer ring of 6 hexagons
    outer_ring = np.array([
        [2*sqrt3, 2.0, 0],     # Far top right
        [2*sqrt3, -2.0, 0],    # Far bottom right
        [0.0, -4.0, 0],        # Far bottom
        [-2*sqrt3, -2.0, 0],   # Far bottom left
        [-2*sqrt3, 2.0, 0],    # Far top left
        [0.0, 4.0, 0],         # Far top
    ], dtype=float)

    # Combine all 12 hexagons
    inner_hex_data = np.vstack([base_ring, outer_ring])

    return inner_hex_data

def estimate_outer_radius(inner_hex_data):
    """Estimate outer hexagon radius based on inner hexagon positions"""
    # Calculate maximum distance from center to any vertex of any hexagon
    max_dist = 0
    for i in range(len(inner_hex_data)):
        x, y, angle = inner_hex_data[i]
        # For a unit hexagon, the distance from center to vertex is 1
        distance = np.sqrt(x*x + y*y) + 1  # +1 for the radius of the hexagon
        max_dist = max(max_dist, distance)
    return max_dist * 1.2  # Add safety margin
