    def simulated_annealing_acceptance(self, old_fitness, new_fitness, temperature):
        """Accept new solution based on simulated annealing criterion."""
        if new_fitness > old_fitness:
            return True
        # Accept with probability based on temperature
        acceptance_prob = math.exp((new_fitness - old_fitness) / temperature)
        return random.random() < acceptance_prob

    def local_search_refinement(self, config):
        """Apply local search refinement to improve a configuration."""

        # Simple local search: try small perturbations and see if they improve fitness
        current_validity, current_fitness = self.evaluate_configuration(refined_config, 0.0, 0.0)
        if not current_validity:
            return config

        # Try several small random perturbations
        best_config = refined_config.copy()
        best_fitness = current_fitness

        # Try various small perturbations
        for _ in range(10):  # 10 perturbation attempts
            # Make small random perturbation
            temp_config = refined_config.copy()
            for i in range(len(temp_config)):
                # Small perturbations to positions
                temp_config[i][0] += random.uniform(-0.1, 0.1)
                temp_config[i][1] += random.uniform(-0.1, 0.1)

            # Evaluate the perturbed configuration

            # If valid and better, accept it
            if validity and fitness > best_fitness:
                best_config = temp_config.copy()
                best_fitness = fitness

        return best_config
