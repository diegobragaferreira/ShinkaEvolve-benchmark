        # Define symmetric groups:
        # Group 0: Central hexagon (index 0)

        # Mutate central hexagon

    def run_evolutionary_optimization(self):
        """Main evolutionary optimization loop."""
        # Generate initial population
        initial_configs = self.generate_multiple_initial_configs()
        population = initial_configs[:]

        best_config = None
        best_inv_radius = 0
        best_time = 0

        # Evolutionary parameters
        generations = 20
        population_size = len(population)
        mutation_rate = 0.3
        elite_count = 2

        for gen in range(generations):
            if self.get_current_time() > self.max_time * 0.95:
                break

            print(f"Evolutionary generation {gen+1}/{generations}")

            # Evaluate current population
            fitness_scores = []
            for config in population:
                validity, inv_radius = self.evaluate_configuration(config, 0.0, 0.0)
                if validity:
                    fitness_scores.append((config, inv_radius))
                else:
                    fitness_scores.append((config, 0))  # Invalid configurations get low fitness

            # Sort by fitness (descending)
            fitness_scores.sort(key=lambda x: x[1], reverse=True)

            # Keep track of best solution
            if fitness_scores[0][1] > best_inv_radius:
                best_inv_radius = fitness_scores[0][1]
                best_config = fitness_scores[0][0].copy()
                best_time = self.get_current_time()

            # Select top individuals (elite)
            elites = [ind for ind, _ in fitness_scores[:elite_count]]

            # Create new population through crossover and mutation
            new_population = elites[:]  # Keep elites

            # Fill rest of population with mutated versions
            while len(new_population) < population_size:
                if len(elites) > 0:
                    # Select parent from elites
                    parent = random.choice(elites)

                    # Apply symmetry-aware mutation
                    child = self.symmetry_aware_mutation(parent, mut_pb=mutation_rate, mut_strength=0.3)

                    # Occasionally add some random perturbations to increase diversity
                    if random.random() < 0.2:  # 20% chance of random perturbation
                        for i in range(len(child)):
                            if random.random() < 0.5:
                                child[i][0] += random.uniform(-0.2, 0.2)
                            if random.random() < 0.5:
                                child[i][1] += random.uniform(-0.2, 0.2)

                    new_population.append(child)
                else:
                    # Fallback to generating completely new configuration
                    new_population.append(self.generate_initial_config())

            population = new_population[:population_size]

        return best_config if best_config is not None else population[0], best_inv_radius
