import numpy as np
from shapely.geometry import Polygon, Point

def hexagon_vertices(center_x, center_y, side_length=1, angle_deg=0):
    """Generate vertices of a regular hexagon given center, side length, and rotation."""
    angle_rad = np.deg2rad(angle_deg)
    angles = np.linspace(0, 2*np.pi, 7) + angle_rad
    vertices = np.array([
        [center_x + side_length * np.cos(a),
         center_y + side_length * np.sin(a)]
        for a in angles
    ])
    return vertices

def evaluate_packing(inner_hex_data, outer_hex_side_length):
    """Evaluate if the packing meets constraints and compute objective."""
    try:
        # Generate vertices for all inner hexagons
        hex_polygons = []
        for i in range(12):
            x, y, angle = inner_hex_data[i]
            vertices = hexagon_vertices(x, y, 1.0, angle)
            hex_polygons.append(Polygon(vertices))

        # Check for overlaps between hexagons
        for i in range(12):
            for j in range(i+1, 12):
                if hex_polygons[i].intersects(hex_polygons[j]):
                    return False, 0

        # Create outer hexagon
        outer_vertices = hexagon_vertices(0, 0, outer_hex_side_length, 0)
        outer_polygon = Polygon(outer_vertices)

        # Check containment
        for i in range(12):
            for vertex in hex_polygons[i].exterior.coords:
                point = Point(vertex[0], vertex[1])
                if not outer_polygon.contains(point):
                    return False, 0

        # If we reach here, packing is valid
        # Calculate objective (1/outer_radius)
        return True, 1.0 / outer_hex_side_length

    except Exception as e:
        return False, 0

class AdaptiveMutationOperator:
    """Mutation operator with adaptive variance that decreases over generations."""

    def __init__(self, initial_variance=0.5, decay_factor=0.95):
        self.initial_variance = initial_variance
        self.decay_factor = decay_factor

    def mutate(self, individual, generation, max_generations):
        """Mutate individual with adaptive variance."""
        mutated = individual.copy()

        # Calculate adaptive variance
        current_variance = self.initial_variance * (self.decay_factor ** generation)
        current_variance = max(current_variance, 0.01)  # Minimum variance

        # Mutate each position and angle
        for i in range(12):
            # Mutate x-coordinate
            mutated[i, 0] += np.random.normal(0, current_variance)
            # Mutate y-coordinate
            mutated[i, 1] += np.random.normal(0, current_variance)
            # Mutate angle (smaller changes for angles)
            mutated[i, 2] += np.random.normal(0, current_variance * 0.5)

        return mutated

def optimize_single_config(base_config, initial_side_length=4.0, max_iter=100):
    """Optimize a single configuration using local optimization."""

    # Flatten the configuration for optimization
    flat_config = base_config.flatten()
    flat_config = np.append(flat_config, initial_side_length)

    def objective_func(config):
        # Unflatten config
        positions = config[:-1].reshape(12, 3)
        side_length = config[-1]

        # Penalize if side length is too small
        if side_length < 1.0:
            return 1e10

        # Evaluate
        is_valid, objective_value = evaluate_packing(positions, side_length)
        if not is_valid:
            return 1e10

        return -objective_value  # Negative because we want to maximize

    # Bounds for optimization (positions: [-10, 10], angles: [0, 360], side length: [1, 10])
    bounds = [(-10, 10)] * 24 + [(1.0, 10.0)]

    try:
        result = minimize(objective_func, flat_config, method='L-BFGSB', bounds=bounds,
                         options={'maxiter': max_iter, 'ftol': 1e-8})
        if result.success:
            final_positions = result.x[:-1].reshape(12, 3)
            final_side_length = result.x[-1]
            return final_positions, final_side_length, True
    except:
        pass

    return base_config, initial_side_length, False
