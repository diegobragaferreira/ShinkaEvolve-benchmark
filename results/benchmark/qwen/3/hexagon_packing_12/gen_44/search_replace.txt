def create_unit_hexagon(center=(0, 0), rotation=0):
    """Create a unit regular hexagon with given center and rotation."""
    angle = rotation * math.pi / 180
    # Vertices of unit hexagon centered at origin with one vertex on positive x-axis
    vertices = []
    for i in range(6):
        theta = angle + i * math.pi / 3
        x = math.cos(theta)
        y = math.sin(theta)
        vertices.append((x + center[0], y + center[1]))
    return Polygon(vertices)


def hexagon_vertices(center, rotation, side_length=1):
    """Get vertices of a hexagon with given center, rotation, and side length."""
    angle = rotation * math.pi / 180
    vertices = []
    for i in range(6):
        theta = angle + i * math.pi / 3
        x = side_length * math.cos(theta) + center[0]
        y = side_length * math.sin(theta) + center[1]
        vertices.append((x, y))
    return vertices

    Compute the minimum radius needed for outer hexagon to contain all inner hexagons.
