import numpy as np

def create_regular_hexagon(center=(0, 0), side_length=1, rotation=0):
    """Create a regular hexagon as a Shapely polygon."""
    angle = rotation * np.pi / 180
    points = []
    for i in range(6):
        theta = angle + i * np.pi / 3
        x = center[0] + side_length * np.cos(theta)
        y = center[1] + side_length * np.sin(theta)
        points.append((x, y))
    return Polygon(points)


def get_hexagon_vertices(center, side_length, rotation):
    """Get vertices of a hexagon."""
    angle = rotation * np.pi / 180
    vertices = []
    for i in range(6):
        theta = angle + i * np.pi / 3
        x = center[0] + side_length * np.cos(theta)
        y = center[1] + side_length * np.sin(theta)
        vertices.append((x, y))
    return np.array(vertices)


def check_containment(hexagon, outer_hexagon):
    """Check if hexagon is fully contained in outer hexagon."""
    h_poly = Polygon(hexagon)
    outer_poly = Polygon(outer_hexagon)
    return outer_poly.contains(h_poly)


def check_overlap(hex1, hex2):
    """Check if two hexagons overlap."""
    h1_poly = Polygon(hex1)
    h2_poly = Polygon(hex2)
    return h1_poly.intersects(h2_poly)


def compute_outer_hexagon_side_length(inner_hex_data, scale=1.0):
    """Estimate the minimum side length of outer hexagon needed."""
    # Get all vertices of all inner hexagons
    all_vertices = []
    for i in range(len(inner_hex_data)):
        center = (inner_hex_data[i][0], inner_hex_data[i][1])
        rotation = inner_hex_data[i][2]
        vertices = get_hexagon_vertices(center, 1, rotation)
        all_vertices.extend(vertices)

    # Find bounding box and extend slightly
    if len(all_vertices) == 0:
        return 1.0

    all_vertices = np.array(all_vertices)
    min_x, max_x = all_vertices[:, 0].min(), all_vertices[:, 0].max()
    min_y, max_y = all_vertices[:, 1].min(), all_vertices[:, 1].max()

    # Calculate distance from center to outermost point
    center_x, center_y = (min_x + max_x) / 2, (min_y + max_y) / 2
    distances = np.sqrt((all_vertices[:, 0] - center_x)**2 + (all_vertices[:, 1] - center_y)**2)

    # Add some margin for safety
    max_distance = distances.max() * 1.1

    # Convert to side length of circumscribed hexagon
    # For a hexagon with circumradius r, side length = r
    return max_distance


def evaluate_configuration(inner_hex_data):
    """Evaluate how well a configuration meets constraints."""
    # Check containment and overlaps
    num_overlaps = 0
    num_out_of_bounds = 0

    # Create reference outer hexagon (centered at origin with calculated size)
    outer_side_length = compute_outer_hexagon_side_length(inner_hex_data)
    outer_hex = get_hexagon_vertices((0, 0), outer_side_length, 0)

    # Check each hexagon
    for i in range(len(inner_hex_data)):
        center = (inner_hex_data[i][0], inner_hex_data[i][1])
        rotation = inner_hex_data[i][2]
        hexagon = get_hexagon_vertices(center, 1, rotation)

        # Check containment
        if not check_containment(hexagon, outer_hex):
            num_out_of_bounds += 1

        # Check overlaps with others
        for j in range(i+1, len(inner_hex_data)):
            center2 = (inner_hex_data[j][0], inner_hex_data[j][1])
            rotation2 = inner_hex_data[j][2]
            hexagon2 = get_hexagon_vertices(center2, 1, rotation2)

            if check_overlap(hexagon, hexagon2):
                num_overlaps += 1

    # Return negative because we want to minimize
    return -(num_overlaps + num_out_of_bounds), outer_side_length
