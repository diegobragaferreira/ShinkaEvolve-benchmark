    @staticmethod
    @jit(nopython=True, fastmath=True)
    def fast_calculate_outer_radius(hex_data: np.ndarray) -> float:
        """Fast calculation of outer radius using numba JIT"""
        max_dist_sq = 0.0
        for i in range(hex_data.shape[0]):
            x = hex_data[i, 0]
            y = hex_data[i, 1]
            dist_sq = x * x + y * y
            if dist_sq > max_dist_sq:
                max_dist_sq = dist_sq
        return np.sqrt(max_dist_sq) + 0.1

    @staticmethod
    @jit(nopython=True, fastmath=True)
    def fast_check_overlap_positions(pos1_x: float, pos1_y: float, pos2_x: float, pos2_y: float) -> bool:
        """Fast overlap check for two hexagon centers"""
        # Hexagons overlap if their centers are less than 2 units apart
        dx = pos1_x - pos2_x
        dy = pos1_y - pos2_y
        distance_sq = dx * dx + dy * dy
