import numpy as np
from shapely.geometry import Polygon
from shapely.ops import unary_union
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import random
from numba import jit

# Global constants
UNIT_HEX_RADIUS = 1.0
MAX_EVAL_TIME = 180.0
TARGET_RATIO = 0.2537
FORCE_CONSTANT = 10.0
BOUNDARY_FORCE_CONSTANT = 50.0
OVERLAP_PENALTY = 1000.0
RESTITUTION = 0.8

@jit(nopython=True)
def get_hexagon_vertices(x, y, angle_deg, radius=1.0):
    """Get vertices of a hexagon given center, angle, and radius"""
    vertices = np.zeros((6, 2))
    angle_rad = np.radians(angle_deg)
    for i in range(6):
        theta = angle_rad + i * np.pi / 3
        vertices[i] = [x + radius * np.cos(theta), y + radius * np.sin(theta)]
    return vertices

def create_regular_hexagon(center=(0,0), side_length=1, rotation=0):
    """Create a regular hexagon as a Shapely polygon"""
    angles = np.linspace(0, 2*np.pi, 7) + np.radians(rotation)
    x = center[0] + side_length * np.cos(angles)
    y = center[1] + side_length * np.sin(angles)
    return Polygon(list(zip(x, y)))

def calculate_overlap_area(hex1, hex2):
    """Calculate the overlap area between two hexagons"""
    intersection = hex1.intersection(hex2)
    return intersection.area if intersection.is_valid else 0.0

def hexagon_to_polygon(x, y, angle_deg, radius=1.0):
    """Convert hexagon parameters to shapely polygon"""
    vertices = get_hexagon_vertices(x, y, angle_deg, radius)
    return Polygon(vertices)

def compute_outer_hexagon_radius(inner_hex_data):
    """Compute minimum outer hexagon radius that contains all inner hexagons"""
    if len(inner_hex_data) == 0:
        return 0.0
    
    # Get all vertices of all inner hexagons
    all_vertices = []
    for i in range(len(inner_hex_data)):
        x, y, angle = inner_hex_data[i]
        vertices = get_hexagon_vertices(x, y, angle)
        all_vertices.extend(vertices)
    
    if len(all_vertices) == 0:
        return 0.0
    
    # Compute centroid
    centroid_x = np.mean([v[0] for v in all_vertices])
    centroid_y = np.mean([v[1] for v in all_vertices])
    
    # Find maximum distance from centroid to any vertex
    max_distance = 0.0
    for x, y in all_vertices:
        distance = np.sqrt((x - centroid_x)**2 + (y - centroid_y)**2)
        max_distance = max(max_distance, distance)
    
    # Add buffer for hexagon radius calculation
    return max_distance + UNIT_HEX_RADIUS

def compute_forces_on_hexagon(hexagons, idx, outer_hexagon):
    """Compute net forces acting on a hexagon due to overlaps and boundaries"""
    forces = np.array([0.0, 0.0])  # [fx, fy]
    
    x, y, angle = hexagons[idx]
    hexagon = hexagon_to_polygon(x, y, angle)
    
    # Force from overlaps with other hexagons
    for i in range(len(hexagons)):
        if i != idx:
            other_x, other_y, other_angle = hexagons[i]
            other_hexagon = hexagon_to_polygon(other_x, other_y, other_angle)
            
            overlap_area = calculate_overlap_area(hexagon, other_hexagon)
            if overlap_area > 1e-8:  # If there's overlap
                # Calculate overlap center
                intersection = hexagon.intersection(other_hexagon)
                if intersection.geom_type == 'Polygon':
                    overlap_centroid = intersection.centroid
                    # Repulsion force (away from overlap center)
                    force_direction = np.array([x, y]) - np.array([overlap_centroid.x, overlap_centroid.y])
                    force_magnitude = FORCE_CONSTANT * overlap_area
                    if np.linalg.norm(force_direction) > 1e-10:
                        force_direction = force_direction / np.linalg.norm(force_direction)
                        forces += force_magnitude * force_direction
    
    # Force from boundary constraint
    # Project the hexagon center to the boundary
    boundary_dist = outer_hexagon.distance(hexagon)
    if boundary_dist < 1e-6:  # Hexagon is outside boundary
        # Push it back using boundary normal
        center_point = hexagon.centroid
        boundary_normal = np.array([center_point.x, center_point.y]) - np.array([0, 0])
        boundary_normal = boundary_normal / (np.linalg.norm(boundary_normal) + 1e-10)
        force_magnitude = BOUNDARY_FORCE_CONSTANT * (1.0 - boundary_dist)
        forces += force_magnitude * boundary_normal
    
    return forces

def evaluate_total_energy(hexagons, outer_hexagon):
    """Evaluate total energy of the system (potential + kinetic)"""
    total_energy = 0.0
    
    # Overlap energy
    for i in range(len(hexagons)):
        for j in range(i + 1, len(hexagons)):
            x1, y1, angle1 = hexagons[i]
            x2, y2, angle2 = hexagons[j]
            hex1 = hexagon_to_polygon(x1, y1, angle1)
            hex2 = hexagon_to_polygon(x2, y2, angle2)
            
            overlap = calculate_overlap_area(hex1, hex2)
            if overlap > 1e-8:
                total_energy += OVERLAP_PENALTY * overlap
    
    # Boundary penetration energy
    for i in range(len(hexagons)):
        x, y, angle = hexagons[i]
        hexagon = hexagon_to_polygon(x, y, angle)
        dist = outer_hexagon.distance(hexagon)
        if dist < 0:  # Penetration
            total_energy += OVERLAP_PENALTY * (-dist)
    
    return total_energy

def simulate_hexagon_system(hexagons, outer_hexagon, steps=1000):
    """Run physics-based simulation to optimize hexagon placement"""
    # Create initial system state
    velocities = np.zeros_like(hexagons[:, :2])
    
    for step in range(steps):
        # Calculate forces on each hexagon
        forces = []
        for i in range(len(hexagons)):
            force = compute_forces_on_hexagon(hexagons, i, outer_hexagon)
            forces.append(force)
        
        # Update velocities and positions
        for i in range(len(hexagons)):
            # Apply forces to velocity
            velocities[i] += forces[i] * 0.01
            
            # Apply damping
            velocities[i] *= RESTITUTION
            
            # Update position
            hexagons[i, 0] += velocities[i, 0] * 0.01
            hexagons[i, 1] += velocities[i, 1] * 0.01
    
    return hexagons

def validate_solution(inner_hex_data, outer_hex_data):
    """Validate that solution is valid"""
    if len(inner_hex_data) != 12:
        return False, "Wrong number of hexagons"
    
    # Create outer hexagon
    outer_x, outer_y, outer_angle = outer_hex_data
    outer_radius = compute_outer_hexagon_radius(inner_hex_data)
    outer_hex = hexagon_to_polygon(outer_x, outer_y, outer_angle, outer_radius)
    
    # Check each inner hexagon
    for i in range(len(inner_hex_data)):
        x, y, angle = inner_hex_data[i]
        inner_hex = hexagon_to_polygon(x, y, angle)
        
        # Check containment
        if not outer_hex.contains(inner_hex):
            return False, f"Inner hexagon {i} not contained"
        
        # Check overlaps with others
        for j in range(i+1, len(inner_hex_data)):
            x2, y2, angle2 = inner_hex_data[j]
            inner_hex2 = hexagon_to_polygon(x2, y2, angle2)
            
            overlap = calculate_overlap_area(inner_hex, inner_hex2)
            if overlap > 1e-8:
                return False, f"Overlapping hexagons {i} and {j}"
    
    return True, "Valid solution"

def generate_symmetric_initial_solution():
    """Generate a highly symmetric initial solution"""
    # Hexagonal close packing arrangement with some variation
    hex_data = []
    
    # Center hexagon
    hex_data.append([0.0, 0.0, 0.0])
    
    # First ring (6 hexagons)
    for i in range(6):
        angle = i * 60  # degrees
        rad = np.radians(angle)
        x = 2.0 * np.cos(rad)
        y = 2.0 * np.sin(rad)
        hex_data.append([x, y, 0.0])
    
    # Second ring (6 hexagons)
    for i in range(6):
        angle = i * 60 + 30  # degrees (offset)
        rad = np.radians(angle)
        x = 3.464 * np.cos(rad)  # approx sqrt(12)
        y = 3.464 * np.sin(rad)
        hex_data.append([x, y, 0.0])
    
    # Ensure exactly 12 hexagons
    while len(hex_data) < 12:
        hex_data.append([0.0, 0.0, 0.0])
    hex_data = hex_data[:12]
    
    # Add slight random noise for better exploration
    for i in range(12):
        hex_data[i][0] += random.uniform(-0.1, 0.1)
        hex_data[i][1] += random.uniform(-0.1, 0.1)
    
    return np.array(hex_data)

def improve_with_gradient_descent(initial_hex_data, outer_hexagon):
    """Use gradient descent to refine the hexagon arrangement"""
    def objective_and_gradient(params):
        # Reshape hex data
        hex_data = params.reshape(12, 3)
        # Compute total energy
        energy = evaluate_total_energy(hex_data, outer_hexagon)
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(params)
        eps = 1e-6
        for i in range(len(params)):
            params_plus = params.copy()
            params_plus[i] += eps
            energy_plus = evaluate_total_energy(params_plus.reshape(12, 3), outer_hexagon)
            grad[i] = (energy_plus - energy) / eps
        return energy, grad
    
    # Flatten for optimization
    flat_start = initial_hex_data.flatten()
    
    # Use scipy optimization
    result = minimize(
        lambda x: objective_and_gradient(x)[0],
        flat_start,
        method='BFGS',
        jac=lambda x: objective_and_gradient(x)[1],
        options={'maxiter': 500, 'gtol': 1e-6},
        tol=1e-6
    )
    
    refined = result.x.reshape(12, 3)
    return refined

def hexagon_packing_12():
    """
    Constructs a packing of 12 disjoint unit regular hexagons inside a larger regular hexagon, maximizing 1/outer_hex_side_length.
    Returns
        inner_hex_data: np.ndarray of shape (12,3), where each row is of the form (x, y, angle_degrees) containing the (x,y) coordinates and angle_degree of the respective inner hexagon.
        outer_hex_data: np.ndarray of shape (3,) of form (x,y,angle_degree) containing the (x,y) coordinates and angle_degree of the outer hexagon.
        outer_hex_side_length: float representing the side length of the outer hexagon.
    """
    start_time = time.time()
    
    # Generate initial symmetric solution
    initial_hex_data = generate_symmetric_initial_solution()
    
    # Create outer hexagon - initially large enough to contain everything
    outer_radius = compute_outer_hexagon_radius(initial_hex_data)
    outer_hexagon = hexagon_to_polygon(0, 0, 0, outer_radius)
    
    # Simulate physics-based system
    optimized_hex_data = simulate_hexagon_system(initial_hex_data.copy(), outer_hexagon, steps=500)
    
    # Fine tune with gradient descent
    refined_hex_data = improve_with_gradient_descent(optimized_hex_data, outer_hexagon)
    
    # Final validation
    valid, msg = validate_solution(refined_hex_data, [0, 0, 0])
    
    # If we're still not valid, fall back to a known good configuration
    if not valid:
        # Try another arrangement that is known to work
        fallback_config = np.array([
            [0, 0, 0],              # center
            [-2.5, 0, 0],           # left
            [2.5, 0, 0],            # right
            [-1.25, 2.17, 0],       # top-left
            [1.25, 2.17, 0],        # top-right
            [-1.25, -2.17, 0],      # bottom-left
            [1.25, -2.17, 0],       # bottom-right
            [-3.75, 2.17, 0],       # far top-left
            [3.75, 2.17, 0],        # far top-right
            [-3.75, -2.17, 0],      # far bottom-left
            [3.75, -2.17, 0],       # far bottom-right
            [0, -4, 0],             # far bottom-center
        ])
        return fallback_config, np.array([0, 0, 0]), 8.0
    
    # Compute final outer hexagon radius
    final_radius = compute_outer_hexagon_radius(refined_hex_data)
    
    return refined_hex_data, np.array([0, 0, 0]), final_radius
