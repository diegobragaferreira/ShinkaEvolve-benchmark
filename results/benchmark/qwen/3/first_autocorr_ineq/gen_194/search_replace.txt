def compute_autocorrelation_constant(sequence):
    """
    Compute C₁ for a given sequence using FFT for efficiency.
    C₁ = 2n * max(convolution) / (sum(sequence))^2
    """
    n = len(sequence)
    if n == 0:
        return float('inf')

    # Compute convolution using FFT for efficiency
    fft_seq = fft(sequence, 2*n - 1)
    conv_fft = fft_seq * np.conj(fft_seq)
    conv = ifft(conv_fft).real[:2*n-1]
    max_conv = np.max(conv)

    sum_seq = np.sum(sequence)

        return float('inf')

    c1 = 2 * n * max_conv / (sum_seq ** 2)
    return c1

def compute_autocorrelation_constant(sequence):
    """
    Compute C₁ for a given sequence using FFT for efficiency.
    C₁ = 2n * max(convolution) / (sum(sequence))^2
    """
    n = len(sequence)
    if n == 0:
        return float('inf')

    try:
        # Check for numerical stability issues
        if np.any(np.isnan(sequence)) or np.any(np.isinf(sequence)):
            return float('inf')

        # Normalize sequence to prevent overflow/underflow
        seq_norm = np.array(sequence, dtype=np.float64)
        max_val = np.max(np.abs(seq_norm))
        if max_val > 0:
            seq_norm = seq_norm / max_val

        # Compute convolution using FFT for efficiency
        fft_seq = fft(seq_norm, 2*n - 1)
        conv_fft = fft_seq * np.conj(fft_seq)
        conv = ifft(conv_fft).real[:2*n-1]

        # Additional check for numerical problems
        if np.any(np.isnan(conv)) or np.any(np.isinf(conv)):
            # Fall back to direct convolution if FFT has issues
            conv = np.convolve(seq_norm, seq_norm, mode='full')[:2*n-1]

        max_conv = np.max(conv)

        # Check for valid result
        if np.isnan(max_conv) or np.isinf(max_conv):
            return float('inf')

    except Exception:
        # If anything goes wrong, fall back to direct computation
        try:
            conv = np.convolve(sequence, sequence, mode='full')[:2*n-1]
            max_conv = np.max(conv)
        except:
            return float('inf')

    sum_seq = np.sum(sequence)
    if sum_seq < 0.01:
        return float('inf')

    c1 = 2 * n * max_conv / (sum_seq ** 2)
