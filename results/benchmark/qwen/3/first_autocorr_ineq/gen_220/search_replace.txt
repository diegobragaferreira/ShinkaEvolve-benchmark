def adaptive_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Adaptive convolution method selection based on sequence properties."""
    n = len(a)

    # For very small sequences, always use direct method
    if n < 50:
        return convolve_direct(a, b)

    # For larger sequences, evaluate if FFT is beneficial
    # Calculate sparsity and decay characteristics
    nonzero_ratio = np.count_nonzero(a) / n
    nonzero_ratio_b = np.count_nonzero(b) / n

    # If sequences are very sparse, direct might be better
    if nonzero_ratio < 0.1 or nonzero_ratio_b < 0.1:
        return convolve_direct(a, b)

    # If sequences have strong decay or structured patterns, direct might be more stable
    # Check if the sequence has significant variation in values
    std_a = np.std(a)
    std_b = np.std(b)

    # For sequences with low standard deviation, FFT might introduce more numerical error
    if std_a < 1e-3 or std_b < 1e-3:
        return convolve_direct(a, b)

    # Otherwise, use FFT (default behavior)
    return convolve_fft(a, b)
