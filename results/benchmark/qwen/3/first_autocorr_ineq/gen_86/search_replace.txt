def compute_c1(sequence):
    """Computes the C1 autocorrelation constant for a sequence."""
    n = len(sequence)
    if n < 1:
        return float('inf')

    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return float('inf')

    # Use FFT for large sequences, direct for small ones
    if n > 100:
        conv = fftconvolve(sequence, sequence, mode='full')
    else:
        conv = np.convolve(sequence, sequence)

    max_conv = np.max(conv)
    c1 = 2 * n * max_conv / (sum_a ** 2)
    return c1

def compute_convolution_stable(sequence):
    """
    Computes convolution with automatic method switching based on numerical stability.
    """
    n = len(sequence)
    # Try FFT first for efficiency
    if n > 100:
        try:
            conv = fftconvolve(sequence, sequence, mode='full')
            conv = conv[:2*n - 1]
            # Check numerical stability
            if n > 1000:  # Only check for very large sequences
                # Compute condition number of the convolution kernel (approximate)
                kernel = np.array([sequence[i] for i in range(n)])
                # If condition number is too high, fall back to direct computation
                if cond(np.eye(n) * kernel) > NUMERICAL_STABILITY_THRESHOLD:
                    conv = np.convolve(sequence, sequence)
            return conv
        except:
            pass
    # Fall back to direct computation
    return np.convolve(sequence, sequence)
