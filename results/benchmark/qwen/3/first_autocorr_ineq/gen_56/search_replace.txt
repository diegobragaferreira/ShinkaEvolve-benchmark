import numpy as np
from scipy import optimize
from scipy.signal import convolve
from deap import base, creator, tools, algorithms
import random
import time

# Configuration constants
MAX_EVALUATIONS = 10000
POP_SIZE = 50
GENERATIONS = 200
TOURNAMENT_SIZE = 3
MUTATION_RATE = 0.1
CROSSOVER_RATE = 0.8
ELITISM_COUNT = 2

def calculate_c1_constant(sequence):
    """Calculate C1 constant for the given sequence."""
    if len(sequence) == 0:
        return float('inf')

    # Ensure sequence has at least one positive element
    sequence = np.array(sequence)
    if np.sum(sequence) < 0.01:
        return float('inf')

    # Compute the auto-correlation (convolution with itself)
    autocorr = convolve(sequence, sequence, mode='full')
    # Only take the relevant part of the convolution (the middle part)
    mid_point = len(autocorr) // 2
    autocorr = autocorr[mid_point:]

    # Calculate C1 = 2 * n * max(autocorr) / (sum(sequence))^2
    n = len(sequence)
    max_autocorr = np.max(autocorr)
    sum_seq = np.sum(sequence)
    if sum_seq == 0:
        return float('inf')

    c1 = 2 * n * max_autocorr / (sum_seq ** 2)
    return c1

def calculate_inv_c1(sequence):
    """Calculate inverse of C1 (what we want to maximize)."""
    c1 = calculate_c1_constant(sequence)
    return 1.0 / c1 if c1 != 0 else 0.0

def get_good_direction_to_move_into(
    sequence: list[float],
) -> list[float] | None:
    """Returns the direction to move into the sequence."""
    n = len(sequence)
    sum_sequence = np.sum(sequence)
    if sum_sequence < 0.01:
        return None

    # Normalize the sequence
    normalized_sequence = [x * np.sqrt(2 * n) / sum_sequence for x in sequence]

    # Compute maximum convolution value
    max_conv = np.max(convolve(normalized_sequence, normalized_sequence, mode='full'))

    # Solve the linear program
    g_fun = solve_convolution_lp(normalized_sequence, max_conv)
    if g_fun is None:
        return None

    # Normalize the result
    sum_g_fun = np.sum(g_fun)
    if sum_g_fun < 0.01:
        return None

    normalized_g_fun = [x * np.sqrt(2 * n) / sum_g_fun for x in g_fun]

    # Apply small step towards the improved direction
    t = 0.01
    new_sequence = [
        (1 - t) * x + t * y for x, y in zip(sequence, normalized_g_fun)
    ]

    # Ensure non-negativity
    new_sequence = [max(0.0, x) for x in new_sequence]
    return new_sequence

def solve_convolution_lp(f_sequence, rhs):
    """Solves the convolution LP for a given sequence and RHS."""
    n = len(f_sequence)
    c = -np.ones(n)
    a_ub = []
    b_ub = []

    # Build the convolution constraint matrix
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Add non-negativity constraints: b_i >= 0
    a_ub_nonneg = -np.eye(n)  # Negative identity matrix for b_i >= 0
    b_ub_nonneg = np.zeros(n)  # Zero vector

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    try:
        result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs')

        if result.success:
            g_sequence = result.x
            # Ensure non-negative values
            g_sequence = np.maximum(g_sequence, 0.0)
            return g_sequence
        else:
            # If optimization fails, try with relaxed constraints
            print('LP optimization failed, attempting fallback.')
            return None
    except Exception as e:
        print(f'Error during LP solving: {e}')
        return None

def fitness_function(individual):
    """Evaluate fitness (inverse of C1) for an individual."""
    # Clip sequence values to reasonable bounds to prevent overflow issues
    clipped_individual = [min(max(x, 0.0), 1000.0) for x in individual]
    inv_c1 = calculate_inv_c1(clipped_individual)
    return (inv_c1,)

def initialize_population(pop_size, min_len=100, max_len=1000):
    """Initialize population with random sequences."""
    population = []
    for _ in range(pop_size):
        n = random.randint(min_len, max_len)
        individual = [random.uniform(0.1, 100.0) for _ in range(n)]
        population.append(individual)
    return population

def crossover_function(ind1, ind2):
    """Perform crossover between two individuals."""
    if len(ind1) != len(ind2):
        # Handle different lengths by padding shorter one
        max_len = max(len(ind1), len(ind2))
        ind1_padded = ind1 + [0.0] * (max_len - len(ind1))
        ind2_padded = ind2 + [0.0] * (max_len - len(ind2))
    else:
        ind1_padded = ind1
        ind2_padded = ind2

    # Simple uniform crossover
    child1 = []
    child2 = []
    for i in range(len(ind1_padded)):
        if random.random() < 0.5:
            child1.append(ind1_padded[i])
            child2.append(ind2_padded[i])
        else:
            child1.append(ind2_padded[i])
            child2.append(ind1_padded[i])

    return child1, child2

def mutate_function(individual, mut_rate=0.1):
    """Mutate an individual by adding small random noise."""
    mutated = []
    for gene in individual:
        if random.random() < mut_rate:
            # Add small Gaussian noise
            mutated.append(max(0.0, gene + random.gauss(0, 0.1 * gene)))
        else:
            mutated.append(gene)
    return mutated

def search_for_best_sequence() -> list[float]:
    """Function to search for the best coefficient sequence using evolutionary optimization."""
    # Set up DEAP framework
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    toolbox = base.Toolbox()
    toolbox.register("individual", tools.initRepeat, creator.Individual,
                     lambda: random.uniform(0.1, 100.0),
                     random.randint(100, 1000))
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", fitness_function)
    toolbox.register("mate", crossover_function)
    toolbox.register("mutate", mutate_function, mut_rate=MUTATION_RATE)
    toolbox.register("select", tools.selTournament, tournsize=TOURNAMENT_SIZE)

    # Initialize population
    population = toolbox.population(n=POP_SIZE)

    # Evaluate initial population
    fitnesses = list(map(toolbox.evaluate, population))
    for ind, fit in zip(population, fitnesses):
        ind.fitness.values = fit

    # Main evolutionary loop
    for generation in range(GENERATIONS):
        # Select next generation
        offspring = toolbox.select(population, len(population))
        offspring = list(map(toolbox.clone, offspring))

        # Apply crossover and mutation
        for child1, child2 in zip(offspring[::2], offspring[1::2]):
            if random.random() < CROSSOVER_RATE:
                child1, child2 = toolbox.mate(child1, child2)
                del child1.fitness.values
                del child2.fitness.values

        for mutant in offspring:
            if random.random() < MUTATION_RATE:
                mutant = toolbox.mutate(mutant)
                del mutant.fitness.values

        # Evaluate invalid individuals
        invalid_ind = [ind for ind in offspring if not ind.fitness.valid]
        fitnesses = map(toolbox.evaluate, invalid_ind)
        for ind, fit in zip(invalid_ind, fitnesses):
            ind.fitness.values = fit

        # Replace old population with new one
        population[:] = offspring

        # Print progress
        if generation % 50 == 0:
            best_ind = tools.selBest(population, 1)[0]
            best_c1 = 1.0 / calculate_inv_c1(best_ind)
            print(f"Generation {generation}: Best C1 = {best_c1:.6f}")

    # Return best individual found
    best_individual = tools.selBest(population, 1)[0]
    return best_individual
