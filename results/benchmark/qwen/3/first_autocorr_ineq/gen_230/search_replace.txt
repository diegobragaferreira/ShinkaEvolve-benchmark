def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Efficient FFT-based convolution for large sequences with numerical stability checks."""
    try:
        # Use FFT-based convolution
        result = fftconvolve(a, b, mode='full')

        # Check for numerical instability
        if np.any(np.isnan(result)) or np.any(np.isinf(result)):
            # Fallback to direct convolution if numerical issues occur
            return convolve_direct(a, b)

        # Clip extreme values to prevent overflow
        result = np.clip(result, -1e10, 1e10)

    except Exception:
        # Fallback to direct convolution on any error
