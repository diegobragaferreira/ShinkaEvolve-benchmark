def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Efficient FFT-based convolution for large sequences."""
    try:
        result = fftconvolve(a, b, mode='full')[:2*len(a)-1]
        # Check for numerical instability
        if np.any(np.isnan(result)) or np.any(np.isinf(result)):
            # Fall back to direct convolution if FFT introduces instability
            return np.convolve(a, b, mode='full')[:2*len(a)-1]
        return result
    except:
        # Fall back to direct convolution if FFT fails
        return np.convolve(a, b, mode='full')[:2*len(a)-1]
