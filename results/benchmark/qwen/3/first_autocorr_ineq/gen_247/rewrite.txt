import numpy as np
from scipy import optimize
from scipy.signal import fftconvolve
import random
from typing import List, Tuple, Optional, Dict, Any
import time
import warnings
import math
from functools import partial

# Suppress scientific notation for cleaner output
np.set_printoptions(suppress=True)

# Configuration constants
MAX_TIME_SECONDS = 180
MIN_SEQ_LENGTH = 100
MAX_SEQ_LENGTH = 1000
BENCHMARK_RATIO = 1.5031
INITIAL_POPULATION_SIZE = 20
GENERATIONS = 40
ELITE_FRACTION = 0.2
PATIENCE_LIMIT = 10
NUM_HISTORICAL_SAMPLES = 3
MAX_LOCAL_REFINEMENT_ITERATIONS = 20
MAX_SMOOTHING_ITERATIONS = 10

# Historical sequences for seeding the algorithm
HISTORICAL_SEQUENCES = [
    [1.0] * 100,
    [1.0] * 50 + [0.0] * 50,
    [1.0, 0.0] * 50,
    [0.0] * 25 + [1.0] * 50 + [0.0] * 25,
    [1.0, 1.0, 0.0, 0.0] * 25,
    [1.0, 0.5, 0.25, 0.125] * 25,
    [1.0, 0.8, 0.6, 0.4, 0.2] * 20,
    [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1] * 10,
]

class AutocorrelationOptimizer:
    def __init__(self):
        self.start_time = time.time()
        self.best_sequence = None
        self.best_inv_c1 = 0.0
        self.patience_counter = 0

    @staticmethod
    def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
        """Efficient FFT-based convolution for large sequences."""
        return fftconvolve(a, b, mode='full')

    @staticmethod
    def convolve_direct(a: np.ndarray, b: np.ndarray) -> np.ndarray:
        """Direct convolution for small sequences."""
        return np.convolve(a, b, mode='full')

    @staticmethod
    def assess_numerical_stability(a: np.ndarray, b: np.ndarray) -> bool:
        """
        Assess whether FFT convolution would be numerically stable for given inputs.
        Returns True if using FFT is likely to be stable, False otherwise.
        """
        n = len(a)

        # For very small sequences, prefer direct convolution
        if n < 30:
            return False

        try:
            # Direct convolution for reference
            direct_conv = AutocorrelationOptimizer.convolve_direct(a, b)

            # FFT convolution
            fft_conv = AutocorrelationOptimizer.convolve_fft(a, b)

            # Compute relative error between methods
            direct_max = np.max(np.abs(direct_conv))
            fft_max = np.max(np.abs(fft_conv))

            # If either is zero, consider stable
            if direct_max == 0 and fft_max == 0:
                return True

            # Relative error metric
            if direct_max > 0:
                rel_error = np.mean(np.abs(direct_conv - fft_conv)) / direct_max
            else:
                rel_error = np.mean(np.abs(direct_conv - fft_conv)) / (fft_max + 1e-15)

            # Also consider the condition number of the discrete convolution matrix
            conv_ratio = np.max(np.abs(direct_conv)) / (np.min(np.abs(direct_conv)) + 1e-15)

            # If error is small and ratio is reasonable, FFT is acceptable
            return rel_error < 0.05 and conv_ratio < 10000

        except Exception:
            # Default to direct method if anything goes wrong
            return False

    @staticmethod
    def compute_c1_constant(sequence: List[float]) -> Tuple[float, float]:
        """Compute C1 constant and 1/C1 value for a given sequence."""
        a = np.array(sequence)
        n = len(a)

        # Determine best convolution method based on stability assessment
        use_fft = n > 100 and AutocorrelationOptimizer.assess_numerical_stability(a, a)

        if use_fft:
            b = AutocorrelationOptimizer.convolve_fft(a, a)
        else:
            b = AutocorrelationOptimizer.convolve_direct(a, a)

        max_conv = np.max(b)
        sum_a = np.sum(a)

        if sum_a < 0.01:
            return float('inf'), 0.0

        c1 = 2 * n * max_conv / (sum_a ** 2)
        inv_c1 = 1.0 / c1 if c1 > 0 else 0.0

        return c1, inv_c1

    @staticmethod
    def solve_convolution_lp(f_sequence: List[float], rhs: float) -> Optional[List[float]]:
        """Solves the convolution LP for a given sequence and RHS."""
        try:
            n = len(f_sequence)
            c = -np.ones(n)
            a_ub = []
            b_ub = []

            # Build constraint matrix for convolution constraints
            for k in range(2 * n - 1):
                row = np.zeros(n)
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        row[j] = f_sequence[i]
                a_ub.append(row)
                b_ub.append(rhs)

            # Add non-negativity constraints
            a_ub_nonneg = -np.eye(n)
            b_ub_nonneg = np.zeros(n)

            a_ub = np.vstack([a_ub, a_ub_nonneg])
            b_ub = np.hstack([b_ub, b_ub_nonneg])

            # Solve the linear program with multiple methods as fallback
            result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'maxiter': 1000})

            if not result.success:
                # Try different method if highs fails
                result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='revised simplex', options={'maxiter': 1000})

            if result.success:
                return result.x.tolist()
            else:
                return None

        except Exception:
            return None

    @staticmethod
    def gradient_descent_step(sequence: List[float], learning_rate: float = 0.001) -> List[float]:
        """Perform a single gradient descent step."""
        a = np.array(sequence)
        n = len(a)
        epsilon = 1e-12

        # Compute convolution
        use_fft = n > 100 and AutocorrelationOptimizer.assess_numerical_stability(a, a)

        if use_fft:
            b = AutocorrelationOptimizer.convolve_fft(a, a)
        else:
            b = AutocorrelationOptimizer.convolve_direct(a, a)

        max_conv = np.max(b)
        sum_a = np.sum(a)

        if sum_a < epsilon:
            return sequence

        # Simplified gradient computation for efficiency
        term1 = - (2 * n * max_conv) / (sum_a**2)**2
        grad_term2 = np.ones(n) * (2 * n) / (sum_a**2)

        gradient = term1 * np.ones(n) + grad_term2

        # Clip gradient to avoid numerical issues
        gradient = np.clip(gradient, -1000, 1000)

        new_sequence = a + learning_rate * gradient
        # Ensure non-negativity and clipping
        new_sequence = np.maximum(new_sequence, 0.0)
        new_sequence = np.clip(new_sequence, 0, 1000)

        return new_sequence.tolist()

    @staticmethod
    def curvature_aware_descent(sequence: List[float], learning_rate: float = 0.001,
                              curvature_weight: float = 0.01) -> List[float]:
        """Perform descent with curvature information to improve convergence."""
        a = np.array(sequence)
        n = len(a)
        epsilon = 1e-5

        # Estimate curvature using a simpler approach
        hessian_diag = np.ones(n) * 0.01  # Simplified curvature estimate

        # Apply curvature-aware gradient descent
        grad = AutocorrelationOptimizer.gradient_descent_step(sequence, learning_rate)
        grad = np.array(grad)

        # Adjust gradient with curvature term
        adjusted_grad = grad + curvature_weight * hessian_diag * grad

        new_sequence = a + learning_rate * adjusted_grad
        new_sequence = np.maximum(new_sequence, 0.0)
        new_sequence = np.clip(new_sequence, 0, 1000)

        return new_sequence.tolist()

    @staticmethod
    def simulated_annealing_refinement(sequence: List[float], max_iterations: int = 20) -> List[float]:
        """Refine sequence using a simulated annealing-inspired approach."""
        current_seq = np.array(sequence)
        current_inv_c1 = AutocorrelationOptimizer.compute_c1_constant(sequence)[1]

        # Initial temperature and cooling schedule
        temp = 1.0
        cooling_rate = 0.95

        for iteration in range(max_iterations):
            # Generate neighbor by small random perturbation
            neighbor = current_seq + np.random.normal(0, 0.1 * np.abs(current_seq) + 1e-5)
            neighbor = np.maximum(neighbor, 0.0)
            neighbor = np.clip(neighbor, 0, 1000)

            # Compute neighbor's inverse C1
            _, neighbor_inv_c1 = AutocorrelationOptimizer.compute_c1_constant(neighbor.tolist())

            # Accept or reject based on Metropolis criterion
            if neighbor_inv_c1 > current_inv_c1:
                current_seq = neighbor
                current_inv_c1 = neighbor_inv_c1
            else:
                # Probability of acceptance for worse solution
                prob_accept = np.exp((neighbor_inv_c1 - current_inv_c1) / temp)
                if np.random.random() < prob_accept:
                    current_seq = neighbor
                    current_inv_c1 = neighbor_inv_c1

            # Cool down temperature
            temp *= cooling_rate

        return current_seq.tolist()

    @staticmethod
    def mutate_sequence(sequence: List[float], mutation_rate: float = 0.1) -> List[float]:
        """Apply random mutation to a sequence."""
        mutated = sequence.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                # Use larger variance for better exploration
                mutated[i] = max(0, mutated[i] + np.random.normal(0, 0.5 * mutated[i]))
        return mutated

    @staticmethod
    def crossover_sequences(seq1: List[float], seq2: List[float]) -> List[float]:
        """Perform uniform crossover between two sequences."""
        # Ensure both sequences are of same length
        min_len = min(len(seq1), len(seq2))
        child = []
        for i in range(min_len):
            if random.random() < 0.5:
                child.append(seq1[i])
            else:
                child.append(seq2[i])

        # Handle length differences
        if len(seq1) > len(seq2):
            child.extend(seq1[min_len:])
        elif len(seq2) > len(seq1):
            child.extend(seq2[min_len:])

        return child

    @staticmethod
    def initialize_population_with_historical_samples(
        population_size: int,
        initial_length_range: Tuple[int, int],
        historical_samples: int = NUM_HISTORICAL_SAMPLES
    ) -> List[List[float]]:
        """Initialize population by sampling from historical sequences."""
        population = []

        # Add some historical sequences
        for _ in range(historical_samples):
            idx = random.randrange(len(HISTORICAL_SEQUENCES))
            seq = HISTORICAL_SEQUENCES[idx].copy()
            # Adjust length to fit range
            n = random.randint(*initial_length_range)
            if n < len(seq):
                seq = seq[:n]
            elif n > len(seq):
                seq = seq + [0.0] * (n - len(seq))
            population.append(seq)

        # Fill the rest with random sequences
        for _ in range(population_size - historical_samples):
            n = random.randint(*initial_length_range)
            individual = [random.random() * 100 for _ in range(n)]
            population.append(individual)

        return population

    @staticmethod
    def adaptive_tournament_selection(population: List[List[float]],
                                      fitness_scores: List[float],
                                      tournament_size: int = 3) -> List[float]:
        """Select an individual using tournament selection."""
        tournament_indices = random.sample(range(len(population)), tournament_size)
        tournament_fitness = [fitness_scores[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmax(tournament_fitness)]
        return population[winner_index].copy()

    @staticmethod
    def adaptive_evolution_phase(population: List[List[float]],
                                fitness_scores: List[float],
                                elite_size: int = 5,
                                mutation_rate: float = 0.1) -> List[List[float]]:
        """Perform evolutionary operations with adaptive parameters."""
        # Sort by fitness descending
        sorted_indices = np.argsort(fitness_scores)[::-1]
        elite_individuals = [population[i] for i in sorted_indices[:elite_size]]

        # Create new population
        new_population = elite_individuals.copy()

        # Generate offspring through crossover and mutation
        while len(new_population) < len(population):
            parent1 = AutocorrelationOptimizer.adaptive_tournament_selection(population, fitness_scores)
            parent2 = AutocorrelationOptimizer.adaptive_tournament_selection(population, fitness_scores)

            # Crossover
            child = AutocorrelationOptimizer.crossover_sequences(parent1, parent2)

            # Adaptive mutation rate based on population diversity
            current_mutation_rate = mutation_rate * (1.0 - len(set(tuple(child))) / len(child))
            child = AutocorrelationOptimizer.mutate_sequence(child, current_mutation_rate)

            # Random initialization for diversity
            if random.random() < 0.05:
                n = random.randint(MIN_SEQ_LENGTH, MAX_SEQ_LENGTH)
                child = [random.random() * 100 for _ in range(n)]

            new_population.append(child)

        return new_population

    def adaptive_local_refinement_phase(self, best_sequence: List[float]) -> List[float]:
        """Refine the best individual using multi-start gradient-based approach."""
        current_seq = best_sequence.copy()
        current_inv_c1 = self.compute_c1_constant(current_seq)[1]

        # Multi-start gradient descent with adaptive learning rates
        best_found = current_seq.copy()
        best_inv_c1_found = current_inv_c1

        # Try multiple starting points
        for start_iter in range(5):
            # Randomly perturb the sequence to create a new starting point
            if start_iter > 0:
                perturbed = self.mutate_sequence(current_seq, mutation_rate=0.1)
                current_seq = perturbed

            # Adaptive learning rate schedule
            initial_lr = 0.001 * (0.5 + 0.5 * random.random())  # Randomize initial LR
            lr = initial_lr

            for iteration in range(MAX_LOCAL_REFINEMENT_ITERATIONS // 5):  # Reduce iterations per start
                # Perform curvature-aware descent
                refined = self.curvature_aware_descent(current_seq, learning_rate=lr, curvature_weight=0.01)
                _, inv_c1_refined = self.compute_c1_constant(refined)

                if inv_c1_refined > current_inv_c1:
                    current_seq = refined
                    current_inv_c1 = inv_c1_refined

                    # Update best found so far
                    if inv_c1_refined > best_inv_c1_found:
                        best_inv_c1_found = inv_c1_refined
                        best_found = refined.copy()

                # Reduce learning rate for later iterations (warm restart)
                lr *= 0.95

                # Early stopping if improvement is minimal
                if iteration > 5 and abs(inv_c1_refined - current_inv_c1) < 1e-8:
                    break

        # Final simulated annealing refinement
        final_refined = self.simulated_annealing_refinement(best_found, max_iterations=MAX_SMOOTHING_ITERATIONS)
        _, final_inv_c1 = self.compute_c1_constant(final_refined)

        return final_refined if final_inv_c1 > best_inv_c1_found else best_found

    @staticmethod
    def adjust_sequence_length(sequence: List[float], target_length: int) -> List[float]:
        """Adjust sequence length to target while preserving mass distribution."""
        current_length = len(sequence)
        if current_length == target_length:
            return sequence
        elif current_length < target_length:
            # Pad with zeros
            return sequence + [0.0] * (target_length - current_length)
        else:
            # Truncate sequence
            return sequence[:target_length]

    def adaptive_search_for_best_sequence(self) -> List[float]:
        """Main function to search for the best coefficient sequence using enhanced approach."""
        # Initialize population with historical sequences
        population = self.initialize_population_with_historical_samples(
            INITIAL_POPULATION_SIZE, (MIN_SEQ_LENGTH, MAX_SEQ_LENGTH)
        )

        for generation in range(GENERATIONS):
            if time.time() - self.start_time > MAX_TIME_SECONDS:
                break

            # Evaluate fitness for all individuals
            fitness_scores = []
            for individual in population:
                _, inv_c1 = self.compute_c1_constant(individual)
                fitness_scores.append(inv_c1)

            # Sort by fitness (descending order)
            sorted_indices = np.argsort(fitness_scores)[::-1]
            current_best, current_best_inv_c1 = population[sorted_indices[0]], fitness_scores[sorted_indices[0]]

            if current_best_inv_c1 > self.best_inv_c1:
                self.best_inv_c1 = current_best_inv_c1
                self.best_sequence = current_best.copy()
                self.patience_counter = 0  # Reset patience counter
            else:
                self.patience_counter += 1

            # Early stopping if no improvement for too long
            if self.patience_counter > PATIENCE_LIMIT:
                break

            # Adaptive evolution
            population = self.adaptive_evolution_phase(
                population,
                fitness_scores,
                int(ELITE_FRACTION * INITIAL_POPULATION_SIZE)
            )

        # Local refinement of best solution
        if self.best_sequence is not None:
            self.best_sequence = self.adaptive_local_refinement_phase(self.best_sequence)

        # Dynamic sequence length exploration
        if self.best_sequence is not None:
            best_final_sequence = self.best_sequence.copy()
            best_final_inv_c1 = self.best_inv_c1

            # Try several sequence lengths around the optimal
            test_lengths = [int(l) for l in np.linspace(MIN_SEQ_LENGTH, MAX_SEQ_LENGTH, 10)]
            for length in test_lengths:
                # Adjust length of sequence
                adjusted_seq = self.adjust_sequence_length(self.best_sequence, length)

                # Compute new C1 value
                _, inv_c1 = self.compute_c1_constant(adjusted_seq)

                # If better, keep the new sequence
                if inv_c1 > best_final_inv_c1:
                    best_final_inv_c1 = inv_c1
                    best_final_sequence = adjusted_seq

            self.best_sequence = best_final_sequence

        return self.best_sequence if self.best_sequence is not None else [1.0]

    def search_for_best_sequence(self) -> List[float]:
        """Function to search for the best coefficient sequence."""
        # Set seed for reproducibility
        random.seed(42)
        np.random.seed(42)

        return self.adaptive_search_for_best_sequence()
