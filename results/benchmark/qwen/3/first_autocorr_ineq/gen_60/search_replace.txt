def solve_convolution_lp(f_sequence, rhs):
    """Solves the convolution LP with improved numerical stability and alternative fallback."""
    n = len(f_sequence)
    if n == 0:
        return None

    try:
        # Create constraint matrix efficiently
        # For each convolution index k, create constraint: sum_{i+j=k} f[i] * x[j] <= rhs
        # We'll build a sparse constraint matrix

        # Using a more stable approach with explicit constraint generation
        a_ub = []
        b_ub = []

        # Generate convolution constraints
        for k in range(2 * n - 1):
            row = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    row[j] = f_sequence[i]
            a_ub.append(row)
            b_ub.append(rhs)

        # Non-negativity constraints
        a_ub_nonneg = -np.eye(n)
        b_ub_nonneg = np.zeros(n)

        a_ub = np.vstack([a_ub, a_ub_nonneg])
        b_ub = np.hstack([b_ub, b_ub_nonneg])

        # Objective: minimize sum of variables (which corresponds to maximizing 1/C1)
        c = np.ones(n)  # Minimize sum(x) to maximize 1/C1

        # Add small regularization term to prevent numerical issues
        c = c + 1e-10

        # Solve with bounds
        bounds = [(0, None) for _ in range(n)]

        # Try high-performance method first
        result = optimize.linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            bounds=bounds,
            method='highs',
            options={'disp': False, 'presolve': True}
        )

        if result.success:
            return result.x
        else:
            # Fallback to alternative method if primary fails
            try:
                result_fallback = optimize.linprog(
                    c,
                    A_ub=a_ub,
                    b_ub=b_ub,
                    bounds=bounds,
                    method='interior-point',
                    options={'disp': False}
                )
                if result_fallback.success:
                    return result_fallback.x
            except:
                pass

            # Final fallback - return a simple uniform distribution
            return np.full(n, 1.0)

    except Exception as e:
        # Last resort - return uniform distribution
        return np.full(n, 1.0)
