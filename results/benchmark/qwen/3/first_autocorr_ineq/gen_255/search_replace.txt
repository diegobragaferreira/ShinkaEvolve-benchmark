        dynamic_t = base_t * (1 - min(iteration / 100, 0.9))

        if n > 10:

            eps = 1e-3
            # Compute approximate Hessian using finite differences for curvature
            hessian_approx = np.zeros((n, n))

                # Perturb sequence in dimension i
                perturbed_seq = normalized_g_fun.copy()
                perturbed_seq[i] += eps

                conv_perturbed = compute_convolution_fft(perturbed_seq)
