def evolve_population(population, fitness_scores, elite_count=5, mutation_rate=0.1):
    """Perform evolution on the population."""
    # Sort population by fitness (ascending order, since we minimize -1/C1)
    sorted_indices = np.argsort(fitness_scores)
    sorted_population = [population[i] for i in sorted_indices]
    sorted_fitness = [fitness_scores[i] for i in sorted_indices]

    # Keep elite individuals
    elite = sorted_population[:elite_count]

    # Create next generation
    next_generation = elite.copy()

    # Generate offspring through selection, crossover, and mutation
    while len(next_generation) < len(population):
        # Tournament selection
        tournament_size = 3
        selected_parents = []
        for _ in range(2):
            tournament_indices = random.sample(range(len(sorted_population)), tournament_size)
            tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]  # Maximize fitness
            selected_parents.append(sorted_population[winner_index])

        # Uniform crossover
        child1, child2 = [], []
        for i in range(len(selected_parents[0])):
            if random.random() < 0.5:
                child1.append(selected_parents[0][i])
                child2.append(selected_parents[1][i])
            else:
                child1.append(selected_parents[1][i])
                child2.append(selected_parents[0][i])

        # Mutation
        for i in range(len(child1)):
            if random.random() < mutation_rate:
                child1[i] = max(0, child1[i] + random.gauss(0, 0.1 * child1[i] + 0.01))
        for i in range(len(child2)):
            if random.random() < mutation_rate:
                child2[i] = max(0, child2[i] + random.gauss(0, 0.1 * child2[i] + 0.01))

        next_generation.extend([child1, child2])

    return next_generation[:len(population)]
