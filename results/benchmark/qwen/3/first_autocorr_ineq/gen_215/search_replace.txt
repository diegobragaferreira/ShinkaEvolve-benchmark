<DESCRIPTION>
I propose improving the autocorrelation optimization by implementing a multi-start gradient-based optimization with adaptive learning rates and incorporating second-order information more effectively. The key enhancements include:

1. Using a multi-start approach to avoid local minima
2. Implementing adaptive learning rates that decrease with iterations
3. Better curvature estimation using a more stable second-order method
4. Adding early stopping criteria based on convergence rate
5. Improving the local refinement phase with simulated annealing-like temperature control

These changes will help discover better sequences that beat the current benchmark (1/C₁ > 1/1.5031) by exploring more effectively in the optimization landscape.

def compute_curvature(sequence: List[float]) -> np.ndarray:
    """Estimate curvature using finite differences for the Hessian."""
    a = np.array(sequence)
    n = len(a)
    epsilon = 1e-4

    # Estimate diagonal Hessian elements
    hessian_diag = np.zeros(n)
    delta = epsilon

    # Compute second derivative using central differences
    for i in range(n):
        # Perturb element i
        a_plus = a.copy()
        a_minus = a.copy()
        a_plus[i] += delta
        a_minus[i] -= delta

        # Compute max convolution values
        use_fft = n > 100 and assess_numerical_stability(a, a)

        if use_fft:
            b_plus = convolve_fft(a_plus, a_plus)
            b_minus = convolve_fft(a_minus, a_minus)
        else:
            b_plus = convolve_direct(a_plus, a_plus)
            b_minus = convolve_direct(a_minus, a_minus)

        max_plus = np.max(b_plus)
        max_minus = np.max(b_minus)
        max_center = np.max(convolve_fft(a, a) if use_fft else convolve_direct(a, a))

        # Estimate second derivative (approximate curvature)
        curvature = (max_plus + max_minus - 2 * max_center) / (delta**2)
        hessian_diag[i] = max(0, curvature)  # Ensuring non-negative curvature

    return hessian_diag

def simulated_annealing_refinement(sequence: List[float], max_iterations: int = 20) -> List[float]:
    """Refine sequence using a simulated annealing-inspired approach."""
    current_seq = np.array(sequence)
    current_inv_c1 = compute_c1_constant(sequence)[1]

    # Initial temperature and cooling schedule
    temp = 1.0
    cooling_rate = 0.95

    for iteration in range(max_iterations):
        # Generate neighbor by small random perturbation
        neighbor = current_seq + np.random.normal(0, 0.1 * np.abs(current_seq) + 1e-5)
        neighbor = np.maximum(neighbor, 0.0)
        neighbor = np.clip(neighbor, 0, 1000)

        # Compute neighbor's inverse C1
        _, neighbor_inv_c1 = compute_c1_constant(neighbor.tolist())

        # Accept or reject based on Metropolis criterion
        if neighbor_inv_c1 > current_inv_c1:
            current_seq = neighbor
            current_inv_c1 = neighbor_inv_c1
        else:
            # Probability of acceptance for worse solution
            prob_accept = np.exp((neighbor_inv_c1 - current_inv_c1) / temp)
            if np.random.random() < prob_accept:
                current_seq = neighbor
                current_inv_c1 = neighbor_inv_c1

        # Cool down temperature
        temp *= cooling_rate

    return current_seq.tolist()

def adaptive_local_refinement_phase(best_sequence: List[float],
                                   max_iterations: int = 15) -> List[float]:
    """Refine the best individual using multi-start gradient-based approach."""
    current_seq = best_sequence.copy()
    current_inv_c1 = compute_c1_constant(current_seq)[1]

    # Multi-start gradient descent with adaptive learning rates
    best_found = current_seq.copy()
    best_inv_c1_found = current_inv_c1

    # Try multiple starting points
    for start_iter in range(5):
        # Randomly perturb the sequence to create a new starting point
        if start_iter > 0:
            perturbed = mutate_sequence(current_seq, mutation_rate=0.1)
            current_seq = perturbed

        # Adaptive learning rate schedule
        initial_lr = 0.001 * (0.5 + 0.5 * random.random())  # Randomize initial LR
        lr = initial_lr

        for iteration in range(max_iterations // 5):  # Reduce iterations per start
            # Perform curvature-aware descent
            refined = curvature_aware_descent(current_seq, learning_rate=lr, curvature_weight=0.01)
            _, inv_c1_refined = compute_c1_constant(refined)

            if inv_c1_refined > current_inv_c1:
                current_seq = refined
                current_inv_c1 = inv_c1_refined

                # Update best found so far
                if inv_c1_refined > best_inv_c1_found:
                    best_inv_c1_found = inv_c1_refined
                    best_found = refined.copy()

            # Reduce learning rate for later iterations (warm restart)
            lr *= 0.95

            # Early stopping if improvement is minimal
            if iteration > 5 and abs(inv_c1_refined - current_inv_c1) < 1e-8:

    # Final simulated annealing refinement
    final_refined = simulated_annealing_refinement(best_found, max_iterations=10)
    _, final_inv_c1 = compute_c1_constant(final_refined)

    return final_refined if final_inv_c1 > best_inv_c1_found else best_found
