def compute_c1_constant(sequence: List[float]) -> Tuple[float, float]:
    """Compute C1 constant and 1/C1 value for a given sequence with improved numerical handling."""
    a = np.array(sequence, dtype=np.float64)
    n = len(a)

    # Guard against empty sequences
    if n == 0:
        return float('inf'), 0.0

    # For very small sequences, use direct computation for better accuracy
    if n <= 50:
        b = np.convolve(a, a, mode='full')
        b = b[:2*n-1]
    else:
        # For larger sequences, use FFT but with proper padding to reduce edge effects
        b = fftconvolve(a, a, mode='full')
        b = b[:2*n-1]

    # Numerical stabilization: clamp convolution values to avoid extreme outliers
    max_conv = np.max(b)

    # Handle case where max_conv might be extremely large due to noise
    if max_conv > 1e12:
        # Apply soft clipping to prevent numerical overflow
        b = np.minimum(b, 1e12)
        max_conv = np.max(b)

    sum_a = np.sum(a)

    # Ensure minimum threshold to avoid division by near-zero values
    if sum_a < 1e-12:
        return float('inf'), 0.0

    # Compute C1 with numerical safeguards
    try:
        c1 = 2 * n * max_conv / (sum_a ** 2)
        # Clamp C1 to reasonable bounds to prevent numerical issues
        c1 = min(c1, 1e6)
        inv_c1 = 1.0 / c1 if c1 > 0 else 0.0
        return c1, inv_c1
    except (FloatingPointError, OverflowError):
