def adaptive_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Adaptive convolution method selection based on sequence properties."""
    n = len(a)

    # For very small sequences, always use direct method
    if n < 50:
        return convolve_direct(a, b)

    # For larger sequences, evaluate if FFT is beneficial based on more detailed criteria

    # Calculate statistical properties to guide method choice
    nonzero_ratio = np.count_nonzero(a) / n
    nonzero_ratio_b = np.count_nonzero(b) / n

    # Check for sequences with very few non-zero elements
    if nonzero_ratio < 0.05 or nonzero_ratio_b < 0.05:
        return convolve_direct(a, b)

    # Calculate signal properties
    std_a = np.std(a)
    std_b = np.std(b)

    # Check for sequences with very low variation (flat signals)
    if std_a < 1e-4 or std_b < 1e-4:
        return convolve_direct(a, b)

    # Analyze the energy distribution
    energy_a = np.sum(a**2)
    energy_b = np.sum(b**2)

    # If either sequence has very low energy relative to others, use direct
    if energy_a < 1e-6 or energy_b < 1e-6:
        return convolve_direct(a, b)

    # Check for sequences with very high dynamic range
    if energy_a > 0 and np.max(a) / np.sqrt(energy_a) > 100:
        return convolve_direct(a, b)

    if energy_b > 0 and np.max(b) / np.sqrt(energy_b) > 100:
        return convolve_direct(a, b)

    # For normal sequences, prefer FFT for efficiency
    return convolve_fft(a, b)
