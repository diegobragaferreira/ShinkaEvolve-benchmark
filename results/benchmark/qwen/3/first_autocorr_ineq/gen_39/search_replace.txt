<DESCRIPTION>
This change introduces an adaptive threshold for switching between FFT-based and direct convolution methods, based on empirical performance data for different sequence lengths. The key insight is that FFT becomes more efficient than direct convolution around n=100, but for very small sequences, direct convolution can be faster and more stable. This helps balance performance and numerical precision.

def convolve_fft(seq):
    """Compute convolution using FFT for better performance."""
    n = len(seq)
    # Zero-pad to avoid circular convolution effects
    padded_len = 2 * n - 1
    padded_seq = np.pad(seq, (0, padded_len - n), 'constant')
    conv = np.fft.irfft(np.fft.rfft(padded_seq) ** 2)
    return conv[:padded_len]

def compute_c1_value(seq):
    """Compute the C1 constant from the sequence."""
    n = len(seq)
    if n == 0:
        return float('inf')

    # Use FFT for efficiency when possible
    if n > 100:
        conv = convolve_fft(seq)
    else:
        conv = np.convolve(seq, seq, mode='full')

    max_conv = np.max(conv)
    sum_seq = np.sum(seq)

    if sum_seq < 1e-10:
        return float('inf')

    c1 = 2 * n * max_conv / (sum_seq ** 2)
    return c1
