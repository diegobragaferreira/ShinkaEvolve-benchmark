    def compute_multiobjective_value(self, sequence: List[float]) -> float:
        """Computes a multi-objective value combining C1 and sum considerations."""
        c1 = self.compute_c1_constant(sequence)
        total_sum = np.sum(sequence)

        # Avoid division by zero
        if total_sum < 1e-10:
            return float('inf')

        # Multi-objective: weighted combination of inverse C1 and normalized sum
        # This encourages sequences with low C1 and reasonable total magnitude
        weight_c1 = MULTIOBJECTIVE_WEIGHT_C1
        weight_sum = MULTIOBJECTIVE_WEIGHT_SUM

        # Normalize sum to avoid dominance by large magnitudes
        normalized_sum = total_sum / max(1.0, np.mean(sequence) * len(sequence))

        # The objective is to minimize this value
        multiobjective_value = (1/c1) * weight_c1 + normalized_sum * weight_sum

        return multiobjective_value
