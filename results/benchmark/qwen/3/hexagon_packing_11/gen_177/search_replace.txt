The main idea is to implement a more sophisticated hierarchical initialization strategy that creates a better starting configuration. Instead of using a simple predefined pattern, this approach generates a structured hexagonal tiling pattern that respects geometric relationships between adjacent hexagons, then adds controlled noise to maintain diversity while ensuring robust starting configurations.

This approach leverages proven effective structured layouts used in successful hexagon packing generations and avoids convergence failures typical of random initialization, while potentially achieving even tighter packings than the current grid-based seeding strategy.

    def generate_hierarchical_initial_config(self, max_radius: float = 10.0) -> np.ndarray:
        """Generate a hierarchical initial configuration based on hexagonal tiling pattern"""
        # Create a structured layout that respects geometric constraints
        # Based on a 3x4 lattice with some geometric optimization

        # Calculate spacing based on hexagon geometry
        # Distance between centers of adjacent hexagons is 2 (unit hexagons)
        spacing = 2.0

        # Create a more sophisticated initial layout
        # Central hexagon plus surrounding ones arranged in layers
        config = []

        # Central hexagon
        config.append([0.0, 0.0, 0.0])

        # First ring around center (6 hexagons)
        for i in range(6):
            angle = i * 60  # 60 degree increments
            x = spacing * np.cos(np.radians(angle))
            y = spacing * np.sin(np.radians(angle))
            config.append([x, y, 0.0])

        # Second ring (12 hexagons)
        for i in range(12):
            angle = i * 30  # 30 degree increments
            # Place at greater distance
            distance = 2 * spacing
            x = distance * np.cos(np.radians(angle))
            y = distance * np.sin(np.radians(angle))
            config.append([x, y, 0.0])

        # Trim to exactly 11 hexagons and add small perturbations for diversity
        if len(config) > 11:
            config = config[:11]
        elif len(config) < 11:
            # Fill remaining slots with random positions but keep structure
            for i in range(11 - len(config)):
                x = np.random.uniform(-max_radius/2, max_radius/2)
                y = np.random.uniform(-max_radius/2, max_radius/2)
                angle = np.random.uniform(0, 360)
                config.append([x, y, angle])

        # Add small random perturbations to avoid degenerate cases
        result = np.array(config)
        for i in range(len(result)):
            # Small random displacements
            result[i][0] += np.random.normal(0, 0.1)
            result[i][1] += np.random.normal(0, 0.1)
            # Random rotations
            result[i][2] += np.random.normal(0, 10)
            result[i][2] = result[i][2] % 360

        return result
