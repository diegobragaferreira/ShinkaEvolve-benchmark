    def evaluate_population_parallel(self, population):
        """Evaluate all individuals in population in parallel"""
        def evaluate_individual(individual):
            hexagons = self.create_hexagons_from_array(individual)
            radius = self.find_optimal_radius(hexagons)
            fitness = self.evaluator.evaluate_fitness(hexagons, radius)
            return fitness, radius

        # Use joblib for parallel evaluation
        results = Parallel(n_jobs=-1)(
            delayed(evaluate_individual)(individual)
            for individual in population
        )

        fitness_scores = [result[0] for result in results]
        radii = [result[1] for result in results]

        return fitness_scores, radii
