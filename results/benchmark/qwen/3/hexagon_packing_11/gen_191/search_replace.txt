def binary_search_outer_radius(inner_hex_data: np.ndarray, min_radius: float,
                              max_radius: float, tolerance: float = 0.0001) -> float:
    """Binary search to find the minimum valid outer radius with adaptive precision."""
    # Track recent fitness improvements to adapt precision
    recent_improvements = []
    prev_penalty = float('inf')

    while max_radius - min_radius > tolerance:
        mid_radius = (min_radius + max_radius) / 2.0
        penalty, is_valid, _ = evaluate_packing(inner_hex_data, mid_radius)

        # Track recent penalties for adaptive tolerance adjustment
        if len(recent_improvements) >= 5:
            recent_improvements.pop(0)
        recent_improvements.append(prev_penalty - penalty if prev_penalty != float('inf') else 0)
        prev_penalty = penalty

        if is_valid:
            max_radius = mid_radius
            # If we're making small improvements, tighten the tolerance
            if len(recent_improvements) >= 2 and recent_improvements[-1] < 0.01:
                tolerance = max(1e-6, tolerance * 0.5)  # More precise search
        else:
            min_radius = mid_radius

        # Prevent infinite loop with minimum tolerance
        tolerance = max(tolerance, 1e-6)

    return max_radius
