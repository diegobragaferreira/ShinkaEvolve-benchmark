    # This approach places hexagons in a pattern that maximizes packing density

    # Create a more sophisticated arrangement:
    # 1. Central hexagon
    # 2. First ring of 6 hexagons (at radius = spacing)
    # 3. Second ring of 12 hexagons (at radius = 2*spacing)
    # 4. Trim to exactly 11 hexagons with strategic placement

    config = []

    # Central hexagon
    config.append([0.0, 0.0, 0.0])

    # First ring - 6 hexagons surrounding the center
    for i in range(6):
        angle = i * 60  # 60 degrees
        rad = np.radians(angle)
        x = spacing * np.cos(rad)
        y = spacing * np.sin(rad)
        config.append([x, y, 0.0])

    # Second ring - 12 hexagons at greater distance
    # But we'll strategically place fewer to get exactly 11

    # Add strategic positions to fill gaps effectively
    config.extend(second_ring_positions)

    # Trim to exactly 11 hexagons and add small perturbations for better optimization
    config = config[:11]
