def create_hierarchical_initial_individual():
    """Create an initial individual using hierarchical hexagonal tiling for better spatial arrangement"""
    # Start with a structured hexagonal lattice pattern
    # Layer 1: central hexagon
    individual = np.array([
        [0, 0, 0],  # center
    ])

    # Layer 2: first ring around center (6 hexagons)
    ring_positions = []
    for i in range(6):
        angle = i * np.pi / 3
        x = 2.0 * np.cos(angle)
        y = 2.0 * np.sin(angle)
        ring_positions.append([x, y, 0])

    individual = np.vstack([individual, ring_positions])

    # Layer 3: second ring (12 hexagons) - we'll use a strategic subset
    # Place hexagons to fill gaps and optimize space
    second_ring_positions = [
        [-3.0, 0, 0],      # far left
        [3.0, 0, 0],       # far right
        [-1.5, 2.6, 0],    # top-left
        [1.5, 2.6, 0],     # top-right
        [-1.5, -2.6, 0],   # bottom-left

        [-3.0, -2.6, 0],   # lower far left
        [3.0, -2.6, 0],    # lower far right
        [-2.5, 1.5, 0],    # near top left
        [2.5, 1.5, 0],     # near top right
    ]

    individual = np.vstack([individual, second_ring_positions[:5]])  # Take 5 of the 12

    # Ensure we have exactly 11 hexagons
    while len(individual) < 11:
        # Fill remaining slots with random positions near existing ones
        individual = np.vstack([individual, [random.uniform(-4, 4), random.uniform(-4, 4), random.uniform(0, 360)]])

    return individual
