def generate_improved_initial_config():
    """Generate a better initial configuration for 11 hexagons using hexagonal lattice pattern."""
    # Create a structured hexagonal lattice approach
    # This follows the mathematical principle of optimal hexagonal packing

    # We'll use a concentric ring approach:
    # Ring 0: 1 hexagon (central)
    # Ring 1: 6 hexagons (first shell)
    # Ring 2: 12 hexagons (second shell) - but we only need 11 total

    # For a hexagonal lattice with unit hexagons, the spacing is 2 units between centers
    # The inradius (distance from center to edge) is sqrt(3)/2 ≈ 0.866
    # The circumradius (distance from center to vertex) is 1

    positions = []

    # Central hexagon
    positions.append([0.0, 0.0, 0.0])

    # First ring - 6 hexagons at distance 2 from center (touching)
    ring_1_radius = 2.0  # Distance between centers of touching hexagons
    for i in range(6):
        angle = i * 60  # 60 degree increments
        rad = math.radians(angle)
        x = ring_1_radius * math.cos(rad)
        y = ring_1_radius * math.sin(rad)
        positions.append([x, y, 0.0])

    # Second ring - additional positions to reach 11 total
    # Using hexagonal lattice spacing of 2*sqrt(3) ≈ 3.464 for second shell
    ring_2_radius = 2 * math.sqrt(3)  # ~3.464 units
    # Place 5 more hexagons to get 11 total (1 + 6 + 4 = 11)
    # But to preserve structure, we'll actually place 4 strategic positions
    # that fill the gaps effectively

    # Strategic positions in the second shell
    # These positions are chosen to maximize packing density while maintaining the structure
    ring_2_positions = [
        (0.0, ring_2_radius, 0.0),     # Top
        (ring_2_radius * 0.5, -ring_2_radius * 0.5, 0.0),  # Bottom-right
        (-ring_2_radius * 0.5, -ring_2_radius * 0.5, 0.0), # Bottom-left
        (ring_2_radius * 0.5, ring_2_radius * 0.5, 0.0),   # Top-right
        (-ring_2_radius * 0.5, ring_2_radius * 0.5, 0.0),  # Top-left
    ]

    # Use 4 of these positions (since we already have 7, we need 4 more for 11 total)
    # We'll select positions that don't overlap with existing ones or are strategically placed
    for i, (x, y, angle) in enumerate(ring_2_positions[:4]):
        # Avoid placing too close to the first ring if possible
        if i == 0 or i == 2:  # Top and Bottom-left
            positions.append([x, y, 0.0])
        else:
            positions.append([x, y, 0.0])

    # Trim to exactly 11 positions
    positions = positions[:11]

    # Add slight random jitter to break symmetry and improve optimization chances
    np.random.seed(42)
    for i in range(len(positions)):
        positions[i][0] += np.random.normal(0, 0.05)
        positions[i][1] += np.random.normal(0, 0.05)

    return np.array(positions)
