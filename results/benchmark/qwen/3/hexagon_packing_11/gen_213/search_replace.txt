    # Binary search to find tightest bounding radius with adaptive precision
    low = max_dist
    high = max_dist * 2.0
    max_iterations = 30

    # Start with relatively loose tolerance and make it adaptive
    tolerance = 1e-2  # Initial tolerance
    prev_diff = float('inf')

    for iteration in range(max_iterations):
        # Adaptive tolerance that decreases as we converge
        current_diff = high - low
        if iteration > 0 and current_diff < prev_diff * 0.9:  # If significant improvement
