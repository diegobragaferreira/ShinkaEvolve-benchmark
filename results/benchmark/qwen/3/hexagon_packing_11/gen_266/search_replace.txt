def gradient_guided_local_optimization(individual, max_iter=100):
    """Enhanced local optimization with adaptive perturbation sizing"""
    best_individual = deepcopy(individual)
    best_fitness, _ = evaluate_fitness(best_individual[:, :2], best_individual[:, 2])

    # Adaptive perturbation sizing based on iteration progress
    for iteration in range(max_iter):
        # Dynamic perturbation size that decreases over time
        progress = iteration / max_iter
        current_displacement = 0.2 * (1 - progress * 0.8)  # Starts at 0.2, ends at ~0.04

        mutated = deepcopy(best_individual)
        idx = random.randint(0, len(mutated)-1)

        # Perturb position with adaptive displacement
        mutated[idx][0] += random.uniform(-current_displacement, current_displacement)
        mutated[idx][1] += random.uniform(-current_displacement, current_displacement)

        # Perturb rotation with adaptive displacement
        rotation_displacement = 5 * (1 - progress * 0.9)  # Starts at 5, ends at ~0.5

        mutated[idx][2] = mutated[idx][2] % 360
