def generate_initial_config():
    """Generate a high-quality initial configuration using optimized hexagonal packing principles"""
    # Use a more sophisticated approach based on known dense hexagonal arrangements
    # This follows principles from mathematical packing theory

    # Create a honeycomb-like structure with better radial distribution
    # Core hexagons arranged in concentric rings
    config = []

    # Center hexagon
    config.append([0.0, 0.0, 0.0])

    # First ring - 6 hexagons at distance of 2 units (centers separated by 2)
    ring1_radius = 2.0
    ring1_angle_offset = 0  # Start with 0 degrees

    for i in range(6):
        angle = ring1_angle_offset + i * 60  # 60-degree increments
        rad_angle = np.radians(angle)
        x = ring1_radius * np.cos(rad_angle)
        y = ring1_radius * np.sin(rad_angle)
        config.append([x, y, 0.0])

    # Second ring - 12 hexagons at distance of 3.464 units (approximately sqrt(12))
    ring2_radius = 2 * np.sqrt(3)  # This gives us proper spacing for dense packing
    ring2_angle_offset = 30  # Offset by 30 degrees to interleave rings

    for i in range(12):
        angle = ring2_angle_offset + i * 30  # 30-degree increments
        rad_angle = np.radians(angle)
        x = ring2_radius * np.cos(rad_angle)
        y = ring2_radius * np.sin(rad_angle)
        config.append([x, y, 0.0])

    # Trim to exactly 11 hexagons (we'll use the first 11)
    # We actually want to remove some positions to get exactly 11
    # Let's keep core + 6 ring1 + 4 ring2 = 11 total
    config = config[:11]

    # Convert to numpy array
    config = np.array(config)

    # Add carefully controlled noise to break symmetry while preserving good packing
    np.random.seed(42)  # Fixed seed for reproducibility
    noise_scale = 0.02  # Reduce noise level to maintain structure
    config[:, 0] += np.random.normal(0, noise_scale, config.shape[0])
    config[:, 1] += np.random.normal(0, noise_scale, config.shape[0])

    # Add slight random rotations to improve chances of finding better local minima
    config[:, 2] += np.random.uniform(-5, 5, config.shape[0])
    config[:, 2] = config[:, 2] % 360  # Keep in [0,360) range

    return config
