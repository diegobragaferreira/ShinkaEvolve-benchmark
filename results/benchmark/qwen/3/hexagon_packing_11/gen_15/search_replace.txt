import numpy as np
from shapely.geometry import Polygon
import math


def create_regular_hexagon(center=(0, 0), side_length=1, rotation=0):
    """Create a regular hexagon as a Shapely polygon"""
    angles = [rotation + i * 60 for i in range(6)]
    points = [
        (
            center[0] + side_length * math.cos(math.radians(angle)),
            center[1] + side_length * math.sin(math.radians(angle))
        )
        for angle in angles
    ]
    return Polygon(points)


def check_hexagon_containment(hexagon, outer_hexagon):
    """Check if a hexagon is completely contained within the outer hexagon"""
    return outer_hexagon.contains(hexagon)


def check_hexagon_overlap(hex1, hex2):
    """Check if two hexagons overlap"""
    return hex1.intersects(hex2)


def calculate_min_outer_radius(inner_hex_data, side_length=1):
    """Calculate the minimum radius needed to contain all hexagons"""
    # Get all vertices of all inner hexagons
    all_vertices = []
    for x, y, angle in inner_hex_data:
        hex_poly = create_regular_hexagon((x, y), side_length, angle)
        all_vertices.extend(list(hex_poly.exterior.coords)[:-1])  # Exclude repeated last point

    # Calculate the centroid of all vertices
    avg_x = sum(v[0] for v in all_vertices) / len(all_vertices)
    avg_y = sum(v[1] for v in all_vertices) / len(all_vertices)

    # Find maximum distance from centroid to any vertex
    max_distance = 0
    for x, y in all_vertices:
        distance = math.sqrt((x - avg_x)**2 + (y - avg_y)**2)
        max_distance = max(max_distance, distance)

    return max_distance + side_length  # Add buffer for hexagon size


def hexagon_packing_11():
    """
    Constructs a packing of 11 disjoint unit regular hexagons inside a larger regular hexagon, maximizing 1/outer_hex_side_length.
    Returns
        inner_hex_data: np.ndarray of shape (11,3), where each row is of the form (x, y, angle_degrees) containing the (x,y) coordinates and angle_degree of the respective inner hexagon.
        outer_hex_data: np.ndarray of shape (3,) of form (x,y,angle_degree) containing the (x,y) coordinates and angle_degree of the outer hexagon.
        outer_hex_side_length: float representing the side length of the outer hexagon.
    """

    # Initial optimized arrangement based on hexagonal lattice
    # Pattern: 3 rows, 4 columns with specific spacing
    inner_hex_data = np.array([
        [0, 0, 0],           # center
        [-2.0, 0, 0],        # left
        [2.0, 0, 0],         # right
        [0, 3.464, 0],       # top (sqrt(3) * 2)
        [0, -3.464, 0],      # bottom (-sqrt(3) * 2)
        [-1.0, 1.732, 0],    # top-left (half row offset)
        [1.0, 1.732, 0],     # top-right
        [-1.0, -1.732, 0],   # bottom-left
        [1.0, -1.732, 0],    # bottom-right
        [-2.0, 3.464, 0],    # far top-left
        [2.0, 3.464, 0],     # far top-right
    ])

    # Refine the arrangement to optimize packing
    best_radius = float('inf')
    best_inner_data = None

    # Try different configurations with small perturbations
    best_config = inner_hex_data.copy()

    # Create initial outer hexagon
    initial_radius = calculate_min_outer_radius(inner_hex_data)

    # Iterative refinement to minimize outer hexagon size
    for _ in range(100):  # limited iterations for performance
        # Check current configuration
        outer_hex = create_regular_hexagon((0, 0), initial_radius)

        # Validate all hexagons are contained within the outer hexagon
        valid = True
        for x, y, angle in inner_hex_data:
            hex_poly = create_regular_hexagon((x, y), 1, angle)
            if not check_hexagon_containment(hex_poly, outer_hex):
                valid = False
                break

        # If valid and better than current best, update
        if valid:
            current_radius = calculate_min_outer_radius(inner_hex_data)
            if current_radius < best_radius:
                best_radius = current_radius
                best_inner_data = inner_hex_data.copy()

        # Make small adjustments to improve packing
        # Move some hexagons closer together
        if not valid:
            break  # Stop if invalid configuration

        # Simple optimization step: slightly adjust positions
        for i in range(len(inner_hex_data)):
            if i == 0:  # Keep center fixed
                continue
            # Make small adjustments
            inner_hex_data[i][0] += np.random.uniform(-0.05, 0.05)
            inner_hex_data[i][1] += np.random.uniform(-0.05, 0.05)

    # Use the best configuration found
    final_inner_data = best_inner_data if best_inner_data is not None else inner_hex_data

    # Calculate final outer radius
    final_radius = calculate_min_outer_radius(final_inner_data)

    outer_hex_data = np.array([0, 0, 0])  # centered at origin
    outer_hex_side_length = final_radius  # optimized size

    return final_inner_data, outer_hex_data, outer_hex_side_length
