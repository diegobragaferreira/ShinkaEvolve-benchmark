def smart_local_optimization(hex_data, max_iterations=100, initial_step_size=0.1):
    """
    Perform adaptive local optimization with multi-stage refinement approach.
    """
    def refine_objective(params):
        hex_data = params.reshape(-1, 3)
        if not validate_individual(hex_data):
            return 1e6
        return -evaluate_fitness(hex_data)

    # Track convergence history
    convergence_history = []
    best_fitness = -float('inf')
    best_params = hex_data.flatten().copy()

    # Stage 1: Aggressive refinement with larger steps
    current_params = hex_data.flatten().copy()
    step_size = initial_step_size

    for iteration in range(max_iterations):
        try:
            # Attempt optimization with current parameters
            result = minimize(
                refine_objective,
                current_params,
                method='L-BFGS-B',
                bounds=[(-10, 10), (-10, 10), (0, 360)] * 11,
                options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-6}
            )

            if result.success:
                current_params = result.x
                current_fitness = -refine_objective(current_params)

                # Update best solution
                if current_fitness > best_fitness:
                    best_fitness = current_fitness
                    best_params = current_params.copy()

                # Track convergence
                convergence_history.append(current_fitness)
                if len(convergence_history) > 10:
                    convergence_history.pop(0)

                # Adjust step size based on recent improvement
                if len(convergence_history) >= 3:
                    recent_improvements = [convergence_history[i] - convergence_history[i-1]
                                         for i in range(1, len(convergence_history))]
                    avg_improvement = sum(recent_improvements) / len(recent_improvements)
                    if abs(avg_improvement) < 1e-6:  # Converged or stagnated
                        if step_size > 1e-4:  # Only reduce if still significant
                            step_size *= 0.5  # Reduce step size

        except Exception:
            # If optimization fails, continue with current parameters
            pass

        # Gradually reduce step size in later stages
        if iteration > max_iterations // 2:
            step_size = max(step_size * 0.95, 1e-5)

    # Final validation and return best result
    final_hex_data = best_params.reshape(-1, 3)
    if not validate_individual(final_hex_data):
        return hex_data  # Return original if invalid

    return final_hex_data
