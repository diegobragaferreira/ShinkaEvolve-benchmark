    """Compute minimum outer hexagon radius that contains all inner hexagons using adaptive binary search."""
    # Compute bounding box of all hexagon vertices
    all_vertices = []
    for hex_params in hex_data:
        vertices = hexagon_vertices(hex_params[0], hex_params[1], 1, hex_params[2])
        all_vertices.extend(vertices)

    # Find the maximum distance from center to any vertex
    max_dist = 0
    for x, y in all_vertices:
        dist = math.sqrt(x*x + y*y)
        max_dist = max(max_dist, dist)

    # Add some buffer for safety
    max_dist += 1e-3

    # Binary search for the exact radius with adaptive tolerance
    low = max_dist
    high = max_dist * 2.0
    tolerance = initial_tolerance

    # Track recent convergence for adaptive tolerance adjustment
    recent_convergence = []
    convergence_stagnation_count = 0

    while high - low > tolerance:
        mid = (low + high) / 2.0
        if check_containment(hex_data, mid):
            high = mid
        else:
            low = mid

        # Adaptive tolerance adjustment based on convergence
        current_diff = high - low
        recent_convergence.append(current_diff)

        # Keep only last 5 differences to avoid memory issues
        if len(recent_convergence) > 5:
            recent_convergence.pop(0)

        # Check if we're converging slowly (stagnation detection)
        if len(recent_convergence) >= 3:
            diffs = [abs(recent_convergence[i] - recent_convergence[i-1]) for i in range(1, len(recent_convergence))]
            avg_diff = sum(diffs) / len(diffs) if diffs else 0

            # If average difference is very small compared to current tolerance
            if avg_diff < tolerance * 0.05:
                convergence_stagnation_count += 1
                if convergence_stagnation_count > 2:  # After 2 consecutive stagnations
                    # Further reduce tolerance to seek tighter solution
                    tolerance = max(tolerance * 0.3, min_tolerance)
                    convergence_stagnation_count = 0  # Reset counter
            else:
                convergence_stagnation_count = 0  # Reset if showing significant progress

    return (low + high) / 2.0
