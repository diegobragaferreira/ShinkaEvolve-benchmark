    left = initial_radius_estimate
    right = 20.0

    # Use adaptive precision that becomes finer as we converge
    max_iterations = 100

    # Track convergence to adjust precision dynamically
    prev_diff = float('inf')
    convergence_count = 0
    min_convergence_steps = 5

    iterations = 0

        # Adjust precision based on convergence behavior
