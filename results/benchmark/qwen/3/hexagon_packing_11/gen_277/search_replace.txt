    def optimize_local(self, individual: np.ndarray, outer_radius: float) -> np.ndarray:
        """Refine solution locally using multiple optimization strategies"""
        def objective(params):
            # Reshape params back to hexagon data
            new_data = individual.copy()
            for i in range(len(new_data)):
                new_data[i][0] = params[i*3]
                new_data[i][1] = params[i*3+1]
                new_data[i][2] = params[i*3+2]

            # Convert to hexagon objects for evaluation
            hexagons = self.create_hexagons_from_array(new_data)

            # Evaluate fitness
            fitness = self.evaluator.evaluate_fitness(hexagons, outer_radius)
            return -fitness  # minimize negative fitness

        # Flatten the data for optimization
        initial_params = []
        for i in range(len(individual)):
            initial_params.extend([individual[i][0], individual[i][1], individual[i][2]])

        # Strategy 1: L-BFGS-B with very strict convergence criteria for high precision
        try:
            result = minimize(objective, initial_params, method='L-BFGS-B',
                            bounds=[(-10, 10), (-10, 10), (0, 360)] * len(individual),
                            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12})
            if result.success:
                # Reshape optimized result back
                refined_data = individual.copy()
                for i in range(len(refined_data)):
                    refined_data[i][0] = result.x[i*3]
                    refined_data[i][1] = result.x[i*3+1]
                    refined_data[i][2] = result.x[i*3+2]
                return refined_data
        except:
            pass

        # Strategy 2: Sequential Nelder-Mead followed by L-BFGS-B for robustness
        try:
            # First try Nelder-Mead for global exploration
            nm_result = minimize(objective, initial_params, method='Nelder-Mead',
                               options={'maxiter': 200, 'disp': False, 'xtol': 1e-8, 'ftol': 1e-8})

            if nm_result.success:
                # Then refine with L-BFGS-B if Nelder-Mead found a good solution
                lbfgs_result = minimize(objective, nm_result.x, method='L-BFGS-B',
                                      bounds=[(-10, 10), (-10, 10), (0, 360)] * len(individual),
                                      options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-10})

                if lbfgs_result.success:
                    # Reshape and validate final result
                    refined_data = individual.copy()
                    for i in range(len(refined_data)):
                        refined_data[i][0] = lbfgs_result.x[i*3]
                        refined_data[i][1] = lbfgs_result.x[i*3+1]
                        refined_data[i][2] = lbfgs_result.x[i*3+2]
                    return refined_data
        except:
            pass

        # Strategy 3: Coordinate descent with adaptive learning rates as last resort
        try:
            current_params = np.array(initial_params)
            current_fitness = objective(current_params)

            # Adaptive learning rates
            lr_pos = 0.1  # Position learning rate
            lr_angle = 2.0  # Angle learning rate

            # Iterative improvement
            for _ in range(100):
                improved = False
                for i in range(len(current_params)):
                    param_copy = current_params.copy()

                    # Apply perturbation
                    if i % 3 < 2:  # x or y coordinate
                        param_copy[i] += np.random.normal(0, lr_pos)
                    else:  # angle
                        param_copy[i] += np.random.normal(0, lr_angle)
                        param_copy[i] = param_copy[i] % 360

                    new_fitness = objective(param_copy)
                    if new_fitness < current_fitness:
                        current_params = param_copy
                        current_fitness = new_fitness
                        improved = True

                # Reduce learning rate if no improvement
                if not improved:
                    lr_pos *= 0.95
                    lr_angle *= 0.95

                    # Stop if learning rates become too small
                    if lr_pos < 1e-8 and lr_angle < 1e-8:
                        break

            # Reconstruct data from final parameters
            refined_data = individual.copy()
            for i in range(len(refined_data)):
                refined_data[i][0] = current_params[i*3]
                refined_data[i][1] = current_params[i*3+1]
                refined_data[i][2] = current_params[i*3+2]
            return refined_data
        except:
            pass

        # If all optimization attempts fail, return original individual
        return individual
