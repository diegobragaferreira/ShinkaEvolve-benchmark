import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time
from shapely.geometry import Polygon, Point
from shapely.ops import cascaded_union

# Constants
HEX_RADIUS = 1.0  # Unit hexagon radius
HEX_APOGEE = HEX_RADIUS * np.sqrt(3) / 2  # Distance from center to flat side
HEX_SIDE = HEX_RADIUS  # Side length of unit hexagon

def create_hexagon_vertices(center=(0,0), rotation=0, radius=1.0):
    """Create vertices of a regular hexagon."""
    angles = np.linspace(0, 2*np.pi, 7) + np.radians(rotation)
    vertices = np.column_stack([
        center[0] + radius * np.cos(angles),
        center[1] + radius * np.sin(angles)
    ])
    return vertices[:-1]  # Remove last vertex to close polygon

def hexagon_contains_point(hex_center, hex_rotation, point):
    """Check if a point is inside a hexagon."""
    hex_vertices = create_hexagon_vertices(hex_center, hex_rotation)
    hex_poly = Polygon(hex_vertices)
    point_obj = Point(point)
    return hex_poly.contains(point_obj) or hex_poly.touches(point_obj)

def check_hexagon_containment(hex_centers, hex_rotations, outer_radius):
    """Check if all inner hexagons are contained within outer hexagon."""
    outer_vertices = create_hexagon_vertices((0, 0), 0, outer_radius)
    outer_poly = Polygon(outer_vertices)

    for center, rotation in zip(hex_centers, hex_rotations):
        hex_vertices = create_hexagon_vertices(center, rotation)
        hex_poly = Polygon(hex_vertices)

        # Check if hexagon is fully contained in outer hexagon
        if not outer_poly.contains(hex_poly):
            return False

    return True

def check_hexagon_collision(hex_centers, hex_rotations):
    """Check if any two hexagons collide."""
    n = len(hex_centers)

    for i in range(n):
        for j in range(i+1, n):
            h1_vertices = create_hexagon_vertices(hex_centers[i], hex_rotations[i])
            h2_vertices = create_hexagon_vertices(hex_centers[j], hex_rotations[j])

            h1_poly = Polygon(h1_vertices)
            h2_poly = Polygon(h2_vertices)

            if h1_poly.intersects(h2_poly):
                return True

    return False

def calculate_outer_hexagon_radius(hex_centers, hex_rotations):
    """Calculate minimum outer hexagon radius that contains all inner hexagons."""
    max_dist = 0.0

    for center, rotation in zip(hex_centers, hex_rotations):
        hex_vertices = create_hexagon_vertices(center, rotation)
        # Find the maximum distance from origin to any vertex
        distances = [np.linalg.norm(vertex) for vertex in hex_vertices]
        max_dist = max(max_dist, max(distances))

    # Add some buffer to ensure complete containment
    return max_dist + 0.1

def objective_function(params):
    """Objective function to minimize (negative of 1/outer_radius)."""
    # params: [x1, y1, rot1, x2, y2, rot2, ..., xn, yn, rotn, outer_radius]
    n = 11
    hex_centers = []
    hex_rotations = []

    for i in range(n):
        hex_centers.append([params[3*i], params[3*i+1]])
        hex_rotations.append(params[3*i+2])

    outer_radius = params[-1]

    # Check constraints
    if check_hexagon_collision(hex_centers, hex_rotations):
        return 1e6  # Large penalty for collisions

    if not check_hexagon_containment(hex_centers, hex_rotations, outer_radius):
        return 1e6  # Large penalty for containment violations

    return -1.0 / outer_radius  # We want to maximize 1/outer_radius

def optimize_hexagon_packing():
    """Optimize the hexagon packing configuration."""
    n = 11
    bounds = []

    initial_guess.append(4.0)

    # Use optimization to find better configuration
    result = differential_evolution(objective_function, bounds, seed=42, maxiter=50)
