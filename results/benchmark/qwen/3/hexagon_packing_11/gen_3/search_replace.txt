import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from shapely.geometry import Polygon
import time

def create_hexagon_vertices(center_x, center_y, side_length=1, rotation=0):
    """Create vertices of a regular hexagon with given center, side length and rotation."""
    angles = np.linspace(0, 2*np.pi, 7) + np.radians(rotation)
    vertices = np.column_stack([
        center_x + side_length * np.cos(angles),
        center_y + side_length * np.sin(angles)
    ])
    return vertices

def check_containment(hexagon_vertices, outer_hex_vertices):
    """Check if a hexagon is fully contained within the outer hexagon."""
    inner_polygon = Polygon(hexagon_vertices)
    outer_polygon = Polygon(outer_hex_vertices)
    return outer_polygon.contains(inner_polygon)

def check_overlap(hex1_vertices, hex2_vertices):
    """Check if two hexagons overlap."""
    poly1 = Polygon(hex1_vertices)
    poly2 = Polygon(hex2_vertices)
    return poly1.intersects(poly2)

def objective_function(params):
    """Objective function to minimize the outer hexagon side length."""
    # Extract parameters
    # First 22 params: x,y positions for 11 hexagons (no rotation for now)
    positions = params[:22].reshape(-1, 2)

    # Last param: outer hexagon side length
    outer_side_length = params[-1]

    # Create outer hexagon vertices
    outer_center = np.array([0, 0])
    outer_vertices = create_hexagon_vertices(outer_center[0], outer_center[1], outer_side_length, 0)

    # Check constraints
    total_penalty = 0

    # Check containment
    for i, pos in enumerate(positions):
        hex_vertices = create_hexagon_vertices(pos[0], pos[1], 1, 0)
        if not check_containment(hex_vertices, outer_vertices):
            total_penalty += 1000  # Large penalty for containment violation

    # Check overlaps
    for i in range(len(positions)):
        for j in range(i+1, len(positions)):
            hex1_vertices = create_hexagon_vertices(positions[i][0], positions[i][1], 1, 0)
            hex2_vertices = create_hexagon_vertices(positions[j][0], positions[j][1], 1, 0)
            if check_overlap(hex1_vertices, hex2_vertices):
                total_penalty += 1000  # Large penalty for overlap

    # Return negative of inverse side length as we want to maximize 1/R
    return total_penalty - 1.0 / outer_side_length if outer_side_length > 0 else 10000

def hexagon_packing_11():
    """
    Constructs a packing of 11 disjoint unit regular hexagons inside a larger regular hexagon, maximizing 1/outer_hex_side_length.
    Returns
        inner_hex_data: np.ndarray of shape (11,3), where each row is of the form (x, y, angle_degrees) containing the (x,y) coordinates and angle_degree of the respective inner hexagon.
        outer_hex_data: np.ndarray of shape (3,) of form (x,y,angle_degree) containing the (x,y) coordinates and angle_degree of the outer hexagon.
        outer_hex_side_length: float representing the side length of the outer hexagon.
    """
    # Use a more informed starting configuration based on known optimal arrangements
    # Start with a central hexagon surrounded by others in a hexagonal pattern
    initial_positions = np.array([
        [0, 0],      # center
        [0, 2],      # top
        [0, -2],     # bottom
        [1.732, 1],  # top-right (sqrt(3) ≈ 1.732)
        [-1.732, 1], # top-left
        [1.732, -1], # bottom-right
        [-1.732, -1], # bottom-left
        [3.464, 0],  # far right (2*sqrt(3))
        [-3.464, 0], # far left
        [1.732, -3], # bottom far
        [-1.732, -3], # bottom far left
    ])

    # Initial guess for optimization
    initial_params = np.concatenate([
        initial_positions.flatten(),
        [4.0]  # initial outer hexagon side length
    ])

    # Define bounds for optimization parameters
    bounds = []
    # Positions bounds (-10, 10) for each coordinate
    for _ in range(22):
        bounds.extend([(-10, 10)] * 2)
    # Outer hexagon side length bound (positive)
    bounds.append((1e-6, 20))

    # Perform optimization
    try:
        result = minimize(
            objective_function,
            initial_params,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )

        if result.success:
            final_positions = result.x[:22].reshape(-1, 2)
            outer_side_length = result.x[-1]
        else:
            # Fallback to original method if optimization fails
            final_positions = initial_positions
            outer_side_length = 4.0

    except Exception as e:
        # Fallback to original method if optimization fails
        print(f"Optimization failed: {e}")
        final_positions = initial_positions
        outer_side_length = 4.0

    # Construct the final result array with rotation set to 0 for simplicity
    inner_hex_data = np.column_stack([final_positions, np.zeros(final_positions.shape[0])])

    # Set outer hexagon at center with appropriate rotation (rotation doesn't matter for regular hexagon)
    outer_hex_data = np.array([0, 0, 0])

    return inner_hex_data, outer_hex_data, outer_side_length
