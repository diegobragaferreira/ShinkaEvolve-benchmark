def compute_min_outer_radius(inner_hexagons):
    """Find minimum outer hexagon radius using binary search to ensure tight fit"""
    # Get all vertices of all hexagons
    all_vertices = []
    for hex_poly in inner_hexagons:
        all_vertices.extend(list(hex_poly.exterior.coords))

    # Find center of all vertices
    xs = [p[0] for p in all_vertices]
    ys = [p[1] for p in all_vertices]
    center_x = (min(xs) + max(xs)) / 2
    center_y = (min(ys) + max(ys)) / 2

    # Compute initial min and max distances
    distances = []
    for x, y in all_vertices:
        dist = math.sqrt((x - center_x)**2 + (y - center_y)**2)
        distances.append(dist)

    # Binary search for the minimal radius
    left = max(distances)  # At least the largest distance
    right = left * 2  # Upper bound for safety

    # Check if the initial radius works
    test_radius = left
    outer_hex = create_regular_hexagon(center_x, center_y, test_radius, 0)
    valid = check_containment_and_overlap(inner_hexagons, outer_hex)

    if not valid:
        # If not valid even at min radius, return a safe upper bound
        return right

    # Binary search until we find the minimum valid radius
    while right - left > 1e-6:
        mid = (left + right) / 2
        outer_hex = create_regular_hexagon(center_x, center_y, mid, 0)
        valid = check_containment_and_overlap(inner_hexagons, outer_hex)

        if valid:
            right = mid
        else:
            left = mid

    return right
