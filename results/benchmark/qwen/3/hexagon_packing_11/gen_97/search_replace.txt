def compute_outer_hexagon_radius(inner_hexagons, padding=0.01):
    """Compute minimum radius needed to contain all inner hexagons with some padding using adaptive binary search"""
    # Get all vertices of all hexagons
    all_vertices = []
    for hex_poly in inner_hexagons:
        all_vertices.extend(list(hex_poly.exterior.coords))

    # Find center of bounding box
    xs = [p[0] for p in all_vertices]
    ys = [p[1] for p in all_vertices]
    center_x = (min(xs) + max(xs)) / 2
    center_y = (min(ys) + max(ys)) / 2

    # Compute max distance from center to any vertex
    max_dist = 0
    for x, y in all_vertices:
        dist = math.sqrt((x - center_x)**2 + (y - center_y)**2)
        max_dist = max(max_dist, dist)

    # Use adaptive binary search for more efficient radius computation
    # Start with a reasonable estimate and refine based on problem scale
    estimated_radius = max_dist + padding

    # For very large radii, use coarser initial steps
    initial_step = max(0.1, estimated_radius * 0.05)  # 5% of radius or 0.1, whichever is larger

    # Binary search with adaptive tolerance
    low = estimated_radius
    high = estimated_radius + 10.0  # Conservative upper bound

    # Adaptive binary search with decreasing tolerance
    tolerance = initial_step * 0.1  # Start with 10% of initial step

    # Limit iterations to avoid infinite loops
    max_iterations = 50
    iteration = 0

    while iteration < max_iterations and (high - low) > tolerance:
        mid = (low + high) / 2

        # Create test outer hexagon
        test_outer_hex = create_regular_hexagon(0, 0, mid, 0)

        # Check if all inner hexagons fit within this radius
        all_contained = True
        for hex_poly in inner_hexagons:
            if not test_outer_hex.contains(hex_poly):
                all_contained = False
                break

        if all_contained:
            high = mid
        else:
            low = mid

        # Decrease tolerance as we get closer to optimal
        tolerance = max(1e-6, initial_step * (0.9 ** iteration))
        iteration += 1

    # Return the final computed radius
    final_radius = (low + high) / 2
    return final_radius
