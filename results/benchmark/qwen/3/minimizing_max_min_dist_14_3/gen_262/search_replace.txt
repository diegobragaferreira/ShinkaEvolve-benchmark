    def improved_local_refinement(self, initial_points, time_limit):
        """Apply advanced local optimization refinement approaches"""
        start_time = time.time()

        # Stage 1: Basin-hopping with aggressive parameters
        try:
            minimizer_kwargs = {"method": "L-BFGS-B", "bounds": [(0, 1) for _ in range(42)]}
            result_bh = basinhopping(
                self.penalty_objective,
                initial_points,
                niter=50,  # More iterations for better search
                T=0.5,     # Lower temperature for more focused search
                stepsize=0.05,  # Smaller step size for finer tuning
                minimizer_kwargs=minimizer_kwargs,
                seed=42,
                callback=lambda x, f, accepted: time.time() - start_time > time_limit - 10
            )

            if result_bh.success:
                return result_bh.x
        except Exception as e:
            warnings.warn(f"Basin-hopping optimization failed: {e}")

        # Stage 2: Gradient-based optimization with tight tolerances
        try:
            result = minimize(
                self.penalty_objective,
                initial_points,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(42)],
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12},
                callback=lambda x: time.time() - start_time > time_limit - 10
            )

            if result.success:
                return result.x
        except Exception as e:
            warnings.warn(f"Gradient optimization failed: {e}")

        # Stage 3: Multi-start local optimization with different initializations
        try:
            best_result = initial_points
            best_value = self.penalty_objective(initial_points)

            # Try different local optimizations with different starting points
            for i in range(5):
                # Add small random perturbations to create diverse starting points
                perturbed_start = initial_points + np.random.normal(0, 0.001, initial_points.shape)
                perturbed_start = np.clip(perturbed_start, 0, 1)

                result = minimize(
                    self.penalty_objective,
                    perturbed_start,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(42)],
                    options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9},
                    callback=lambda x: time.time() - start_time > time_limit - 10
                )

                if result.success:
                    current_value = self.penalty_objective(result.x)
                    if current_value < best_value:
                        best_value = current_value
                        best_result = result.x

            return best_result
        except Exception as e:
            warnings.warn(f"Multi-start local optimization failed: {e}")

        return initial_points
