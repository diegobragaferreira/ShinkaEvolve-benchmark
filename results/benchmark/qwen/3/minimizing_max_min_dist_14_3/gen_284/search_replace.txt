        """Run differential evolution with adaptive population sizing."""
        start_time = time.time()
        best_solution = None
        best_value = float('inf')

        # Multi-start optimization with different initial configurations
        configs = self.generate_initial_configurations()

        # Track convergence for adaptive population sizing
        prev_best = float('inf')
        stagnation_counter = 0
        max_stagnation = 5
        improvement_history = []
        min_improvement_threshold = 1e-8

        for i, (config_name, config) in enumerate(configs):
            if time.time() - start_time > time_limit - 15:  # Leave buffer time
                break

            base_popsize = 15
            popsize = base_popsize + i * 2  # Gradually increase popsize

            # Further increase population size if there's stagnation or poor convergence
            if len(improvement_history) >= 3:
                recent_improvements = improvement_history[-3:]
                avg_improvement = np.mean(recent_improvements)
                if avg_improvement < min_improvement_threshold:
                    popsize = min(popsize + 5, 50)  # Increase population for better exploration

            # Even more aggressive population sizing for later iterations
            if i >= 3:
                popsize = min(popsize + 5, 50)
