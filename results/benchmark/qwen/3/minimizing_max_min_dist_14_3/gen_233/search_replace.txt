def create_spherical_code_initialization(n_points: int = 14) -> np.ndarray:
    """
    Create initial point configuration based on spherical code principles.
    Uses optimized spherical code for 14 points.
    """
    # Known good configuration for 14 points on sphere from literature
    # These coordinates are normalized to unit sphere
    spherical_points = np.array([
        [0.0000, 0.0000, 1.0000],
        [0.0000, 0.0000, -1.0000],
        [0.9343, 0.0000, 0.3564],
        [-0.9343, 0.0000, 0.3564],
        [0.0000, 0.9343, 0.3564],
        [0.0000, -0.9343, 0.3564],
        [0.0000, 0.9343, -0.3564],
        [0.0000, -0.9343, -0.3564],
        [0.9343, 0.0000, -0.3564],
        [-0.9343, 0.0000, -0.3564],
        [0.3564, 0.9343, 0.0000],
        [-0.3564, 0.9343, 0.0000],
        [0.3564, -0.9343, 0.0000],
        [-0.3564, -0.9343, 0.0000]
    ])

    # Normalize to unit sphere if needed
    norms = np.linalg.norm(spherical_points, axis=1, keepdims=True)
    spherical_points = spherical_points / np.where(norms == 0, 1, norms)

    # Generate multiple candidate configurations with different perturbations
    candidates = []
    best_ratio = -float('inf')
    best_candidate = spherical_points.copy()

    # Try multiple random perturbations to find better starting points
    for perturbation_seed in range(10):
        np.random.seed(perturbation_seed)
        perturbation = np.random.normal(0, 0.02, spherical_points.shape)  # Increased perturbation
        perturbed_points = spherical_points + perturbation

        # Normalize again after perturbation
        norms = np.linalg.norm(perturbed_points, axis=1, keepdims=True)
        normalized_points = perturbed_points / np.where(norms == 0, 1, norms)

        # Calculate quality metric for this candidate
        distances = pdist(normalized_points)
        if len(distances) > 0:
            min_dist = np.min(distances)
            max_dist = np.max(distances)
            if max_dist > 0:
                ratio = min_dist / max_dist
                candidates.append((normalized_points, ratio))
                if ratio > best_ratio:
                    best_ratio = ratio
                    best_candidate = normalized_points.copy()

    return best_candidate
