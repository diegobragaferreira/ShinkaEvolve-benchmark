    def generate_diverse_initial_points():
        """Generate diverse initial point configurations"""
        initial_points_list = []

        # Strategy 1: Fibonacci sphere
        fib_points = fibonacci_sphere(14)
        np.random.seed(42)
        perturbation = np.random.normal(0, 0.05, fib_points.shape)
        points1 = fib_points + perturbation
        norms = np.linalg.norm(points1, axis=1, keepdims=True)
        points1 = points1 / np.maximum(norms, 1e-10) * 0.99
        initial_points_list.append(points1)

        # Strategy 2: Random points on sphere with structured distribution
        points2 = np.random.randn(14, 3)
        norms = np.linalg.norm(points2, axis=1, keepdims=True)
        points2 = points2 / np.maximum(norms, 1e-10) * 0.99
        initial_points_list.append(points2)

        # Strategy 3: Axis-based distribution
        points3 = np.zeros((14, 3))
        for i in range(min(14, 6)):
            # Distribute along coordinate axes
            axis = i % 3
            sign = 1 if i < 3 else -1
            points3[i] = np.eye(3)[axis] * sign * 0.9
        points3[6:] = np.random.randn(8, 3)
        norms = np.linalg.norm(points3, axis=1, keepdims=True)
        points3 = points3 / np.maximum(norms, 1e-10) * 0.99
        initial_points_list.append(points3)

        # Strategy 4: Perturbed Fibonacci with larger variance
        fib_points = fibonacci_sphere(14)
        np.random.seed(43)
        perturbation = np.random.normal(0, 0.1, fib_points.shape)
        points4 = fib_points + perturbation
        norms = np.linalg.norm(points4, axis=1, keepdims=True)
        points4 = points4 / np.maximum(norms, 1e-10) * 0.99
        initial_points_list.append(points4)

        # Strategy 5: Spherical Voronoi distribution for even spreading
        try:
            # Generate random points and create SphericalVoronoi diagram
            random_points = np.random.randn(14, 3)
            norms = np.linalg.norm(random_points, axis=1, keepdims=True)
            normalized_random = random_points / np.maximum(norms, 1e-10)

            # Create spherical Voronoi diagram and get generators
            sv = SphericalVoronoi(normalized_random)
            # Use the centers of Voronoi cells as our point configuration
            voronoi_points = sv.vertices
            # Take first 14 points (or pad if needed)
            if len(voronoi_points) >= 14:
                points5 = voronoi_points[:14]
            else:
                points5 = np.vstack([voronoi_points, np.random.randn(14-len(voronoi_points), 3)])
                norms = np.linalg.norm(points5, axis=1, keepdims=True)
                points5 = points5 / np.maximum(norms, 1e-10) * 0.99
            initial_points_list.append(points5)
        except:
            # Fallback to regular random points if SphericalVoronoi fails
            points5 = np.random.randn(14, 3)
            norms = np.linalg.norm(points5, axis=1, keepdims=True)
            points5 = points5 / np.maximum(norms, 1e-10) * 0.99
            initial_points_list.append(points5)

        return initial_points_list
