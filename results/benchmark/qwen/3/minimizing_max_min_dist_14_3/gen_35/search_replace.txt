1. Enhanced initialization strategy: Instead of just using Fibonacci spiral, I'm implementing a multi-strategy initialization that includes:
   - Spherical Fibonacci arrangement (existing)
   - Random uniform distribution (fallback)
   - Perturbed regularized arrangement (additional diversity)

2. Adaptive optimization parameters: Adding a dynamic approach to adjust optimization parameters based on convergence behavior, specifically:
   - Monitoring the improvement rate during differential evolution
   - Increasing population size when convergence stalls
   - Using early stopping criteria to prevent unnecessary computation
   - Combining multiple optimization strategies with probabilistic selection

These changes aim to create a more robust and efficient optimization pipeline that can better escape local optima and converge toward the target ratio of ~0.4898.

import time

import warnings

class PointInitializer:
    """Handles various point initialization strategies."""

    @staticmethod
    def fibonacci_spherical(n: int) -> np.ndarray:
        """Initialize points using Fibonacci spiral on sphere."""
        points = []
        for i in range(n):
            phi = np.arccos(1 - 2 * (i / (n - 1)))
            theta = np.sqrt(n) * phi
            x = np.sin(phi) * np.cos(theta)
            y = np.sin(phi) * np.sin(theta)
            z = np.cos(phi)
            points.append([x, y, z])

        return np.array(points)

    @staticmethod
    def random_uniform(n: int, d: int = 3, seed: int = 42) -> np.ndarray:
        """Initialize points using uniform random distribution."""
        np.random.seed(seed)
        return np.random.rand(n, d)

    @staticmethod
    def regularized_arrangement(n: int, d: int = 3, seed: int = 42) -> np.ndarray:
        """Initialize points using a regularized arrangement for better spread."""
        np.random.seed(seed)
        # Create a grid-like structure with some randomness
        points = np.random.rand(n, d)
        # Add slight perturbations to ensure good distribution
        points += np.random.normal(0, 0.02, points.shape)
        return np.clip(points, 0, 1)

    @classmethod
    def initialize_points(cls, n: int = 14, d: int = 3, seed: int = 42) -> np.ndarray:
        """Initialize points with multiple strategies and selection based on diversity."""
        np.random.seed(seed)

        # Strategy 1: Fibonacci spiral on sphere
        fib_points = cls.fibonacci_spherical(n)
        # Normalize to unit sphere
        norms = np.linalg.norm(fib_points, axis=1, keepdims=True)
        fib_points = fib_points / np.max(norms)
        # Scale and shift to [0,1]^3
        fib_points = fib_points * 0.5 + 0.5
        # Add small random perturbation for better mixing
        fib_points += np.random.normal(0, 0.01, fib_points.shape)
        fib_points = np.clip(fib_points, 0, 1)

        # Strategy 2: Random uniform
        rand_points = cls.random_uniform(n, d, seed)

        # Strategy 3: Regularized arrangement
        reg_points = cls.regularized_arrangement(n, d, seed)

        # Evaluate which initialization gives better spread
        # Using minimum distance as proxy for spread quality
        def compute_min_distance(points):
            if len(points) < 2:
                return 0.0
            try:
                distances = pdist(points)
                if len(distances) == 0:
                    return 0.0
                return np.min(distances)
            except:
                return 0.0

        fib_quality = compute_min_distance(fib_points)
        rand_quality = compute_min_distance(rand_points)
        reg_quality = compute_min_distance(reg_points)

        # Select the best initialization based on quality
        if fib_quality >= rand_quality and fib_quality >= reg_quality:
            return fib_points
        elif rand_quality >= reg_quality:
            return rand_points
        else:
            return reg_points

        return ratio + penalty

    def adaptive_differential_evolution_optimize(self, initial_points: np.ndarray,
                                              max_time: float = 300.0) -> np.ndarray:
        """Optimize using differential evolution with adaptive parameters."""
        x0 = initial_points.flatten()
        bounds = [(0, 1)] * self.total_variables

        # Initialize adaptive parameters
        current_popsize = 15
        maxiter = 500
        stagnation_counter = 0
        best_value = float('inf')
        stagnation_threshold = 10
        max_stagnation = 20

        # History tracking for convergence
        history = []

        try:
            # Run multiple rounds of DE with adaptive population size
            for round_num in range(3):
                # Check if we've exceeded time limits
                if len(history) > 0 and len(history) % 50 == 0:
                    # Check for convergence
                    if len(history) >= 2:
                        recent_change = abs(history[-1] - history[-2])
                        if recent_change < 1e-8:
                            stagnation_counter += 1
                        else:
                            stagnation_counter = 0

                    # Increase population size if stagnated
                    if stagnation_counter >= stagnation_threshold and current_popsize < 30:
                        current_popsize = min(current_popsize + 5, 30)
                        stagnation_counter = 0  # Reset counter after adjustment

                    # Stop if too many stagnations
                    if stagnation_counter >= max_stagnation:
                        break

                # Run differential evolution with current settings
                de_options = {
                    'maxiter': maxiter // 3,
                    'popsize': current_popsize,
                    'tol': 1e-6,

                    'seed': 42,
                    'disp': False

                result = differential_evolution(
                    self._penalty_objective,
                    bounds,
                    **de_options
                )

                # Track history
                history.append(result.fun)

                # Update best value
                if result.fun < best_value:
                    best_value = result.fun
                    stagnation_counter = 0  # Reset stagnation counter on improvement

            return result.x.reshape(-1, self.dimension)
