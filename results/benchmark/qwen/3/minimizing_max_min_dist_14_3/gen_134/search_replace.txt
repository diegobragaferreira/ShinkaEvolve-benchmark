    def evaluate_initialization(points):
        """Fast evaluation of initialization quality with better numerical stability"""
        distances = pdist(points)
        if len(distances) == 0:
            return 0
        d_min = np.min(distances)
        d_max = np.max(distances)
        # Use a small epsilon to avoid numerical issues
        epsilon = 1e-12
        if d_max > epsilon:
            return d_min / d_max
        return 0

    def voronoi_uniformity_score(points):
        """Evaluate how uniformly distributed points are using Voronoi cell areas"""
        try:
            # For small point sets, we approximate uniformity by examining distances
            # This is a simplified approach that works well for our case
            distances = pdist(points)
            if len(distances) == 0:
                return 0
            # High uniformity means distances vary less - we'll use inverse variance
            return 1.0 / (np.var(distances) + 1e-12)
        except:
            return 0
