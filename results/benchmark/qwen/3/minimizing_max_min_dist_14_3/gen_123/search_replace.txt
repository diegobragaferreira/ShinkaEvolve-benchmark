    def spherical_objective(x):
        """Objective function that keeps points constrained to unit sphere"""
        points = x.reshape(-1, 3)

        # Project all points onto unit sphere to maintain constraint
        norms = np.linalg.norm(points, axis=1, keepdims=True)
        # Avoid division by zero
        norms = np.where(norms == 0, 1, norms)
        points_on_sphere = points / norms

        # Calculate pairwise distances on sphere
        distances = pdist(points_on_sphere)

        # Handle edge cases
        if len(distances) == 0:
            return -1.0

        # Remove any NaN or infinite values
        distances = distances[np.isfinite(distances)]

        if len(distances) == 0:
            return -1.0

        # Calculate min and max distances
        d_min = np.min(distances)
        d_max = np.max(distances)

        # Return negative ratio to maximize (since we're minimizing)
        if d_max <= 0:
            return -1.0
        return -d_min / d_max

    def project_to_unit_cube(points):
        """Project points to unit cube [0,1]^3 with proper scaling"""
        # Center points around origin
        centered = points - np.mean(points, axis=0)
        # Scale to fit within unit cube
        max_coord = np.max(np.abs(centered))
        if max_coord > 0:
            scaled = centered / (2 * max_coord) + 0.5
        else:
            scaled = centered + 0.5
        return np.clip(scaled, 0, 1)

    def get_spherical_initialization():
        """Create better initial configuration that respects spherical constraints"""
        # Start with Fibonacci spiral on sphere
        np.random.seed(42)
        points = []

            radius = np.sqrt(1 - y * y)  # radius at y

            theta = phi * i  # golden angle increment

            x = np.cos(theta) * radius
            z = np.sin(theta) * radius

            points.append([x, y, z])

        initial_points = np.array(points)

        norms = np.where(norms == 0, 1, norms)
        initial_points = initial_points / norms
