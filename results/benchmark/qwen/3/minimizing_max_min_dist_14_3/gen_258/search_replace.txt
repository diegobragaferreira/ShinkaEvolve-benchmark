def create_symmetric_variants(points: np.ndarray, num_variants: int = 12) -> np.ndarray:
    """
    Create diverse symmetric variants including rotations, reflections, and permutations.
    """
    variants = [points]

    # Create multiple rotation variants using Rodrigues' rotation formula
    for i in range(num_variants // 3):
        # Random rotation around random axis
        angle = 2 * np.pi * np.random.random()
        axis = np.random.randn(3)
        axis = axis / np.linalg.norm(axis)

        # Rodrigues' rotation formula
        cos_a = np.cos(angle)
        sin_a = np.sin(angle)
        kx, ky, kz = axis
        rot_matrix = np.array([
            [cos_a + kx*kx*(1-cos_a), kx*ky*(1-cos_a) - kz*sin_a, kx*kz*(1-cos_a) + ky*sin_a],
            [kx*ky*(1-cos_a) + kz*sin_a, cos_a + ky*ky*(1-cos_a), ky*kz*(1-cos_a) - kx*sin_a],
            [kx*kz*(1-cos_a) - ky*sin_a, ky*kz*(1-cos_a) + kx*sin_a, cos_a + kz*kz*(1-cos_a)]
        ])

        rotated = points @ rot_matrix.T
        variants.append(rotated)

    # Create reflection variants
    reflections = [
        np.array([[-1, 0, 0], [0, 1, 0], [0, 0, 1]]),  # Reflect across yz-plane
        np.array([[1, 0, 0], [0, -1, 0], [0, 0, 1]]),  # Reflect across xz-plane
        np.array([[1, 0, 0], [0, 1, 0], [0, 0, -1]]),  # Reflect across xy-plane
        np.array([[-1, 0, 0], [0, -1, 0], [0, 0, 1]]),  # Reflect across origin (xz)
        np.array([[-1, 0, 0], [0, 1, 0], [0, 0, -1]]),  # Reflect across yz-plane then xy-plane
        np.array([[1, 0, 0], [0, -1, 0], [0, 0, -1]]),  # Reflect across xz-plane then xy-plane
    ]

    for ref_matrix in reflections:
        reflected = points @ ref_matrix.T
        variants.append(reflected)

    # Add permutation variants (swap coordinates)
    perms = [
        [0, 1, 2],  # identity
        [0, 2, 1],  # swap y and z
        [1, 0, 2],  # swap x and y
        [1, 2, 0],  # cyclic perm x->y, y->z, z->x
        [2, 0, 1],  # cyclic perm x->z, z->y, y->x
        [2, 1, 0],  # swap x and z
    ]

    for perm in perms:
        permuted = points[:, perm]
        variants.append(permuted)
