    def local_refinement(points, max_iter=150):
        """Local refinement using dual L-BFGS-B optimization strategies."""
        try:
            # Convert to flat array for optimization
            x0 = points.flatten()

            def obj(x):
                points_refined = x.reshape((14, 3))
                # Ensure points are on sphere
                points_refined = normalize_to_sphere(points_refined)
                cube_points = project_to_cube(points_refined)
                distances = pdist(cube_points)
                min_dist = np.min(distances)
                max_dist = np.max(distances)
                if max_dist > 0:
                    return -min_dist / max_dist
                else:
                    return 0

            # First refinement with loose tolerances for quick convergence
            result1 = minimize(
                obj,
                x0,
                method='L-BFGS-B',
                bounds=[(None, None)] * 42,  # No bounds for internal optimization
                options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': max_iter//2},
                tol=1e-10
            )

            # Second refinement with tight tolerances for precision
            result2 = minimize(
                obj,
                result1.x,
                method='L-BFGS-B',
                bounds=[(None, None)] * 42,  # No bounds for internal optimization
                options={'ftol': 1e-14, 'gtol': 1e-14, 'maxiter': max_iter//2},
                tol=1e-14
            )

            refined_points = result2.x.reshape((14, 3))
            refined_points = normalize_to_sphere(refined_points)
            return refined_points

        except Exception:
            return points
