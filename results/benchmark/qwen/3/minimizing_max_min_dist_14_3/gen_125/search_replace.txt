    def local_optimize(self, initial_points: np.ndarray,
                      max_time: float = 60.0) -> np.ndarray:
        """Optimize using L-BFGS-B for local refinement with adaptive tolerance tightening."""
        x0 = initial_points.flatten()
        bounds = [(0, 1)] * self.total_variables

        try:
            # Start with loose tolerances for faster initial convergence
            ftol_start = 1e-6
            gtol_start = 1e-6
            ftol_final = 1e-12
            gtol_final = 1e-12

            # Try with initial loose tolerances
            result = minimize(
                self._penalty_objective,
                x0,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': ftol_start, 'gtol': gtol_start}
            )

            # If we achieved good improvement, try with tighter tolerances
            if result.success and abs(result.fun) > 1e-10:
                # Check if we have room for improvement by running with tighter tolerances
                result_tight = minimize(
                    self._penalty_objective,
                    result.x,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 500, 'ftol': ftol_final, 'gtol': gtol_final}
                )

                if result_tight.success and result_tight.fun < result.fun:
                    return result_tight.x.reshape(-1, self.dimension)

            return result.x.reshape(-1, self.dimension)
        except Exception:
            return initial_points.reshape(-1, self.dimension)
