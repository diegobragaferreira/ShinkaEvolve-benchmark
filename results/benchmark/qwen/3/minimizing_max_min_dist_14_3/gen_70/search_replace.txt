    def initialize_cube_grid_points(n_points):
        """Initialize points in a 3D cube grid with better spacing"""
        # Find appropriate grid size
        grid_size = int(np.ceil(n_points**(1/3)))
        # Use a slightly larger grid to ensure we have enough points
        grid_size = max(grid_size, 3)
        coords = np.linspace(0, 1, grid_size)
        grid_points = []

        # Create a more evenly spaced grid pattern
        for i in range(grid_size):
            for j in range(grid_size):
                for k in range(grid_size):
                    if len(grid_points) < n_points:
                        # Add slight jitter to avoid perfect grid artifacts
                        jitter_i = np.random.uniform(-0.1, 0.1) if len(grid_points) < n_points else 0
                        jitter_j = np.random.uniform(-0.1, 0.1) if len(grid_points) < n_points else 0
                        jitter_k = np.random.uniform(-0.1, 0.1) if len(grid_points) < n_points else 0

                        grid_points.append([
                            np.clip(coords[i] + jitter_i, 0, 1),
                            np.clip(coords[j] + jitter_j, 0, 1),
                            np.clip(coords[k] + jitter_k, 0, 1)
                        ])

        return np.array(grid_points[:n_points])

        recent_improvements = []

            if stagnation_count > 3 and current_popsize < 30:
                current_popsize = min(current_popsize + 5, 30)

                result = differential_evolution(

                    mutation=(0.5, 1.0),  # Adaptive mutation range
                    recombination=0.9,    # Higher recombination for better exploration
                    disp=False
                )

                # Fall back to smaller population if needed
                try:
                    result = differential_evolution(
                        objective_func,
                        bounds,
                        seed=42 + iteration,
                        maxiter=10,
                        popsize=max(5, current_popsize - 5),

                        popsize=10,
                        tol=1e-12,
                        mutation=(0.5, 1.0),
                        recombination=0.7,
                        disp=False
                    )

            # Check for improvement
            current_best = -result.fun
            improvement = current_best - prev_best

            recent_improvements.append(improvement)
            if len(recent_improvements) > 5:
                recent_improvements.pop(0)

            if len(recent_improvements) == 5 and all(abs(impr) < min_improvement for impr in recent_improvements):

            if improvement > improvement_threshold:
                stagnation_count = 0
            else:
                stagnation_count += 1

            prev_best = current_best
