    Args:

    Returns:
        Initial point configuration
    """
    np.random.seed(42)

    # Strategy 1: Start with vertices of a regular icosahedron (good base distribution)
    # Vertices of regular icosahedron scaled to unit sphere
    phi = (1 + np.sqrt(5)) / 2  # golden ratio
    vertices = np.array([
        [-1,  phi,  0],
        [ 1,  phi,  0],
        [-1, -phi,  0],
        [ 1, -phi,  0],
        [ 0, -1,  phi],
        [ 0,  1,  phi],
        [ 0, -1, -phi],
        [ 0,  1, -phi],
        [ phi,  0, -1],
        [ phi,  0,  1],
        [-phi,  0, -1],
        [-phi,  0,  1]
    ])

    # If we need more than 12 points, use Fibonacci spiral for additional points
    if n <= 12:
        points = vertices[:n].copy()
    else:
        # Start with icosahedron vertices and add Fibonacci spiral points
        points = vertices.copy()

        # Add remaining points using Fibonacci spiral with careful distribution
        extra_points_needed = n - 12
        for i in range(extra_points_needed):
            # Distribute remaining points more evenly
            phi = np.arccos(1 - 2 * ((12 + i) / (n - 1)))
            theta = np.sqrt(n) * phi
            x = np.sin(phi) * np.cos(theta)
            y = np.sin(phi) * np.sin(theta)
            z = np.cos(phi)
            points = np.vstack([points, [x, y, z]])

    # Normalize to unit sphere
    norms = np.linalg.norm(points, axis=1, keepdims=True)
    points = points / np.max(norms)

    # Improve distribution using spherical Voronoi-inspired method
    # Create a better spread by using spherical Voronoi centroids when possible
    if n > 12:
        # For more points, refine the distribution using iterative method
        # This creates a more uniform distribution that's beneficial for our optimization
        for _ in range(3):  # Multiple refinement steps
            # Create spherical Voronoi diagram (conceptually)
            # We'll do a simple iterative improvement focusing on spacing
            for i in range(len(points)):
                # Move point slightly away from neighbors to improve spacing
                neighbor_indices = [(j, np.linalg.norm(points[i] - points[j]))
                                  for j in range(len(points)) if i != j]
                neighbor_indices.sort(key=lambda x: x[1])  # Sort by distance

                # Move point away from closest neighbors
                if neighbor_indices:
                    # Simple repulsion from close neighbors
                    repulsion = np.zeros(3)
                    for j, dist in neighbor_indices[:3]:  # Consider 3 nearest neighbors
                        if dist < 1.0:  # Only repel when very close
                            repulsion += (points[i] - points[j]) * (1.0 - dist)

                    if np.linalg.norm(repulsion) > 0:
                        points[i] += 0.01 * repulsion / np.linalg.norm(repulsion)

            # Keep points on sphere
            norms = np.linalg.norm(points, axis=1, keepdims=True)
            points = points / np.maximum(norms, 1e-10)  # Avoid division by zero

    # Scale and shift to [0,1]^3
    points = points * 0.5 + 0.5

    # Add small random perturbation to avoid symmetry issues
    points += np.random.normal(0, 0.005, points.shape)

    # Clip to ensure within bounds
    points = np.clip(points, 0, 1)

    return points
