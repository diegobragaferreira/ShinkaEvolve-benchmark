    def _constraint_sphere(self, x):
        """Constraint function for unit sphere: sum of squares should be 1"""
        points = x.reshape(-1, 3)
        norms_squared = np.sum(points**2, axis=1)
        return norms_squared - 1.0

    def _local_refinement(self, points):
        """Apply local refinement to improve final solution"""
        def objective_local(x):
            points_local = x.reshape(-1, 3)
            # Normalize to unit sphere
            norms = np.linalg.norm(points_local, axis=1, keepdims=True)
            points_local = points_local / np.maximum(norms, 1e-10)
            distances = pdist(points_local)

            if len(distances) == 0:
                return -np.inf

            d_min = np.min(distances)
            d_max = np.max(distances)

            if d_max > 1e-12:
                return -(d_min / d_max)
            else:
                return -np.inf

        try:
            x0_refine = points.flatten()
            bounds = [(0, 1)] * self.n_points * 3

            # Create constraint for unit sphere
            cons = [{'type': 'eq', 'fun': self._constraint_sphere}]

            result_refine = minimize(
                objective_local,
                x0_refine,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,

            return points
