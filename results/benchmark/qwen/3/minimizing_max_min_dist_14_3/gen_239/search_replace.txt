    def fibonacci_sphere_sampling(n):
        """Generate points on a unit sphere using Fibonacci spiral method with better distribution"""
        points = []
        phi = np.pi * (3 - np.sqrt(5))  # golden angle

        for i in range(n):
            y = 1 - (i / (n - 1)) * 2  # y goes from 1 to -1
            radius = np.sqrt(1 - y * y)  # radius at y

            theta = phi * i  # golden angle increment

            x = np.cos(theta) * radius
            z = np.sin(theta) * radius

            points.append([x, y, z])

    def icosahedron_sphere_sampling(n):
        """Generate points on a unit sphere using icosahedron-based method for optimal distribution"""
        # Create icosahedron vertices
        phi = (1 + np.sqrt(5)) / 2  # golden ratio
        vertices = np.array([
            [-1,  phi,  0],
            [ 1,  phi,  0],
            [-1, -phi,  0],
            [ 1, -phi,  0],
            [ 0, -1,  phi],
            [ 0,  1,  phi],
            [ 0, -1, -phi],
            [ 0,  1, -phi],
            [ phi,  0, -1],
            [ phi,  0,  1],
            [-phi,  0, -1],
            [-phi,  0,  1]
        ])

        # Normalize vertices to unit sphere
        vertices = vertices / np.linalg.norm(vertices, axis=1, keepdims=True)

        # For 14 points, we'll use the icosahedron vertices plus additional points
        # But since we want exactly 14 points, we'll use a subset and add a few more
        if n <= 12:
            # Just use some vertices of the icosahedron
            selected_vertices = vertices[:n]
        else:
            # Use all 12 vertices and add extra points
            selected_vertices = vertices.copy()
            # Add points along edges or faces for additional points
            # For simplicity, add some perturbed vertices
            np.random.seed(42)
            for i in range(n - 12):
                # Pick two random vertices and place a point between them
                idx1, idx2 = np.random.choice(12, 2, replace=False)
                point = (vertices[idx1] + vertices[idx2]) / 2
                # Normalize to unit sphere
                point = point / np.linalg.norm(point)
                selected_vertices = np.vstack([selected_vertices, point])

        # Ensure we have exactly n points
        if len(selected_vertices) > n:
            selected_vertices = selected_vertices[:n]
        elif len(selected_vertices) < n:
            # Fill with random points on sphere
            np.random.seed(42)
            extra_points = np.random.rand(n - len(selected_vertices), 3)
            # Normalize to unit sphere
            extra_points = extra_points / np.linalg.norm(extra_points, axis=1, keepdims=True)
            selected_vertices = np.vstack([selected_vertices, extra_points])

        return selected_vertices
