def initialize_points(n: int = 14, d: int = 3) -> np.ndarray:
    """
    Initialize points using an enhanced approach that combines Fibonacci distribution
    with convex hull optimization for better initial spread.

    Args:
        n: number of points
        d: dimensionality

    Returns:
        Initial point configuration
    """
    np.random.seed(42)

    # Strategy 1: Fibonacci spiral on sphere
    points = []
    for i in range(n):
        phi = np.arccos(1 - 2 * (i / (n - 1)))
        theta = np.sqrt(n) * phi

        # Add structured perturbation to avoid perfect symmetry
        perturbation = np.random.normal(0, 0.05) if i % 3 == 0 else np.random.normal(0, 0.02)
        theta += perturbation

        x = np.sin(phi) * np.cos(theta)
        y = np.sin(phi) * np.sin(theta)
        z = np.cos(phi)
        points.append([x, y, z])

    points = np.array(points)

    # Normalize to unit sphere
    norms = np.linalg.norm(points, axis=1, keepdims=True)
    points = points / np.max(norms)

    # Apply convex hull-based optimization to improve initial spread
    # Generate multiple candidate configurations and select the best
    best_points = points.copy()
    best_min_dist = 0

    # Try different perturbation strategies
    for perturbation_seed in range(5):
        np.random.seed(perturbation_seed * 100)
        perturbed_points = points.copy()

        # Add random perturbations to create variation
        noise = np.random.normal(0, 0.03, points.shape)
        perturbed_points += noise

        # Normalize again after perturbation
        norms = np.linalg.norm(perturbed_points, axis=1, keepdims=True)
        perturbed_points = perturbed_points / np.max(norms)

        # Convert to cube coordinates
        cube_coords = (perturbed_points + 1) / 2

        # Calculate minimum distance
        if len(cube_coords) >= 2:
            distances = pdist(cube_coords)
            if len(distances) > 0:
                min_dist = np.min(distances)
                if min_dist > best_min_dist:
                    best_min_dist = min_dist
                    best_points = perturbed_points.copy()

    # Scale and shift to [0,1]^3
    points = best_points * 0.5 + 0.5

    # Final clip to ensure within bounds
    points = np.clip(points, 0, 1)

    return points
