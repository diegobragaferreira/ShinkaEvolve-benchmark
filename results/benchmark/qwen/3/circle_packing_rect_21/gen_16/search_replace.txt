import numpy as np


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions - since perimeter = 4, width + height = 2
    # Using a 1x1 square for simplicity
    rect_width = 1.0
    rect_height = 1.0

    # Better initialization - place circles in a structured way
    circles = np.zeros((21, 3))

    # Create a more systematic distribution
    # Use a pattern that tries to maximize space utilization
    rows = 3
    cols = 7

    # Calculate spacing
    x_spacing = rect_width / (cols + 1)
    y_spacing = rect_height / (rows + 1)

    # Fill positions
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= 21:
                break
            x = (j + 1) * x_spacing
            y = (i + 1) * y_spacing
            # Set small initial radius
            r = 0.03
            circles[idx] = [x, y, r]
            idx += 1

    # Simple optimization: try to increase radii while maintaining constraints
    for _ in range(100):  # Limited iterations for speed
        improved = False
        for i in range(21):
            x, y, r = circles[i]

            # Calculate maximum possible radius at this location
            max_radius = min(x, rect_width - x, y, rect_height - y)

            # Check collisions with other circles
            for j in range(21):
                if i != j:
                    x2, y2, r2 = circles[j]
                    distance = np.sqrt((x - x2)**2 + (y - y2)**2)
                    max_radius = min(max_radius, distance - r2)

            # If we can increase the radius, do it
            if max_radius > r and max_radius > 0:
                circles[i][2] = max_radius
                improved = True

        # If no improvement, stop early
        if not improved:
            break

    # Final validation to ensure all constraints are satisfied
    for i in range(21):
        x, y, r = circles[i]
        # Ensure circles stay within bounds
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            # Reset to a safe value
            circles[i] = [0.5, 0.5, 0.02]

    return circles
