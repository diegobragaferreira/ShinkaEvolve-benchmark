import numpy as np
from scipy.optimize import differential_evolution
import random

# Problem parameters
RECT_PERIMETER = 4
RECT_WIDTH = 1.0  # Square container for simplicity
RECT_HEIGHT = 1.0
NUM_CIRCLES = 21

def objective_function(vars):
    """
    Objective function to maximize the sum of radii
    vars: flattened array of [x1, y1, r1, x2, y2, r2, ..., x21, y21, r21]
    """
    # Reshape variables into circles
    circles = vars.reshape(-1, 3)
    total_radius = np.sum(circles[:, 2])

    # Penalty for constraints violation
    penalty = 0

    # Boundary constraint penalty
    for x, y, r in circles:
        if x - r < 0 or x + r > RECT_WIDTH or y - r < 0 or y + r > RECT_HEIGHT:
            penalty += 10000000

    # Overlap constraint penalty
    for i in range(len(circles)):
        for j in range(i+1, len(circles)):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            distance = np.sqrt((x2-x1)**2 + (y2-y1)**2)
            if distance < (r1 + r2):  # Circles overlapping
                penalty += 100000000

    # Return negative because we minimize in scipy but want to maximize
    return -total_radius + penalty

def create_initial_guess():
    """Create an initial guess for the optimization"""
    # Start with a grid-based heuristic approach
    circles = []

    # Calculate grid dimensions
    grid_size = int(np.ceil(np.sqrt(NUM_CIRCLES)))

    # Place circles in a grid pattern
    radius = 0.1
    spacing = radius * 2.2

    for i in range(grid_size):
        for j in range(grid_size):
            if len(circles) >= NUM_CIRCLES:
                break
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            # Ensure we don't go out of bounds
            x = max(radius, min(RECT_WIDTH - radius, x))
            y = max(radius, min(RECT_HEIGHT - radius, y))
            circles.append([x, y, radius])

    # Fill remaining slots
    while len(circles) < NUM_CIRCLES:
        x = random.uniform(radius, RECT_WIDTH - radius)
        y = random.uniform(radius, RECT_HEIGHT - radius)
        circles.append([x, y, radius * 0.5])

    return np.array(circles).flatten()

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Set seeds for reproducibility
    np.random.seed(42)
    random.seed(42)

    # Create bounds for each variable (x, y, r for each of 21 circles)
    bounds = []
    for _ in range(NUM_CIRCLES):
        # x bounds
        bounds.extend([(0.001, RECT_WIDTH - 0.001)])
        # y bounds
        bounds.extend([(0.001, RECT_HEIGHT - 0.001)])
        # r bounds (positive radius, limited by container size)
        bounds.extend([(0.001, 0.4)])  # Max radius of 0.4 should be safe

    try:
        # Run differential evolution optimization
        result = differential_evolution(
            objective_function,
            bounds,
            maxiter=100,
            popsize=15,
            tol=1e-6,
            seed=42,
            callback=None,
            disp=False
        )

        # Extract the best solution
        best_circles = result.x.reshape(-1, 3)

        # Clip values to ensure they're within bounds
        best_circles[:, 0] = np.clip(best_circles[:, 0], 0.001, RECT_WIDTH - 0.001)
        best_circles[:, 1] = np.clip(best_circles[:, 1], 0.001, RECT_HEIGHT - 0.001)
        best_circles[:, 2] = np.clip(best_circles[:, 2], 0.001, 0.4)

        return best_circles

    except Exception as e:
        # Fallback to initial heuristic if optimization fails
        print(f"Optimization failed: {e}")
        initial_guess = create_initial_guess().reshape(-1, 3)
        return initial_guess
