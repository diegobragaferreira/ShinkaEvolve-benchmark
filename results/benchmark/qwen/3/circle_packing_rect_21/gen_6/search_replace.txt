import numpy as np
import random
import time

# Global constants
RECT_PERIMETER = 4.0
RECT_WIDTH = 1.0
RECT_HEIGHT = 1.0
NUM_CIRCLES = 21
MAX_ITERATIONS = 5000

def check_collision(circle1, circle2):
    """Check if two circles collide."""
    x1, y1, r1 = circle1
    x2, y2, r2 = circle2
    distance = np.sqrt((x2 - x1)**2 + (y2 - y1)**2)
    return distance < (r1 + r2)

def is_valid_placement(circle):
    """Check if a circle is within bounds."""
    x, y, r = circle
    return (r >= 0 and
            x >= r and x <= RECT_WIDTH - r and
            y >= r and y <= RECT_HEIGHT - r)

def validate_and_fix_placement(circles):
    """Ensure all circles are valid and non-overlapping."""
    # Try to fix collisions by adjusting positions
    improved = True
    iterations = 0

    while improved and iterations < 100:
        improved = False
        for i in range(len(circles)):
            # Check if circle is valid
            if not is_valid_placement(circles[i]):
                # Fix invalid position
                circles[i][0] = random.uniform(circles[i][2], RECT_WIDTH - circles[i][2])
                circles[i][1] = random.uniform(circles[i][2], RECT_HEIGHT - circles[i][2])
                improved = True

            # Check for collisions with other circles
            for j in range(len(circles)):
                if i != j and check_collision(circles[i], circles[j]):
                    # Move circle to avoid collision
                    circles[i][0] += random.uniform(-0.1, 0.1)
                    circles[i][1] += random.uniform(-0.1, 0.1)
                    # Clip to bounds
                    circles[i][0] = np.clip(circles[i][0], circles[i][2], RECT_WIDTH - circles[i][2])
                    circles[i][1] = np.clip(circles[i][1], circles[i][2], RECT_HEIGHT - circles[i][2])
                    improved = True
        iterations += 1

    return circles

def calculate_total_radius(circles):
    """Calculate total sum of radii."""
    return np.sum(circles[:, 2])

def generate_random_solution():
    """Generate a random valid solution."""
    circles = np.zeros((NUM_CIRCLES, 3))

    # Generate random positions and radii
    for i in range(NUM_CIRCLES):
        # Start with a small radius
        r = random.uniform(0.01, 0.15)
        # Random position within bounds
        x = random.uniform(r, RECT_WIDTH - r)
        y = random.uniform(r, RECT_HEIGHT - r)
        circles[i] = [x, y, r]

    # Fix any initial collisions
    circles = validate_and_fix_placement(circles)

    return circles

def local_improvement(circles, max_iterations=100):
    """Try to locally improve the solution by adjusting positions."""
    best_circles = circles.copy()
    best_radius_sum = calculate_total_radius(best_circles)

    for iteration in range(max_iterations):
        # Try to slightly increase radii while maintaining constraints
        test_circles = best_circles.copy()
        improved = False

            old_radius = test_circles[i][2]

            new_radius = old_radius * 1.05
            test_circles[i][2] = new_radius

            # Check if we're still valid
            if is_valid_placement(test_circles[i]):
                # Check if we can place other circles without overlap
                temp_circles = test_circles.copy()
                valid = True
                for j in range(NUM_CIRCLES):
                    if j != i:

                        temp_circles[j][1] += random.uniform(-0.02, 0.02)

                        temp_circles[j][0] = np.clip(temp_circles[j][0], temp_circles[j][2], RECT_WIDTH - temp_circles[j][2])
                        temp_circles[j][1] = np.clip(temp_circles[j][1], temp_circles[j][2], RECT_HEIGHT - temp_circles[j][2])

            if current_radius_sum > best_radius_sum:
                best_circles = test_circles.copy()
                best_radius_sum = current_radius_sum
        else:
            # If no improvement, try repositioning some circles

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    random.seed(42)  # For reproducibility
    np.random.seed(42)

    start_time = time.time()

    # Start with a good initial solution
    best_solution = generate_random_solution()
    best_radius_sum = calculate_total_radius(best_solution)

    print(f"Initial solution radius sum: {best_radius_sum:.6f}")

    # Iteratively improve the solution
    for iteration in range(MAX_ITERATIONS):
        # Generate a new candidate solution
        candidate_solution = generate_random_solution()

        # Local improvement step
        candidate_solution = local_improvement(candidate_solution)

        candidate_radius_sum = calculate_total_radius(candidate_solution)

        # Accept better solutions
        if candidate_radius_sum > best_radius_sum:
            best_solution = candidate_solution.copy()
            best_radius_sum = candidate_radius_sum
            print(f"New best at iteration {iteration}: {best_radius_sum:.6f}")

        # Occasionally reset to avoid getting stuck
        if iteration % 500 == 0 and iteration > 0:
            print(f"Iteration {iteration}: Current best = {best_radius_sum:.6f}")
            # Reset to a fresh solution occasionally
            current_solution = generate_random_solution()
            current_solution = local_improvement(current_solution)
            current_radius = calculate_total_radius(current_solution)
            if current_radius > best_radius_sum:
                best_solution = current_solution.copy()
                best_radius_sum = current_radius

    end_time = time.time()
    print(f"Optimization completed in {end_time - start_time:.2f} seconds")
    print(f"Final sum of radii: {best_radius_sum:.6f}")

    return best_solution
