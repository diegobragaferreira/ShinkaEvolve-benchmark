def generate_voronoi_lloyd_initialization(n_points: int, width: float, height: float,
                                       max_iterations: int = 100) -> np.ndarray:
    """
    Generate well-distributed points using Voronoi-based Lloyd's algorithm.
    This creates a more uniform initial configuration than simple grids.
    """
    # Start with random points
    points = np.random.rand(n_points, 2) * 0.8 + 0.1  # Keep away from edges
    points[:, 0] *= width
    points[:, 1] *= height

    # Apply Lloyd's algorithm
    for _ in range(max_iterations):
        try:
            # Create Voronoi diagram
            vor = Voronoi(points)

            # Compute centroids of Voronoi cells
            new_points = []
            for i in range(len(points)):
                # Get vertices of Voronoi cell for point i
                region = vor.regions[vor.point_region[i]]
                if -1 in region or len(region) < 3:
                    # Skip invalid regions or infinite ones
                    new_points.append(points[i])
                    continue

                # Get vertices of the region
                vertices = np.array([vor.vertices[r] for r in region if r >= 0])

                # Filter vertices within bounds
                valid_vertices = vertices[
                    (vertices[:, 0] >= 0) &
                    (vertices[:, 0] <= width) &
                    (vertices[:, 1] >= 0) &
                    (vertices[:, 1] <= height)
                ]

                if len(valid_vertices) < 3:
                    new_points.append(points[i])
                    continue

                # Compute centroid
                centroid = np.mean(valid_vertices, axis=0)
                new_points.append(centroid)

            # Update points
            new_points = np.array(new_points)

            # Convergence check
            if np.allclose(points, new_points, atol=1e-6):
                break

            points = new_points
        except Exception:
            # Fallback to simple random if Voronoi computation fails
            points = np.random.rand(n_points, 2) * 0.8 + 0.1
            points[:, 0] *= width
            points[:, 1] *= height

    # Ensure all points are within bounds
    points[:, 0] = np.clip(points[:, 0], 0.05, width - 0.05)
    points[:, 1] = np.clip(points[:, 1], 0.05, height - 0.05)

    return points
