    def gradient_based_local_search(circles: np.ndarray, rect_width: float, rect_height: float,
                                  max_iterations: int = 200) -> np.ndarray:
        """Advanced local search using numerical gradient computation to maximize radii"""
        current = circles.copy()
        eps = 1e-5  # Small epsilon for gradient calculation

        for iteration in range(max_iterations):
            updated = False
            new_circles = current.copy()

            # For each circle, compute gradient and move in direction of steepest ascent
            for i in range(len(current)):
                x, y, r = current[i]

                # Compute numerical gradient using finite differences
                grad_x = 0.0
                grad_y = 0.0

                # Calculate partial derivative w.r.t. x
                test_x_pos = x + eps
                test_x_neg = x - eps

                # Ensure points are within bounds
                test_x_pos = np.clip(test_x_pos, 0.001, rect_width - 0.001)
                test_x_neg = np.clip(test_x_neg, 0.001, rect_width - 0.001)

                radius_x_pos = compute_max_radius_at_position(test_x_pos, y, current, rect_width, rect_height)
                radius_x_neg = compute_max_radius_at_position(test_x_neg, y, current, rect_width, rect_height)
                grad_x = (radius_x_pos - radius_x_neg) / (2 * eps)

                # Calculate partial derivative w.r.t. y
                test_y_pos = y + eps
                test_y_neg = y - eps

                # Ensure points are within bounds
                test_y_pos = np.clip(test_y_pos, 0.001, rect_height - 0.001)
                test_y_neg = np.clip(test_y_neg, 0.001, rect_height - 0.001)

                radius_y_pos = compute_max_radius_at_position(x, test_y_pos, current, rect_width, rect_height)
                radius_y_neg = compute_max_radius_at_position(x, test_y_neg, current, rect_width, rect_height)
                grad_y = (radius_y_pos - radius_y_neg) / (2 * eps)

                # Update using gradient ascent (since we want to maximize radius)
                step_size = 0.02

                # Adaptive step size based on gradient magnitude
                grad_magnitude = np.sqrt(grad_x**2 + grad_y**2)
                if grad_magnitude > 1e-8:
                    step_size = min(0.05, 0.05 / grad_magnitude)

                new_x = x + step_size * grad_x
                new_y = y + step_size * grad_y

                # Keep within bounds
                new_x = np.clip(new_x, 0.001, rect_width - 0.001)
                new_y = np.clip(new_y, 0.001, rect_height - 0.001)

                # Test the new position
                test_radius = compute_max_radius_at_position(new_x, new_y, current, rect_width, rect_height)

                # Accept the move if it increases radius
                if test_radius > r:
                    new_circles[i] = [new_x, new_y, test_radius]
                    updated = True

            current = new_circles

            # Early stopping if no significant improvement
            if not updated and iteration > 50:
                break

        return current
