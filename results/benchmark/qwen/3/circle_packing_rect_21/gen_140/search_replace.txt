def calculate_multi_objective_radius(circles, index, x, y, width, height, current_radius):
    """Calculate 'multi-objective' radius for circle at given position considering both gain and stability."""
    # Maximum radius based on container boundaries
    max_radius_bound = min(x, y, width - x, height - y)

    # Maximum radius based on other circles
    max_radius_overlap = float('inf')

    # Keep track of overlap violations for penalty calculation
    overlap_violation = 0

    for i, (cx, cy, cr) in enumerate(circles):
        if i != index:
            # Distance to other circle center
            dist = np.sqrt((x - cx)**2 + (y - cy)**2)

            # Calculate overlap violation (negative values indicate overlap)
            overlap = dist - (cr + 0.001)  # Add small buffer
            if overlap < 0:
                overlap_violation = min(overlap_violation, overlap)

            # Max radius that avoids overlap
            max_radius_for_this_circle = dist - cr

            if max_radius_for_this_circle < max_radius_overlap:
                max_radius_overlap = max_radius_for_this_circle

    max_radius = min(max_radius_bound, max_radius_overlap)

    # Apply penalty for overlap violations to discourage unstable moves
    if overlap_violation < 0:
        # Penalty for overlap - reduce the effective max radius
        penalty_factor = 1.0 + abs(overlap_violation) * 0.5  # Adjust penalty strength
        max_radius = max_radius / penalty_factor

    # Ensure minimum radius
    return max(max_radius, 0.001)


def calculate_max_radius_with_tolerance(circles, index, width, height, overlap_tolerance=0.0):
    """Calculate maximum radius for circle at given index without overlapping others, considering tolerance."""
    x, y, current_radius = circles[index]

    # Maximum radius based on container boundaries
    max_radius_bound = min(x, y, width - x, height - y)

    # Maximum radius based on other circles using vectorized computation
    if len(circles) > 1:
        # Get all other circles (excluding the target circle)
        other_circles = np.concatenate([circles[:index], circles[index+1:]])

        # Vectorized distance calculation
        if len(other_circles) > 0:
            distances = np.sqrt(np.sum((other_circles[:, :2] - [x, y])**2, axis=1))
            max_radius_for_others = np.min(distances - other_circles[:, 2] - overlap_tolerance)
            max_radius_overlap = max_radius_for_others
        else:
            max_radius_overlap = float('inf')
    else:
        max_radius_overlap = float('inf')

    max_radius = min(max_radius_bound, max_radius_overlap)
    return max(max_radius, 0.001)  # Ensure minimum radius
