import numpy as np
from scipy.spatial.distance import cdist
from scipy.spatial import Voronoi
import random
import math

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    # Optimize width/height ratio - try 2:1 ratio which often works well for circle packing
    width, height = 1.2, 0.8

    # Set seed for reproducibility
    np.random.seed(42)
    random.seed(42)

    # Phase 1: Enhanced initialization with hexagonal packing and strategic seeding
    n_circles = 21
    circles = np.zeros((n_circles, 3))

    # Hexagonal grid initialization
    rows = 5
    cols = 5
    
    # Calculate spacing for hexagonal packing
    estimated_radius = 0.1
    spacing_x = 2 * estimated_radius
    spacing_y = np.sqrt(3) * estimated_radius

    # Create hexagonal grid points
    hex_points = []
    for i in range(rows):
        for j in range(cols):
            if len(hex_points) >= n_circles:
                break
            x = j * spacing_x + (i % 2) * spacing_x / 2  # Offset every other row
            y = i * spacing_y
            
            # Add point if it's within bounds
            if 0 <= x <= width and 0 <= y <= height:
                hex_points.append([x, y])

    # If we don't have enough points, add boundary and center points
    while len(hex_points) < n_circles:
        # Corner points
        corner_points = [
            [0.1, 0.1], [width-0.1, 0.1], 
            [0.1, height-0.1], [width-0.1, height-0.1],
            [width/2, height/2]
        ]
        for pt in corner_points:
            if len(hex_points) < n_circles and pt not in hex_points:
                hex_points.append(pt)
        # Add random points if still not enough
        if len(hex_points) < n_circles:
            for _ in range(n_circles - len(hex_points)):
                x = np.random.uniform(0.05, width - 0.05)
                y = np.random.uniform(0.05, height - 0.05)
                hex_points.append([x, y])

    # Initialize with small radii
    points = np.array(hex_points[:n_circles])
    for i in range(n_circles):
        circles[i] = [points[i][0], points[i][1], 0.01]

    # Phase 2: Adaptive local optimization with Voronoi guidance
    def compute_max_radius(circles, index, width, height):
        """Calculate maximum radius for circle at given index without overlapping others."""
        x, y, current_radius = circles[index]

        # Maximum radius based on container boundaries
        max_radius_bound = min(x, y, width - x, height - y)

        # Maximum radius based on other circles
        max_radius_overlap = float('inf')

        for i, (cx, cy, cr) in enumerate(circles):
            if i != index:
                # Distance to other circle center
                dist = np.sqrt((x - cx)**2 + (y - cy)**2)
                # Max radius that avoids overlap
                max_radius_for_this_circle = dist - cr

                if max_radius_for_this_circle < max_radius_overlap:
                    max_radius_overlap = max_radius_for_this_circle

        max_radius = min(max_radius_bound, max_radius_overlap)
        return max(max_radius, 0.001)  # Ensure minimum radius

    # Multi-phase optimization with improved cooling schedule
    best_sum = 0
    best_circles = circles.copy()

    # Phase A: Coarse search with high exploration
    for iteration in range(500):
        improved = False
        
        # Randomize order for diverse search
        indices = list(range(n_circles))
        random.shuffle(indices)
        
        for i in indices:
            # Find maximum possible radius for circle i
            max_radius = compute_max_radius(circles, i, width, height)

            if max_radius > circles[i][2]:
                circles[i][2] = max_radius
                improved = True
                
        # Check for improvement
        current_sum = np.sum(circles[:, 2])
        if current_sum > best_sum:
            best_sum = current_sum
            best_circles = circles.copy()
            
        if not improved:
            break

    # Phase B: Refinement with Voronoi-guided search
    for iteration in range(300):
        current_circles = circles.copy()
        improved = False
        
        # For each circle, try to improve using Voronoi-guided local search
        indices = list(range(n_circles))
        random.shuffle(indices)
        
        for i in indices:
            orig_x, orig_y, orig_r = current_circles[i]
            
            # Calculate current constraints and available space
            max_radius_current = min(orig_x, width - orig_x, orig_y, height - orig_y)
            
            # Use Voronoi-based approach to guide search
            best_x, best_y, best_r = orig_x, orig_y, orig_r
            best_delta_sum = 0
            
            # Try multiple strategies for finding better positions
            strategies = []
            
            # Strategy 1: Direct gradient-like search in 4 directions
            for dx, dy in [(0, 0), (-0.01, 0), (0.01, 0), (0, -0.01), (0, 0.01)]:
                strategies.append((dx, dy))
            
            # Strategy 2: Larger steps
            for dx, dy in [(-0.02, 0), (0.02, 0), (0, -0.02), (0, 0.02)]:
                strategies.append((dx, dy))
            
            # Strategy 3: Diagonal moves
            for dx, dy in [(-0.01, -0.01), (-0.01, 0.01), (0.01, -0.01), (0.01, 0.01)]:
                strategies.append((dx, dy))
            
            # Strategy 4: Voronoi-based neighborhood (try to move towards unoccupied Voronoi regions)
            try:
                temp_points = np.array([[c[0], c[1]] for c in current_circles])
                vor = Voronoi(temp_points)
                # Get the Voronoi region for this circle
                region_vertices = vor.vertices[vor.regions[vor.point_region[i]]]
                # Try moving towards center of Voronoi region (if meaningful)
                if len(region_vertices) > 0:
                    # Move towards centroid of the region
                    centroid_x = np.mean(region_vertices[:, 0])
                    centroid_y = np.mean(region_vertices[:, 1])
                    # Normalize direction and scale it
                    dx = centroid_x - orig_x
                    dy = centroid_y - orig_y
                    # Normalize and scale
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist > 0.001:
                        dx, dy = dx/dist * 0.01, dy/dist * 0.01
                        strategies.append((dx, dy))
            except:
                pass  # Ignore Voronoi errors gracefully
            
            # Evaluate different move candidates
            for dx, dy in strategies:
                new_x = orig_x + dx
                new_y = orig_y + dy
                
                # Keep within bounds
                new_x = max(0.05, min(width - 0.05, new_x))
                new_y = max(0.05, min(height - 0.05, new_y))
                
                # Compute max radius at new location
                new_r = min(new_x, width - new_x, new_y, height - new_y)
                
                # Check for conflicts with all other circles
                valid = True
                for j in range(n_circles):
                    if i != j:
                        px, py, pr = current_circles[j]
                        dist = np.sqrt((new_x - px)**2 + (new_y - py)**2)
                        if dist < (new_r + pr):
                            valid = False
                            break
                
                if valid:
                    # Calculate change in sum
                    delta_sum = new_r - orig_r
                    
                    # Prefer moves that increase radius significantly
                    if delta_sum > best_delta_sum:
                        best_delta_sum = delta_sum
                        best_x, best_y, best_r = new_x, new_y, new_r
                        improved = True
            
            # Apply the best move if there was an improvement
            if best_delta_sum > 0:
                current_circles[i] = [best_x, best_y, best_r]
        
        # Check for improvement
        new_sum = np.sum(current_circles[:, 2])
        if new_sum > best_sum:
            best_sum = new_sum
            best_circles = current_circles.copy()
            improved = True
        elif not improved:
            break
            
        circles = current_circles.copy()

    # Phase C: Final refinement using grid sampling
    for _ in range(100):
        # Randomly sample some circles to try to improve
        indices = np.random.choice(n_circles, size=min(10, n_circles), replace=False)

        for idx in indices:
            old_x, old_y, old_r = circles[idx]

            # Find best new position from nearby grid points
            best_x, best_y = old_x, old_y
            best_r = old_r
            best_sum = np.sum(circles[:, 2])

            # Sample around current position
            search_range = 0.1
            fine_grid_size = 6
            
            # Create fine grid around current point
            fine_x = np.linspace(max(0.05, old_x - search_range), 
                               min(width - 0.05, old_x + search_range), fine_grid_size)
            fine_y = np.linspace(max(0.05, old_y - search_range), 
                               min(height - 0.05, old_y + search_range), fine_grid_size)

            for fx in fine_x:
                for fy in fine_y:
                    # Ensure it's within bounds
                    new_x = max(0.05, min(width - 0.05, fx))
                    new_y = max(0.05, min(height - 0.05, fy))

                    # Compute max radius
                    new_r = min(new_x, width - new_x, new_y, height - new_y)

                    # Check for conflicts
                    valid = True
                    for j in range(n_circles):
                        if j != idx:
                            px, py, pr = circles[j]
                            dist = np.sqrt((new_x - px)**2 + (new_y - py)**2)
                            if dist < (new_r + pr):
                                valid = False
                                break

                    if valid:
                        # See if this is better
                        new_sum = np.sum(circles[:, 2]) - old_r + new_r
                        if new_sum > best_sum:
                            best_sum = new_sum
                            best_x, best_y, best_r = new_x, new_y, new_r

            # Apply improvement if found
            if best_sum > np.sum(circles[:, 2]):
                circles[idx] = [best_x, best_y, best_r]

    return best_circles
