import numpy as np
from scipy.spatial import Voronoi, distance_matrix
from scipy.spatial.distance import cdist
from sklearn.cluster import KMeans
import math
from numba import jit
import random

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Container setup (perimeter = 4, so width + height = 2)
    container_width, container_height = 1.2, 0.8

    # Number of circles
    n = 21

    # Initialize circles array
    circles = np.zeros((n, 3))

    # Phase 1: Smart Voronoi-based initialization with geometric patterns
    # Generate strategic anchor points (corners, edges, and center)
    anchors = []
    # Corners
    anchors.extend([(0.05, 0.05), (container_width - 0.05, 0.05),
                   (0.05, container_height - 0.05), (container_width - 0.05, container_height - 0.05)])
    # Edge centers
    anchors.extend([(container_width/2, 0.05), (container_width/2, container_height - 0.05),
                   (0.05, container_height/2), (container_width - 0.05, container_height/2)])
    # Center
    anchors.append((container_width/2, container_height/2))

    # Create a structured grid of candidate points
    grid_density = 16
    candidate_points = []
    for i in range(grid_density):
        for j in range(grid_density):
            x = (i + 0.5) / grid_density * container_width
            y = (j + 0.5) / grid_density * container_height
            # Filter out points too close to edges
            if 0.1 <= x <= container_width - 0.1 and 0.1 <= y <= container_height - 0.1:
                candidate_points.append((x, y))

    # Add some random points for diversity
    random.seed(42)
    for _ in range(50):
        x = np.random.uniform(0.1, container_width - 0.1)
        y = np.random.uniform(0.1, container_height - 0.1)
        candidate_points.append((x, y))

    # Combine and cluster to select optimal initial positions
    all_points = anchors + candidate_points
    kmeans = KMeans(n_clusters=min(len(all_points), n), init='k-means++', n_init=5, random_state=42)
    all_array = np.array(all_points)
    kmeans.fit(all_array)
    selected_centroids = kmeans.cluster_centers_

    # Select actual points closest to centroids
    selected_positions = []
    for centroid in selected_centroids:
        distances = np.linalg.norm(all_array - centroid, axis=1)
        closest_idx = np.argmin(distances)
        selected_positions.append(tuple(all_array[closest_idx]))

    # Ensure exact count
    if len(selected_positions) < n:
        remaining_needed = n - len(selected_positions)
        for i in range(min(remaining_needed, len(candidate_points))):
            selected_positions.append(candidate_points[i])
    elif len(selected_positions) > n:
        selected_positions = selected_positions[:n]

    # Initialize with positions and small radii
    for i in range(n):
        x, y = selected_positions[i]
        circles[i] = [x, y, 0.01]

    # Phase 2: Multi-resolution Voronoi Force Field Optimization
    max_iterations = 2000
    best_sum_radii = 0
    best_circles = None

    # Optimization parameters
    force_strength = 1.0
    boundary_repulsion = 3.0
    max_radius_limit = 0.3
    time_step = 0.05
    
    # Precomputed neighbor lists for efficiency - Voronoi-based
    neighbor_cache = {}
    
    @jit(nopython=True)
    def calculate_voronoi_forces_2d(positions, radii, n, container_width, container_height, neighbor_list=None):
        """Calculate net forces for all circles using efficient neighbor identification"""
        forces = np.zeros((n, 2))
        
        # Calculate forces - simplified but efficient direct approach
        # For large n, we'll approximate using a spatial hash or neighbor list
        # But for now, we'll do direct calculation with smart limits
        for i in range(n):
            for j in range(n):
                if i != j:
                    dx = positions[i, 0] - positions[j, 0]
                    dy = positions[i, 1] - positions[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    
                    if dist > 1e-8 and dist < (radii[i] + radii[j]):
                        # Repulsive force
                        force_magnitude = (radii[i] + radii[j] - dist) / (dist + 1e-8) * force_strength
                        forces[i, 0] -= force_magnitude * dx / dist
                        forces[i, 1] -= force_magnitude * dy / dist
                    
                    # Boundary repulsion (stronger forces near walls)
                    x, y = positions[i, 0], positions[i, 1]
                    r = radii[i]
                    
                    # Left boundary
                    if x < r:
                        forces[i, 0] += (r - x) * boundary_repulsion
                    # Right boundary
                    if x + r > container_width:
                        forces[i, 0] -= (x + r - container_width) * boundary_repulsion
                    # Bottom boundary
                    if y < r:
                        forces[i, 1] += (r - y) * boundary_repulsion
                    # Top boundary
                    if y + r > container_height:
                        forces[i, 1] -= (y + r - container_height) * boundary_repulsion
                        
        return forces

    @jit(nopython=True)
    def compute_max_radius_fast(positions, radii, index, width, height, n):
        """Fast calculation of max radius using direct neighbor analysis"""
        x, y = positions[index, 0], positions[index, 1]
        
        # Boundary constraints
        max_radius = min(x, y, width - x, height - y)
        
        # Overlap constraints with direct comparison - optimized version
        for i in range(n):
            if i != index:
                dx = x - positions[i, 0]
                dy = y - positions[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist > 1e-8:
                    max_radius_for_this_circle = dist - radii[i]
                    if max_radius_for_this_circle < max_radius:
                        max_radius = max_radius_for_this_circle
                        
        return max(max_radius, 0.001)

    @jit(nopython=True)
    def compute_max_radius_vectorized(positions, radii, index, width, height):
        """Vectorized max radius calculation for better performance"""
        x, y = positions[index, 0], positions[index, 1]
        
        # Boundary constraints
        max_radius = min(x, y, width - x, height - y)
        
        # Vectorized overlap checking with other circles
        if len(positions) > 1:
            # Compute distances to all other circles
            other_positions = np.delete(positions, index, axis=0)
            other_radii = np.delete(radii, index)
            
            # Calculate all distances
            dx = x - other_positions[:, 0]
            dy = y - other_positions[:, 1]
            distances = np.sqrt(dx*dx + dy*dy)
            
            # Find max radius constraint from all overlaps
            overlap_constraints = distances - other_radii
            valid_constraints = overlap_constraints[overlap_constraints > 0]  # Only positive constraints
            
            if len(valid_constraints) > 0:
                max_radius = min(max_radius, np.min(valid_constraints))
                
        return max(max_radius, 0.001)

    # Main optimization loop with multi-resolution approach
    for iteration in range(max_iterations):
        # Progressively reduce time step and force strength
        current_time_step = time_step * (1.0 - iteration/max_iterations * 0.5)
        current_force_strength = force_strength * (1.0 - iteration/max_iterations * 0.3)
        
        positions = circles[:, :2]
        radii = circles[:, 2]

        # Calculate all forces
        forces = calculate_voronoi_forces_2d(positions, radii, n, container_width, container_height)

        # Update positions and radii
        for i in range(n):
            x, y, r = circles[i]
            
            # Update radius based on available space
            max_radius = compute_max_radius_vectorized(positions, radii, i, container_width, container_height)
            max_radius = min(max_radius, max_radius_limit)
            
            if max_radius > r:
                # Increase radius gradually with adaptive scaling
                growth_rate = 0.01 + 0.005 * (1.0 - iteration/max_iterations)
                new_radius = min(r + growth_rate * (max_radius - r), max_radius)
                circles[i, 2] = new_radius
            
            # Apply forces to position with boundaries
            fx, fy = forces[i, 0], forces[i, 1]
            
            # Update position with adaptive time step
            new_x = max(r, min(container_width - r, x + current_time_step * fx))
            new_y = max(r, min(container_height - r, y + current_time_step * fy))
            
            circles[i, 0] = new_x
            circles[i, 1] = new_y

        # Occasionally apply global refinement with adaptive perturbation
        if iteration % 150 == 0 and iteration > 0:
            for i in range(n):
                x, y, r = circles[i]
                # Adaptive perturbation based on iteration progress
                perturbation_scale = max(0.001, 0.015 * (1.0 - iteration/max_iterations))
                circles[i] = [max(r, min(container_width - r, x + np.random.normal(0, perturbation_scale))),
                              max(r, min(container_height - r, y + np.random.normal(0, perturbation_scale))),
                              r]

        # Apply local grid search refinements every 300 iterations
        if iteration % 300 == 0 and iteration > 0:
            for i in range(n):
                old_x, old_y, old_r = circles[i]
                
                # Try small adjustments in grid pattern
                best_x, best_y = old_x, old_y
                best_r = old_r
                best_sum = np.sum(circles[:, 2])
                
                # Grid search around current position
                search_steps = [-0.03, -0.015, 0, 0.015, 0.03]
                for dx in search_steps:
                    for dy in search_steps:
                        new_x = old_x + dx
                        new_y = old_y + dy
                        
                        if (0.05 <= new_x <= container_width - 0.05 and 
                            0.05 <= new_y <= container_height - 0.05):
                            
                            # Compute max radius at this new location
                            temp_circles = circles.copy()
                            temp_circles[i] = [new_x, new_y, old_r]
                            new_max_radius = compute_max_radius_vectorized(
                                temp_circles[:, :2], temp_circles[:, 2], i, 
                                container_width, container_height
                            )
                            
                            # Check if this configuration is valid and improves sum
                            temp_circles[i] = [new_x, new_y, min(new_max_radius, old_r * 1.1)]
                            
                            new_sum = np.sum(temp_circles[:, 2])
                            if new_sum > best_sum:
                                best_sum = new_sum
                                best_x, best_y, best_r = new_x, new_y, temp_circles[i, 2]
                
                circles[i] = [best_x, best_y, best_r]

        # Save best configuration periodically
        current_sum = np.sum(radii)
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_circles = circles.copy()

    # Return best configuration or final state
    if best_circles is not None:
        return best_circles

    # Final cleanup
    for i in range(n):
        x, y, r = circles[i]
        x = max(r, min(container_width - r, x))
        y = max(r, min(container_height - r, y))
        circles[i] = [x, y, r]

    return circles
