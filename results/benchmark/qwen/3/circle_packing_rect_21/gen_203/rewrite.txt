# EVOLVE-BLOCK-START
import numpy as np
from scipy.spatial.distance import cdist
import random
import math
from typing import Tuple, List, Optional
from dataclasses import dataclass
from enum import Enum

class OptimizationPhase(Enum):
    EXPLORATION = 1
    EXPLOITATION = 2
    LOCAL_REFINEMENT = 3

@dataclass
class OptimizationConfig:
    """Configuration for the optimization process"""
    rect_width: float = 1.5
    rect_height: float = 0.5
    num_circles: int = 21
    max_iterations: int = 1000
    population_size: int = 10
    elite_size: int = 3
    exploration_steps: int = 300
    exploitation_steps: int = 400
    local_refinement_steps: int = 300
    adaptive_step_reduction: float = 0.95
    min_step_size: float = 0.001

class CirclePacker:
    """Main circle packing optimizer with evolutionary approach"""
    
    def __init__(self, config: OptimizationConfig):
        self.config = config
        random.seed(42)
        np.random.seed(42)
    
    def initialize_population(self, num_individuals: int) -> List[np.ndarray]:
        """Generate initial population of circle configurations"""
        population = []
        for _ in range(num_individuals):
            circles = self._create_hybrid_initialization()
            population.append(circles)
        return population
    
    def _create_hybrid_initialization(self) -> np.ndarray:
        """Create initial configuration using hybrid approach"""
        circles = np.zeros((self.config.num_circles, 3))
        
        # Corner and edge positions to establish boundaries
        corner_positions = [
            (self.config.rect_width * 0.1, self.config.rect_height * 0.1),    # bottom-left
            (self.config.rect_width * 0.9, self.config.rect_height * 0.1),    # bottom-right
            (self.config.rect_width * 0.1, self.config.rect_height * 0.9),    # top-left
            (self.config.rect_width * 0.9, self.config.rect_height * 0.9),    # top-right
            (self.config.rect_width * 0.5, self.config.rect_height * 0.1),    # bottom-middle
            (self.config.rect_width * 0.5, self.config.rect_height * 0.9),    # top-middle
            (self.config.rect_width * 0.1, self.config.rect_height * 0.5),    # left-middle
            (self.config.rect_width * 0.9, self.config.rect_height * 0.5),    # right-middle
        ]

        # Place some circles at strategic positions
        for i in range(min(len(corner_positions), self.config.num_circles)):
            x, y = corner_positions[i]
            circles[i] = [x, y, 0.03]

        # Fill remaining positions with hexagonal grid
        remaining = self.config.num_circles - len(corner_positions)
        if remaining > 0:
            rows = int(math.ceil(math.sqrt(remaining)))
            cols = int(math.ceil(remaining / rows))

            cell_width = self.config.rect_width / (cols + 1)
            cell_height = self.config.rect_height / (rows + 1)

            idx = len(corner_positions)
            for i in range(rows):
                for j in range(cols):
                    if idx >= self.config.num_circles:
                        break
                    x_offset = 0.0 if i % 2 == 0 else 0.5
                    x = (j + 1 + x_offset) * cell_width
                    y = (i + 1) * cell_height
                    # Ensure within bounds
                    x = max(0.01, min(self.config.rect_width - 0.01, x))
                    y = max(0.01, min(self.config.rect_height - 0.01, y))
                    circles[idx] = [x, y, 0.02]
                    idx += 1
                    if idx >= self.config.num_circles:
                        break

        return circles
    
    def compute_max_radius_at_position_vectorized(self, x: float, y: float, 
                                               existing_circles: np.ndarray) -> float:
        """Vectorized computation of maximum possible radius for a circle at given position"""
        # Distance to boundaries
        min_bound = min(x, self.config.rect_width - x, y, self.config.rect_height - y)

        # Vectorized distance calculation to all existing circles
        if len(existing_circles) > 0:
            positions = existing_circles[:, :2]
            radii = existing_circles[:, 2]

            # Calculate distances to all existing circles
            dx = positions[:, 0] - x
            dy = positions[:, 1] - y
            distances = np.sqrt(dx*dx + dy*dy)

            # Min distance minus sum of radii
            min_dist_to_others = np.min(distances - radii)
            actual_min_dist = min_dist_to_others
        else:
            actual_min_dist = float('inf')

        # Take minimum of boundary and other-circle distances
        max_radius = min(min_bound, actual_min_dist if actual_min_dist < float('inf') else float('inf'))
        return max(0.001, max_radius)
    
    def is_valid_configuration_vectorized(self, circles: np.ndarray) -> bool:
        """Vectorized validation of circle configuration"""
        # Check boundary constraints efficiently
        if np.any(circles[:, 0] - circles[:, 2] < 0) or \
           np.any(circles[:, 0] + circles[:, 2] > self.config.rect_width) or \
           np.any(circles[:, 1] - circles[:, 2] < 0) or \
           np.any(circles[:, 1] + circles[:, 2] > self.config.rect_height):
            return False

        # Check overlap constraints efficiently using vectorized computation
        if len(circles) < 2:
            return True

        # Use vectorized computation for overlap detection
        positions = circles[:, :2]
        radii = circles[:, 2]

        # Create distance matrix
        dist_matrix = cdist(positions, positions)

        # Set diagonal to infinity (self-distances)
        np.fill_diagonal(dist_matrix, float('inf'))

        # Minimum distances between circles
        min_distances = np.min(dist_matrix, axis=1)

        # Minimum sum of radii for each circle pair
        radii_sums = radii[:, np.newaxis] + radii[np.newaxis, :]

        # Check overlaps - vectorized operation
        overlap_mask = min_distances < np.min(radii_sums, axis=0)

        return not np.any(overlap_mask)
    
    def evaluate_fitness(self, circles: np.ndarray) -> float:
        """Evaluate fitness as sum of radii"""
        return np.sum(circles[:, 2])
    
    def mutate_individual(self, individual: np.ndarray, phase: OptimizationPhase, 
                         iteration: int) -> np.ndarray:
        """Apply mutation to an individual"""
        mutated = individual.copy()
        
        # Determine mutation strength based on phase and iteration
        if phase == OptimizationPhase.EXPLORATION:
            step_size = 0.15
            mutation_prob = 0.7
        elif phase == OptimizationPhase.EXPLOITATION:
            step_size = 0.08
            mutation_prob = 0.5
        else:  # LOCAL_REFINEMENT
            step_size = 0.03
            mutation_prob = 0.3
        
        # Apply mutations to circles
        for i in range(len(mutated)):
            if random.random() < mutation_prob:
                # Randomly perturb position and radius
                dx = random.uniform(-step_size, step_size)
                dy = random.uniform(-step_size, step_size)
                
                new_x = max(0.001, min(self.config.rect_width - 0.001, mutated[i, 0] + dx))
                new_y = max(0.001, min(self.config.rect_height - 0.001, mutated[i, 1] + dy))
                
                # Compute new radius
                temp_circles = mutated.copy()
                temp_circles[i] = [new_x, new_y, 0.01]  # Temporary small radius
                max_r = self.compute_max_radius_at_position_vectorized(
                    new_x, new_y, temp_circles
                )
                
                new_r = min(max_r, max(0.001, mutated[i, 2] + random.uniform(-0.02, 0.02)))
                
                mutated[i] = [new_x, new_y, new_r]
        
        return mutated
    
    def select_parents(self, population: List[np.ndarray], fitnesses: List[float]) -> List[np.ndarray]:
        """Select parents using tournament selection"""
        selected = []
        tournament_size = 3
        
        for _ in range(len(population)):
            # Tournament selection
            tournament_indices = random.sample(range(len(population)), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        return selected
    
    def crossover(self, parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
        """Simple uniform crossover"""
        child = parent1.copy()
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child
    
    def evolve_population(self, population: List[np.ndarray], 
                         phase: OptimizationPhase, iteration: int) -> List[np.ndarray]:
        """Perform evolution on population"""
        # Evaluate fitness
        fitnesses = [self.evaluate_fitness(individual) for individual in population]
        
        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitnesses)[::-1]
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite individuals
        elite = sorted_population[:self.config.elite_size]
        
        # Select parents
        parents = self.select_parents(sorted_population, sorted_fitnesses)
        
        # Create new population through crossover and mutation
        new_population = elite.copy()
        
        while len(new_population) < len(population):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            
            child = self.crossover(parent1, parent2)
            mutated_child = self.mutate_individual(child, phase, iteration)
            
            # Validate and repair if necessary
            if self.is_valid_configuration_vectorized(mutated_child):
                new_population.append(mutated_child)
            else:
                # If invalid, keep the original individual
                new_population.append(child)
        
        return new_population[:len(population)]
    
    def optimize(self) -> np.ndarray:
        """Main optimization routine"""
        # Initialize population
        population = self.initialize_population(self.config.population_size)
        
        # Track best solution
        best_solution = None
        best_fitness = -float('inf')
        last_improvement = 0
        
        # Phase 1: Exploration
        for i in range(self.config.exploration_steps):
            population = self.evolve_population(population, OptimizationPhase.EXPLORATION, i)
            
            # Track best
            fitnesses = [self.evaluate_fitness(individual) for individual in population]
            current_best_idx = np.argmax(fitnesses)
            current_best_fitness = fitnesses[current_best_idx]
            
            if current_best_fitness > best_fitness:
                best_fitness = current_best_fitness
                best_solution = population[current_best_idx].copy()
                last_improvement = i
        
        # Phase 2: Exploitation
        for i in range(self.config.exploitation_steps):
            population = self.evolve_population(population, OptimizationPhase.EXPLOITATION, i)
            
            # Track best
            fitnesses = [self.evaluate_fitness(individual) for individual in population]
            current_best_idx = np.argmax(fitnesses)
            current_best_fitness = fitnesses[current_best_idx]
            
            if current_best_fitness > best_fitness:
                best_fitness = current_best_fitness
                best_solution = population[current_best_idx].copy()
                last_improvement = i + self.config.exploration_steps
                
        # Phase 3: Local Refinement
        for i in range(self.config.local_refinement_steps):
            population = self.evolve_population(population, OptimizationPhase.LOCAL_REFINEMENT, i)
            
            # Track best
            fitnesses = [self.evaluate_fitness(individual) for individual in population]
            current_best_idx = np.argmax(fitnesses)
            current_best_fitness = fitnesses[current_best_idx]
            
            if current_best_fitness > best_fitness:
                best_fitness = current_best_fitness
                best_solution = population[current_best_idx].copy()
                last_improvement = i + self.config.exploration_steps + self.config.exploitation_steps
        
        # Final validation and cleanup
        if best_solution is not None:
            # Ensure all circles are valid
            while True:
                valid = self.is_valid_configuration_vectorized(best_solution)
                if valid:
                    break
                # If invalid, reinitialize
                best_solution = self._create_hybrid_initialization()
                
            # Final cleanup - ensure bounds and minimum radius
            for i in range(len(best_solution)):
                best_solution[i, 2] = max(best_solution[i, 2], 0.001)
                best_solution[i, 0] = np.clip(best_solution[i, 0], 0.001, self.config.rect_width - 0.001)
                best_solution[i, 1] = np.clip(best_solution[i, 1], 0.001, self.config.rect_height - 0.001)
        
        return best_solution

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: width + height = 2, using 1.5 x 0.5 for good aspect ratio
    config = OptimizationConfig(rect_width=1.5, rect_height=0.5)
    
    packer = CirclePacker(config)
    return packer.optimize()
