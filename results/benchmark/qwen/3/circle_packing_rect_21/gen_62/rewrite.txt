import numpy as np
from scipy.spatial.distance import cdist
import time

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    width, height = 1.0, 1.0  # Optimal square configuration
    
    # Set seed for reproducibility
    np.random.seed(42)
    
    # Physics-based circle packing with force-directed optimization
    n_circles = 21
    
    # Initialize circles with random positions and small radii
    circles = np.zeros((n_circles, 3))
    
    # Initial placement using hexagonal lattice approximation for good starting point
    # Grid spacing based on desired density
    grid_size = int(np.ceil(np.sqrt(n_circles)))
    spacing = min(width, height) / (grid_size + 1)
    
    idx = 0
    for i in range(grid_size):
        for j in range(grid_size):
            if idx >= n_circles:
                break
            x = (j + 0.5) * spacing
            y = (i + 0.5) * spacing
            
            # Adjust for boundary constraints
            x = max(0.05, min(width - 0.05, x))
            y = max(0.05, min(height - 0.05, y))
            
            # Initial small radius
            r = min(x, width - x, y, height - y) * 0.2
            circles[idx] = [x, y, r]
            idx += 1
            
        if idx >= n_circles:
            break
    
    # Fill remaining positions with random placements
    for i in range(idx, n_circles):
        attempts = 0
        while attempts < 100:
            x = np.random.uniform(0.05, width - 0.05)
            y = np.random.uniform(0.05, height - 0.05)
            r = min(x, width - x, y, height - y) * 0.3
            
            # Check collision with existing circles
            valid = True
            for j in range(i):
                px, py, pr = circles[j]
                dist = np.sqrt((x - px)**2 + (y - py)**2)
                if dist < (r + pr):
                    valid = False
                    break
                    
            if valid:
                circles[i] = [x, y, r]
                break
            attempts += 1

    # Physics simulation parameters
    max_iterations = 1000
    dt = 0.01
    repulsion_strength = 100.0
    attraction_strength = 0.1
    boundary_repulsion = 1000.0
    
    # Optimization loop with physics simulation
    best_sum = np.sum(circles[:, 2])
    best_circles = circles.copy()
    
    for iteration in range(max_iterations):
        # Calculate forces between circles
        forces = np.zeros((n_circles, 2))
        
        # Circle-to-circle repulsion forces
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                
                dx = x2 - x1
                dy = y2 - y1
                dist = np.sqrt(dx*dx + dy*dy)
                
                if dist > 0 and dist < (r1 + r2):
                    # Overlapping circles - strong repulsion
                    force_magnitude = repulsion_strength * (r1 + r2 - dist) / dist
                elif dist > 0:
                    # Non-overlapping circles - inverse square repulsion
                    force_magnitude = repulsion_strength / (dist * dist)
                else:
                    force_magnitude = 0
                    
                forces[i, 0] += force_magnitude * dx
                forces[i, 1] += force_magnitude * dy
                forces[j, 0] -= force_magnitude * dx
                forces[j, 1] -= force_magnitude * dy
        
        # Boundary repulsion forces (stronger near boundaries)
        for i in range(n_circles):
            x, y, r = circles[i]
            
            # Repulsion from boundaries
            left_force = boundary_repulsion * max(0, 0.05 - x) / (0.05 - x + 1e-8)
            right_force = boundary_repulsion * max(0, x - (width - 0.05)) / (x - (width - 0.05) + 1e-8)
            bottom_force = boundary_repulsion * max(0, 0.05 - y) / (0.05 - y + 1e-8)
            top_force = boundary_repulsion * max(0, y - (height - 0.05)) / (y - (height - 0.05) + 1e-8)
            
            forces[i, 0] += left_force - right_force
            forces[i, 1] += bottom_force - top_force
        
        # Update positions with velocity and boundary checks
        for i in range(n_circles):
            # Apply forces to update velocity and position
            x, y, r = circles[i]
            
            # Velocity update (simplified physics)
            vx = forces[i, 0] * dt
            vy = forces[i, 1] * dt
            
            # Update position
            new_x = x + vx
            new_y = y + vy
            
            # Enforce boundary conditions with reflection
            if new_x <= 0.05:
                new_x = 0.05
                # Reflect velocity
                vx = abs(vx) * 0.5
            elif new_x >= width - 0.05:
                new_x = width - 0.05
                # Reflect velocity
                vx = -abs(vx) * 0.5
                
            if new_y <= 0.05:
                new_y = 0.05
                # Reflect velocity
                vy = abs(vy) * 0.5
            elif new_y >= height - 0.05:
                new_y = height - 0.05
                # Reflect velocity
                vy = -abs(vy) * 0.5
            
            # Update position
            circles[i, 0] = new_x
            circles[i, 1] = new_y
            
            # Update radius to maximize while respecting constraints
            max_radius = min(new_x, width - new_x, new_y, height - new_y)
            
            # Make sure we don't go too far from current radius
            current_radius = circles[i, 2]
            # Allow some adjustment but stay conservative to avoid instability
            if max_radius > current_radius:
                # Only increase radius if there's enough space and no overlap
                safe_radius = max_radius
                
                # Check overlap with all other circles
                for j in range(n_circles):
                    if i != j:
                        px, py, pr = circles[j]
                        dist = np.sqrt((new_x - px)**2 + (new_y - py)**2)
                        if dist < (safe_radius + pr):
                            # Reduce radius to avoid overlap
                            overlap_amount = dist - (safe_radius + pr)
                            safe_radius -= overlap_amount * 0.5
                            if safe_radius <= 0:
                                safe_radius = current_radius * 0.9
                            break
                
                # Update radius with some smoothing
                circles[i, 2] = safe_radius
            
            # Apply additional fine-tuning with local optimization
            if iteration % 10 == 0 and iteration > 0:
                # Local refinement around each circle
                x, y, r = circles[i]
                
                # Sample nearby positions to find better placement
                best_x, best_y, best_r = x, y, r
                best_sum_local = np.sum(circles[:, 2])
                
                # Test several positions around current circle
                test_positions = []
                step = min(0.1, r * 0.5)
                for dx in [-step, -step/2, 0, step/2, step]:
                    for dy in [-step, -step/2, 0, step/2, step]:
                        test_positions.append((dx, dy))
                
                for dx, dy in test_positions:
                    new_x, new_y = x + dx, y + dy
                    # Keep within bounds
                    new_x = max(0.05, min(width - 0.05, new_x))
                    new_y = max(0.05, min(height - 0.05, new_y))
                    
                    # Compute max radius at new location
                    new_r = min(new_x, width - new_x, new_y, height - new_y)
                    
                    # Check for conflicts with all other circles
                    valid = True
                    for j in range(n_circles):
                        if i != j:
                            px, py, pr = circles[j]
                            dist = np.sqrt((new_x - px)**2 + (new_y - py)**2)
                            if dist < (new_r + pr):
                                valid = False
                                break
                    
                    if valid:
                        # Calculate sum if we made this change
                        sum_with_change = best_sum_local - r + new_r
                        if sum_with_change > best_sum_local:
                            best_sum_local = sum_with_change
                            best_x, best_y, best_r = new_x, new_y, new_r
                
                # Apply the best local improvement
                if best_sum_local > np.sum(circles[:, 2]):
                    circles[i] = [best_x, best_y, best_r]
        
        # Periodically check for improvement
        current_sum = np.sum(circles[:, 2])
        if current_sum > best_sum:
            best_sum = current_sum
            best_circles = circles.copy()
        
        # Adaptive stopping criteria
        if iteration > 100 and iteration % 50 == 0:
            if abs(current_sum - best_sum) < 1e-6:
                break
    
    # Final refinement pass with gradient-like local search
    for _ in range(100):
        improved = False
        for i in range(n_circles):
            x, y, r = circles[i]
            
            # Get current sum
            old_sum = np.sum(circles[:, 2])
            
            # Try moving in different directions
            best_x, best_y, best_r = x, y, r
            best_sum = old_sum
            
            # Test movement in 8 directions
            directions = [
                (-0.05, 0), (0.05, 0), (0, -0.05), (0, 0.05),
                (-0.03, -0.03), (-0.03, 0.03), (0.03, -0.03), (0.03, 0.03)
            ]
            
            for dx, dy in directions:
                new_x, new_y = x + dx, y + dy
                # Keep within bounds
                new_x = max(0.05, min(width - 0.05, new_x))
                new_y = max(0.05, min(height - 0.05, new_y))
                
                # Compute max radius at new location
                new_r = min(new_x, width - new_x, new_y, height - new_y)
                
                # Check for conflicts
                valid = True
                for j in range(n_circles):
                    if i != j:
                        px, py, pr = circles[j]
                        dist = np.sqrt((new_x - px)**2 + (new_y - py)**2)
                        if dist < (new_r + pr):
                            valid = False
                            break
                
                if valid:
                    new_sum = old_sum - r + new_r
                    if new_sum > best_sum:
                        best_sum = new_sum
                        best_x, best_y, best_r = new_x, new_y, new_r
            
            # Apply improvement
            if best_sum > old_sum:
                circles[i] = [best_x, best_y, best_r]
                improved = True
        
        if not improved:
            break
    
    return best_circles
