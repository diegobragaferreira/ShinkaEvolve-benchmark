import numpy as np
from scipy.spatial.distance import cdist
import time
from deap import base, creator, tools, algorithms
import random

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    width, height = 1.0, 1.0  # Optimal square configuration

    # Set seed for reproducibility
    np.random.seed(42)
    random.seed(42)

    # Define constants
    n_circles = 21
    bounds_min = 0.05
    bounds_max = width - 0.05

    def calculate_radius_sum(circles_array):
        """Calculate the sum of all radii"""
        return np.sum(circles_array[:, 2])

    def is_valid_configuration(circles_array):
        """Check if the configuration is valid (no overlaps and within bounds)"""
        # Check boundary constraints
        for i in range(n_circles):
            x, y, r = circles_array[i]
            if x - r < bounds_min or x + r > bounds_max or y - r < bounds_min or y + r > bounds_max:
                return False

        # Check overlaps
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                x1, y1, r1 = circles_array[i]
                x2, y2, r2 = circles_array[j]
                dist = np.sqrt((x2-x1)**2 + (y2-y1)**2)
                if dist < (r1 + r2):
                    return False
        return True

    def evaluate_individual(individual):
        """Evaluate the fitness of an individual (solution)"""
        # Convert individual to circles array (x, y, r)
        circles = np.zeros((n_circles, 3))
        for i in range(n_circles):
            circles[i] = [individual[3*i], individual[3*i+1], individual[3*i+2]]

        # Check validity
        if not is_valid_configuration(circles):
            return (-1000000,)  # Invalid solution penalty

        # Fitness is the sum of radii (we want to maximize)
        radius_sum = calculate_radius_sum(circles)
        return (radius_sum,)

    def create_individual():
        """Create a random individual with valid initial configuration"""
        individual = []
        # Create positions and radii with better distribution
        for i in range(n_circles):
            # Try to place circles in a grid-like fashion with some randomness
            x = np.random.uniform(bounds_min, bounds_max)
            y = np.random.uniform(bounds_min, bounds_max)
            # Radius - smaller to allow more circles
            r = np.random.uniform(0.01, 0.15)
            individual.extend([x, y, r])
        return individual

    def mutate_individual(individual):
        """Mutate an individual with different strategies"""
        mutpb = 0.2  # Mutation probability

        for i in range(len(individual)):
            if random.random() < mutpb:
                if i % 3 == 0:  # x coordinate
                    individual[i] = np.random.uniform(bounds_min, bounds_max)
                elif i % 3 == 1:  # y coordinate
                    individual[i] = np.random.uniform(bounds_min, bounds_max)
                else:  # radius
                    individual[i] = np.random.uniform(0.01, 0.15)
        return individual,

    def crossover_individuals(ind1, ind2):
        """Crossover two individuals"""
        # Simple uniform crossover
        for i in range(len(ind1)):
            if random.random() < 0.5:
                ind1[i], ind2[i] = ind2[i], ind1[i]
        return ind1, ind2

    # Phase 1: Initialize with better starting configuration using strategic points
    circles = np.zeros((n_circles, 3))

    # Create a strategic grid-based initialization
    grid_size = 5
    x_coords = np.linspace(0.1, width - 0.1, grid_size)
    y_coords = np.linspace(0.1, height - 0.1, grid_size)

    # Sample positions from grid
    points = []
    for x in x_coords:
        for y in y_coords:
            if len(points) < n_circles:
                points.append((x, y))

    # Add some random points to ensure diversity
    while len(points) < n_circles:
        points.append((np.random.uniform(0.1, width-0.1), np.random.uniform(0.1, height-0.1)))

    # Assign circles with reasonable max possible radius at each point
    for i, (x, y) in enumerate(points[:n_circles]):
        max_radius = min(x, width-x, y, height-y)
        # Adjust for the fact we might want more circles fitting
        circles[i] = [x, y, max_radius * 0.7]  # Use smaller fraction to leave room for others

    # Phase 2: Evolutionary Optimization with DEAP
    # Set up evolutionary algorithm components
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    toolbox = base.Toolbox()
    toolbox.register("individual", create_individual)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate_individual)
    toolbox.register("mate", crossover_individuals)
    toolbox.register("mutate", mutate_individual)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Create initial population
    pop_size = 30
    population = toolbox.population(n=pop_size)

    # Run evolutionary algorithm with statistics
    n_generations = 30
    hof = tools.HallOfFame(1)
    stats = tools.Statistics(lambda ind: ind.fitness.values[0])
    stats.register("avg", np.mean)
    stats.register("min", np.min)
    stats.register("max", np.max)

    # Run evolution
    best_fitness_history = []
    for gen in range(n_generations):
        # Evaluate population
        invalid_ind = [ind for ind in population if not ind.fitness.valid]
        fitnesses = toolbox.map(toolbox.evaluate, invalid_ind)
        for ind, fit in zip(invalid_ind, fitnesses):
            ind.fitness.values = fit

        # Update Hall of Fame
        hof.update(population)

        # Record best fitness
        best_fitness = np.max([ind.fitness.values[0] for ind in population])
        best_fitness_history.append(best_fitness)

        # Early stopping if no improvement in last few generations
        if len(best_fitness_history) > 5:
            if best_fitness_history[-1] - best_fitness_history[-5] < 1e-6:
                break

        # Select next generation
        offspring = toolbox.select(population, len(population))
        offspring = list(map(toolbox.clone, offspring))

        # Apply crossover and mutation
        for child1, child2 in zip(offspring[::2], offspring[1::2]):
            if random.random() < 0.8:
                toolbox.mate(child1, child2)
                del child1.fitness.values
                del child2.fitness.values

        for mutant in offspring:
            toolbox.mutate(mutant)
            del mutant.fitness.values

        # Replace population
        population[:] = offspring

    # Get the best individual
    best_individual = hof[0]

    # Convert back to circles array
    final_circles = np.zeros((n_circles, 3))
    for i in range(n_circles):
        final_circles[i] = [best_individual[3*i], best_individual[3*i+1], best_individual[3*i+2]]

    # Phase 3: Local refinement of best solution
    # Try to improve using local search with better neighborhood exploration
    current_solution = final_circles.copy()
    best_sum = calculate_radius_sum(current_solution)

    # Local search with better perturbation strategies
    max_local_iter = 100
    for iter_num in range(max_local_iter):
        improved = False

        # Try to improve each circle
        for i in range(n_circles):
            # Store original values
            orig_x, orig_y, orig_r = current_solution[i]

            # Try multiple strategies for improvement
            best_x, best_y, best_r = orig_x, orig_y, orig_r
            best_sum_current = best_sum

            # Strategy 1: Small random perturbations around current position
            for _ in range(5):
                dx = np.random.uniform(-0.02, 0.02)
                dy = np.random.uniform(-0.02, 0.02)
                dr = np.random.uniform(-0.01, 0.01)

                new_x = max(bounds_min, min(bounds_max, orig_x + dx))
                new_y = max(bounds_min, min(bounds_max, orig_y + dy))
                new_r = max(0.01, min(0.2, orig_r + dr))

                # Check if valid and better
                temp_config = current_solution.copy()
                temp_config[i] = [new_x, new_y, new_r]

                if is_valid_configuration(temp_config):
                    new_sum = calculate_radius_sum(temp_config)
                    if new_sum > best_sum_current:
                        best_sum_current = new_sum
                        best_x, best_y, best_r = new_x, new_y, new_r
                        improved = True

            # Strategy 2: Try maximizing radius at current position
            max_radius = min(orig_x, width-orig_x, orig_y, height-orig_y)
            if max_radius > orig_r:
                # Try to increase radius to maximum while keeping configuration valid
                temp_config = current_solution.copy()
                temp_config[i] = [orig_x, orig_y, max_radius]

                if is_valid_configuration(temp_config):
                    new_sum = calculate_radius_sum(temp_config)
                    if new_sum > best_sum_current:
                        best_sum_current = new_sum
                        best_x, best_y, best_r = orig_x, orig_y, max_radius
                        improved = True

            # Apply the best improvement
            if improved:
                current_solution[i] = [best_x, best_y, best_r]
                best_sum = best_sum_current

        # If no improvement made, stop
        if not improved:
            break

    return current_solution
