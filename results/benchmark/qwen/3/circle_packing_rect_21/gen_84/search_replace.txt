import numpy as np
from scipy.spatial.distance import cdist
import time
from deap import base, creator, tools, algorithms
from sklearn.cluster import KMeans
import random

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    width, height = 1.0, 1.0  # Optimal square configuration

    # Set seed for reproducibility
    np.random.seed(42)
    random.seed(42)

    # Problem parameters
    n_circles = 21
    max_iterations = 1000

    # Helper function to calculate maximum possible radius at a given position
    def calculate_max_radius(x, y, circles_array):
        # Compute max radius considering boundaries
        max_radius = min(x, width - x, y, height - y)

        # Consider conflicts with existing circles
        for i in range(len(circles_array)):
            px, py, pr = circles_array[i]
            dist = np.sqrt((x - px)**2 + (y - py)**2)
            # If overlapping, reduce radius to prevent overlap
            if dist < pr:
                max_radius = 0  # Invalid position
                break
            # If would overlap, reduce radius accordingly
            elif dist < (pr + max_radius):
                max_radius = max(0, dist - pr)

        return max_radius

    # Fitness function for evolution
    def evaluate(individual):
        # Convert individual to circles array
        circles = np.zeros((n_circles, 3))
        for i in range(n_circles):
            circles[i] = [individual[3*i], individual[3*i+1], individual[3*i+2]]

        # Check constraints
        valid = True
        total_radius = 0

        for i in range(n_circles):
            x, y, r = circles[i]
            # Boundary check
            if x <= 0.05 or x >= width - 0.05 or y <= 0.05 or y >= height - 0.05:
                valid = False
                break

            # Check overlap with other circles
            for j in range(i):
                px, py, pr = circles[j]
                dist = np.sqrt((x - px)**2 + (y - py)**2)
                if dist < (r + pr):
                    valid = False
                    break

            if not valid:
                break

            total_radius += r

        # Return negative because we want to maximize (DEAP minimizes by default)
        return (-total_radius,) if valid else (float('inf'),)

    # Initialize population with better heuristic
    def init_population():
        # Use k-means clustering for better initial spread
        # Generate random points in a grid pattern
        grid_points = []
        grid_size = 6
        margin = 0.05

        for i in range(grid_size):
            for j in range(grid_size):
                x = margin + (width - 2*margin) * i / (grid_size - 1)
                y = margin + (height - 2*margin) * j / (grid_size - 1)
                grid_points.append([x, y])

        # Add some randomness and select best initial points
        initial_points = []
        for _ in range(n_circles):
            # Choose a point from grid with some randomness
            if grid_points:
                point = random.choice(grid_points)
                initial_points.append(point)
                # Remove chosen point to encourage diversity
                grid_points.remove(point)
            else:
                # Fallback to random points
                x = np.random.uniform(0.05, width - 0.05)
                y = np.random.uniform(0.05, height - 0.05)
                initial_points.append([x, y])

        # Generate initial population with heuristic placement
        population = []
        for _ in range(30):
            individual = []
            for i, (x, y) in enumerate(initial_points):
                # Add small random offset to get diversity
                dx = np.random.uniform(-0.05, 0.05)
                dy = np.random.uniform(-0.05, 0.05)
                new_x = np.clip(x + dx, 0.05, width - 0.05)
                new_y = np.clip(y + dy, 0.05, height - 0.05)

                # Compute max radius at that position (considering existing circles in this individual)
                temp_circles = np.array(individual).reshape(-1, 3) if individual else np.zeros((0, 3))
                r = calculate_max_radius(new_x, new_y, temp_circles)
                if r <= 0:
                    r = np.random.uniform(0.01, 0.1)  # fallback to small radius

                individual.extend([new_x, new_y, r])

            # Make sure the individual is valid
            if len(individual) == 3*n_circles:
                population.append(individual)

        return population

    # Setup DEAP
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    toolbox = base.Toolbox()
    toolbox.register("evaluate", evaluate)
    toolbox.register("population", init_population)
    toolbox.register("mate", tools.cxUniform, indpb=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.02, indpb=0.2)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Create initial population
    population = toolbox.population()

    # Evolution parameters
    n_generations = 50
    cxpb = 0.8  # crossover probability
    mutpb = 0.2  # mutation probability

    best_fitness = float('-inf')
    best_individual = None

    # Evolution loop
    for generation in range(n_generations):
        # Evaluate fitness
        fitnesses = list(map(toolbox.evaluate, population))
        for ind, fit in zip(population, fitnesses):
            ind.fitness.values = fit

        # Update best solution
        for ind in population:
            if ind.fitness.values[0] != float('inf') and ind.fitness.values[0] > best_fitness:
                best_fitness = ind.fitness.values[0]
                best_individual = ind[:]

        # Selection
        offspring = toolbox.select(population, len(population))
        offspring = list(map(toolbox.clone, offspring))

        # Crossover and mutation
        for child1, child2 in zip(offspring[::2], offspring[1::2]):
            if random.random() < cxpb:
                toolbox.mate(child1, child2)
                del child1.fitness.values
                del child2.fitness.values

        for mutant in offspring:
            if random.random() < mutpb:
                toolbox.mutate(mutant)
                del mutant.fitness.values

        # Replace population
        population[:] = offspring

    # If no good solution found by evolution, fall back to local refinement
    if best_individual is None:
        # Use the previous method as fallback
        return _fallback_method(width, height, n_circles)

    # Convert best individual to circles array
    circles = np.zeros((n_circles, 3))
    for i in range(n_circles):
        circles[i] = [best_individual[3*i], best_individual[3*i+1], best_individual[3*i+2]]

    # Local refinement phase to improve the solution further
    # Use the local search approach but with the evolved solution as starting point
    circles = _local_refinement(circles, width, height, max_iterations//10)

    return circles
