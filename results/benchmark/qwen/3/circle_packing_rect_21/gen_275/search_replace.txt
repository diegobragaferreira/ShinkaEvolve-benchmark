    def multi_scale_optimization(self, points: np.ndarray,
                               max_iterations: int = 300,
                               initial_tolerance: float = 2.0,
                               final_tolerance: float = 1.0) -> np.ndarray:
        """Enhanced multi-scale optimization with adaptive step sizes and momentum-based movement"""
        current_points = points.copy()
        momentum_history = [np.zeros_like(points)] * 3  # Track recent momentum
        best_improvement = 0.0
        prev_best_radius_sum = 0.0

        # Phase-based optimization with adaptive strategies
        phases = [
            {"name": "exploration", "iterations": 50, "step_sizes": [-0.15, -0.1, 0, 0.1, 0.15], "intensity": 1.0},
            {"name": "medium_exploration", "iterations": 70, "step_sizes": [-0.1, -0.05, 0, 0.05, 0.1], "intensity": 0.8},
            {"name": "refinement", "iterations": 80, "step_sizes": [-0.05, -0.025, 0, 0.025, 0.05], "intensity": 0.6},
            {"name": "fine_tune", "iterations": 100, "step_sizes": [-0.02, -0.01, 0, 0.01, 0.02], "intensity": 0.3}
        ]

        for phase_idx, phase in enumerate(phases):
            phase_steps = phase["iterations"]
            step_sizes = phase["step_sizes"]
            intensity = phase["intensity"]

            for iteration in range(phase_steps):
                # Adaptive tolerance based on progress
                current_tolerance = initial_tolerance + (final_tolerance - initial_tolerance) * (iteration / phase_steps)

                # Calculate adaptive step size based on convergence
                adaptive_factor = 1.0
                if iteration > 5 and phase_idx >= 2:  # Start adapting in later phases
                    recent_improvements = [prev_best_radius_sum - best_improvement]  # Simplified for now
                    if len(recent_improvements) > 0 and max(recent_improvements) < 0.001:
                        adaptive_factor = 0.7  # Reduce step size if progress stalls

                # Momentum-based adaptive movement
                momentum_factor = 0.3 if phase_idx >= 2 else 0.1

                for i in range(len(current_points)):
                    current_point = current_points[i].copy()
                    best_point = current_point.copy()
                    best_radius = self.compute_circle_radius_with_tolerance(current_point, current_points, current_tolerance)
                    best_delta = 0.0

                    # Use momentum from previous iterations
                    momentum = momentum_history[-1][i] if len(momentum_history) > 0 else np.zeros(2)

                    # Try different step combinations
                    step_combinations = []
                    for dx in step_sizes:
                        for dy in step_sizes:
                            step_combinations.append((dx * intensity * adaptive_factor * momentum_factor,
                                                   dy * intensity * adaptive_factor * momentum_factor))

                    # Add momentum-based direction
                    if phase_idx >= 2:  # Only use momentum in later phases
                        step_combinations.append((momentum[0] * 0.5, momentum[1] * 0.5))

                    # Try each combination
                    for dx, dy in step_combinations:
                        new_x = current_point[0] + dx
                        new_y = current_point[1] + dy

                        # Keep within bounds
                        if (0.05 <= new_x <= self.rect_width - 0.05 and
                            0.05 <= new_y <= self.rect_height - 0.05):

                            test_point = np.array([new_x, new_y])
                            test_radius = self.compute_circle_radius_with_tolerance(test_point, current_points, current_tolerance)

                            delta = test_radius - best_radius
                            if delta > best_delta:
                                best_delta = delta
                                best_point = test_point

                    # Update momentum
                    if len(momentum_history) > 2:
                        momentum_history.pop(0)
                    momentum_history.append(best_point - current_point)

                    current_points[i] = best_point

        return current_points
