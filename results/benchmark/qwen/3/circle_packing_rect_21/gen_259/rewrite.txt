def circle_packing21() -> np.ndarray:

    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.

    # Optimize width/height ratio using a grid search approach
    best_ratio = 1.0
    best_sum = 0

    ratios_to_test = [0.5, 0.7, 1.0, 1.3, 1.5, 1.7, 2.0]
    
    for ratio in ratios_to_test:
        width = 2.0 * ratio / (1 + ratio)
        height = 2.0 / (1 + ratio)
        
        # Quick estimation of performance for this ratio
        if ratio < 0.5 or ratio > 2.0:  # Skip extreme cases
            continue
            
        try:
            test_circles = initialize_adaptive_layout(width, height, 21)
            test_sum = np.sum(test_circles[:, 2])
            if test_sum > best_sum:
                best_sum = test_sum
                best_ratio = ratio
        except:
            continue
    
    # Use the best ratio found
    width = 2.0 * best_ratio / (1 + best_ratio)
    height = 2.0 / (1 + best_ratio)

    np.random.seed(42)
    random.seed(42)

    n = 21

    max_iterations = 1500

    improvement_history = []

    phase1_iterations = 300

            for dx, dy in strategies:

                if 0.01 <= new_x <= width - 0.01 and 0.01 <= new_y <= height - 0.01:

                    max_radius = calculate_max_radius_at_position_with_voronoi(

    for i in range(n):

        r = min(r, x - 0.01, width - x - 0.01, y - 0.01, height - y - 0.01)

    return circles


def initialize_adaptive_layout(width: float, height: float, n: int) -> np.ndarray:
    """Initialize circles using adaptive Voronoi-based layout with better spatial distribution."""
    circles = np.zeros((n, 3))
    
    # Try to create a better initial distribution using Voronoi concept
    # First, generate points that are likely to produce good Voronoi cells
    
    # Use a honeycomb-like pattern with some randomness
    rows = 5
    cols = 5
    
    # Generate hexagonal grid points
    hex_points = []
    for i in range(rows):
        for j in range(cols):
            # Hexagonal pattern with offset rows
            x = (j + 0.5) * (width / (cols + 1))
            y = (i + 0.5) * (height / (rows + 1))
            if i % 2 == 1:
                x += width / (2 * (cols + 1))
                
            # Only keep points that fall within the rectangle
            if 0.01 <= x <= width - 0.01 and 0.01 <= y <= height - 0.01:
                hex_points.append([x, y])
    
    # Add corner and center points for better boundary coverage
    corner_points = [
        [0.1, 0.1], [width-0.1, 0.1], [0.1, height-0.1], [width-0.1, height-0.1],
        [width/2, 0.1], [width/2, height-0.1], [0.1, height/2], [width-0.1, height/2],
        [width/2, height/2]
    ]
    
    # Combine and possibly add more points
    all_points = hex_points + corner_points
    
    # Fill with initial positions
    for i in range(n):
        if i < len(all_points):
            x, y = all_points[i]
        else:
            # Random placement with boundary constraints
            x = np.random.uniform(0.05, width - 0.05)
            y = np.random.uniform(0.05, height - 0.05)
        circles[i] = [x, y, 0.01]
    
    # Now compute initial max radii for all positions
    for i in range(n):
        x, y = circles[i][0], circles[i][1]
        circles[i][2] = compute_initial_max_radius(x, y, width, height, circles[:i])
    
    return circles


def compute_initial_max_radius(x: float, y: float, width: float, height: float, existing_circles: np.ndarray) -> float:
    """Compute initial maximum radius with improved constraint checking."""
    # Boundary constraints
    min_dist_from_edge = min(x, width - x, y, height - y)
    
    if min_dist_from_edge <= 0:
        return 0.001
    
    # Overlap constraints with existing circles (more conservative)
    min_dist_from_others = float('inf')
    
    for circle in existing_circles:
        if circle[2] > 0:  # Only consider placed circles
            cx, cy, cr = circle
            dist = np.sqrt((x - cx)**2 + (y - cy)**2)
            # Conservative constraint - allow some overlap for exploration
            min_dist_from_others = min(min_dist_from_others, dist - cr * 0.95)
    
    # Take minimum of boundary and overlap constraints
    max_radius = min(min_dist_from_edge, min_dist_from_others)
    
    return max(0.001, max_radius)


def calculate_max_radius_with_voronoi_heuristic(circles, index, width, height, overlap_tolerance=0.99):
    """Calculate maximum radius using Voronoi heuristic for smarter positioning."""
    x, y, current_radius = circles[index]

    # Maximum radius based on container boundaries
    max_radius_bound = min(x, y, width - x, height - y)

    # Maximum radius based on other circles (with improved tolerance handling)
    max_radius_overlap = float('inf')
    
    # Use cKDTree for efficient neighbor search
    if len(circles) > 1:
        tree = cKDTree(circles[:, :2])
        # Find neighbors within a reasonable distance
        neighbors = tree.query_ball_point([x, y], 2.0 * max_radius_bound)
        # Only consider actual neighbors (not including self)
        neighbors = [n for n in neighbors if n != index]
        
        for i in neighbors:
            if i != index:
                cx, cy, cr = circles[i]
                # Distance to other circle center
                dist = np.sqrt((x - cx)**2 + (y - cy)**2)
                # Max radius that avoids overlap with tolerance
                max_radius_for_this_circle = dist * overlap_tolerance - cr

                if max_radius_for_this_circle < max_radius_overlap:
                    max_radius_overlap = max_radius_for_this_circle
    else:
        # If no neighbors, just use boundary constraints
        max_radius_overlap = max_radius_bound

    max_radius = min(max_radius_bound, max_radius_overlap)
    return max(max_radius, 0.001)  # Ensure minimum radius
