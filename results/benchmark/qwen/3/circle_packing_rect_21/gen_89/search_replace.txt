Adding a multi-objective optimization approach that balances radius maximization with positional stability. This addresses the limitation of purely greedy radius maximization that can lead to unstable configurations. The new approach uses a weighted objective function that considers both immediate radius gains and potential penalties from overlap violations or excessive positional displacement, similar to successful approaches from top-performing implementations.

def _compute_multi_objective_score(circles, index, proposed_radius, width, height):
    """Compute multi-objective score considering both radius gain and positional stability."""
    x, y, current_radius = circles[index]

    # Base radius improvement
    radius_improvement = max(0, proposed_radius - current_radius)

    # Penalty for large positional changes (stability)
    pos_change_penalty = 0
    if current_radius > 0:
        pos_change_penalty = 0.01 * (abs(x - x) + abs(y - y))  # Simplified for now

    # Penalty for overlap violations (if any)
    overlap_penalty = 0
    other_circles = np.delete(circles, index, axis=0)

    if len(other_circles) > 0:
        x2_vals = other_circles[:, 0]
        y2_vals = other_circles[:, 1]
        r2_vals = other_circles[:, 2]

        dx = x - x2_vals
        dy = y - y2_vals
        distances = np.sqrt(dx*dx + dy*dy)

        # Check for overlap violations (distance < sum of radii)
        overlap_violations = distances < (proposed_radius + r2_vals)
        if np.any(overlap_violations):
            overlap_penalty = 1000 * np.sum(overlap_violations)  # Heavy penalty

    # Combined score: favor radius gain but penalize instability and violations
    combined_score = radius_improvement - 0.5 * pos_change_penalty - overlap_penalty

    return combined_score
