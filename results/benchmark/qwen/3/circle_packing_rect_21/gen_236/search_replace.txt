    def initialize_positions(self) -> np.ndarray:
        """Initialize circle positions using Voronoi-based distribution and strategic seeding."""
        # Strategy 1: Voronoi-based seeding for even distribution
        voronoi_seed_points = self._generate_voronoi_seeds()

        # Strategy 2: Additional strategic points (corners, edges, center)
        strategic_points = []
        # Corners
        strategic_points.extend([
            [0.05, 0.05], [self.width - 0.05, 0.05],
            [0.05, self.height - 0.05], [self.width - 0.05, self.height - 0.05]
        ])
        # Edges (midpoints)
        strategic_points.extend([
            [self.width/2, 0.05], [self.width/2, self.height - 0.05],
            [0.05, self.height/2], [self.width - 0.05, self.height/2]
        ])
        # Center
        strategic_points.append([self.width/2, self.height/2])

        # Strategy 3: Fill remaining spots with random but bounded points
        remaining_spots = self.n_circles - len(voronoi_seed_points) - len(strategic_points)

        # Combine all strategies
        positions = list(voronoi_seed_points)
        positions.extend(strategic_points[:min(remaining_spots + len(strategic_points), self.n_circles - len(positions))])

        # Add remaining random points if needed
        while len(positions) < self.n_circles:
            x = random.uniform(0.05, self.width - 0.05)
            y = random.uniform(0.05, self.height - 0.05)
            positions.append([x, y])

        # Trim to exact number
        positions = positions[:self.n_circles]

        return np.array(positions)

    def _generate_voronoi_seeds(self) -> list:
        """Generate seed points using Voronoi diagram approach to ensure even distribution."""
        # Generate a coarse grid of points
        grid_size = 8
        points = []
        for i in range(grid_size):
            for j in range(grid_size):
                x = (i + 0.5) / grid_size * self.width
                y = (j + 0.5) / grid_size * self.height
                points.append([x, y])

        # Add boundary points for better coverage
        boundary_points = []
        for i in range(4):
            for j in range(3):
                if i == 0:  # Top edge
                    boundary_points.append([j * self.width / 3, 0.05])
                elif i == 1:  # Right edge
                    boundary_points.append([self.width - 0.05, j * self.height / 3])
                elif i == 2:  # Bottom edge
                    boundary_points.append([j * self.width / 3, self.height - 0.05])
                else:  # Left edge
                    boundary_points.append([0.05, j * self.height / 3])

        points.extend(boundary_points[:min(len(boundary_points), 10)])

        # Use Voronoi to get representative points
        if len(points) >= 3:
            try:
                vor = Voronoi(points)
                # Take Voronoi vertices as seeds (but filter to be within bounds)
                seeds = []
                for vertex in vor.vertices:
                    if (0.05 <= vertex[0] <= self.width - 0.05 and
                        0.05 <= vertex[1] <= self.height - 0.05):
                        seeds.append(vertex.tolist())
                # If we don't have enough seeds, add more grid points
                if len(seeds) < self.n_circles:
                    # Add some additional grid points
                    additional_points = []
                    for i in range(6):

                            y = 0.1 + (j * 0.8 / 5) * self.height
                            additional_points.append([x, y])
                    seeds.extend(additional_points[:self.n_circles - len(seeds)])
                return seeds[:self.n_circles]
            except:
                # Fallback to simple grid if Voronoi fails
                pass

        # Fallback to simple grid
        seeds = []
        rows = max(2, int(np.sqrt(self.n_circles)))
        cols = max(2, int(np.ceil(self.n_circles / rows)))
        col_spacing = self.width / (cols + 1)
        row_spacing = self.height / (rows + 1)

        for row in range(rows):
            for col in range(cols):
                if len(seeds) < self.n_circles:
                    x = (col + 1) * col_spacing
                    y = (row + 1) * row_spacing
                    seeds.append([x, y])

        return seeds[:self.n_circles]
