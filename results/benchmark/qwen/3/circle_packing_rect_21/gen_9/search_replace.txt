import numpy as np
import random
from scipy.spatial.distance import cdist

# Rectangle dimensions (perimeter = 4, so width + height = 2)
RECT_WIDTH = 1.0
RECT_HEIGHT = 1.0

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Initialize with a regular grid pattern
    n = 21
    circles = np.zeros((n, 3))

    # Create a hexagonal-like grid pattern
    rows = 5
    cols = 5
    spacing_x = RECT_WIDTH / (cols + 1)
    spacing_y = RECT_HEIGHT / (rows + 1)

    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x = spacing_x * (j + 1)
            y = spacing_y * (i + 1)

            # Set initial radius based on proximity to edges
            min_dist_to_edge = min(x, y, RECT_WIDTH - x, RECT_HEIGHT - y)
            initial_radius = min(0.15, min_dist_to_edge / 2)

            circles[idx] = [x, y, initial_radius]
            idx += 1

        if idx >= n:
            break

    # Fill remaining positions
    for i in range(idx, n):
        x = random.uniform(0.05, RECT_WIDTH - 0.05)
        y = random.uniform(0.05, RECT_HEIGHT - 0.05)
        min_dist_to_edge = min(x, y, RECT_WIDTH - x, RECT_HEIGHT - y)
        circles[i] = [x, y, min(0.1, min_dist_to_edge / 2)]

    # Local optimization to maximize radii
    # Gradient ascent approach: try to increase each radius while respecting constraints
    max_iter = 1000
    for iteration in range(max_iter):
        improved = False

        # Try to increase each circle's radius
        for i in range(n):
            current_x, current_y, current_radius = circles[i]

            # Calculate maximum possible radius
            max_possible_radius = min(
                current_x, current_y,
                RECT_WIDTH - current_x, RECT_HEIGHT - current_y
            )

            # Check overlaps with other circles
            overlap_penalty = 0
            for j in range(n):
                if i != j:
                    x2, y2, r2 = circles[j]
                    distance = np.sqrt((current_x - x2)**2 + (current_y - y2)**2)
                    if distance < (current_radius + r2):
                        overlap_penalty += 1000 * (current_radius + r2 - distance)

            # If we can safely increase radius
            old_radius = current_radius
            if max_possible_radius > current_radius:
                # Try to increase radius slightly
                new_radius = min(current_radius + 0.01, max_possible_radius)

                # Check if new configuration is valid
                valid = True
                for j in range(n):
                    if i != j:
                        x2, y2, r2 = circles[j]
                        distance = np.sqrt((current_x - x2)**2 + (current_y - y2)**2)
                        if distance < (new_radius + r2):
                            valid = False
                            break

                if valid and overlap_penalty < 1:
                    circles[i][2] = new_radius
                    improved = True

        if not improved:
            break

    # Final refinement: try local adjustments
    for _ in range(100):
        for i in range(n):
            current_x, current_y, current_radius = circles[i]

            # Small random perturbation
            delta_x = random.uniform(-0.02, 0.02)
            delta_y = random.uniform(-0.02, 0.02)

            new_x = max(0.01, min(RECT_WIDTH - 0.01, current_x + delta_x))
            new_y = max(0.01, min(RECT_HEIGHT - 0.01, current_y + delta_y))

            # Check if new position is valid
            valid = True
            new_radius = current_radius

            # Check boundary constraints
            if (new_x - new_radius < 0 or new_x + new_radius > RECT_WIDTH or
                new_y - new_radius < 0 or new_y + new_radius > RECT_HEIGHT):
                continue

            # Check overlap with others
            for j in range(n):
                if i != j:
                    x2, y2, r2 = circles[j]
                    distance = np.sqrt((new_x - x2)**2 + (new_y - y2)**2)
                    if distance < (new_radius + r2):
                        valid = False
                        break

            if valid:
                circles[i][0] = new_x
                circles[i][1] = new_y

    return circles
