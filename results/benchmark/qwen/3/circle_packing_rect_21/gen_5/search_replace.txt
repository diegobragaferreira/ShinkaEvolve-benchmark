import numpy as np
import random
import time

# Set seed for reproducibility
np.random.seed(42)
random.seed(42)

def is_valid_position(circle, circles):
    """Check if a circle position is valid (within bounds and not overlapping)"""
    x, y, r = circle

    # Check boundary constraints (rectangle 0,0 to 2,2 for perimeter 4)
    if x - r < 0 or x + r > 2 or y - r < 0 or y + r > 2:
        return False

    # Check collision with existing circles
    for existing_circle in circles:
        ex, ey, er = existing_circle
        distance = np.sqrt((x - ex)**2 + (y - ey)**2)
        if distance < (r + er):
            return False

    return True

def calculate_fitness(circles):
    """Calculate total radius sum with penalties for invalid configurations"""
    # Basic fitness is sum of radii
    fitness = np.sum(circles[:, 2])

    # Penalize invalid configurations
    for i, circle in enumerate(circles):
        if not is_valid_position(circle, circles[:i]):
            fitness -= 1000  # Heavy penalty for invalid placement

    return fitness

def mutate_individual(circles, mutation_strength=0.1):
    """Create a mutated version of the circles configuration"""
    mutated = circles.copy()

    # Mutate some circles
    num_mutations = max(1, int(len(circles) * 0.3))  # Mutate about 30% of circles

    for _ in range(num_mutations):
        idx = np.random.randint(len(circles))
        circle = mutated[idx]

        # Slightly adjust position and radius
        mutated[idx] = [
            max(0.01, min(1.99, circle[0] + np.random.normal(0, mutation_strength))),
            max(0.01, min(1.99, circle[1] + np.random.normal(0, mutation_strength))),
            max(0.01, circle[2] * np.random.normal(1, mutation_strength * 0.5))
        ]

    return mutated

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    start_time = time.time()

    # Parameters for evolution
    population_size = 30
    generations = 150
    elite_size = 5

    # Generate initial population
    population = [generate_initial_solution() for _ in range(population_size)]

    best_fitness = -float('inf')
    best_solution = None

    for generation in range(generations):
        # Calculate fitness for each individual
        fitness_scores = []
        for individual in population:
            fitness = calculate_fitness(individual)
            fitness_scores.append(fitness)

            if fitness > best_fitness:
                best_fitness = fitness
                best_solution = individual.copy()

        # Sort by fitness descending
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]

        # Print progress
        if generation % 30 == 0:
            print(f"Generation {generation}: Best fitness = {best_fitness:.6f}")

        # Create new population with elitism
        new_population = population[:elite_size]

        # Generate offspring through mutation
        while len(new_population) < population_size:
            # Select parents (tournament selection)
            tournament_size = 3
            parent_indices = np.random.choice(len(population), tournament_size)
            parent_fitness = [fitness_scores[i] for i in parent_indices]
            parent_idx = parent_indices[np.argmax(parent_fitness)]

            # Mutate selected parent
            mutated = mutate_individual(population[parent_idx])
            new_population.append(mutated)

        population = new_population[:population_size]

    # Refine the best solution further
    if best_solution is not None:
        # Apply local search to the best solution
        for _ in range(50):
            improved = False
            mutated = mutate_individual(best_solution, mutation_strength=0.05)
            if calculate_fitness(mutated) > calculate_fitness(best_solution):
                best_solution = mutated
                improved = True

            if not improved:
                break

    print(f"Final best fitness: {calculate_fitness(best_solution):.6f}")
    print(f"Execution time: {time.time() - start_time:.2f} seconds")

    return best_solution
