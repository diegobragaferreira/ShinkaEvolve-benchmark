import numpy as np
from scipy.spatial import Voronoi, distance_matrix
from scipy.spatial.distance import cdist
from sklearn.cluster import KMeans
import math
from numba import jit

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Container setup (perimeter = 4, so width + height = 2)
    container_width, container_height = 1.2, 0.8

    # Number of circles
    n = 21

    # Initialize circles array
    circles = np.zeros((n, 3))

    # Phase 1: Voronoi-based force field initialization
    # Generate strategic anchor points (corners and edges)
    anchors = []
    # Corners
    anchors.extend([(0.05, 0.05), (container_width - 0.05, 0.05),
                   (0.05, container_height - 0.05), (container_width - 0.05, container_height - 0.05)])
    # Edge centers
    anchors.extend([(container_width/2, 0.05), (container_width/2, container_height - 0.05),
                   (0.05, container_height/2), (container_width - 0.05, container_height/2)])

    # Generate dense candidate points
    candidate_points = []
    grid_density = 64
    for i in range(grid_density):
        for j in range(grid_density):
            x = (i + 0.5) / grid_density * container_width
            y = (j + 0.5) / grid_density * container_height
            candidate_points.append((x, y))

    # Add random points
    np.random.seed(42)
    for _ in range(100):
        x = np.random.uniform(0.05, container_width - 0.05)
        y = np.random.uniform(0.05, container_height - 0.05)
        candidate_points.append((x, y))

    # Voronoi-based selection using k-means
    all_points = anchors + candidate_points
    kmeans = KMeans(n_clusters=min(len(all_points), n), init='k-means++', n_init=10, random_state=42)
    all_array = np.array(all_points)
    kmeans.fit(all_array)
    selected_centroids = kmeans.cluster_centers_

    # Select actual points closest to centroids
    selected_positions = []
    for centroid in selected_centroids:
        distances = np.linalg.norm(all_array - centroid, axis=1)
        closest_idx = np.argmin(distances)
        selected_positions.append(tuple(all_array[closest_idx]))

    # Ensure exact count
    if len(selected_positions) < n:
        remaining_needed = n - len(selected_positions)
        for i in range(min(remaining_needed, len(candidate_points))):
            selected_positions.append(candidate_points[i])
    elif len(selected_positions) > n:
        selected_positions = selected_positions[:n]

    # Initialize with positions and small radii
    for i in range(n):
        x, y = selected_positions[i]
        circles[i] = [x, y, 0.01]

    # Phase 2: Force Field Optimization with Voronoi Partitioning
    max_iterations = 3000
    best_sum_radii = 0
    best_circles = None

    # Force field parameters
    force_strength = 0.5
    boundary_repulsion = 2.0
    max_radius_limit = 0.4
    time_step = 0.1
    
    # Precompute Voronoi regions for efficiency
    voronoi_regions = [None] * n

    @jit(nopython=True)
    def calculate_voronoi_forces(positions, radii, n, container_width, container_height):
        """Calculate net forces for all circles using efficient neighbor identification"""
        forces = np.zeros((n, 2))
        
        # Precompute all distances
        distances = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i != j:
                    dx = positions[i, 0] - positions[j, 0]
                    dy = positions[i, 1] - positions[j, 1]
                    distances[i, j] = np.sqrt(dx*dx + dy*dy)
                else:
                    distances[i, j] = np.inf
        
        # Calculate forces - only direct interaction
        for i in range(n):
            for j in range(n):
                if i != j:
                    dist = distances[i, j]
                    if dist < (radii[i] + radii[j]) and dist > 1e-8:
                        # Repulsive force
                        force_magnitude = (radii[i] + radii[j] - dist) / (dist + 1e-8)
                        dx = positions[j, 0] - positions[i, 0]
                        dy = positions[j, 1] - positions[i, 1]
                        forces[i, 0] -= force_magnitude * dx
                        forces[i, 1] -= force_magnitude * dy
        
        # Boundary repulsion (stronger forces near walls)
        for i in range(n):
            x, y = positions[i, 0], positions[i, 1]
            r = radii[i]
            
            # Left boundary
            if x < r:
                forces[i, 0] += (r - x) * boundary_repulsion
            # Right boundary
            if x + r > container_width:
                forces[i, 0] -= (x + r - container_width) * boundary_repulsion
            # Bottom boundary
            if y < r:
                forces[i, 1] += (r - y) * boundary_repulsion
            # Top boundary
            if y + r > container_height:
                forces[i, 1] -= (y + r - container_height) * boundary_repulsion
                
        return forces

    @jit(nopython=True)
    def compute_max_radius_fast(positions, radii, index, width, height, n):
        """Fast calculation of max radius using direct neighbor analysis"""
        x, y = positions[index, 0], positions[index, 1]
        
        # Boundary constraints
        max_radius = min(x, y, width - x, height - y)
        
        # Overlap constraints with direct comparison
        for i in range(n):
            if i != index:
                dx = x - positions[i, 0]
                dy = y - positions[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist > 1e-8:
                    max_radius_for_this_circle = dist - radii[i]
                    if max_radius_for_this_circle < max_radius:
                        max_radius = max_radius_for_this_circle
                        
        return max(max_radius, 0.001)

    # Main optimization loop
    for iteration in range(max_iterations):
        # Recalculate Voronoi regions every 50 iterations for efficiency
        if iteration % 50 == 0:
            # Prepare points for Voronoi
            points = circles[:, :2].copy()
            try:
                vor = Voronoi(points)
                # For simplicity, we'll just compute the max radius efficiently for now
                # In a more advanced version this would be used for neighbor identification
                pass
            except:
                pass  # Fall back to regular computation if Voronoi fails

        positions = circles[:, :2]
        radii = circles[:, 2]

        # Calculate all forces efficiently
        forces = calculate_voronoi_forces(positions, radii, n, container_width, container_height)

        # Update positions and radii
        for i in range(n):
            x, y, r = circles[i]
            
            # Update radius based on available space
            max_radius = compute_max_radius_fast(positions, radii, i, container_width, container_height, n)
            max_radius = min(max_radius, max_radius_limit)
            
            if max_radius > r:
                # Increase radius gradually
                new_radius = min(r + 0.005 * (max_radius - r), max_radius)
                circles[i, 2] = new_radius
            
            # Apply forces to position with boundaries
            fx, fy = forces[i, 0], forces[i, 1]
            
            # Update position
            new_x = max(r, min(container_width - r, x + time_step * fx))
            new_y = max(r, min(container_height - r, y + time_step * fy))
            
            circles[i, 0] = new_x
            circles[i, 1] = new_y

        # Occasionally apply global refinement
        if iteration % 200 == 0 and iteration > 0:
            for i in range(n):
                x, y, r = circles[i]
                # Random perturbation with adaptive scale
                perturbation_scale = max(0.001, 0.02 * (1.0 - iteration/max_iterations))
                circles[i] = [max(r, min(container_width - r, x + np.random.normal(0, perturbation_scale))),
                              max(r, min(container_height - r, y + np.random.normal(0, perturbation_scale))),
                              r]

        # Save best configuration
        current_sum = np.sum(radii)
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_circles = circles.copy()

    # Return best configuration or final state
    if best_circles is not None:
        return best_circles

    # Final cleanup
    for i in range(n):
        x, y, r = circles[i]
        x = max(r, min(container_width - r, x))
        y = max(r, min(container_height - r, y))
        circles[i] = [x, y, r]

    return circles
