import numpy as np
import random
from scipy.spatial.distance import cdist

def check_collision(circle1, circle2):
    """Check if two circles collide"""
    x1, y1, r1 = circle1
    x2, y2, r2 = circle2
    distance = np.sqrt((x2-x1)**2 + (y2-y1)**2)
    return distance < (r1 + r2)

def check_boundary(circle, width, height):
    """Check if circle is within bounds"""
    x, y, r = circle
    return (r <= x <= width-r) and (r <= y <= height-r)

def is_valid_configuration(circles, width, height):
    """Check if configuration is valid (no collisions, all within bounds)"""
    for i, circle in enumerate(circles):
        if not check_boundary(circle, width, height):
            return False
        for j in range(i):
            if check_collision(circles[i], circles[j]):
                return False
    return True

def evaluate_fitness(circles):
    """Evaluate fitness as sum of radii"""
    return np.sum(circles[:, 2])

def initialize_hexagonal_pack(width, height, n_circles):
    """Initialize circles using hexagonal packing pattern"""
    # Start with a reasonable initial radius
    max_radius = min(width, height) * 0.1
    circles = []

    # Try to place circles in a hexagonal pattern
    row_height = max_radius * 2 * 0.866  # sqrt(3)/2
    col_width = max_radius * 2

    row = 0
    col = 0
    placed_count = 0

    while placed_count < n_circles:
        x = col_width * col + max_radius
        y = row_height * row + max_radius

        # Adjust for staggered rows
        if row % 2 == 1:
            x += col_width / 2

        # Check bounds
        if x <= width - max_radius and y <= height - max_radius:
            circles.append([x, y, max_radius])
            placed_count += 1

        col += 1
        if col * col_width + max_radius > width:
            col = 0
            row += 1
            if row * row_height + max_radius > height:
                break

    # Fill remaining positions with smaller circles if needed
    while len(circles) < n_circles:
        # Try to place additional circles with small random radii
        x = random.uniform(max_radius, width - max_radius)
        y = random.uniform(max_radius, height - max_radius)
        r = random.uniform(0.001, max_radius * 0.5)
        circles.append([x, y, r])

    return np.array(circles[:n_circles])

def mutate_circle(circle, mutation_rate=0.1):
    """Mutate a single circle"""
    mutated = circle.copy()
    if random.random() < mutation_rate:
        mutated[0] += random.gauss(0, 0.05)  # x coordinate
    if random.random() < mutation_rate:
        mutated[1] += random.gauss(0, 0.05)  # y coordinate
    if random.random() < mutation_rate:
        mutated[2] *= random.uniform(0.9, 1.1)  # radius
        mutated[2] = max(mutated[2], 0.001)  # ensure positive radius
    return mutated

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: width + height = 2
    # Optimal aspect ratio for circle packing is typically around 3:1
    width, height = 1.5, 0.5

    # Initialize with hexagonal packing
    circles = initialize_hexagonal_pack(width, height, 21)

    # Simple local optimization approach
    best_fitness = evaluate_fitness(circles)
    best_config = circles.copy()

    # Try a few iterations of improvement
    for iteration in range(100):
        # Create a new candidate configuration by mutating
        new_circles = circles.copy()

        # Mutate a few circles
        mutate_count = random.randint(3, 8)
        for _ in range(mutate_count):
            idx = random.randint(0, 20)
            new_circles[idx] = mutate_circle(new_circles[idx])

        # Ensure all circles stay within bounds
        for i in range(len(new_circles)):
            x, y, r = new_circles[i]
            # Keep radius positive and within bounds
            r = max(0.001, min(r, width/2, height/2))
            # Keep center within bounds
            x = max(r, min(width - r, x))
            y = max(r, min(height - r, y))
            new_circles[i] = [x, y, r]

        # Check if new configuration is valid
        if is_valid_configuration(new_circles, width, height):
            new_fitness = evaluate_fitness(new_circles)
            if new_fitness > best_fitness:
                best_fitness = new_fitness
                best_config = new_circles.copy()
                circles = new_circles.copy()
            elif random.random() < 0.1:  # Accept some worse solutions occasionally
                circles = new_circles.copy()

    return best_config
