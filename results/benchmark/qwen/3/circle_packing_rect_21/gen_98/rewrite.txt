def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    # Using width=1.2, height=0.8 for optimal packing efficiency
    width, height = 1.2, 0.8
    
    # Set random seed for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    # Step 1: Generate initial Voronoi-based configuration
    initial_centers = generate_voronoi_seeds(width, height, 21)
    
    # Step 2: Initialize circles with maximum possible radii at Voronoi vertices
    circles = initialize_from_voronoi(initial_centers, width, height)
    
    # Step 3: Apply constrained optimization to maximize sum of radii
    optimized_circles = optimize_voronoi_based(circles, width, height)
    
    # Step 4: Final refinement with boundary correction
    final_circles = finalize_configuration(optimized_circles, width, height)
    
    return final_circles

def generate_voronoi_seeds(width: float, height: float, n: int) -> np.ndarray:
    """Generate initial seed points using Voronoi-based strategy."""
    # Create a mixture of structured grid points and strategic placements
    points = []
    
    # Regular grid points for structured coverage
    grid_rows, grid_cols = 5, 5
    for i in range(grid_rows):
        for j in range(grid_cols):
            if len(points) >= n:
                break
            x = 0.1 + (j + 1) * (width - 0.2) / (grid_cols + 1)
            y = 0.1 + (i + 1) * (height - 0.2) / (grid_rows + 1)
            points.append([x, y])
    
    # Corner points for better boundary coverage
    if len(points) < n:
        corners = [
            [0.1, 0.1], [width-0.1, 0.1], 
            [0.1, height-0.1], [width-0.1, height-0.1],
            [width/2, 0.1], [width/2, height-0.1],
            [0.1, height/2], [width-0.1, height/2]
        ]
        for corner in corners:
            if len(points) < n:
                points.append(corner)
    
    # Random points to fill remaining space
    while len(points) < n:
        x = np.random.uniform(0.1, width - 0.1)
        y = np.random.uniform(0.1, height - 0.1)
        points.append([x, y])
    
    return np.array(points[:n])

def initialize_from_voronoi(centers: np.ndarray, width: float, height: float) -> np.ndarray:
    """Initialize circles based on Voronoi cells with computed maximum radii."""
    circles = np.zeros((len(centers), 3))
    
    # Generate Voronoi diagram
    vor = Voronoi(centers)
    
    # Try to place circles at Voronoi vertices or centroids
    for i, center in enumerate(centers):
        x, y = center
        
        # Make sure point is within bounds
        x = max(0.1, min(width - 0.1, x))
        y = max(0.1, min(height - 0.1, y))
        
        # Compute maximum radius at this position
        max_radius = compute_max_radius_at_point(x, y, width, height, circles[:i])
        
        circles[i] = [x, y, max_radius]
    
    return circles

def compute_max_radius_at_point(x: float, y: float, width: float, height: float, existing_circles: np.ndarray) -> float:
    """Compute maximum radius for circle at (x,y) given existing circles and container boundaries."""
    # Boundary constraints
    min_dist_from_edge = min(x, width - x, y, height - y)
    
    if min_dist_from_edge <= 0:
        return 0
    
    # Overlap constraints with existing circles
    min_dist_from_others = float('inf')
    
    for circle in existing_circles:
        if circle[2] > 0:  # Only consider placed circles
            cx, cy, cr = circle
            dist = sqrt((x - cx)**2 + (y - cy)**2)
            min_dist_from_others = min(min_dist_from_others, dist - cr)
    
    # Take minimum of boundary and overlap constraints
    max_radius = min(min_dist_from_edge, min_dist_from_others)
    
    return max(0.001, max_radius)

def optimize_voronoi_based(circles: np.ndarray, width: float, height: float) -> np.ndarray:
    """Optimize using constrained optimization approach with Voronoi structure preservation."""
    
    def objective(params):
        """Minimize negative of sum of radii (maximize sum of radii)."""
        # params contains [x1,y1,r1,x2,y2,r2,...]
        total_radius = 0
        for i in range(21):
            x = params[3*i]
            y = params[3*i+1]
            r = params[3*i+2]
            total_radius += r
        return -total_radius  # Negative because we want to maximize
    
    def constraint_function(params):
        """Ensure no overlaps and boundary constraints."""
        # Extract positions and radii
        positions = params.reshape(-1, 3)[:, :2]  # x,y pairs
        radii = params.reshape(-1, 3)[:, 2]      # radii
        
        # Constraint violations - return negative values for violations
        violations = []
        
        # Boundary constraints
        for i in range(21):
            x, y, r = positions[i][0], positions[i][1], radii[i]
            if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
                violations.append(-1)  # Violation
            else:
                violations.append(0)   # Valid
        
        # Overlap constraints
        for i in range(21):
            for j in range(i+1, 21):
                x1, y1, r1 = positions[i][0], positions[i][1], radii[i]
                x2, y2, r2 = positions[j][0], positions[j][1], radii[j]
                dist = sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2:  # Overlap violation
                    violations.append(-dist/(r1+r2))  # Ratio of overlap to sum of radii
        
        return np.array(violations)
    
    # Initial guess from existing circles
    initial_params = circles.flatten()
    
    # Create simple optimization bounds
    bounds = []
    for i in range(21):
        # x bounds
        bounds.append((0.1, width - 0.1))
        # y bounds  
        bounds.append((0.1, height - 0.1))
        # r bounds
        bounds.append((0.001, min(width, height) * 0.5))
    
    # Use a simple optimization approach with manual constraint handling
    best_circles = circles.copy()
    best_sum = np.sum(circles[:, 2])
    
    # Multi-stage optimization
    for stage in range(3):
        # Stage 1: Global search with random perturbations
        if stage == 0:
            iterations = 100
        elif stage == 1:
            iterations = 150
        else:
            iterations = 200
            
        for iteration in range(iterations):
            # Random perturbation of positions
            new_circles = best_circles.copy()
            
            # Select a few circles to perturb
            indices = list(range(21))
            random.shuffle(indices)
            perturbed_count = min(5, len(indices))
            
            for i in range(perturbed_count):
                idx = indices[i]
                # Add small random movement
                dx = random.uniform(-0.05, 0.05)
                dy = random.uniform(-0.05, 0.05)
                dr = random.uniform(-0.02, 0.02)
                
                new_x = max(0.1, min(width - 0.1, new_circles[idx, 0] + dx))
                new_y = max(0.1, min(height - 0.1, new_circles[idx, 1] + dy))
                new_r = max(0.001, min(0.5, new_circles[idx, 2] + dr))
                
                new_circles[idx, 0] = new_x
                new_circles[idx, 1] = new_y
                new_circles[idx, 2] = new_r
                
                # Recompute maximum radius for this circle
                max_rad = compute_max_radius_at_point(new_x, new_y, width, height, 
                                                     np.vstack([new_circles[:idx], new_circles[idx+1:]]))
                new_circles[idx, 2] = max_rad
            
            # Validate and accept if better
            if is_valid_configuration(new_circles, width, height):
                new_sum = np.sum(new_circles[:, 2])
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_circles = new_circles.copy()
    
    return best_circles

def finalize_configuration(circles: np.ndarray, width: float, height: float) -> np.ndarray:
    """Final refinement with boundary correction and optimization."""
    # Ensure all circles are valid
    final_circles = circles.copy()
    
    # Perform one final optimization pass focusing on radius maximization
    for iteration in range(50):
        improved = False
        for i in range(21):
            old_r = final_circles[i, 2]
            # Recompute maximum radius for position
            max_radius = compute_max_radius_at_point(
                final_circles[i, 0], final_circles[i, 1], width, height,
                np.vstack([final_circles[:i], final_circles[i+1:]])
            )
            
            if max_radius > old_r + 1e-6:
                final_circles[i, 2] = max_radius
                improved = True
        
        if not improved:
            break
    
    # Final boundary correction
    for i in range(21):
        x, y, r = final_circles[i]
        # Ensure circle stays within bounds
        r = min(r, x - 0.01, width - x - 0.01, y - 0.01, height - y - 0.01)
        r = max(r, 0.001)
        final_circles[i] = [x, y, r]
    
    return final_circles

def is_valid_configuration(circles: np.ndarray, width: float, height: float) -> bool:
    """Check if all circles satisfy constraints."""
    for i in range(len(circles)):
        x, y, r = circles[i]
        # Check boundary conditions
        if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
            return False
            
        # Check overlap with other circles
        for j in range(i + 1, len(circles)):
            x2, y2, r2 = circles[j]
            distance = sqrt((x - x2)**2 + (y - y2)**2)
            if distance < r + r2:
                return False
                
    return True
