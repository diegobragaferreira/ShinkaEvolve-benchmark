import numpy as np

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions - since perimeter = 4, width + height = 2
    # Optimized aspect ratio found to work well for circle packing
    rect_width, rect_height = 1.2, 0.8
    
    # Set seeds for reproducibility
    np.random.seed(42)
    random.seed(42)

    # Phase 1: Advanced Voronoi-based Initialization
    circles = np.zeros((21, 3))
    
    # Generate strategic initial points using Voronoi tessellation
    # Create a dense grid of candidate points
    grid_points = []
    grid_size = 15
    for i in range(grid_size):
        for j in range(grid_size):
            x = 0.05 + (i / (grid_size - 1)) * (rect_width - 0.1)
            y = 0.05 + (j / (grid_size - 1)) * (rect_height - 0.1)
            grid_points.append([x, y])
    
    # Create Voronoi diagram to identify good distribution points
    try:
        vor = Voronoi(np.array(grid_points))
        # Use Voronoi centroids that are within bounds
        voronoi_centroids = []
        for point in vor.points:
            if 0.05 <= point[0] <= rect_width - 0.05 and 0.05 <= point[1] <= rect_height - 0.05:
                voronoi_centroids.append(tuple(point))
        
        # Select first 21 centroid points
        selected_centroids = voronoi_centroids[:21]
    except:
        # Fallback to grid if Voronoi fails
        selected_centroids = [(0.05 + (i / (grid_size - 1)) * (rect_width - 0.1), 
                              0.05 + (j / (grid_size - 1)) * (rect_height - 0.1)) 
                             for i in range(5) for j in range(5)][:21]
    
    # Initialize with Voronoi-based positions and small radii
    for i, (x, y) in enumerate(selected_centroids):
        circles[i] = [x, y, 0.02]

    # Phase 2: Multi-phase Optimization with Enhanced Strategies

    best_radius_sum = 0
    best_circles = None

    # Multi-start approach with enhanced diversity
    for start_iter in range(20):  # Increased from 15 to 20
        # Reset current circles for this iteration
        current_circles = circles.copy()
        
        # Phase 2a: Coarse exploration with aggressive radius growth (first 150 iterations)
        for iteration in range(150):
            improved = False
            
            # Process circles in random order for better exploration
            circle_indices = list(range(21))
            random.shuffle(circle_indices)
            
            for i in circle_indices:
                # Compute maximum possible radius at this location
                max_radius = _compute_max_radius_optimized(current_circles, i, rect_width, rect_height)
                
                # Stochastic acceptance with dynamic temperature
                old_r = current_circles[i, 2]
                
                if max_radius > old_r:
                    # Dynamic temperature that starts high and decreases gradually
                    temp = max(0.01, 1.0 - (iteration / 150.0) * 0.9)
                    prob = math.exp((max_radius - old_r) / (temp + 1e-8))
                    if random.random() < prob:
                        current_circles[i, 2] = max_radius
                        improved = True
                elif max_radius > old_r + 1e-6:
                    current_circles[i, 2] = max_radius
                    improved = True
            
            # Periodic perturbation to escape local minima
            if iteration % 25 == 0 and iteration > 0:
                for _ in range(3):
                    i = random.randint(0, 20)
                    # Apply perturbations with adaptive magnitude
                    pert_mag = 0.05 * (1.0 - iteration/150.0)  # Decreasing magnitude
                    current_circles[i, 0] += random.uniform(-pert_mag, pert_mag)
                    current_circles[i, 1] += random.uniform(-pert_mag, pert_mag)
                    
                    # Clamp to bounds
                    current_circles[i, 0] = np.clip(current_circles[i, 0], 0.01, rect_width - 0.01)
                    current_circles[i, 1] = np.clip(current_circles[i, 1], 0.01, rect_height - 0.01)
                    
                    # Recompute max radius after perturbation
                    max_radius = _compute_max_radius_optimized(current_circles, i, rect_width, rect_height)
                    current_circles[i, 2] = max_radius

        # Phase 2b: Medium refinement with balanced approach (next 250 iterations)
        for iteration in range(250):
            improved = False
            
            # Process circles in shuffled order
            circle_indices = list(range(21))
            random.shuffle(circle_indices)
            
            for i in circle_indices:
                # Use enhanced radius computation
                max_radius = _compute_max_radius_enhanced(current_circles, i, rect_width, rect_height)
                
                # Accept with dynamic probability
                old_r = current_circles[i, 2]
                
                if max_radius > old_r:
                    temp = max(0.01, 0.3 - (iteration / 250.0) * 0.25)
                    prob = math.exp((max_radius - old_r) / (temp + 1e-8))
                    if random.random() < prob:
                        current_circles[i, 2] = max_radius
                        improved = True
                elif max_radius > old_r + 1e-6:
                    current_circles[i, 2] = max_radius
                    improved = True
            
            # Perturbation with medium intensity
            if iteration % 20 == 0 and iteration > 0:
                for _ in range(4):
                    i = random.randint(0, 20)
                    # Apply moderate perturbations
                    pert_mag = 0.03 * (1.0 - iteration/250.0)
                    current_circles[i, 0] += random.uniform(-pert_mag, pert_mag)
                    current_circles[i, 1] += random.uniform(-pert_mag, pert_mag)
                    
                    # Clamp to bounds
                    current_circles[i, 0] = np.clip(current_circles[i, 0], 0.01, rect_width - 0.01)
                    current_circles[i, 1] = np.clip(current_circles[i, 1], 0.01, rect_height - 0.01)
                    
                    # Recompute max radius after perturbation
                    max_radius = _compute_max_radius_optimized(current_circles, i, rect_width, rect_height)
                    current_circles[i, 2] = max_radius

        # Phase 2c: Fine tuning with maximum exploitation (next 350 iterations)
        for iteration in range(350):
            improved = False
            
            # Process circles in shuffled order
            circle_indices = list(range(21))
            random.shuffle(circle_indices)
            
            for i in circle_indices:
                # Compute maximum radius using optimized method
                max_radius = _compute_max_radius_optimized(current_circles, i, rect_width, rect_height)
                
                # Direct acceptance for fine tuning
                if max_radius > current_circles[i, 2]:
                    current_circles[i, 2] = max_radius
                    improved = True
            
            # Fine perturbations
            if iteration % 15 == 0 and iteration > 0:
                for _ in range(5):
                    i = random.randint(0, 20)
                    # Small perturbations
                    current_circles[i, 0] += random.uniform(-0.005, 0.005)
                    current_circles[i, 1] += random.uniform(-0.005, 0.005)
                    
                    # Clamp to bounds
                    current_circles[i, 0] = np.clip(current_circles[i, 0], 0.01, rect_width - 0.01)
                    current_circles[i, 1] = np.clip(current_circles[i, 1], 0.01, rect_height - 0.01)
                    
                    # Recompute max radius after perturbation
                    max_radius = _compute_max_radius_optimized(current_circles, i, rect_width, rect_height)
                    current_circles[i, 2] = max_radius

        # Final validation and scoring
        if _validate_configuration(current_circles, rect_width, rect_height):
            radius_sum = np.sum(current_circles[:, 2])
            if radius_sum > best_radius_sum:
                best_radius_sum = radius_sum
                best_circles = current_circles.copy()

    # Ensure we have a valid solution even if optimizations failed
    if best_circles is None:
        # Fallback to improved grid-based initialization
        circles = np.zeros((21, 3))
        rows, cols = 5, 5  # Changed to 5x5 grid
        x_spacing = rect_width / (cols + 1)
        y_spacing = rect_height / (rows + 1)

        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx >= 21:
                    break
                x = (j + 1) * x_spacing + random.uniform(-0.01, 0.01)
                y = (i + 1) * y_spacing + random.uniform(-0.01, 0.01)
                r = 0.025
                circles[idx] = [x, y, r]
                idx += 1

        # Local optimization on fallback
        for _ in range(400):  # More iterations for fallback
            improved = False
            for i in range(21):
                max_radius = _compute_max_radius_optimized(circles, i, rect_width, rect_height)
                if max_radius > circles[i, 2]:
                    circles[i, 2] = max_radius
                    improved = True
            if not improved:
                break

        best_circles = circles

    # Final correction to ensure valid constraints
    for i in range(21):
        x, y, r = best_circles[i]
        # Ensure circles are within bounds and radius is reasonable
        r = min(r, x, rect_width - x, y, rect_height - y)
        if r <= 0.001:
            r = 0.015
        best_circles[i] = [x, y, r]

    return best_circles

def _compute_max_radius_optimized(circles, index, width, height):
    """Optimized maximum radius computation using vectorized operations."""
    x, y, _ = circles[index]

    # Minimum distance to boundaries
    min_dist_to_boundaries = min(x, width - x, y, height - y)

    # Check collisions with other circles using vectorized operations
    min_dist_to_others = float('inf')
    
    # Efficient vectorized computation
    other_circles = np.delete(circles, index, axis=0)
    
    if len(other_circles) > 0:
        # Vectorized operations for all other circles
        x2_vals = other_circles[:, 0]
        y2_vals = other_circles[:, 1]
        r2_vals = other_circles[:, 2]

        # Compute distances using vectorized operations
        dx = x - x2_vals
        dy = y - y2_vals
        distances = np.sqrt(dx*dx + dy*dy)

        # Calculate constraints - distance minus other radius
        radius_constraints = distances - r2_vals

        # Filter out negative or zero constraints (overlapping circles)
        positive_constraints = radius_constraints[radius_constraints > 1e-10]
        if len(positive_constraints) > 0:
            min_dist_to_others = np.min(positive_constraints)

    # Return the minimum of all constraints
    max_radius = min(min_dist_to_boundaries, min_dist_to_others)

    # Ensure minimum positive radius
    return max(0.001, max_radius)

def _compute_max_radius_enhanced(circles, index, width, height):
    """Enhanced maximum radius computation with better numerical handling."""
    x, y, _ = circles[index]

    # Minimum distance to boundaries
    min_dist_to_boundaries = min(x, width - x, y, height - y)

    # Check collisions with other circles using optimized vectorized operations
    min_dist_to_others = float('inf')
    
    other_circles = np.delete(circles, index, axis=0)
    
    if len(other_circles) > 0:
        # Vectorized distance computation
        x2_vals = other_circles[:, 0]
        y2_vals = other_circles[:, 1]
        r2_vals = other_circles[:, 2]

        # Use broadcasting to compute all distances at once
        dx = np.abs(x - x2_vals)
        dy = np.abs(y - y2_vals)
        distances = np.sqrt(dx*dx + dy*dy)

        # Compute constraints
        radius_constraints = distances - r2_vals

        # Find minimum constraint with proper filtering
        positive_constraints = radius_constraints[radius_constraints > 1e-8]
        if len(positive_constraints) > 0:
            min_dist_to_others = np.min(positive_constraints)

    # Return optimized maximum radius with enhanced stability
    max_radius = min(min_dist_to_boundaries, min_dist_to_others)
    
    # Add numerical safety margin
    return max(0.001, max_radius * 0.995)

def _validate_configuration(circles, width, height):
    """Validate configuration with improved numerical precision."""
    for i in range(len(circles)):
        x, y, r = circles[i]
        # Check boundary conditions
        if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
            return False

        # Check overlap with other circles with robust tolerance
        for j in range(i + 1, len(circles)):
            x2, y2, r2 = circles[j]
            distance = np.sqrt((x - x2)**2 + (y - y2)**2)
            # Allow for numerical precision errors but prevent actual overlap
            if distance < r + r2 - 1e-8:
                return False

    return True
