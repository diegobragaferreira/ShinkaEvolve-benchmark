    if outer_side_length <= 0:

    # Create outer hexagon vertices once
    outer_vertices = hexagon_vertices(0, 0, 0, outer_side_length)

    total_penalty = 0

    inner_positions = params[:-1].reshape(-1, 3)
    inner_hex_vertices = []

        x, y, rot = inner_positions[i]

        inner_hex_vertices.append(hex_vertices)

        if not check_containment_single(hex_vertices, outer_polygon):
            total_penalty += 1e8  # Large penalty for containment failure

    if total_penalty == 0:
        # Use parallel processing for collision detection
        def check_pair_collision(i, j):
            return check_collision_single(inner_hex_vertices[i], inner_hex_vertices[j])

        collision_checks = Parallel(n_jobs=-1)(
            delayed(check_pair_collision)(i, j)

        )

        # Count collisions
        collisions = sum(collision_checks)
        if collisions > 0:

def objective_function(params):
    """Optimized objective function to minimize (negative inverse of outer hex side length)."""
    # params: [x1, y1, rot1, x2, y2, rot2, ..., x11, y11, rot11, R]
    n = 11
    outer_side_length = params[-1]

    if outer_side_length <= 0:
        return 1e10  # Penalty for invalid outer hex

    outer_vertices = hexagon_vertices(0, 0, 0, outer_side_length)
    outer_polygon = prep(Polygon(outer_vertices))  # Prepare for fast intersection tests

    # Check containment and collisions in parallel
    total_penalty = 0

    # Check containment of all inner hexagons (parallel check)
    inner_positions = params[:-1].reshape(-1, 3)
    inner_hex_vertices = []

    # Generate hexagon vertices in parallel
    def generate_hex_vertex(i):
        x, y, rot = inner_positions[i]
        return hexagon_vertices(x, y, rot, 1)

    inner_hex_vertices = Parallel(n_jobs=-1)(
        delayed(generate_hex_vertex)(i) for i in range(n)
    )

    # Check containment in parallel
    def check_containment_parallel(i):
        return check_containment_single(inner_hex_vertices[i], outer_polygon)

    containment_checks = Parallel(n_jobs=-1)(
        delayed(check_containment_parallel)(i) for i in range(n)
    )

    # Sum up containment failures
    containment_failures = sum(1 for check in containment_checks if not check)
    total_penalty += containment_failures * 1e8  # Large penalty for containment failure

    # Check collisions between all pairs of inner hexagons in parallel
    if total_penalty == 0:
        # Use parallel processing for collision detection
        def check_pair_collision(i, j):
            return check_collision_single(inner_hex_vertices[i], inner_hex_vertices[j])

        # Only check necessary pairs (avoid redundant checks and early termination)

            delayed(check_pair_collision)(i, j)
            for i in range(n) for j in range(i+1, n)

        collisions = sum(collision_checks)
