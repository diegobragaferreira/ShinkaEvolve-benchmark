def check_containment_single(hex_vertices, outer_center_x, outer_center_y, outer_radius):
    """Check if all vertices of hexagon are within the outer hexagon using SAT approach."""
    # Create a buffer for floating point precision issues
    buffer = 1e-6
    outer_center = (outer_center_x, outer_center_y)

    # Check if all vertices are within the outer hexagon's circumscribed circle
    for vertex in hex_vertices:
        dist = math.sqrt((vertex[0] - outer_center[0])**2 + (vertex[1] - outer_center[1])**2)
        # With buffer to account for floating point precision
        if dist > outer_radius - buffer:
            return False
    return True

def check_overlap_single(hex1_vertices, hex2_vertices):
    """Check if two hexagons overlap using SAT (Separating Axis Theorem)."""
    def get_edges(vertices):
        edges = []
        n = len(vertices)
        for i in range(n):
            edges.append((vertices[i][0] - vertices[(i+1)%n][0],
                         vertices[i][1] - vertices[(i+1)%n][1]))
        return edges

    def dot_product(v1, v2):
        return v1[0]*v2[0] + v1[1]*v2[1]

    def project_polygon_onto_axis(vertices, axis):
        projections = []
        for vertex in vertices:
            proj = dot_product(vertex, axis)
            projections.append(proj)
        return min(projections), max(projections)

    # Get edges from both hexagons
    edges1 = get_edges(hex1_vertices)
    edges2 = get_edges(hex2_vertices)

    # Generate normals for all edges (perpendicular vectors)
    normals = []
    for edge in edges1 + edges2:
        # Normal vector (perpendicular to edge)
        normal = (-edge[1], edge[0])
        # Normalize
        length = math.sqrt(normal[0]**2 + normal[1]**2)
        if length > 1e-10:  # Avoid division by zero
            normal = (normal[0]/length, normal[1]/length)
        normals.append(normal)

    # Test all axes
    for axis in normals:
        min1, max1 = project_polygon_onto_axis(hex1_vertices, axis)
        min2, max2 = project_polygon_onto_axis(hex2_vertices, axis)

        # Check for overlap
        if max1 < min2 or max2 < min1:
            return False  # No overlap along this axis

    return True  # Overlap detected
