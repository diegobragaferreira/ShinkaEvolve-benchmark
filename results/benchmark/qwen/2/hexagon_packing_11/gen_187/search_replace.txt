    """SAT-based overlap detection - improved version with better numerical stability"""
    # Get edges for both polygons
    edges1 = get_edges(hex1_vertices)
    edges2 = get_edges(hex2_vertices)

    # Get normals to all edges (perpendicular vectors)
    normals1 = []
    normals2 = []

    # Process edges and compute normals with better numerical handling
    for edge in edges1:
        # Normal vector (perpendicular to edge)
        normal = np.array([-edge[1], edge[0]])
        # Normalize with better epsilon checking
        norm_len = np.sqrt(normal[0]**2 + normal[1]**2)
        if norm_len > 1e-12:  # Use stricter epsilon
            normal = normal / norm_len
        else:
            continue  # Skip zero-length edges
        normals1.append(normal)

    for edge in edges2:
        # Normal vector (perpendicular to edge)
        normal = np.array([-edge[1], edge[0]])
        # Normalize with better epsilon checking
        norm_len = np.sqrt(normal[0]**2 + normal[1]**2)
        if norm_len > 1e-12:  # Use stricter epsilon
            normal = normal / norm_len
        else:
            continue  # Skip zero-length edges
        normals2.append(normal)

    # Test all axes - early termination optimization
    all_normals = normals1 + normals2

    # Early termination: if we have too few normals, skip SAT test
    if len(all_normals) == 0:
        return True  # Assume overlap if no valid normals

    # Test all axes with early termination
    for axis in all_normals:
        min1, max1 = project_polygon_onto_axis(hex1_vertices, axis)
        min2, max2 = project_polygon_onto_axis(hex2_vertices, axis)

        # Check for overlap with better numerical handling
        # Use small epsilon to account for floating point errors
