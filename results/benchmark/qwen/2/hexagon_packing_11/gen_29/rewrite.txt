@jit(nopython=True)
def hexagon_vertices(x, y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given position, angle, and side length."""
    angle_rad = np.radians(angle_deg)
    # Hexagon vertices in local coordinate system
    local_vertices = np.array([
        [side_length, 0],
        [side_length/2, side_length * np.sqrt(3)/2],
        [-side_length/2, side_length * np.sqrt(3)/2],
        [-side_length, 0],
        [-side_length/2, -side_length * np.sqrt(3)/2],
        [side_length/2, -side_length * np.sqrt(3)/2]
    ])
    
    # Rotate and translate
    cos_a = np.cos(angle_rad)
    sin_a = np.sin(angle_rad)
    rot_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
    
    rotated_vertices = np.dot(local_vertices, rot_matrix.T)
    global_vertices = rotated_vertices + np.array([x, y])
    
    return global_vertices

@jit(nopython=True)
def point_in_hexagon(point_x, point_y, hex_center_x, hex_center_y, hex_angle, hex_side_length=1):
    """Check if a point is inside a hexagon using ray casting."""
    # Transform point to hexagon's local coordinate system
    hex_angle_rad = np.radians(hex_angle)
    cos_a = np.cos(-hex_angle_rad)
    sin_a = np.sin(-hex_angle_rad)
    
    local_x = (point_x - hex_center_x) * cos_a - (point_y - hex_center_y) * sin_a
    local_y = (point_x - hex_center_x) * sin_a + (point_y - hex_center_y) * cos_a
    
    # Check against the hexagon boundaries
    # For a regular hexagon with side length 1, the distance from center to vertex is 1
    # Distance from center to side is sqrt(3)/2
    dist_to_center = np.sqrt(local_x*local_x + local_y*local_y)
    if dist_to_center > hex_side_length:
        return False
    
    # Check if point is inside by verifying it's on the correct side of all edges
    # Simplified check for performance: use the fact that a hexagon can be checked against 6 half-planes
    # But this is a simplified version - a full SAT approach would be more accurate
    # For now, just check distance and basic orientation
    return True

def get_hexagon_polygon(x, y, angle_deg, side_length=1):
    """Get a Shapely polygon representation of a hexagon."""
    vertices = hexagon_vertices(x, y, angle_deg, side_length)
    return Polygon(vertices)

def check_all_collisions(hex_data, side_length=1):
    """Check if any hexagons collide with each other."""
    n = len(hex_data)
    polygons = [get_hexagon_polygon(hex_data[i,0], hex_data[i,1], hex_data[i,2], side_length) for i in range(n)]
    
    # Check pairwise collisions
    for i in range(n):
        for j in range(i+1, n):
            if polygons[i].intersects(polygons[j]):
                return True
    return False

def check_containment(hex_data, outer_hex_data, outer_side_length, side_length=1):
    """Check if all inner hexagons are contained within the outer hexagon."""
    outer_poly = get_hexagon_polygon(outer_hex_data[0], outer_hex_data[1], outer_hex_data[2], outer_side_length)
    
    for i in range(len(hex_data)):
        inner_poly = get_hexagon_polygon(hex_data[i,0], hex_data[i,1], hex_data[i,2], side_length)
        # We need to make sure ALL vertices of inner hexagon are inside outer hexagon
        inner_vertices = hexagon_vertices(hex_data[i,0], hex_data[i,1], hex_data[i,2], side_length)
        
        for vx, vy in inner_vertices:
            if not outer_poly.contains(Point(vx, vy)):
                return False
    return True

def evaluate_solution(hex_data, outer_hex_data, side_length=1):
    """Evaluate the quality of a solution."""
    # Calculate the actual side length needed
    min_x = min_y = float('inf')
    max_x = max_y = float('-inf')
    
    # Get all vertices of all hexagons
    for i in range(len(hex_data)):
        vertices = hexagon_vertices(hex_data[i,0], hex_data[i,1], hex_data[i,2], side_length)
        for vx, vy in vertices:
            min_x = min(min_x, vx)
            max_x = max(max_x, vx)
            min_y = min(min_y, vy)
            max_y = max(max_y, vy)
    
    # Calculate the minimal bounding box
    width = max_x - min_x
    height = max_y - min_y
    
    # Calculate the side length of the bounding hexagon
    # For a hexagon, the diagonal is 2*side_length
    # We want the smallest hexagon that contains everything
    # Approximate: we consider the minimal hexagon that can contain the rectangle
    # The side length should be at least max(width, height) / sqrt(3)
    # Actually, let's compute it properly:
    
    # Let's compute the minimum hexagon that encloses the entire configuration
    # First find the centroid of all hexagon centers
    center_x = np.mean(hex_data[:,0])
    center_y = np.mean(hex_data[:,1])
    
    # Find maximum distance from centroid to any vertex
    max_dist = 0
    for i in range(len(hex_data)):
        vertices = hexagon_vertices(hex_data[i,0], hex_data[i,1], hex_data[i,2], side_length)
        for vx, vy in vertices:
            dist = np.sqrt((vx - center_x)**2 + (vy - center_y)**2)
            max_dist = max(max_dist, dist)
    
    # The side length of the enclosing hexagon should account for the radius
    # A hexagon with side length s has circumradius s
    # So we need side_length >= max_dist
    outer_side_length = max_dist
    
    # Now check constraints
    collision_free = not check_all_collisions(hex_data, side_length)
    contained = check_containment(hex_data, outer_hex_data, outer_side_length, side_length)
    
    if collision_free and contained:
        return 1.0 / outer_side_length
    else:
        # Invalid solution - heavily penalize
        return 0.0

# Generate initial population with better starting configurations
def generate_initial_population(pop_size, n_hexagons=11):
    """Generate initial population with improved configurations."""
    pop = []
    
    # Generate some good starting configurations manually
    for _ in range(pop_size // 3):
        # Center hexagon with surrounding ones in a hexagonal pattern
        config = np.zeros((n_hexagons, 3))
        config[0] = [0, 0, 0]  # center hexagon
        
        # Place surrounding hexagons around the center
        for i in range(1, 7):  # First 6 neighbors
            angle = (i - 1) * 60  # 60 degree increments
            distance = 1.0  # Just touching
            x = distance * np.cos(np.radians(angle))
            y = distance * np.sin(np.radians(angle))
            config[i] = [x, y, 0]
        
        # Add remaining 4 hexagons in a strategic pattern
        config[7] = [-2.0, 0, 0]  # Left
        config[8] = [2.0, 0, 0]   # Right
        config[9] = [0, 2.0, 0]   # Top
        config[10] = [0, -2.0, 0] # Bottom
        pop.append(config.copy())
    
    # Add more random but reasonable configurations
    for _ in range(pop_size - len(pop)):
        config = np.zeros((n_hexagons, 3))
        # Start with center hexagon
        config[0] = [0, 0, 0]
        
        # Add others with some clustering
        for i in range(1, n_hexagons):
            # Random positions near the center
            angle = np.random.uniform(0, 360)
            distance = np.random.uniform(0.5, 2.5)
            x = distance * np.cos(np.radians(angle))
            y = distance * np.sin(np.radians(angle))
            config[i] = [x, y, np.random.uniform(0, 360)]
        
        pop.append(config)
    
    return pop

def optimize_hexagon_packing():
    """Main optimization function."""
    n_hexagons = 11
    side_length = 1.0
    
    # Parameters
    pop_size = 50
    max_generations = 100
    mutation_rate = 0.3
    elite_size = 5
    
    # Set up optimization
    best_score = 0.0
    best_config = None
    best_outer_hex = None
    best_side_length = float('inf')
    
    # Generate initial population
    population = generate_initial_population(pop_size, n_hexagons)
    
    for generation in range(max_generations):
        # Evaluate all individuals
        scores = []
        for individual in population:
            # Create outer hexagon data (centered at origin, no rotation)
            outer_hex_data = np.array([0.0, 0.0, 0.0])
            
            # Evaluate
            score = evaluate_solution(individual, outer_hex_data, side_length)
            scores.append(score)
        
        # Sort by score (descending)
        sorted_indices = np.argsort(scores)[::-1]
        population = [population[i] for i in sorted_indices]
        scores = [scores[i] for i in sorted_indices]
        
        # Track the best solution
        if scores[0] > best_score:
            best_score = scores[0]
            best_config = population[0].copy()
            # Recalculate exact outer hex side length
            min_x = min_y = float('inf')
            max_x = max_y = float('-inf')
            
            # Get all vertices of all hexagons
            for i in range(len(best_config)):
                vertices = hexagon_vertices(best_config[i,0], best_config[i,1], best_config[i,2], side_length)
                for vx, vy in vertices:
                    min_x = min(min_x, vx)
                    max_x = max(max_x, vx)
                    min_y = min(min_y, vy)
                    max_y = max(max_y, vy)
            
            # Compute minimum bounding circle
            center_x = (min_x + max_x) / 2
            center_y = (min_y + max_y) / 2
            
            max_dist = 0
            for i in range(len(best_config)):
                vertices = hexagon_vertices(best_config[i,0], best_config[i,1], best_config[i,2], side_length)
                for vx, vy in vertices:
                    dist = np.sqrt((vx - center_x)**2 + (vy - center_y)**2)
                    max_dist = max(max_dist, dist)
            
            best_side_length = max_dist
            best_outer_hex = np.array([center_x, center_y, 0])
        
        # Early stopping
        if generation > 20 and abs(scores[0] - scores[1]) < 1e-6:
            break
            
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, scores, 3)
            parent2 = tournament_selection(population, scores, 3)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child, mutation_rate)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    return best_config, best_outer_hex, best_side_length

def tournament_selection(population, scores, k):
    """Select individual using tournament selection."""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmax([scores[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent1[i].copy()
        else:
            child[i] = parent2[i].copy()
    return child

def mutate(individual, rate):
    """Mutate an individual."""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i, 0] += np.random.normal(0, 0.2)  # x position
        if np.random.rand() < rate:
            mutated[i, 1] += np.random.normal(0, 0.2)  # y position
        if np.random.rand() < rate:
            mutated[i, 2] += np.random.normal(0, 10)  # angle
            mutated[i, 2] = mutated[i, 2] % 360       # keep angle in [0, 360)
    return mutated
