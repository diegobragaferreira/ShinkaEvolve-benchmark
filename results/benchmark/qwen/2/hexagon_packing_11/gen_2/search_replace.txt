import numpy as np

def hexagon_vertices(center_x, center_y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given center, angle, and side length"""
    angle_rad = np.radians(angle_deg)
    # Vertices of a regular hexagon with side length 1, centered at origin
    base_vertices = np.array([
        [1, 0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1, 0],
        [-0.5, -np.sqrt(3)/2],
        [0.5, -np.sqrt(3)/2]
    ])

    # Rotate and translate
    cos_a = np.cos(angle_rad)
    sin_a = np.sin(angle_rad)
    rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])

    rotated_vertices = base_vertices @ rotation_matrix.T
    return rotated_vertices + np.array([center_x, center_y])


def check_containment(hex_vertices, outer_center_x, outer_center_y, outer_radius):
    """Check if all vertices of a hexagon are within the outer hexagon"""
    # Create outer hexagon vertices
    outer_vertices = hexagon_vertices(outer_center_x, outer_center_y, 0, outer_radius)

    # Check if all inner vertices are inside outer hexagon using point-in-polygon test
    for vertex in hex_vertices:
        if not point_in_hexagon(vertex[0], vertex[1], outer_center_x, outer_center_y, outer_radius):
            return False
    return True

def hexagon_distance(hex1_vertices, hex2_vertices):
    """Calculate minimum distance between two hexagons"""
    min_dist = float('inf')
    for v1 in hex1_vertices:
        for v2 in hex2_vertices:
            dist = np.linalg.norm(v1 - v2)
            min_dist = min(min_dist, dist)
    return min_dist


def is_valid_configuration(inner_data, outer_center_x, outer_center_y, outer_radius):
    """Check if configuration satisfies all constraints"""
    # Generate vertices for all inner hexagons
    hex_vertices_list = []
    for i in range(len(inner_data)):
        center_x, center_y, angle = inner_data[i]
        vertices = hexagon_vertices(center_x, center_y, angle, 1)
        hex_vertices_list.append(vertices)

    # Check containment for each inner hexagon
    for vertices in hex_vertices_list:
        if not check_containment(vertices, outer_center_x, outer_center_y, outer_radius):
            return False

    # Check pairwise non-overlap
    for i in range(len(hex_vertices_list)):
        for j in range(i+1, len(hex_vertices_list)):
            # Minimum distance should be at least 2 (since each hexagon has diameter 2)
            min_dist = hexagon_distance(hex_vertices_list[i], hex_vertices_list[j])
            if min_dist < 2 - 1e-6:  # Allow small numerical tolerance
                return False

    return True


def objective_function(params):
    """Minimize negative of 1/outer_radius (maximize 1/outer_radius)"""
    # params: [x1, y1, angle1, ..., x11, y11, angle11, outer_radius, outer_center_x, outer_center_y]
    n = 11
    inner_params = params[:3*n]
    outer_radius = params[3*n]
    outer_center_x = params[3*n + 1]
    outer_center_y = params[3*n + 2]

    # Reshape inner parameters
    inner_data = inner_params.reshape(n, 3)

    # Check if configuration is valid
    if not is_valid_configuration(inner_data, outer_center_x, outer_center_y, outer_radius):
        # Return large penalty for invalid configurations
        return 1e10

    # Return negative of the inverse outer radius (to maximize 1/outer_radius)
    return -1.0 / outer_radius
