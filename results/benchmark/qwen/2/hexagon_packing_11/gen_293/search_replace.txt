<DESCRIPTION>
The current collision detection relies on Shapely's intersection tests which can be computationally expensive and prone to floating-point precision issues. I'll implement a more robust SAT (Separating Axis Theorem) based collision detection system that provides better performance and numerical stability for hexagon-hexagon intersection tests. This will involve creating a dedicated SAT collision checker that uses geometric properties of hexagons to quickly determine if they intersect.

Additionally, I'll enhance the constraint checking with a tighter geometric validation layer that uses small buffers to prevent false positives from floating-point artifacts during containment checks.

@jit(nopython=True, cache=True)
def distance_point_to_line_segment(px, py, x1, y1, x2, y2):
    """Calculate the distance from point to line segment."""
    # Vector from (x1,y1) to (x2,y2)
    dx = x2 - x1
    dy = y2 - y1

    # Length squared of the segment
    length_sq = dx*dx + dy*dy

    # Handle degenerate case
    if length_sq == 0.0:
        return np.sqrt((px - x1)**2 + (py - y1)**2)

    # Project point onto line
    t = ((px - x1) * dx + (py - y1) * dy) / length_sq

    # Clamp t to [0, 1] to stay within segment
    t = max(0.0, min(1.0, t))

    # Closest point on segment
    closest_x = x1 + t * dx
    closest_y = y1 + t * dy

    # Return distance
    return np.sqrt((px - closest_x)**2 + (py - closest_y)**2)

@jit(nopython=True, cache=True)
def hexagon_overlaps_sat(hex1_vertices, hex2_vertices):
    """
    Fast SAT-based overlap check for hexagons using separating axis theorem.
    Returns True if hexagons overlap, False otherwise.
    """
    # Get axes for SAT (normal vectors of edges)
    axes = np.empty((12, 2), dtype=np.float64)

    # For first hexagon (6 edges)
    for i in range(6):
        x1, y1 = hex1_vertices[i]
        x2, y2 = hex1_vertices[(i + 1) % 6]
        # Edge vector
        edge_x = x2 - x1
        edge_y = y2 - y1
        # Normal vector (perpendicular)
        axes[i, 0] = -edge_y
        axes[i, 1] = edge_x
        # Normalize
        length = np.sqrt(axes[i, 0]**2 + axes[i, 1]**2)
        if length > 1e-12:
            axes[i, 0] /= length
            axes[i, 1] /= length

    # For second hexagon (6 edges)
    for i in range(6):
        x1, y1 = hex2_vertices[i]
        x2, y2 = hex2_vertices[(i + 1) % 6]
        # Edge vector
        edge_x = x2 - x1
        edge_y = y2 - y1
        # Normal vector (perpendicular)
        axes[i + 6, 0] = -edge_y
        axes[i + 6, 1] = edge_x
        # Normalize
        length = np.sqrt(axes[i + 6, 0]**2 + axes[i + 6, 1]**2)
        if length > 1e-12:
            axes[i + 6, 0] /= length
            axes[i + 6, 1] /= length

    # Check each axis
    for i in range(12):
        axis_x = axes[i, 0]
        axis_y = axes[i, 1]

        # Project both hexagons onto this axis
        min1 = max1 = hex1_vertices[0, 0] * axis_x + hex1_vertices[0, 1] * axis_y
        min2 = max2 = hex2_vertices[0, 0] * axis_x + hex2_vertices[0, 1] * axis_y

        for j in range(1, 6):
            proj1 = hex1_vertices[j, 0] * axis_x + hex1_vertices[j, 1] * axis_y
            proj2 = hex2_vertices[j, 0] * axis_x + hex2_vertices[j, 1] * axis_y
            min1 = min(min1, proj1)
            max1 = max(max1, proj1)
            min2 = min(min2, proj2)
            max2 = max(max2, proj2)

        # Check for separation
        if max1 < min2 or max2 < min1:
            return False  # Separating axis found, no overlap

    return True  # Overlapping
