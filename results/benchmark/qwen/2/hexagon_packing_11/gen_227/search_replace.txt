This approach leverages multiple cores to run several DE instances with different initializations and parameters, increasing the chances of finding a better global optimum. The key insight is that the previous implementation only ran one optimization attempt, while parallel execution allows for more thorough exploration of the solution space.

    # Try multiple approaches in parallel using differential evolution
    def run_de_optimization(initial_guess, seed):
        try:

            for _ in range(11):
                bounds.extend([(-10, 10), (-10, 10), (-180, 180)])
            bounds.append((1.0, 20.0))  # Outer hex side length

                compute_objective_with_penalties,
                bounds,
                maxiter=80,
                popsize=20,
                mutation=(0.8, 1),
                recombination=0.9,
                seed=seed,
                disp=False,
                polish=False
            )

            # Refine with L-BFGS-B
            refined_result = minimize(
                compute_objective_with_penalties,
                de_result.x,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 40, 'ftol': 1e-8, 'gtol': 1e-8},
                args=(True,)
            )

            return refined_result.x if refined_result.success else None
        except Exception:
            return None

    from joblib import Parallel, delayed
