def calculate_outer_hex_side_length(inner_hex_data):
    """Calculate minimum outer hexagon side length that contains all inner hexagons using precise geometric calculation"""
    if len(inner_hex_data) == 0:
        return 1000

    # Get all vertices of inner hexagons
    all_vertices = []
    for i in range(len(inner_hex_data)):
        center_x, center_y, angle = inner_hex_data[i]
        vertices = create_hexagon_vertices(center_x, center_y, angle)
        all_vertices.extend(vertices)

    if not all_vertices:
        return 1000

    # Calculate the centroid of all vertices
    centroid_x = sum(v[0] for v in all_vertices) / len(all_vertices)
    centroid_y = sum(v[1] for v in all_vertices) / len(all_vertices)

    # Calculate maximum distance from centroid to any vertex
    max_dist = 0
    for x, y in all_vertices:
        dist = math.sqrt((x - centroid_x)**2 + (y - centroid_y)**2)
        max_dist = max(max_dist, dist)

    # For a regular hexagon circumscribing a circle of radius r:
    # The side length of the hexagon equals the radius of the circumscribed circle
    # However, we need to account for the fact that our hexagons have rotation and
    # may not be optimally aligned with the outer hexagon
    side_length = max_dist * 1.05  # Add margin for safety and proper containment

    # Further refinement: consider that hexagons have specific orientation effects
    # Calculate the bounding circle more carefully by checking extreme directions
    directions = [(1, 0), (0.5, math.sqrt(3)/2), (-0.5, math.sqrt(3)/2),
                  (-1, 0), (-0.5, -math.sqrt(3)/2), (0.5, -math.sqrt(3)/2)]

    max_radius_in_dir = 0
    for dx, dy in directions:
        # Project all vertices onto this direction
        projections = [(x * dx + y * dy) for x, y in all_vertices]
        min_proj, max_proj = min(projections), max(projections)
        # Distance from center along this direction
        radius = max(abs(min_proj), abs(max_proj))
        max_radius_in_dir = max(max_radius_in_dir, radius)

    # Use the more conservative estimate
    side_length = max(side_length, max_radius_in_dir)

    return side_length
