import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist
from shapely.geometry import Polygon
import time


def get_hexagon_vertices(center_x, center_y, angle_deg, side_length=1):
    """Get vertices of a regular hexagon given center, angle, and side length."""
    angle_rad = np.radians(angle_deg)
    # Vertices of a regular hexagon with side length 1, centered at origin
    base_vertices = np.array([
        [1.0, 0.0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1.0, 0.0],
        [-0.5, -np.sqrt(3)/2],
        [0.5, -np.sqrt(3)/2]
    ])

    # Rotate and translate
    cos_a = np.cos(angle_rad)
    sin_a = np.sin(angle_rad)
    rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])

    rotated_vertices = base_vertices @ rotation_matrix.T
    translated_vertices = rotated_vertices + np.array([center_x, center_y])

    return translated_vertices


def check_containment(hexagon_vertices, outer_hex_vertices):
    """Check if hexagon vertices are fully contained within outer hexagon."""
    outer_polygon = Polygon(outer_hex_vertices)
    for vertex in hexagon_vertices:
        if not outer_polygon.contains(Point(vertex[0], vertex[1])):
            return False
    return True


def check_overlap(hex1_vertices, hex2_vertices):
    """Check if two hexagons overlap using Shapely."""
    poly1 = Polygon(hex1_vertices)
    poly2 = Polygon(hex2_vertices)
    return poly1.intersects(poly2)


def calculate_outer_hexagon_radius(inner_hex_data, buffer=0.1):
    """Estimate the minimum radius needed for outer hexagon."""
    # Get all hexagon centers and their max distance from origin
    centers = inner_hex_data[:, :2]
    max_dist = 0

    for i in range(len(inner_hex_data)):
        center = centers[i]
        # Calculate distance from center to hexagon vertices
        hex_vertices = get_hexagon_vertices(center[0], center[1], inner_hex_data[i, 2])
        for vertex in hex_vertices:
            dist = np.linalg.norm(vertex - center)
            # Add some buffer for safety
            total_dist = np.linalg.norm(vertex)
            if total_dist > max_dist:
                max_dist = total_dist

    return max_dist + buffer


def objective_function(params):
    """Objective function to maximize 1/outer_radius."""
    # Unpack parameters: 11 hexagons * 3 parameters each = 33 parameters
    # But we'll use a different approach with constrained optimization
    # params contains [x1, y1, theta1, x2, y2, theta2, ..., x11, y11, theta11, outer_radius]

    num_inner = 11
    # First 33 parameters are the hexagon positions and rotations
    inner_params = params[:3*num_inner]
    outer_radius = params[-1]  # Last parameter is outer radius

    # Reshape inner parameters
    inner_positions_angles = inner_params.reshape((num_inner, 3))

    # Calculate outer hexagon vertices
    outer_vertices = get_hexagon_vertices(0, 0, 0, outer_radius)

    # Check all constraints
    for i in range(num_inner):
        center_x, center_y, angle = inner_positions_angles[i]
        hex_vertices = get_hexagon_vertices(center_x, center_y, angle)

        # Check containment
        if not check_containment(hex_vertices, outer_vertices):
            return -1e10  # Invalid - return very negative value

        # Check overlap with other hexagons
        for j in range(i + 1, num_inner):
            other_center_x, other_center_y, other_angle = inner_positions_angles[j]
            other_hex_vertices = get_hexagon_vertices(other_center_x, other_center_y, other_angle)

            if check_overlap(hex_vertices, other_hex_vertices):
                return -1e10  # Invalid - return very negative value

    # If we got here, everything is valid - return the inverse of the radius to maximize
    # We want to minimize outer_radius, so we maximize 1/outer_radius
    return 1.0 / outer_radius


def evaluate_solution(inner_hex_data, outer_hex_side_length):
    """Evaluate the solution for validation and scoring."""
    # Calculate all inner hexagon vertices
    inner_vertices = []
    for i in range(len(inner_hex_data)):
        center_x, center_y, angle = inner_hex_data[i]
        vertices = get_hexagon_vertices(center_x, center_y, angle)
        inner_vertices.append(vertices)

    # Calculate outer hexagon vertices
    outer_hex_vertices = get_hexagon_vertices(0, 0, 0, outer_hex_side_length)

    # Check containment
    for vertices in inner_vertices:
        if not check_containment(vertices, outer_hex_vertices):
            return 0.0  # Invalid

    # Check overlap
    for i in range(len(inner_vertices)):
        for j in range(i + 1, len(inner_vertices)):
            if check_overlap(inner_vertices[i], inner_vertices[j]):
                return 0.0  # Invalid

    return 1.0 / outer_hex_side_length  # Valid solution
