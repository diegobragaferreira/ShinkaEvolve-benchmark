def check_containment_single(hex_vertices, outer_polygon):
    """Check if all vertices of a hexagon are inside the outer hexagon."""
    # Add a small buffer to ensure containment with floating-point precision
    BUFFER = 1e-6

    # Check if all vertices are within the outer hexagon with buffer
    for vertex in hex_vertices:
        point = Point(vertex)
        # Use buffer to prevent numerical precision issues
        if not outer_polygon.contains(point):
            # Double-check with a slightly larger buffer for safety
            buffered_polygon = outer_polygon.buffer(BUFFER)
            if not buffered_polygon.contains(point):
