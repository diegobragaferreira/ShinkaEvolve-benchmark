def hexagon_vertices(center_x, center_y, rotation_degrees, side_length=1):
    """Generate vertices of a regular hexagon given center, rotation, and side length."""
    angle_rad = np.radians(rotation_degrees)
    # Vertices of a unit hexagon centered at origin
    unit_vertices = np.array([
        [1, 0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1, 0],
        [-0.5, -np.sqrt(3)/2],
        [0.5, -np.sqrt(3)/2]
    ])
    
    # Rotate and translate
    cos_a, sin_a = np.cos(angle_rad), np.sin(angle_rad)
    rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
    rotated_vertices = unit_vertices @ rotation_matrix.T
    return rotated_vertices * side_length + np.array([center_x, center_y])

def get_hexagon_bounding_box(vertices):
    """Get bounding box of hexagon vertices."""
    vertices_array = np.array(vertices)
    min_x, max_x = vertices_array[:, 0].min(), vertices_array[:, 0].max()
    min_y, max_y = vertices_array[:, 1].min(), vertices_array[:, 1].max()
    return min_x, max_x, min_y, max_y

def check_containment_single(hex_vertices, outer_polygon):
    """Check if all vertices of a hexagon are inside the outer hexagon (optimized)."""
    # Add small buffer to handle floating point precision issues
    buffer_size = 1e-6
    buffered_outer = outer_polygon.buffer(buffer_size)
    
    for vertex in hex_vertices:
        point = Point(vertex)
        if not buffered_outer.contains(point):
            return False
    return True

def check_collision_single_fast(hex1_vertices, hex2_vertices):
    """Fast collision check using approximate bounds first."""
    # Quick bounding box check
    min1_x, max1_x, min1_y, max1_y = get_hexagon_bounding_box(hex1_vertices)
    min2_x, max2_x, min2_y, max2_y = get_hexagon_bounding_box(hex2_vertices)
    
    # If bounding boxes don't intersect, no collision
    if max1_x < min2_x or max2_x < min1_x or max1_y < min2_y or max2_y < min1_y:
        return False
    
    # Fall back to precise Shapely check with buffer for numerical stability
    buffer_size = 1e-6
    poly1 = Polygon(hex1_vertices).buffer(buffer_size)
    poly2 = Polygon(hex2_vertices).buffer(buffer_size)
    return poly1.intersects(poly2)

def calculate_outer_hex_side_length(inner_hex_params):
    """Calculate the minimal outer hexagon side length needed to contain all inner hexagons."""
    # Get all vertices of all inner hexagons
    all_vertices = []
    for i in range(11):
        x, y, rot = inner_hex_params[3*i], inner_hex_params[3*i+1], inner_hex_params[3*i+2]
        hex_vertices = hexagon_vertices(x, y, rot, 1)
        all_vertices.extend(hex_vertices)
    
    if len(all_vertices) == 0:
        return 100.0
        
    # Calculate bounding box
    all_vertices = np.array(all_vertices)
    min_x, max_x = all_vertices[:, 0].min(), all_vertices[:, 0].max()
    min_y, max_y = all_vertices[:, 1].min(), all_vertices[:, 1].max()
    
    # Calculate diagonal distance from center to corner
    center_x = (min_x + max_x) / 2
    center_y = (min_y + max_y) / 2
    
    # Find maximum distance from center to any vertex
    max_dist = 0
    for vertex in all_vertices:
        dist = np.sqrt((vertex[0] - center_x)**2 + (vertex[1] - center_y)**2)
        max_dist = max(max_dist, dist)
    
    # For a hexagon, the side length is max_dist / sqrt(3)
    # But we want to ensure our hexagon can contain everything with some margin
    return max_dist * 2 / np.sqrt(3) * 1.05  # Tighter margin for better packing

def objective_function(params):
    """Optimized objective function to minimize (negative inverse of outer hex side length)."""
    # params: [x1, y1, rot1, x2, y2, rot2, ..., x11, y11, rot11, R]
    n = 11
    outer_side_length = params[-1]
    
    # Check if outer hexagon is large enough
    if outer_side_length <= 0:
        return 1e10  # Penalty for invalid outer hex
    
    # Create outer hexagon vertices once
    outer_vertices = hexagon_vertices(0, 0, 0, outer_side_length)
    outer_polygon = prep(Polygon(outer_vertices))  # Prepare for fast intersection tests
    
    # Check containment and collisions in parallel
    total_penalty = 0
    
    # Check containment of all inner hexagons (vectorized check)
    inner_positions = params[:-1].reshape(-1, 3)
    inner_hex_vertices = []
    
    for i in range(n):
        x, y, rot = inner_positions[i]
        hex_vertices = hexagon_vertices(x, y, rot, 1)
        inner_hex_vertices.append(hex_vertices)
        
        # Check containment for this hexagon
        if not check_containment_single(hex_vertices, outer_polygon):
            total_penalty += 1e9  # Large penalty for containment failure
    
    # If containment failed, early exit
    if total_penalty > 0:
        return -(1.0 / outer_side_length) + total_penalty
    
    # Check collisions between all pairs of inner hexagons using optimized approach
    if total_penalty == 0:
        # Build KD-tree for neighbors to speed up collision checks
        centers = []
        for i in range(n):
            x, y, rot = inner_positions[i]
            centers.append([x, y])
        centers = np.array(centers)
        
        tree = KDTree(centers)
        
        # Check collisions with neighbors only (more efficient)
        collision_pairs = []
        for i in range(n):
            # Find nearby points within some radius (tighter than previous)
            nearby_indices = tree.query_ball_point(centers[i], 3.0)  # Adjust radius as needed
            for j in nearby_indices:
                if i < j:  # Avoid duplicate pairs and self-comparison
                    collision_pairs.append((i, j))
        
        # Perform collision checks only for likely candidates
        collision_penalties = 0
        for i, j in collision_pairs:
            hex1_vertices = inner_hex_vertices[i]
            hex2_vertices = inner_hex_vertices[j]
            
            # Use fast collision check for initial screening
            if check_collision_single_fast(hex1_vertices, hex2_vertices):
                collision_penalties += 1e9  # Large penalty for collisions
        
        total_penalty += collision_penalties
    
    # Return negative inverse of outer hex side length plus penalties
    return -(1.0 / outer_side_length) + total_penalty

def adaptive_differential_evolution(func, bounds, maxiter=100, popsize=15, mutation=(0.5, 1), recombination=0.7, seed=None, disp=False, polish=True):
    """
    Enhanced differential evolution with adaptive parameters
    """
    # Adaptive mutation rate that decreases over time
    def adaptive_mutation_rate(iteration, max_iter):
        # Start with higher mutation for exploration, decrease to 0.1 for exploitation
        return max(0.1, mutation[0] + (mutation[1] - mutation[0]) * (1 - iteration/max_iter))
    
    # Store best solution found so far
    best_solution = None
    best_value = float('inf')
    
    # Run multiple trials with different seeds for robustness
    for trial in range(5):  # Multiple trials
        np.random.seed(seed + trial if seed is not None else trial)
        
        # Create a copy of bounds to modify for each trial
        bounds_copy = list(bounds)
        
        # Run DE with adaptive parameters
        result = differential_evolution(
            func,
            bounds_copy,
            maxiter=maxiter,
            popsize=popsize,
            mutation=(adaptive_mutation_rate(0, maxiter), mutation[1]),
            recombination=recombination,
            seed=seed + trial if seed is not None else trial,
            disp=disp,
            polish=polish
        )
        
        # Update best solution if this trial was better
        if result.fun < best_value:
            best_value = result.fun
            best_solution = result.x
    
    return type('Result', (), {'x': best_solution, 'fun': best_value})()
