def is_valid_placement(circles, x, y, r, tree=None):
    """Check if placing a circle at (x,y) with radius r is valid."""
    # Check boundary constraints
    if r > x or r > y or r > (1-x) or r > (1-y):
        return False

    # If we have a KDTree, use it for efficient overlap detection
    if tree is not None:
        # Find potential overlapping circles within distance 2*(r+max_radius)
        # This is a rough upper bound to avoid unnecessary distance calculations
        # We'll check the actual distance for the candidates
        candidates = tree.query_ball_point([x, y], 2*r)
        for i in candidates:
            cx, cy, cr = circles[i]
            distance = np.sqrt((x - cx)**2 + (y - cy)**2)
            if distance < (r + cr):
                return False
    else:
        # Fallback to original method if no tree provided
        for i in range(len(circles)):
            cx, cy, cr = circles[i]
            distance = np.sqrt((x - cx)**2 + (y - cy)**2)
            if distance < (r + cr):
                return False

    return True

def compute_local_density(circles, point, k=5):
    """Compute local density around a point using k nearest neighbors."""
    if len(circles) < 2:
        return 0

    # Convert circles to array for KDTree query
    circle_points = np.array([[c[0], c[1]] for c in circles])
    tree = cKDTree(circle_points)

    # Query k nearest neighbors
    dists, _ = tree.query(point, k=min(k, len(circle_points)))

    # Return average distance to nearest neighbors
    return np.mean(dists[dists > 0]) if np.any(dists > 0) else 1.0

def initialize_circles_hexagonal(n=32, grid_size=None):
    """Initialize circle positions using a hexagonal grid pattern."""
    if grid_size is None:
        grid_size = int(np.ceil(np.sqrt(n)))

    circles = []

    # Create hexagonal grid points
    spacing_x = 1.0 / (grid_size + 1)
    spacing_y = spacing_x * np.sqrt(3) / 2

    # Generate hexagonal grid positions with staggered rows
    for i in range(grid_size):
        for j in range(grid_size):
            if len(circles) >= n:
                break
            # Offset every other row for hexagonal pattern
            x_offset = spacing_x / 2 * (i % 2)
            x = (j + 0.5 + x_offset) * spacing_x
            y = (i + 0.5) * spacing_y

            # Clamp to valid range
            x = np.clip(x, 0.01, 0.99)
            y = np.clip(y, 0.01, 0.99)

            # Initial radius based on boundary distance
            r_min = min(x, y, 1-x, 1-y)
            r = min(r_min * 0.3, 0.15)

            # Only add if valid
            if is_valid_placement(circles, x, y, r):
                circles.append([x, y, r])

    # Pad to exact number required
    while len(circles) < n:
        x = np.random.uniform(0.05, 0.95)
        y = np.random.uniform(0.05, 0.95)
        r = min(x, y, 1-x, 1-y) * 0.2
        if is_valid_placement(circles, x, y, r):
            circles.append([x, y, r])

    return np.array(circles[:n])

def evaluate_fitness(circles):
    """Evaluate fitness of circle configuration with proper penalties."""
    n = len(circles)
    total_radius = np.sum(circles[:, 2])

    # Penalty for constraint violations
    penalty = 0

    # Check containment constraints
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            penalty += 1000  # Large penalty for containment violation

    # Check overlap constraints efficiently using vectorized operations
    positions = circles[:, :2]
    radii = circles[:, 2]

    # Compute pairwise distances
    distances = np.sqrt(((positions[:, None] - positions[None, :])**2).sum(axis=2))

    # Create mask for pairs that would overlap (excluding diagonal)
    overlap_mask = distances < (radii[:, None] + radii[None, :])
    np.fill_diagonal(overlap_mask, False)

    # Apply penalty for overlaps
    overlap_penalty = np.sum((radii[:, None] + radii[None, :] - distances)[overlap_mask])
    penalty += 1000 * overlap_penalty

    # Return negative fitness (since we want to maximize total radius)
    return -(total_radius - penalty)

def optimize_circles(circles, max_iter=1000):
    """Optimize circle positions using scipy optimization."""
    def objective(params):
        # Reshape params back to circles array
        circles_flat = params.reshape(-1, 3)
        return evaluate_fitness(circles_flat)  # Negative because we want to maximize

    # Flatten the circles array for optimization
    initial_params = circles.flatten()

    # Define bounds for optimization (radius: 0.001-0.5, positions: 0.001-0.999)
    bounds = []
    for i in range(len(initial_params)):
        if i % 3 == 2:  # radius parameter
            bounds.append((0.001, 0.5))  # Radius between 0.001 and 0.5
        else:  # x and y parameters
            bounds.append((0.001, 0.999))  # Position within the unit square

    try:
        # Use L-BFGS-B which handles bounds well
        result = minimize(objective, initial_params, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': max_iter})

        if result.success:
            optimized_circles = result.x.reshape(-1, 3)
            return optimized_circles
    except Exception as e:
        print(f"Optimization error: {e}")

    # Return original if optimization failed
    return circles

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 32

    # Multi-start optimization: try several different initializations
    best_circles = None
    best_sum_radii = -np.inf

    # Try different seed values for diversity
    seeds = [42, 123, 456, 789, 999, 1000, 2023, 2024]

    for seed in seeds:
        np.random.seed(seed)

        # Try different initialization approaches
        # 1. Hexagonal grid with perturbations
        circles_1 = initialize_circles_hexagonal(n)

        # 2. Heuristic initialization
        circles_2 = initialize_circles_heuristic(n)

        # Try optimizing from both initializations
        for circles_init in [circles_1, circles_2]:
            # Add slight randomization to avoid symmetric solutions
            circles_init += np.random.normal(0, 0.001, circles_init.shape)

            # Optimize the configuration
            optimized_circles = optimize_circles(circles_init)

            # Evaluate the result
            sum_radii = np.sum(optimized_circles[:, 2])

            if sum_radii > best_sum_radii:
                best_sum_radii = sum_radii
                best_circles = optimized_circles.copy()

    # If we couldn't find anything better, return the heuristic version
    if best_circles is None:
        best_circles = initialize_circles_heuristic(n)

    return best_circles
