import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

# You can define functions outside the main function below.
# Remember that any function used in parallel computation must be defined globally and not locally.

def hexagonal_grid_initialization(n_circles):
    """Initialize circles in a hexagonal grid pattern"""
    # Determine grid dimensions
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))

    # Create hexagonal grid
    positions = []
    for i in range(rows):
        for j in range(cols):
            if len(positions) >= n_circles:
                break
            # Offset odd rows
            x_offset = j + 0.5 * (i % 2)
            y_offset = i * np.sqrt(3) / 2
            positions.append([x_offset, y_offset])

    # Normalize to fit in unit square
    if positions:
        positions = np.array(positions)
        # Scale down to fit within square
        max_x = np.max(positions[:, 0])
        max_y = np.max(positions[:, 1])
        scale_factor = min(1.0/max_x, 1.0/max_y) if max_x > 0 and max_y > 0 else 1.0
        positions *= scale_factor

        # Center the grid
        center_offset = 0.5 - np.mean(positions, axis=0)
        positions += center_offset

        # Ensure all positions are valid (within unit square)
        positions[:, 0] = np.clip(positions[:, 0], 0, 1)
        positions[:, 1] = np.clip(positions[:, 1], 0, 1)

    return positions[:n_circles]

def evaluate_constraints(circles):
    """Evaluate constraint violations"""
    n = len(circles)
    if n == 0:
        return 0

    # Check containment constraints
    containment_violations = 0
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            containment_violations += 1

    # Check overlap constraints
    overlap_violations = 0
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            distance = np.sqrt((x1-x2)**2 + (y1-y2)**2)
            if distance < (r1 + r2):
                overlap_violations += 1

    return containment_violations + overlap_violations

def objective_function(circles_flat):
    """Objective function to maximize sum of radii"""
    n = len(circles_flat) // 3
    circles = circles_flat.reshape(-1, 3)
    # Return negative because we want to maximize
    return -np.sum(circles[:, 2])

def constraint_containment(circles_flat):
    """Constraint: circles must be fully contained in unit square"""
    n = len(circles_flat) // 3
    circles = circles_flat.reshape(-1, 3)
    result = []
    for i in range(n):
        x, y, r = circles[i]
        # These should be >= 0 for feasibility (containment)
        result.extend([
            x - r,              # x - r >= 0
            1 - x - r,          # 1 - x - r >= 0
            y - r,              # y - r >= 0
            1 - y - r           # 1 - y - r >= 0
        ])
    return np.array(result)

def constraint_overlap(circles_flat):
    """Constraint: circles must not overlap"""
    n = len(circles_flat) // 3
    circles = circles_flat.reshape(-1, 3)
    result = []
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            distance = np.sqrt((x1-x2)**2 + (y1-y2)**2)
            # Distance >= r1 + r2 (we're using -d + r1 + r2 <= 0)
            result.append(-(distance - r1 - r2))
    return np.array(result)

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 32

    # Initialize using hexagonal grid
    initial_positions = hexagonal_grid_initialization(n)

    # Initialize with equal small radii
    initial_radii = np.full(n, 0.05)
    circles_flat = np.column_stack([initial_positions, initial_radii]).flatten()

    # Set bounds for optimization (x, y, r for each circle)
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0.001, 0.5)])  # x, y, r bounds

    # Define constraints for optimization
    cons = [
        {'type': 'ineq', 'fun': lambda x: constraint_containment(x)},
        {'type': 'ineq', 'fun': lambda x: constraint_overlap(x)}
    ]

    # Optimize using SLSQP method
    try:
        result = minimize(
            objective_function,
            circles_flat,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-6}
        )

        if result.success:
            optimized_circles = result.x.reshape(-1, 3)
            return optimized_circles
    except Exception:
        pass

    # If optimization fails, return initial configuration
    return np.column_stack([initial_positions, initial_radii])
