This approach improves upon the previous version by being more systematic in its density assessment and better distributing circles across the space.

The key insight is that we want circles to be larger in sparser regions and smaller in denser regions, but we also want to ensure good distribution patterns. The enhanced approach uses a multi-step process that creates a better starting configuration for optimization.

def initialize_circles_heuristic(n=32):
    """Initialize circle positions using a density-adaptive heuristic approach."""
    circles = []

    # Step 1: Create a hexagonal grid pattern for initial distribution
    # Calculate grid size to fit approximately n circles
    grid_size = int(np.ceil(np.sqrt(n * 1.5)))  # Slightly more to account for irregularities
    spacing = 1.0 / (grid_size + 1)

    # Fill the grid with circles
    for i in range(grid_size):
        for j in range(grid_size):
            if len(circles) >= n:
                break
            # Create a hexagonal offset pattern
            x = (i + 1) * spacing
            y = (j + 1) * spacing
            if i % 2 == 1:  # Offset every other row
                x += spacing * 0.5

            # Initial radius estimate based on available space
            r_min = min(x, y, 1-x, 1-y)

            # Very conservative initial radius to ensure validity
            r = min(r_min * 0.4, 0.15)

            # Only add if valid
            if is_valid_placement(circles, x, y, r):
                circles.append([x, y, r])

                # Sample from corners and center areas
                corner_probs = [0.25, 0.25, 0.25, 0.25]
                corner_idx = np.random.choice(4, p=corner_probs)
                if corner_idx == 0:
                    x = np.random.uniform(0.05, 0.3)
                    y = np.random.uniform(0.05, 0.3)

                    x = np.random.uniform(0.7, 0.95)
                    y = np.random.uniform(0.05, 0.3)
                elif corner_idx == 2:
                    x = np.random.uniform(0.05, 0.3)
                    y = np.random.uniform(0.7, 0.95)

                    x = np.random.uniform(0.7, 0.95)
                    y = np.random.uniform(0.7, 0.95)

            # Estimate max radius at this location
            r_max = min(x, y, 1-x, 1-y)
            if r_max <= 0:

            # For better density estimation, sample a few nearby points
            sample_density = 0
            sample_count = 10
            for _ in range(sample_count):

                point = np.array([px, py])
                density = compute_local_density(circles, point)
                sample_density += density
            avg_density = sample_density / sample_count if sample_count > 0 else 0

            # Adjust radius based on how dense the area is
            # In sparse areas, allow larger radii; in dense areas, smaller
            r_factor = max(0.1, 1.0 / (1.0 + avg_density * 10))

            test_radii = np.linspace(0.005, adjusted_r_max * 0.7, 15)

                        best_x, best_y = x, y
                        break

            circles.append([best_x, best_y, best_r])
        else:
            # If we couldn't place another circle, try again with slightly smaller target
            if len(circles) > 0:
                # Remove last circle and retry with reduced radius
                circles = circles[:-1]

    # Use a simple greedy approach to remove overlaps

    if len(refined_circles) < n * 0.8:  # Keep at least 80%
