import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

# You can define functions outside the main function below.
# Remember that any function used in parallel computation must be defined globally and not locally.

def initialize_hexagonal_grid(n_circles, square_size=1.0):
    """Initialize circles in a hexagonal grid pattern."""
    # Calculate grid parameters
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))

    # Start with a reasonable spacing
    spacing = square_size / max(rows, cols)
    radius = spacing / 2.0

    # Generate hexagonal grid positions
    positions = []
    for i in range(rows):
        for j in range(cols):
            if len(positions) >= n_circles:
                break
            # Offset odd rows
            x = (j + 0.5 * (i % 2)) * spacing
            y = i * spacing * np.sqrt(3) / 2
            positions.append([x, y])

    # Trim to exact number of circles needed
    positions = positions[:n_circles]

    # Adjust positions to fit in unit square
    if positions:
        min_x = min(pos[0] for pos in positions)
        max_x = max(pos[0] for pos in positions)
        min_y = min(pos[1] for pos in positions)
        max_y = max(pos[1] for pos in positions)

        scale_x = square_size / (max_x - min_x) if max_x > min_x else 1
        scale_y = square_size / (max_y - min_y) if max_y > min_y else 1

        scale = min(scale_x, scale_y)

        # Center the grid in the unit square
        center_x = (min_x + max_x) / 2
        center_y = (min_y + max_y) / 2

        adjusted_positions = []
        for x, y in positions:
            new_x = (x - center_x) * scale + square_size / 2
            new_y = (y - center_y) * scale + square_size / 2
            adjusted_positions.append([new_x, new_y])

        return adjusted_positions, radius

    return [[square_size/2, square_size/2]], radius

def check_containment(circle, square_size=1.0):
    """Check if circle is fully contained in the square."""
    x, y, r = circle
    return (r <= x <= square_size - r and
            r <= y <= square_size - r)

def check_overlap(circle1, circle2):
    """Check if two circles overlap."""
    x1, y1, r1 = circle1
    x2, y2, r2 = circle2
    distance = np.sqrt((x2 - x1)**2 + (y2 - y1)**2)
    return distance < (r1 + r2)

def evaluate_constraints(circles, square_size=1.0):
    """Evaluate constraint violations."""
    penalty = 0

    # Check containment
    for circle in circles:
        if not check_containment(circle, square_size):
            penalty += 1000  # Large penalty for containment violation

    # Check overlaps
    n = len(circles)
    for i in range(n):
        for j in range(i+1, n):
            if check_overlap(circles[i], circles[j]):
                penalty += 1000  # Large penalty for overlap violation

    return penalty

def objective_function(circles_flat):
    """Objective function to maximize sum of radii."""
    # Reshape flat array back to circles
    n = len(circles_flat) // 3
    circles = circles_flat.reshape(n, 3)

    # Sum of radii (we negate because we want to maximize)
    sum_radii = -np.sum(circles[:, 2])

    # Add penalty for constraint violations
    penalty = evaluate_constraints(circles)

    return sum_radii + penalty

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 32
    square_size = 1.0

    # Initialize with hexagonal grid
    positions, initial_radius = initialize_hexagonal_grid(n, square_size)

    # Create initial circles array
    circles = np.zeros((n, 3))
    for i, (x, y) in enumerate(positions):
        circles[i] = [x, y, initial_radius]

    # Flatten for optimization
    circles_flat = circles.flatten()

    # Define bounds (x, y, r) for each circle
    bounds = []
    for i in range(n):
        # x and y bounds: [radius, 1-radius]
        bounds.extend([(initial_radius, 1-initial_radius),
                      (initial_radius, 1-initial_radius),
                      (initial_radius, 1-initial_radius)])

    # Optimization settings
    options = {'maxiter': 1000}

    try:
        # Run optimization
        result = minimize(objective_function, circles_flat, method='L-BFGS-B',
                         bounds=bounds, options=options, tol=1e-6)

        # Extract optimized circles
        optimized_circles = result.x.reshape(n, 3)

        # Ensure final containment
        for i in range(n):
            x, y, r = optimized_circles[i]
            # Clamp to valid range
            x = np.clip(x, r, 1-r)
            y = np.clip(y, r, 1-r)
            optimized_circles[i] = [x, y, r]

        return optimized_circles

    except Exception as e:
        print(f"Optimization failed: {e}")
        # Return initial hexagonal grid if optimization fails
        return circles
