def is_valid_placement(circles, x, y, r, tree=None):
    """Check if placing a circle at (x,y) with radius r is valid."""
    # Check boundary constraints
    if r > x or r > y or r > (1-x) or r > (1-y):
        return False

    # If we have a KDTree, use it for efficient overlap detection
    if tree is not None:
        # Find potential overlapping circles within distance 2*(r+max_radius)
        # This is a rough upper bound to avoid unnecessary distance calculations
        # We'll check the actual distance for the candidates
        candidates = tree.query_ball_point([x, y], 2*r)
        for i in candidates:
            cx, cy, cr = circles[i]
            distance = np.sqrt((x - cx)**2 + (y - cy)**2)
