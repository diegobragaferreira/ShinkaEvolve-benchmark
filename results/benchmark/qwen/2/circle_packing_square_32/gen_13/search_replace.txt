import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

# You can define functions outside the main function below.

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 32

    # Create initial hexagonal grid layout
    def create_hexagonal_layout(n_circles):
        # Determine grid dimensions
        rows = int(np.ceil(np.sqrt(n_circles)))
        cols = int(np.ceil(n_circles / rows))

        # Calculate spacing
        spacing_x = 1.0 / cols
        spacing_y = 1.0 / rows

        # Create grid points
        y_coords = np.linspace(spacing_y/2, 1-spacing_y/2, rows)
        x_coords = np.linspace(spacing_x/2, 1-spacing_x/2, cols)

        # Generate positions
        positions = []
        for i, y in enumerate(y_coords):
            for j, x in enumerate(x_coords):
                # Offset every other row
                offset = spacing_x/2 if i % 2 == 1 else 0
                x_pos = x + offset
                if len(positions) < n_circles:
                    positions.append([x_pos, y])
                else:
                    break
            if len(positions) >= n_circles:
                break

        return np.array(positions[:n_circles])

    # Initialize positions
    positions = create_hexagonal_layout(n)

    # Initialize radii (start with small values)
    initial_radii = np.full(n, 0.02)

    # Combine positions and radii for optimization
    initial_params = np.concatenate([positions.flatten(), initial_radii])

    def objective(params):
        # Extract positions and radii
        pos = params[:2*n].reshape(-1, 2)
        radii = params[2*n:]

        # Maximize sum of radii (minimize negative sum)
        return -np.sum(radii)

    def constraints_func(params):
        pos = params[:2*n].reshape(-1, 2)
        radii = params[2*n:]

        # Check containment constraints
        containment_violations = []
        for i in range(n):
            x, y = pos[i]
            r = radii[i]
            # Check if circle is fully inside unit square
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                containment_violations.append(1000.0)  # Large penalty for containment violation
            else:
                containment_violations.append(0.0)

        # Check overlap constraints
        overlap_violations = []
        for i in range(n):
            for j in range(i+1, n):
                x1, y1 = pos[i]
                x2, y2 = pos[j]
                r1 = radii[i]
                r2 = radii[j]

                # Distance between centers
                dist = np.sqrt((x1-x2)**2 + (y1-y2)**2)

                # Check if circles overlap
                if dist < r1 + r2:
                    overlap_violations.append(1000.0)  # Large penalty for overlap violation
                else:
                    overlap_violations.append(0.0)

        # Return combined violations
        return np.array(containment_violations + overlap_violations)

    def constraint_func(params):
        pos = params[:2*n].reshape(-1, 2)
        radii = params[2*n:]

        # Check containment constraints
        for i in range(n):
            x, y = pos[i]
            r = radii[i]
            # Check if circle is fully inside unit square (penalty approach)
            if x - r < 0:
                return 1000.0 + (0 - (x - r))**2
            if x + r > 1:
                return 1000.0 + ((x + r) - 1)**2
            if y - r < 0:
                return 1000.0 + (0 - (y - r))**2
            if y + r > 1:
                return 1000.0 + ((y + r) - 1)**2

        # Check overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                x1, y1 = pos[i]
                x2, y2 = pos[j]
                r1 = radii[i]
                r2 = radii[j]

                # Distance between centers
                dist = np.sqrt((x1-x2)**2 + (y1-y2)**2)

                # If circles overlap, apply penalty
                if dist < r1 + r2:
                    overlap_penalty = 1000.0 + (r1 + r2 - dist)**2
                    return overlap_penalty

        return 0.0  # No violations

    # Define bounds for parameters (x, y, r)
    bounds = []
    for i in range(n):  # Positions
        bounds.extend([(0, 1), (0, 1)])  # x, y bounds
    for i in range(n):  # Radii
        bounds.append((0.001, 0.5))  # Radius bounds

    # Use scipy.optimize.minimize with SLSQP method
    try:
        result = minimize(objective, initial_params, method='SLSQP',
                         bounds=bounds, options={'maxiter': 1000})

        # Extract optimized results
        final_positions = result.x[:2*n].reshape(-1, 2)
        final_radii = result.x[2*n:]

        # Create final circles array
        circles = np.column_stack([final_positions, final_radii])

        # Verify the solution quality
        sum_radii = np.sum(final_radii)
        print(f"Sum of radii: {sum_radii}")

    except Exception as e:
        # Fallback to simple initialization if optimization fails
        print(f"Optimization failed: {e}")
        # Create simple configuration with uniform small radii
        circles = np.zeros((n, 3))
        for i in range(n):
            circles[i, 0] = 0.1 + 0.8 * (i % 10) / 9.0
            circles[i, 1] = 0.1 + 0.8 * (i // 10) / 2.0
            circles[i, 2] = 0.02
            if i >= 30:  # Set some larger radii for last few circles
                circles[i, 2] = 0.05

    return circles
