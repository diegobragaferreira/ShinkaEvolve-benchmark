def _initialize_hexagonal_grid(n):
    """Initialize circles using a hexagonal grid pattern with proper spacing"""
    # Calculate grid dimensions for n circles
    rows = int(math.sqrt(n))
    cols = int(math.ceil(n / rows))

    # Adjust to ensure we have enough slots
    while rows * cols < n:
        rows += 1

    # Create hexagonal grid points
    spacing_x = 1.0 / cols
    spacing_y = 1.0 / rows

    # Adjust spacing to create proper hexagonal packing
    # In a hexagonal lattice, horizontal spacing should be 2*r and vertical spacing should be sqrt(3)*r
    # So we'll use spacing to determine appropriate radius
    hex_spacing = min(spacing_x, spacing_y)
    target_radius = hex_spacing * 0.45  # Leave some margin for optimization

    points = []
    for i in range(rows):
        for j in range(cols):
            if len(points) >= n:
                break
            x = (j + (i % 2) * 0.5) * hex_spacing
            y = i * hex_spacing * math.sqrt(3) / 2
            points.append([x, y])

    # Ensure we have exactly n points
    points = points[:n]

    # Initialize radii with proper values based on spacing
    radii = [target_radius] * n

    # Add small random perturbations to avoid getting stuck in symmetric local optima
    np.random.seed(42)  # For reproducibility
    for i in range(n):
        # Add small random displacement to positions (within 1% of spacing)
        random_factor = 0.01
        points[i][0] += (np.random.rand() - 0.5) * random_factor * hex_spacing
        points[i][1] += (np.random.rand() - 0.5) * random_factor * hex_spacing

        # Also add small random variation to radii (±5%)
        radii[i] *= (0.95 + np.random.rand() * 0.1)

    return np.array(points), radii
