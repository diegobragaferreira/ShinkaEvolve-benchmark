def compute_local_density(circles, point, k=5):
    """Compute local density around a point using k-nearest neighbors."""
    if len(circles) < 2:
        return 0.0

    # Convert to numpy array for efficient processing
    pts = np.array(circles)[:, :2]
    tree = cKDTree(pts)

    # Query k nearest neighbors (excluding the point itself if it exists)
    distances, indices = tree.query(point, k=min(k+1, len(pts)), p=2)

    # Average distance to neighbors (excluding self if present)
    if len(distances) > 1:
        avg_distance = np.mean(distances[1:])  # Skip the first (distance to itself)
    else:
        avg_distance = distances[0]

    # Density is inversely proportional to average distance
    if avg_distance > 0:
        return 1.0 / avg_distance
    else:
        return float('inf')

def calculate_constraints(circles):
    """Calculate constraint violations for all circles."""
    violations = []
    n = len(circles)

    # Boundary violations
    for i in range(n):
        x, y, r = circles[i]
        boundary_violation = 0

        # Check all boundaries
        if r > x:
            boundary_violation += (r - x)
        if r > y:
            boundary_violation += (r - y)
        if r > (1 - x):
            boundary_violation += (r - (1 - x))
        if r > (1 - y):
            boundary_violation += (r - (1 - y))

        violations.append(boundary_violation)

    # Overlap violations
    tree = cKDTree(circles[:, :2])
    for i in range(n):
        x, y, r = circles[i]

        # Find nearby circles using cKDTree for efficiency
        neighbors = tree.query_ball_point([x, y], 2 * r)
        for j in neighbors:
            if i != j:
                x2, y2, r2 = circles[j]
                distance = np.sqrt((x - x2)**2 + (y - y2)**2)
                overlap_violation = max(0, (r + r2) - distance)

    return violations

def calculate_constraints(circles):
    """Calculate constraint violations for all circles."""
    n = len(circles)

    # Calculate violations for each circle
    boundary_violations = np.zeros(n)
    overlap_violations = np.zeros(n)

    # Boundary violations
    for i in range(n):
        x, y, r = circles[i]
        # Calculate how much each boundary is violated
        boundary_violations[i] = (max(0, r - x) +
                                max(0, r - y) +
                                max(0, r - (1 - x)) +
                                max(0, r - (1 - y)))

    # Overlap violations using cKDTree for efficiency
    tree = cKDTree(circles[:, :2])
    for i in range(n):
        x, y, r = circles[i]

        # Find nearby circles within a reasonable search radius
        search_radius = 2 * max(0.1, r)  # At least 0.1 for small circles
        neighbors = tree.query_ball_point([x, y], search_radius)

        for j in neighbors:
            if i != j:
                x2, y2, r2 = circles[j]
                distance = np.sqrt((x - x2)**2 + (y - y2)**2)
                overlap_violation = max(0, (r + r2) - distance)
                if overlap_violation > 0:
                    overlap_violations[i] += overlap_violation

    return boundary_violations, overlap_violations

def refine_configuration(circles, max_iterations=50, tolerance=1e-6):
    """Refine the circle configuration using iterative optimization with better convergence criteria."""
    n = len(circles)

    # Track improvement for early stopping
    previous_fitness = evaluate_fitness(circles)
    no_improvement_count = 0

    for iteration in range(max_iterations):
        # Store old configuration
        old_circles = circles.copy()

        # Optimize each circle individually
        for i in range(n):
            new_radius = optimize_individual_circle(i, circles)
            circles[i, 2] = new_radius

        # Check for convergence based on fitness improvement
        current_fitness = evaluate_fitness(circles)
        improvement = current_fitness - previous_fitness

        # Update tracking
        if improvement < tolerance:
            no_improvement_count += 1
        else:
            no_improvement_count = 0

        previous_fitness = current_fitness

        # Stop if no significant improvement for several iterations
        if no_improvement_count >= 5:
            break

        # Stop if we've reached very small improvements

            break
