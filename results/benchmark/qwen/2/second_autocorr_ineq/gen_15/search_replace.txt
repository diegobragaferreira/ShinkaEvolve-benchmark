<DESCRIPTION>
This improvement implements adaptive peak shaping by dynamically adjusting the noise level based on the computed autoconvolution norms during function generation. The key insight is to use the current norm values to control how much noise is added - specifically, when the infinity norm (peak) becomes too dominant, we reduce noise to prevent numerical instability that could hurt the C2 ratio. This builds on the Gaussian-based construction but adds intelligent control over the noise injection process to maintain optimal balance between peak height and distribution smoothness.

def construct_function() -> list[float]:
    """
    Construct step function with optimized Gaussian-based peak placement.
    """
    # Number of steps
    n_steps = np.random.randint(1000, 5000)

    # Generate positions using Gaussian distribution around center
    # This creates a bell-curve shaped distribution of peak heights
    positions = np.random.normal(0, 0.15, n_steps)
    # Clip to [-0.25, 0.25] to stay within domain
    positions = np.clip(positions, -0.25, 0.25)

    # Sort positions for proper step function construction
    positions = np.sort(positions)

    # Generate amplitudes based on distance from center
    # Peaks closer to center get higher amplitude
    amplitudes = np.exp(-20 * positions**2)

    # Add adaptive noise to break symmetry but preserve structure
    # Start with moderate noise
    noise_level = 0.1
    amplitudes += np.random.normal(0, noise_level, n_steps)

    # Compute initial norms to check if noise level needs adjustment
    # We do a quick norm computation to see if the resulting function is reasonable
    test_amplitudes = np.maximum(amplitudes, 0)
    if np.sum(test_amplitudes) > 0:
        test_amplitudes = test_amplitudes / np.sum(test_amplitudes) * 10

        # Quick norm computation to check quality
        temp_f = test_amplitudes
        temp_g = np.convolve(temp_f, temp_f, mode='full')
        temp_g = temp_g[len(temp_f)-1:2*len(temp_f)-1]
        temp_norm_1 = np.sum(np.abs(temp_g))
        temp_norm_inf = np.max(np.abs(temp_g)) if np.max(np.abs(temp_g)) > 0 else 1e-15
        temp_norm_2_squared = 0.0
        dx = 0.5 / len(temp_f)
        for i in range(len(temp_g)-1):
            y0, y1 = temp_g[i], temp_g[i+1]
            temp_norm_2_squared += (dx/3) * (y0**2 + y0*y1 + y1**2)

        # If the peak is too dominant (which would hurt C2), reduce noise
        if temp_norm_inf > 10 * temp_norm_1:
            # Reduce noise level significantly to prevent overly dominant peaks
            noise_level = max(0.01, noise_level * 0.5)
            amplitudes = np.random.normal(0, noise_level, n_steps)
            amplitudes += np.exp(-20 * positions**2)

    # Ensure non-negative amplitudes
    amplitudes = np.maximum(amplitudes, 0)

    # Normalize amplitudes to make them reasonable
    if np.sum(amplitudes) > 0:
        amplitudes = amplitudes / np.sum(amplitudes) * 10

    # Return as list
    return amplitudes.tolist()
