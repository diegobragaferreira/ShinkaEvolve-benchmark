def compute_autoconvolution_norms_fast(f_values: list[float]) -> tuple[float, float, float]:
    """
    Fast computation of the three norms needed for C2 calculation using piecewise linear integration.
    """
    # Convert to numpy array for easier manipulation
    f = np.array(f_values)
    n_steps = len(f)

    if n_steps == 0:
        return 0.0, 0.0, 0.0

    # Step width
    dx = 0.5 / n_steps

    # Compute autoconvolution using discrete convolution
    g = np.convolve(f, f, mode='full')
    # Trim g to the correct size (this accounts for the convolution)
    g = g[len(f)-1:2*len(f)-1]

    # Compute L2 norm squared using piecewise linear integration
    norm_2_squared = 0.0
    for i in range(len(g)-1):
        # Trapezoidal-like integration for quadratic function
        # Using formula for integral of ax^2 + bx + c over [x0,x1]
        # But here we approximate with piecewise linear segments
        # So we use: (dx/3)(y0^2 + y0*y1 + y1**2)
        y0, y1 = g[i], g[i+1]
        norm_2_squared += (dx/3) * (y0**2 + y0*y1 + y1**2)

    # L1 norm (sum of absolute values)
    norm_1 = np.sum(np.abs(g))

    # Infinity norm
    norm_inf = np.max(np.abs(g))

    # Handle numerical edge cases
    if norm_1 <= 1e-15:
        norm_1 = 1e-15
    if norm_inf <= 1e-15:
        norm_inf = 1e-15

    return norm_2_squared, norm_1, norm_inf

def create_initial_peak_parameters(n_peaks: int, domain_width: float = 0.5) -> list[float]:

def optimize_peak_parameters_direct(peak_params: list[float], n_steps: int = 2000) -> list[float]:
    """
    Direct optimization of peak parameters using differential evolution on just the peak parameters
    """
    domain_width = 0.5

    # Define bounds for optimization (amplitude, center, width for each peak)
    bounds = []
    n_peaks = len(peak_params) // 3
    for i in range(n_peaks):
        # Amplitude bounds
        bounds.extend([(10.0, 100.0)])
        # Center bounds
        bounds.extend([(-domain_width/2, domain_width/2)])
        # Width bounds
        bounds.extend([(0.01, 0.15)])

    def objective(params):
        # Convert flat parameter list back to peak structure
        new_peak_params = list(params)
        # Check if we need to adjust number of peaks - simplest approach: keep same number
        if len(new_peak_params) != len(peak_params):
            # Reconstruct from scratch if needed
            return -compute_c2_for_peaks(peak_params, n_steps)

        c2 = compute_c2_for_peaks(new_peak_params, n_steps)
        return -c2  # Negative because we minimize

    try:
        # Use differential evolution for global optimization
        result = differential_evolution(
            objective,
            bounds,
            maxiter=20,  # Reduced iterations for speed
            popsize=8,   # Smaller population for faster convergence
            seed=42,
            strategy='best1bin',
            atol=1e-6,
            rtol=1e-6
        )

        # If optimization was successful, return the optimized parameters
        if hasattr(result, 'success') and result.success:
            return list(result.x)
    except:
        pass

    # If optimization failed, return original parameters
    return peak_params

def compute_c2_for_peaks(peak_params: list[float], n_steps: int = 2000) -> float:
    """
    Compute C2 value directly from peak parameters for faster evaluation during optimization
    """
