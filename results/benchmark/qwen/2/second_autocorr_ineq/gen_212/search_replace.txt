This change implements a selective evolutionary approach that focuses on optimizing only the most impactful peak parameters (positions and amplitudes) rather than mutating all function values. This dimensional reduction technique should significantly improve convergence speed while maintaining solution quality. The key insight is that in step function optimization, the peak locations and heights are the primary determinants of the autoconvolution shape, so we should concentrate optimization efforts there.

The approach works by:
1. Starting with a spectral-guided peak configuration
2. Identifying the most influential peaks that contribute significantly to the autoconvolution
3. Evolving only these key parameters instead of the full function vector
4. Using a hybrid approach that combines evolutionary optimization with local refinement

        # Pass 2: Medium adjustments
        elif pass_num == 1:
            step_size = 0.02
            iterations = 50

        else:
            step_size = 0.01

def compute_c2(func_vals: List[float]) -> float:
    """
    Compute C₂ value with optimized numerical integration.
    """
    f = np.array(func_vals)

    # Autoconvolution using convolution
    g = np.convolve(f, f, mode='full')
    g = g[len(g)//2:]  # Take middle portion

    # Adjust for correct length
    if len(g) > len(f):
        g = g[:len(f)]

    # Compute norms with improved numerical accuracy
    # ||g||₂² (L2 norm squared) - using piecewise quadratic integration
    dx = 0.5 / (len(f) - 1) if len(f) > 1 else 0.5
    norm_2_sq = 0

        # Trapezoidal rule for better precision
        area = dx * (g[i]**2 + g[i+1]**2) / 2
        norm_2_sq += area

    norm_1 = np.sum(np.abs(g)) * dx

    # ||g||∞ (infinity norm)
    norm_inf = np.max(np.abs(g))

    # Prevent division by zero or extremely small values
    if norm_1 <= 1e-15 or norm_inf <= 1e-15:
        return 0.0

def evolve_peak_parameters(initial_func: List[float], n_steps: int) -> List[float]:
    """
    Selective evolutionary optimization focusing only on peak parameters (positions and heights)
    rather than all function values to improve convergence speed and quality.
    """
    x = np.linspace(-0.25, 0.25, n_steps)
    current_func = np.array(initial_func)

    # First, identify key peak characteristics from the current function
    # This helps us understand which parameters matter most for autoconvolution
    peak_locations = identify_key_peaks(current_func, x)

    # Create a representative peak parameter vector
    # Format: [position_1, height_1, width_1, position_2, height_2, width_2, ...]
    peak_params = []
    for loc in peak_locations:
        # Find the peak height at this location
        idx = np.argmin(np.abs(x - loc))
        height = current_func[idx]
        # Use a typical width for initial guess
        width = 0.025
        peak_params.extend([loc, height, width])

    # Evolutionary optimization of peak parameters only
    best_params = peak_params.copy()
    best_c2 = compute_c2(current_func)

    # Simple evolutionary algorithm focused on peak parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1

    # Initialize population
    population = [best_params.copy()]
    for _ in range(population_size - 1):
        individual = [p + random.gauss(0, 0.05 * abs(p) if p != 0 else 0.05)
                     for p in best_params]
        population.append(individual)

    for generation in range(generations):
        # Evaluate population
        fitness_scores = []
        evaluated_individuals = []

        for individual in population:
            # Reconstruct function from peak parameters
            reconstructed_func = reconstruct_from_peaks(individual, x)
            reconstructed_func = np.maximum(reconstructed_func, 0)

            # Compute C2 for this individual
            c2 = compute_c2(reconstructed_func)
            fitness_scores.append(c2)
            evaluated_individuals.append((individual, c2))

        # Find best individual in this generation
        best_idx = np.argmax(fitness_scores)
        if fitness_scores[best_idx] > best_c2:
            best_c2 = fitness_scores[best_idx]
            best_params = evaluated_individuals[best_idx][0].copy()

        # Create new population through selection and mutation
        # Tournament selection
        new_population = []
        for _ in range(population_size):
            # Select two individuals
            tournament_size = 3
            tournament_indices = random.choices(range(population_size), k=tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]

            # Clone and mutate
            new_individual = population[winner_idx].copy()
            for i in range(len(new_individual)):
                if random.random() < mutation_rate:
                    # Apply mutation with appropriate scaling
                    mutation_strength = 0.1 * abs(new_individual[i]) if new_individual[i] != 0 else 0.1
                    new_individual[i] += random.gauss(0, mutation_strength)

            new_population.append(new_individual)

        population = new_population

    # Return the best reconstructed function
    best_function = reconstruct_from_peaks(best_params, x)
    return best_function.tolist()

def identify_key_peaks(func_vals: np.ndarray, x: np.ndarray, min_height_ratio: float = 0.1) -> List[float]:
    """
    Identify key peak locations in the function for evolutionary optimization.
    """
    # Find local maxima
    peaks = []
    for i in range(1, len(func_vals)-1):
        if func_vals[i] > func_vals[i-1] and func_vals[i] > func_vals[i+1]:
            # Only consider significant peaks
            if func_vals[i] > min_height_ratio * np.max(func_vals):
                peaks.append(x[i])

    # If no peaks found, return some strategic positions
    if not peaks:
        peaks = [x[len(x)//2], x[len(x)//4], x[3*len(x)//4]]

    return sorted(set(peaks))  # Remove duplicates and sort

def reconstruct_from_peaks(peak_params: List[float], x: np.ndarray) -> np.ndarray:
    """
    Reconstruct function from peak parameters.
    """
    func = np.zeros_like(x)

    # Peak parameters come in groups of 3: [position, height, width]
    for i in range(0, len(peak_params), 3):
        if i + 2 < len(peak_params):
            pos = peak_params[i]
            height = peak_params[i+1]
            width = peak_params[i+2]

            # Create Gaussian peak
            gaussian_peak = height * np.exp(-0.5 * ((x - pos) / width)**2)
            func += gaussian_peak

    return func
