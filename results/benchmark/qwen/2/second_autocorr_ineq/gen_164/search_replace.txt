    def create_gaussian_peaks(n_peaks=8, peak_width=0.05, domain_width=0.5):
        """Create Gaussian-shaped peaks with optimized spacing"""
        f = np.zeros(2000)  # Fixed resolution
        x = np.linspace(-domain_width/2, domain_width/2, len(f))

        # Use logarithmic spacing for peak positions across domain
        # This ensures better spread across multiple scales and frequency ranges
        peak_positions = []

        # Create logarithmically spaced positions
        # Using log distribution to sample more points in smaller regions
        log_min = np.log(0.005)  # Wider range to capture more detail
        log_max = np.log(0.25)   # Max half-width
        log_spacings = np.logspace(log_min, log_max, n_peaks + 1)

        # Distribute peaks more evenly with log spacing
        for i in range(n_peaks):
            # Position based on logarithmic distribution
            if i == 0:
                pos = -0.25 + log_spacings[0] * 0.5  # Start near left boundary
            else:
                # Place at log-spaced intervals
                pos = -0.25 + np.sum(log_spacings[:i+1])  # cumulative log spacing
            # Add small randomness to avoid perfect periodicity
            pos += np.random.uniform(-0.005, 0.005)
