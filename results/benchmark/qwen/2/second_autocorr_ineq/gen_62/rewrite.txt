import numpy as np
import torch
import time
from numba import jit, prange
import warnings

@jit(nopython=True)
def compute_autoconvolution_numba(f):
    """Numba-accelerated autoconvolution computation"""
    n = len(f)
    g = np.zeros(2*n - 1)
    
    # Manual convolution loop for speed
    for i in range(n):
        for j in range(n):
            g[i + j] += f[i] * f[j]
    
    return g[n-1:]  # Return positive lags only

@jit(nopython=True)
def compute_norms_numba(g):
    """Numba-accelerated norm computations"""
    n = len(g)
    
    # Compute norms
    norm_1 = 0.0
    norm_2_sq = 0.0
    norm_inf = 0.0
    
    for i in range(n):
        abs_g = abs(g[i])
        norm_1 += abs_g
        norm_2_sq += abs_g * abs_g
        if abs_g > norm_inf:
            norm_inf = abs_g
    
    return norm_1, norm_2_sq, norm_inf

@jit(nopython=True)
def compute_c2_numba(norm_1, norm_2_sq, norm_inf):
    """Numba-accelerated C2 computation"""
    if norm_1 < 1e-12 or norm_inf < 1e-12:
        return 0.0
    return norm_2_sq / (norm_1 * norm_inf)

def evaluate_function_torch(f_np):
    """Evaluate function using torch for gradients and fast computation"""
    try:
        # Convert to tensor
        f_tensor = torch.tensor(f_np, dtype=torch.float32, requires_grad=True)
        
        # Compute autoconvolution manually for consistency
        n = len(f_np)
        g = np.zeros(2*n - 1)
        
        # Manual convolution loop
        for i in range(n):
            for j in range(n):
                g[i + j] += f_np[i] * f_np[j]
        
        g_np = g[n-1:]  # Positive lags only
        
        # Compute norms using numpy (more stable)
        norm_1, norm_2_sq, norm_inf = compute_norms_numba(g_np)
        
        # Avoid division by zero
        if norm_1 < 1e-12 or norm_inf < 1e-12:
            c2 = 0.0
        else:
            c2 = norm_2_sq / (norm_1 * norm_inf)
            
        return float(c2), g_np
    except Exception:
        return 0.0, np.array([0.0])

def construct_function() -> list[float]:
    """
    Construct a step function that maximizes C2 = ||g||₂² / (||g||₁ · ||g||∞)
    where g = f*f (autoconvolution) and f is the step function.
    """
    
    # Set seed for reproducibility
    np.random.seed(42)
    torch.manual_seed(42)
    
    # Parameters
    max_time = 90.0  # seconds
    start_time = time.time()
    
    # Initialize best solution
    best_c2 = 0.0
    best_f = None
    
    # Phase 1: Coarse grid search to find promising regions
    coarse_resolutions = [100, 200, 300, 400, 500, 600]
    best_coarse = None
    best_coarse_c2 = 0.0
    
    for res in coarse_resolutions:
        if time.time() - start_time > max_time * 0.3:
            break
            
        # Generate multiple candidate functions with different structures  
        for trial in range(5):
            # Create diverse initial patterns
            f = np.zeros(res)
            
            # Pattern type 1: Multi-peak structure with geometric spacing
            if trial % 3 == 0:
                # Create peaks using geometric distribution
                n_peaks = max(2, min(10, res // 50))
                positions = np.geomspace(0.1, res-0.1, n_peaks)
                positions = np.clip(positions.astype(int), 0, res-1)
                
                for pos in positions:
                    height = 0.5 + np.random.random() * 0.5
                    f[pos] = max(f[pos], height)
                    
            # Pattern type 2: Sparse high-amplitude peaks
            elif trial % 3 == 1:
                n_peaks = max(1, min(5, res // 100))
                positions = np.sort(np.random.choice(res, n_peaks, replace=False))
                
                for pos in positions:
                    # High amplitude peaks with some randomness
                    height = 1.0 + np.random.random() * 2.0
                    f[pos] = max(f[pos], height)  
            
            # Pattern type 3: Uniform background with sparse bumps
            else:
                f[:] = 0.2 + np.random.random(res) * 0.3
                # Add some sparse high-value points
                n_bumps = max(1, res // 50)
                bump_positions = np.random.choice(res, n_bumps, replace=False)
                for pos in bump_positions:
                    f[pos] += 1.0 + np.random.random() * 2.0
                    
            # Ensure non-negative values
            f = np.maximum(f, 0)
                
            # Normalize to reasonable scale
            if np.sum(f) > 0:
                f = f / np.sum(f) * 5.0
            else:
                f = np.ones_like(f) * 0.1
                
            c2, _ = evaluate_function_torch(f)
            
            if c2 > best_coarse_c2:
                best_coarse_c2 = c2
                best_coarse = f.copy()
    
    # Phase 2: Gradient-guided refinement of best coarse solution
    if best_coarse is not None:
        # Start with best coarse solution
        current_f = best_coarse.copy()
        
        # Convert to PyTorch tensor for optimization
        f_tensor = torch.tensor(current_f, dtype=torch.float32, requires_grad=True)
        
        # Optimizer settings
        optimizer = torch.optim.Adam([f_tensor], lr=0.05, weight_decay=1e-6)
        
        # Use momentum to help escape local minima
        scheduler = torch.optim.lr_scheduler.StepLR(optimizer, step_size=20, gamma=0.8)
        
        # Optimization loop
        for epoch in range(100):  # Reduced iterations for time constraints
            if time.time() - start_time > max_time * 0.8:
                break
                
            optimizer.zero_grad()
            
            # Ensure non-negativity with soft constraint
            f_positive = torch.clamp(f_tensor, min=0.0)
            
            # Convert back to numpy for evaluation
            f_np = f_positive.detach().numpy()
            
            # Evaluate C2
            c2_score, _ = evaluate_function_torch(f_np)
            
            # Use negative because we're maximizing (minimizing negative)
            loss = -torch.tensor(c2_score, dtype=torch.float32)
            
            # Backward pass
            loss.backward()
            
            # Update parameters
            optimizer.step()
            
            # Adjust learning rate
            scheduler.step()
            
            # Ensure non-negativity again post-update
            with torch.no_grad():
                f_tensor.clamp_(min=0.0)
            
            # Track best
            if c2_score > best_c2:
                best_c2 = c2_score
                best_f = f_positive.detach().numpy().copy()
        
        # Phase 3: Multi-scale refinement 
        if best_f is not None:
            # Try with higher resolution
            high_res = max(1000, len(best_f) * 2)
            f_high_res = np.interp(np.linspace(0, len(best_f)-1, high_res), 
                                 np.arange(len(best_f)), best_f)
            
            # Normalize
            if np.sum(f_high_res) > 0:
                f_high_res = f_high_res / np.sum(f_high_res) * 5.0
                
            c2_high, _ = evaluate_function_torch(f_high_res)
            
            if c2_high > best_c2:
                best_c2 = c2_high
                best_f = f_high_res.copy()
    
    # Phase 4: Local optimization around best so far
    if best_f is not None:
        # Implement simulated annealing approach with local search
        current_f = best_f.copy()
        current_c2, _ = evaluate_function_torch(current_f)
        
        best_local_f = current_f.copy()
        best_local_c2 = current_c2
        
        # Simulated annealing parameters
        temp = 1.0
        cooling_rate = 0.98
        min_temp = 0.01
        
        # Number of moves per temperature
        moves_per_temp = 50
        
        for iteration in range(1000):
            if time.time() - start_time > max_time * 0.95:
                break
                
            # Create neighbor solution
            neighbor_f = current_f.copy()
            
            # Perturb a few random positions
            num_perturb = max(1, len(current_f) // 20)
            indices = np.random.choice(len(current_f), num_perturb, replace=False)
            
            for idx in indices:
                # Add small random perturbation
                delta = np.random.normal(0, 0.1)
                neighbor_f[idx] += delta
                neighbor_f[idx] = max(0, neighbor_f[idx])
                
            # Normalize
            if np.sum(neighbor_f) > 0:
                neighbor_f = neighbor_f / np.sum(neighbor_f) * 5.0
                
            # Evaluate neighbor
            neighbor_c2, _ = evaluate_function_torch(neighbor_f)
            
            # Accept or reject based on Metropolis criterion
            if neighbor_c2 > current_c2:
                current_f = neighbor_f.copy()
                current_c2 = neighbor_c2
            else:
                # Probability of accepting worse solution
                prob_accept = np.exp((neighbor_c2 - current_c2) / temp)
                if np.random.random() < prob_accept:
                    current_f = neighbor_f.copy()
                    current_c2 = neighbor_c2
                    
            # Update best
            if current_c2 > best_local_c2:
                best_local_c2 = current_c2
                best_local_f = current_f.copy()
                
            # Cool down
            temp = max(min_temp, temp * cooling_rate)
            
            # Occasionally reset temperature to escape local optima
            if iteration % 100 == 0 and temp < 0.1:
                temp = 1.0
                
        if best_local_c2 > best_c2:
            best_c2 = best_local_c2
            best_f = best_local_f.copy()
    
    # Final fallback if no good solution was found
    if best_f is None:
        # Create a well-balanced function
        n = 500
        best_f = np.zeros(n)
        
        # Place several well-spaced peaks with varying heights
        n_peaks = 8
        positions = np.linspace(0.1*n, 0.9*n, n_peaks)
        for i, pos in enumerate(positions):
            idx = int(pos)
            height = 0.8 + np.random.random() * 0.4
            if 0 <= idx < n:
                best_f[idx] = height
                
        # Normalize
        if np.sum(best_f) > 0:
            best_f = best_f / np.sum(best_f) * 5.0
        else:
            best_f = np.ones(n) * 0.2
    
    # Ensure all values are non-negative
    result = [max(0, x) for x in best_f]
    
    return result
