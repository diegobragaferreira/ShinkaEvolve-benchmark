            # Create x coordinates from -1/4 to 1/4
            x = np.linspace(-0.25, 0.25, length)

            # Use adaptive peak distribution: more peaks near center where they matter most
            individual = np.zeros(length)

            center_region_width = 0.1  # Center region where peaks are denser
            edge_region_width = 0.15   # Edge regions where peaks are sparser

            peak_positions = []
