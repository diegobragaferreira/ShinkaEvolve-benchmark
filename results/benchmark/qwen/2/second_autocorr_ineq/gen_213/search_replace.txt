def refine_function_spectral_guided(initial_func: List[float], n_steps: int) -> List[float]:
    """
    Multi-scale refinement approach guided by spectral properties.
    """
    x = np.linspace(-0.25, 0.25, n_steps)
    current_func = np.array(initial_func)

    # Multi-pass refinement with decreasing step sizes
    for pass_num in range(3):
        # Pass 1: Large adjustments
        if pass_num == 0:
            step_size = 0.05
            iterations = 100
        # Pass 2: Medium adjustments
        elif pass_num == 1:
            step_size = 0.02
            iterations = 50
        # Pass 3: Fine adjustments
        else:
            step_size = 0.01
            iterations = 30

        for _ in range(iterations):

def evolutionary_refine_function(initial_func: List[float], n_steps: int) -> List[float]:
    """
    Evolutionary optimization of function using DEAP library to refine peak parameters.
    """
    # Create individual representation as flattened list of peak parameters
    # Each peak represented as [position, height, width]

    # First, determine number of peaks from the initial function
    # We'll extract peaks using a simple detection method
    x = np.linspace(-0.25, 0.25, n_steps)
    func_array = np.array(initial_func)

    # Detect peaks to determine how many to optimize
    peaks = []
    for i in range(1, len(func_array)-1):
        if func_array[i] > func_array[i-1] and func_array[i] > func_array[i+1]:
            peaks.append((i, func_array[i]))

    # Sort by height and take top 15
    peaks.sort(key=lambda x: x[1], reverse=True)
    num_peaks = min(15, len(peaks))

    # Prepare initial peak parameters (position, height, width)
    peak_params = []
    for i in range(num_peaks):
        peak_idx, peak_height = peaks[i]
        peak_pos = x[peak_idx]
        # Determine appropriate width based on function characteristics
        peak_width = 0.02 + 0.04 * (1.0 - abs(peak_pos)/0.25)
        peak_params.extend([peak_pos, peak_height, peak_width])

    # Define evolutionary algorithm parameters
    # Individual: [pos1, height1, width1, pos2, height2, width2, ...]
    IND_SIZE = num_peaks * 3

    # Create DEAP structures
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    toolbox = base.Toolbox()

    # Attribute generator
    def gen_peak_param():
        # Generate random peak parameter values
        return random.uniform(-0.25, 0.25)  # Position
        # We'll generate separate bounds for each parameter type

    # Structure definitions for different parameter types
    def gen_position():
        return random.uniform(-0.25, 0.25)

    def gen_height():
        return random.uniform(0.5, 3.0)

    def gen_width():
        return random.uniform(0.01, 0.08)

    # Register operators
    toolbox.register("attr_pos", gen_position)
    toolbox.register("attr_height", gen_height)
    toolbox.register("attr_width", gen_width)

    # Individual and population creation
    def create_individual():
        individual = []
        for i in range(num_peaks):
            individual.extend([gen_position(), gen_height(), gen_width()])
        return creator.Individual(individual)

    toolbox.register("individual", create_individual)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    def eval_function(individual):
        """Evaluate fitness of individual"""
        # Recreate function from peak parameters
        func_copy = np.zeros_like(func_array)

        for i in range(num_peaks):
            pos = individual[i*3]
            height = individual[i*3 + 1]
            width = individual[i*3 + 2]
            # Ensure width is positive
            width = max(0.001, width)
            gaussian = height * np.exp(-0.5 * ((x - pos) / width)**2)
            func_copy += gaussian

        # Add supplementary structure
        add_supplementary_structure(func_copy, x)

        # Ensure non-negativity
        func_copy = np.maximum(func_copy, 0)

        # Compute C2
        try:
            c2_value = compute_c2(func_copy.tolist())
            return (c2_value,)
        except:
            return (0,)

    toolbox.register("evaluate", eval_function)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.02, indpb=0.2)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Run evolution
    try:
        # Create initial population
        pop = toolbox.population(n=50)

        # Evolve for 20 generations
        for gen in range(20):
            offspring = algorithms.varAnd(pop, toolbox, cxpb=0.7, mutpb=0.3)
            fits = toolbox.map(toolbox.evaluate, offspring)
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = fit
            pop = toolbox.select(offspring, k=len(pop))

        # Get best individual
        best_ind = tools.selBest(pop, 1)[0]

        # Reconstruct function with best parameters
        best_func = np.zeros_like(func_array)
        for i in range(num_peaks):
            pos = best_ind[i*3]
            height = best_ind[i*3 + 1]
            width = best_ind[i*3 + 2]
            width = max(0.001, width)
            gaussian = height * np.exp(-0.5 * ((x - pos) / width)**2)
            best_func += gaussian

        # Add supplementary structure
        add_supplementary_structure(best_func, x)
        best_func = np.maximum(best_func, 0)

        return best_func.tolist()

    except Exception as e:
        warnings.warn(f"Evolutionary optimization failed: {str(e)}")
        # Return original function if evolution fails
        return initial_func

def compute_c2(func_vals: List[float]) -> float:
    """
    Compute C₂ value with optimized numerical integration.
    """
    f = np.array(func_vals)

    # Autoconvolution using convolution
    g = np.convolve(f, f, mode='full')
    g = g[len(g)//2:]  # Take middle portion

    # Adjust for correct length
    if len(g) > len(f):
        g = g[:len(f)]

    # Compute norms with improved numerical accuracy
    # ||g||₂² (L2 norm squared) - using piecewise quadratic integration
    dx = 0.5 / (len(f) - 1) if len(f) > 1 else 0.5
    norm_2_sq = 0
    for i in range(len(g)-1):
        # Trapezoidal rule for better precision
        area = dx * (g[i]**2 + g[i+1]**2) / 2
        norm_2_sq += area

    # ||g||₁ (L1 norm)
    norm_1 = np.sum(np.abs(g)) * dx

    # ||g||∞ (infinity norm)
    norm_inf = np.max(np.abs(g))

    # Prevent division by zero or extremely small values
    if norm_1 <= 1e-15 or norm_inf <= 1e-15:
        return 0.0

    return norm_2_sq / (norm_1 * norm_inf)
