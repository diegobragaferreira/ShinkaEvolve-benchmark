def enforce_minimum_peak_spacing(peak_params: list[float], domain_width: float, min_distance_ratio: float = 0.05) -> list[float]:
    """
    Enforce minimum distance between Gaussian peaks to prevent narrow autoconvolution interference.
    This helps maintain flatter autoconvolution profiles which improve C2 values.
    """
    if len(peak_params) < 3:
        return peak_params

    # Create list of peaks [amplitude, center, width]
    peaks = []
    for i in range(0, len(peak_params), 3):
        peaks.append([peak_params[i], peak_params[i+1], peak_params[i+2]])

    # Sort by center position
    peaks.sort(key=lambda x: x[1])

    # More sophisticated peak spacing enforcement using iterative optimization
    min_distance = min_distance_ratio * domain_width

    # Create working copy of peaks
    working_peaks = [peak[:] for peak in peaks]  # Deep copy

    # Iteratively adjust peak positions to satisfy minimum spacing
    changed = True
    max_iterations = 100  # Prevent infinite loops
    iterations = 0

    while changed and iterations < max_iterations:
        changed = False
        iterations += 1

        # Check all adjacent pairs
        for i in range(len(working_peaks) - 1):
            amp1, center1, width1 = working_peaks[i]
            amp2, center2, width2 = working_peaks[i + 1]

            current_distance = abs(center2 - center1)

            if current_distance < min_distance:
                # Calculate adjustment needed
                adjustment = min_distance - current_distance

                # Determine direction to move peaks
                # Prefer moving rightmost peak to maintain structure
                if center1 <= center2:
                    # Move right peak to the right, left peak to the left
                    move_right = adjustment / 2.0
                    move_left = adjustment / 2.0
                else:
                    # Move left peak to the left, right peak to the right
                    move_right = adjustment / 2.0
                    move_left = adjustment / 2.0

                # Apply adjustments with boundary checking
                new_center1 = center1 - move_left
                new_center2 = center2 + move_right

                # Clip to domain bounds
                new_center1 = np.clip(new_center1, -domain_width/2, domain_width/2)
                new_center2 = np.clip(new_center2, -domain_width/2, domain_width/2)

                # Check if adjustment would make peaks overlap again
                if abs(new_center2 - new_center1) < min_distance:
                    # If they still overlap, push them apart more
                    total_overlap = min_distance - abs(new_center2 - new_center1)
                    half_overlap = total_overlap / 2.0

                    if center1 <= center2:
                        new_center1 = center1 - half_overlap
                        new_center2 = center2 + half_overlap
                    else:
                        new_center1 = center1 + half_overlap
                        new_center2 = center2 - half_overlap

                    # Final boundary clipping
                    new_center1 = np.clip(new_center1, -domain_width/2, domain_width/2)
                    new_center2 = np.clip(new_center2, -domain_width/2, domain_width/2)

                # Update if changes occurred

                    working_peaks[i + 1] = [amp2, new_center2, width2]
                    changed = True

        working_peaks[i] = [amp, np.clip(center, -domain_width/2, domain_width/2), width]
