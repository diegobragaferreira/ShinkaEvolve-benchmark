import numpy as np
from scipy import signal
from scipy.optimize import differential_evolution

def compute_autoconvolution_norms(f_values):
    """Compute the norms needed for C2 calculation"""
    # Convert to numpy array
    f = np.array(f_values)

    # Compute autoconvolution
    g = np.convolve(f, f, mode='full')

    # Adjust for proper normalization (autoconvolution of normalized function)
    g = g[len(f)-1:]  # Keep only the relevant part

    # Compute norms
    g_squared = g**2
    norm_2_squared = np.sum(g_squared)
    norm_1 = np.sum(np.abs(g))
    norm_inf = np.max(np.abs(g))

    return norm_2_squared, norm_1, norm_inf

def adaptive_gaussian_construction():
    """Construct function using adaptive Gaussian peaks"""
    np.random.seed(42)  # For reproducibility

    # Try several configurations and pick the best
    best_c2 = 0
    best_f = None

    # Generate multiple candidate functions with different parameters
    for attempt in range(10):
        try:
            # Determine number of peaks (between 5 and 25)
            n_peaks = np.random.randint(5, 26)

            # Create domain
            x = np.linspace(-0.25, 0.25, 2000)
            f_values = np.zeros_like(x)

            # Place peaks adaptively
            peak_positions = np.random.uniform(-0.25, 0.25, n_peaks)
            peak_widths = np.random.uniform(0.01, 0.05, n_peaks)
            peak_amplitudes = np.random.uniform(0.5, 2.0, n_peaks)

            # Add Gaussian peaks
            for i in range(n_peaks):
                # Gaussian peak centered at peak_position with given width and amplitude
                gaussian_peak = peak_amplitudes[i] * np.exp(-0.5 * ((x - peak_positions[i]) / peak_widths[i])**2)
                f_values += gaussian_peak

            # Apply smoothing to reduce any sharp edges
            window_size = max(3, int(len(x) / 100))
            if window_size % 2 == 0:
                window_size += 1
            f_values = signal.savgol_filter(f_values, window_size, 3)

            # Normalize the function
            f_values = np.maximum(f_values, 0)  # Ensure non-negative

            # Compute C2
            norm_2_sq, norm_1, norm_inf = compute_autoconvolution_norms(f_values)

            # Avoid division by zero
            if norm_1 > 1e-10 and norm_inf > 1e-10:
                c2 = norm_2_sq / (norm_1 * norm_inf)

                # Only consider valid solutions (not too extreme)
                if 0.8 < c2 < 2.0:
                    if c2 > best_c2:
                        best_c2 = c2
                        best_f = f_values.tolist()

        except Exception as e:
            continue

    # If we didn't find anything good, fallback to a basic approach
    if best_f is None:
        f_values = [np.random.random() for _ in range(100)]
        # Apply basic smoothing
        f_values = [max(0, x) for x in f_values]
        best_f = f_values

    # Final refinement with local optimization
    try:
        def objective(params):
            # Reconstruct function from params (assuming we're optimizing peak positions/amplitudes)
            # This is a simplified version - full optimization would be complex
            return -compute_autoconvolution_norms(params)[0] / (compute_autoconvolution_norms(params)[1] * compute_autoconvolution_norms(params)[2] + 1e-12)

        # Simple gradient-free optimization to refine the best found solution
        refined_params = differential_evolution(objective,
                                               bounds=[(0, 1)] * len(best_f),
                                               maxiter=50, popsize=10,
                                               seed=42).x

        # Re-compute with refined parameters
        _, norm_1, norm_inf = compute_autoconvolution_norms(refined_params)
        if norm_1 > 1e-10 and norm_inf > 1e-10:
            norm_2_sq, _, _ = compute_autoconvolution_norms(refined_params)
            final_c2 = norm_2_sq / (norm_1 * norm_inf)
            if final_c2 > best_c2:
                best_c2 = final_c2
                best_f = refined_params.tolist()
    except:
        pass

    return best_f

def construct_function() -> list[float]:
    """Function to construct step-function with high C2 value."""
    # Use adaptive Gaussian construction for better results
    return adaptive_gaussian_construction()
