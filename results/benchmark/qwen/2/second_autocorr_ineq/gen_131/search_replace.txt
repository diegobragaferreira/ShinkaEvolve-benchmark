These enhancements aim to create better initial populations that are more likely to lead to high C₂ values by avoiding problematic configurations while maintaining diversity.

        """Create diverse initial population using enhanced Gaussian-based strategies"""
        population = []

        # Strategy 1: Enhanced Gaussian-based approach with strategic positioning
        for _ in range(self.population_size // 2):
            length = np.random.randint(min_length, max_length)

            # Use logarithmic spacing for peak positions to ensure good distribution
            num_peaks = max(2, min(8, length // 100))  # Adaptive peak count based on length

            # Generate logarithmically spaced positions to avoid clustering
            log_min = np.log(0.01)
            log_max = np.log(0.2)  # Cap at 20% of domain width
            log_positions = np.logspace(log_min, log_max, num_peaks)

            # Create x coordinates from -1/4 to 1/4
            x = np.linspace(-0.25, 0.25, length)

            # Initialize function
            individual = np.zeros(length)

            # Add peaks with strategic positioning and amplitude scaling
            for i, log_pos in enumerate(log_positions):
                # Position peaks in the middle third to avoid edge effects
                side = np.random.choice([-1, 1])
                base_pos = side * log_pos * 0.5

                # Add variation to avoid perfect symmetry
                peak_pos = base_pos + np.random.uniform(-0.015, 0.015)

                # Ensure peak stays within domain bounds

                        amp_factor = 1.3
                    else:  # Outer areas
