    def _generate_initial_solution(self) -> np.ndarray:
        """Generate an enhanced diverse initial population with symmetry-aware configurations."""
        # Generate multiple symmetric base configurations
        base_configs = self._generate_symmetric_base_configurations()

        # Weights can be based on known performance characteristics or random selection
        weights = [0.4, 0.3, 0.3]  # Prefer first configuration slightly
