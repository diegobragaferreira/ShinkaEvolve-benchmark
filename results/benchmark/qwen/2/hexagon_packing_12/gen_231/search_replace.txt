class SpatialHashGrid:
    """Spatial hash grid for efficient neighbor lookups with adaptive cell sizing."""

    def __init__(self, cell_size=2.5, adaptive=False):
        self.cell_size = cell_size
        self.adaptive = adaptive
        self.grid = {}
        self.hex_centers = {}  # Store centers for adaptive sizing

    def clear(self):
        self.grid.clear()
        self.hex_centers.clear()

    def insert(self, hex_id, center_x, center_y):
        """Insert hexagon into spatial hash grid."""
        cell_x = int(center_x // self.cell_size)
        cell_y = int(center_y // self.cell_size)
        key = (cell_x, cell_y)
        if key not in self.grid:
            self.grid[key] = []
        self.grid[key].append(hex_id)
        self.hex_centers[hex_id] = (center_x, center_y)

    def update_cell_size(self, positions):
        """Adaptively adjust cell size based on hexagon distribution."""
        if not self.adaptive or len(positions) < 2:
            return

        # Calculate average distance between hexagons
        centers = np.array([(pos[0], pos[1]) for pos in positions])
        if len(centers) < 2:
            return

        # Compute pairwise distances
        distances = []
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                dist = np.linalg.norm(centers[i] - centers[j])
                distances.append(dist)

        if distances:
            avg_dist = np.mean(distances)
            # Cell size should be roughly 1.5x the average distance to ensure good coverage
            self.cell_size = max(1.0, avg_dist * 1.5)

    def get_neighbors(self, center_x, center_y, radius=None):
        """Get all hexagons in the same and neighboring cells."""
        if radius is None:
            # Standard 3x3 neighborhood
            cell_x = int(center_x // self.cell_size)
            cell_y = int(center_y // self.cell_size)

            neighbors = []
            for dx in [-1, 0, 1]:
                for dy in [-1, 0, 1]:
                    key = (cell_x + dx, cell_y + dy)
                    if key in self.grid:
                        neighbors.extend(self.grid[key])
        else:
            # Radius-based neighborhood (used for overlap checking)
            neighbors = []
            min_cell_x = int((center_x - radius) // self.cell_size)
            max_cell_x = int((center_x + radius) // self.cell_size)
            min_cell_y = int((center_y - radius) // self.cell_size)
            max_cell_y = int((center_y + radius) // self.cell_size)

            for cx in range(min_cell_x, max_cell_x + 1):
                for cy in range(min_cell_y, max_cell_y + 1):
                    key = (cx, cy)
                    if key in self.grid:
                        neighbors.extend(self.grid[key])
