def generate_symmetric_initial_population(pop_size, num_hexagons=12):
    """Generate intelligent initial population with symmetry patterns."""
    population = []

    # Base symmetric configuration: hexagonal pattern around center
    # Uses mathematical group theory for hexagonal symmetry
    base_angles = np.linspace(0, 2*np.pi, 7, endpoint=False)  # 6 surrounding positions + 1 center

    # Define the symmetry group: D6 (dihedral group of order 12)
    # This is what we want to exploit: 6 rotations + 6 reflections

    for _ in range(pop_size):
        # Start with a symmetric layout respecting D6 structure
        hex_config = []

        # Central hexagon
        hex_config.append([0, 0, 0])  # Fixed at center, no rotation needed

        # First ring around center (6 hexagons)
        # Distribute evenly around circle of radius ~2.0
        for i in range(1, 7):
            radius = 2.0
            angle = base_angles[i]  # 0, π/3, 2π/3, π, 4π/3, 5π/3
            x = radius * np.cos(angle)
            y = radius * np.sin(angle)
            # Keep rotation fixed for now (can be randomized later)
            hex_config.append([x, y, 0])

        # Second ring around first ring (5 hexagons)
        # Place them along radial lines from center but with different spacing
        for i in range(7, 12):
            # Use 60-degree increments but place further out
            angle_index = (i - 7) * 60  # 0°, 60°, 120°, 180°, 240°
            angle_rad = np.radians(angle_index)
            # Place further out to avoid overlap
            radius = 3.2
            x = radius * np.cos(angle_rad)
            y = radius * np.sin(angle_rad)
            hex_config.append([x, y, 0])

        # Add controlled random perturbations
        for i in range(len(hex_config)):
            # Perturb positions slightly but maintain relative distances
            hex_config[i][0] += np.random.normal(0, 0.15)
            hex_config[i][1] += np.random.normal(0, 0.15)
            # Allow small rotation adjustments but not too much to preserve symmetry
            hex_config[i][2] += np.random.uniform(-10, 10)

        # Apply symmetry transformation to create a more structured symmetric pattern
        # This ensures that even though we introduce randomness, we maintain a symmetric core
        population.append(np.array(hex_config).flatten())

    return population
