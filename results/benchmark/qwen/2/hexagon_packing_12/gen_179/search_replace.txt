def local_search_refinement(solution_array: np.ndarray, evaluator: SolutionEvaluator,
                           geometry: HexagonGeometry, validator: ConstraintValidator,
                           max_iterations: int = 50) -> np.ndarray:
    """Apply local search refinement to improve final solution."""
    from scipy.optimize import minimize

    # Create a copy of the solution to work with
    current_solution = solution_array.copy()
    current_inner_hex_data = current_solution.reshape(-1, 3)

    # Calculate initial objective
    min_side_length, centroid = evaluator.calculate_min_enclosing_hexagon(current_inner_hex_data, 1.05)
    initial_objective = -1.0 / min_side_length

    # Optimization bounds (same as original bounds)
    bounds = []
    for i in range(12):
        bounds.extend([(-8.0, 8.0), (-8.0, 8.0), (0.0, 360.0)])

    # Try different local optimization approaches
    for iteration in range(max_iterations):
        # Create a new solution by making small changes
        # Focus on improving positions and rotations
        new_solution = current_solution.copy()

        # Make small random perturbations to some hexagons
        hex_to_modify = np.random.choice(12, size=3, replace=False)  # Modify 3 hexagons

        for hex_id in hex_to_modify:
            # Modify position with small random changes
            new_solution[hex_id*3] += np.random.normal(0, 0.1)  # x position
            new_solution[hex_id*3 + 1] += np.random.normal(0, 0.1)  # y position
            new_solution[hex_id*3 + 2] += np.random.normal(0, 5.0)  # rotation (degrees)
            # Keep rotation in [0, 360) range
            new_solution[hex_id*3 + 2] = new_solution[hex_id*3 + 2] % 360

        # Check if the new solution is valid
        new_inner_hex_data = new_solution.reshape(-1, 3)

        # Validate constraints
        valid = True
        try:
            # Check containment
            for i in range(len(new_inner_hex_data)):
                center_x, center_y, angle = new_inner_hex_data[i]
                vertices = geometry.vertices(center_x, center_y, angle)
                if not validator.is_contained(vertices, centroid[0], centroid[1], min_side_length):
                    valid = False
                    break

            # Check overlaps
            if valid:
                for i in range(len(new_inner_hex_data)):
                    for j in range(i+1, len(new_inner_hex_data)):
                        vertices1 = geometry.vertices(*new_inner_hex_data[i])
                        vertices2 = geometry.vertices(*new_inner_hex_data[j])
                        if validator.has_overlap_fast(vertices1, vertices2):
                            valid = False
                            break
        except:
            valid = False

        if valid:
            # Evaluate the new solution
            try:
                new_min_side_length, _ = evaluator.calculate_min_enclosing_hexagon(new_inner_hex_data, 1.05)
                new_objective = -1.0 / new_min_side_length

                # Accept if better
                if new_objective < initial_objective:
                    current_solution = new_solution.copy()
                    initial_objective = new_objective
                    min_side_length = new_min_side_length
            except:
                pass  # Skip invalid solutions

    return current_solution
