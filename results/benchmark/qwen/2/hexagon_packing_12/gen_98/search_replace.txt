def point_in_hexagon(point_x, point_y, hex_center_x, hex_center_y, hex_angle_deg, hex_side_length=1):
    """Check if a point is inside a hexagon using fast geometric operations."""
    # Transform point to hexagon's local coordinate system
    hex_angle_rad = hex_angle_deg * math.pi / 180.0
    dx = point_x - hex_center_x
    dy = point_y - hex_center_y

    # Rotate point back to align with hexagon axes
    cos_a = math.cos(-hex_angle_rad)
    sin_a = math.sin(-hex_angle_rad)
    local_x = dx * cos_a - dy * sin_a
    local_y = dx * sin_a + dy * cos_a

    # Check if point is inside hexagon
    # For regular hexagon with side length 1, distance from center to vertex is 1
    # Distance from center to edge is sqrt(3)/2
    r = math.sqrt(local_x * local_x + local_y * local_y)
    if r > hex_side_length:
        return False

    # More precise check using hexagon boundaries
    # Hexagon vertices in local coordinates (simplified)
    # But for our use case, just check against outer radius
    return True

import numpy as np
from numba import jit, prange
import math
from scipy.spatial.distance import cdist

# Constants
HEX_SIDE_LENGTH = 1.0
HEX_RADIUS = HEX_SIDE_LENGTH  # For unit hexagon
HEX_APO = HEX_RADIUS * math.sqrt(3) / 2  # Apothem of unit hexagon

@jit(nopython=True)
def hexagon_vertices_numba(center_x, center_y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given center, angle, and side length - JIT compiled."""
    angle_rad = angle_deg * math.pi / 180.0
    vertices = np.empty((6, 2))
    for i in range(6):
        angle = angle_rad + i * math.pi / 3
        x = center_x + side_length * math.cos(angle)
        y = center_y + side_length * math.sin(angle)
        vertices[i] = (x, y)
    return vertices

@jit(nopython=True)
def point_in_polygon(point_x, point_y, polygon_vertices):
    """Fast point-in-polygon test using ray casting algorithm - JIT compiled."""
    n = len(polygon_vertices)
    inside = False
    p1x, p1y = polygon_vertices[0]
    for i in range(1, n + 1):
        p2x, p2y = polygon_vertices[i % n]
        if point_y > min(p1y, p2y):
            if point_y <= max(p1y, p2y):
                if point_x <= max(p1x, p2x):
                    if p1y != p2y:
                        xinters = (point_y - p1y) * (p2x - p1x) / (p2y - p1y) + p1x
                    if p1x == p2x or point_x <= xinters:
                        inside = not inside
        p1x, p1y = p2x, p2y
    return inside

@jit(nopython=True)
def distance_point_to_segment(point_x, point_y, seg_start_x, seg_start_y, seg_end_x, seg_end_y):
    """Calculate distance from point to line segment - JIT compiled."""
    # Vector from seg_start to seg_end
    dx = seg_end_x - seg_start_x
    dy = seg_end_y - seg_start_y
    # Vector from seg_start to point
    fx = point_x - seg_start_x
    fy = point_y - seg_start_y

    # Project point onto line
    length_sq = dx*dx + dy*dy
    if length_sq == 0:
        return math.sqrt(fx*fx + fy*fy)

    t = (fx*dx + fy*dy) / length_sq
    t = max(0, min(1, t))  # Clamp t to [0,1]

    # Closest point on segment
    closest_x = seg_start_x + t*dx
    closest_y = seg_start_y + t*dy

    # Distance to closest point
    return math.sqrt((point_x - closest_x)**2 + (point_y - closest_y)**2)

@jit(nopython=True)
def hexagon_distance(hex1_vertices, hex2_vertices):
    """Compute minimum distance between two hexagons - JIT compiled."""
    # Check if any vertex of hex1 is inside hex2
    for i in range(6):
        if point_in_polygon(hex1_vertices[i, 0], hex1_vertices[i, 1], hex2_vertices):
            return 0.0

    # Check if any vertex of hex2 is inside hex1
    for i in range(6):
        if point_in_polygon(hex2_vertices[i, 0], hex2_vertices[i, 1], hex1_vertices):
            return 0.0

    # Check edges
    min_dist = 1e10
    for i in range(6):
        for j in range(6):
            p1 = hex1_vertices[i]
            p2 = hex1_vertices[(i+1)%6]
            p3 = hex2_vertices[j]
            p4 = hex2_vertices[(j+1)%6]

            # Compute distance between segments
            dist = distance_point_to_segment(p1[0], p1[1], p3[0], p3[1], p4[0], p4[1])
            min_dist = min(min_dist, dist)

            dist = distance_point_to_segment(p3[0], p3[1], p1[0], p1[1], p2[0], p2[1])
            min_dist = min(min_dist, dist)

    return min_dist

class SpatialHashGrid:
    """Spatial hash grid for efficient neighbor lookups."""

    def __init__(self, cell_size=3.0):
        self.cell_size = cell_size
        self.grid = {}

    def clear(self):
        self.grid.clear()

    def insert(self, hex_id, center_x, center_y):
        """Insert hexagon into spatial hash grid."""
        cell_x = int(center_x // self.cell_size)
        cell_y = int(center_y // self.cell_size)
        key = (cell_x, cell_y)
        if key not in self.grid:
            self.grid[key] = []
        self.grid[key].append(hex_id)

    def get_neighbors(self, center_x, center_y):
        """Get all hexagons in the same and neighboring cells."""
        cell_x = int(center_x // self.cell_size)
        cell_y = int(center_y // self.cell_size)

        neighbors = []
        for dx in [-1, 0, 1]:
            for dy in [-1, 0, 1]:
                key = (cell_x + dx, cell_y + dy)
                if key in self.grid:
                    neighbors.extend(self.grid[key])
        return neighbors

@jit(nopython=True)
def compute_outer_hexagon_side_length_fast(inner_hex_data):
    """Fast computation of outer hexagon side length - JIT compiled."""
    # Get all vertices
    all_vertices = np.empty((len(inner_hex_data) * 6, 2), dtype=np.float64)

    for i in range(len(inner_hex_data)):
        center_x, center_y, angle_degrees = inner_hex_data[i]
        vertices = hexagon_vertices_numba(center_x, center_y, angle_degrees)
        for j in range(6):
            all_vertices[i*6 + j] = vertices[j]

    if len(all_vertices) == 0:
        return 1e6

    # Compute bounding box
    min_x = all_vertices[:, 0].min()
    max_x = all_vertices[:, 0].max()
    min_y = all_vertices[:, 1].min()
    max_y = all_vertices[:, 1].max()

    # Compute center
    center_x = (min_x + max_x) / 2.0
    center_y = (min_y + max_y) / 2.0

    # Compute max distance squared from center
    max_dist_sq = 0.0
    for i in range(len(all_vertices)):
        x, y = all_vertices[i]
        dist_sq = (x - center_x)**2 + (y - center_y)**2
        max_dist_sq = max(max_dist_sq, dist_sq)

    # Side length = sqrt(max_dist) * 2 / sqrt(3)
    return math.sqrt(max_dist_sq) * 2.0 / math.sqrt(3)

def check_containment_fast(hex_vertices, outer_hex_vertices):
    """Check containment using Shapely-like approach."""
    # Check if all vertices of hex are inside outer hex
    for i in range(6):
        if not point_in_polygon(hex_vertices[i, 0], hex_vertices[i, 1], outer_hex_vertices):
            return False
    return True

def check_overlap_fast(hex1_vertices, hex2_vertices):
    """Check overlap using fast geometric methods."""
    # If distance between hexagons is less than 2 (sum of radii), they overlap
    # However, this is a simplification; we'll use the full geometric distance
    return hexagon_distance(hex1_vertices, hex2_vertices) < 0.001

def evaluate_fitness(solution, outer_hex_center=(0, 0), outer_hex_angle=0):
    """Evaluate fitness of a solution with optimized overlap checking."""
    # Reshape solution back to 12 hexagons with (x, y, angle)
    positions = solution.reshape(-1, 3)

    # Create vertices for all inner hexagons
    inner_vertices = []
    for i in range(len(positions)):
        center_x, center_y, angle_degrees = positions[i]
        vertices = hexagon_vertices_numba(center_x, center_y, angle_degrees)
        inner_vertices.append(vertices)

    # Find outer hexagon side length
    side_length = compute_outer_hexagon_side_length_fast(positions)

    # Check containment and overlaps efficiently using spatial hashing
    total_penalty = 0
    num_inner = len(inner_vertices)

    # Outer hexagon vertices (centered at origin with given angle)
    outer_hex_vertices = hexagon_vertices_numba(outer_hex_center[0], outer_hex_center[1], outer_hex_angle)

    # Build spatial hash for fast neighbor lookup
    spatial_grid = SpatialHashGrid(cell_size=3.0)
    for i in range(num_inner):
        center_x, center_y, _ = positions[i]
        spatial_grid.insert(i, center_x, center_y)

    # Check containment
    for i in range(num_inner):
        if not check_containment_fast(inner_vertices[i], outer_hex_vertices):
            # Large penalty for containment violations
            total_penalty += 1e6

    # Check overlaps using spatial hash for efficiency
    for i in range(num_inner):
        center_x, center_y, _ = positions[i]
        neighbors = spatial_grid.get_neighbors(center_x, center_y)

        for j in neighbors:
            if i >= j:  # Avoid duplicate checks and self-checking
                continue
            if check_overlap_fast(inner_vertices[i], inner_vertices[j]):
                # Penalize overlap more heavily
                total_penalty += 1e5

    # Objective: maximize 1/side_length
    # So minimize negative log of side_length plus penalties
    if side_length < 1e-6:
        fitness = -1e10 - total_penalty
    else:
        fitness = -math.log(side_length) - total_penalty

    return fitness, side_length
