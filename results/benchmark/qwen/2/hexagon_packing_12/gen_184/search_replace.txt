    def evaluate_fitness(self, params: np.ndarray) -> float:
        """Evaluate the fitness of a given configuration with adaptive penalty weighting"""
        # Extract parameters
        hexagons = []
        idx = 0
        for i in range(12):
            hexagons.append((params[idx], params[idx+1], params[idx+2]))
            idx += 3

        outer_side_length = params[-1]

        # Validate constraints
        validation_result = self.validator.validate_all_constraints(hexagons, outer_side_length)

        # Calculate adaptive penalty based on violation severity
        containment_violations = len(validation_result['containment_violations'])
        overlap_violations = len(validation_result['overlap_violations'])

        # Adaptive penalty weights - higher penalty for containment violations
        containment_weight = 1e7 if containment_violations > 0 else 1e6
        overlap_weight = 1e6 if overlap_violations > 0 else 1e5

        penalty = containment_violations * containment_weight + overlap_violations * overlap_weight

        # Inverse side length (negative because we minimize)
        # We want to maximize inverse side length, so minimize negative value
        objective_value = -1.0 / outer_side_length + penalty

        return objective_value
