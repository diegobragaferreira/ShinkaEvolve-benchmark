    for i in range(num_inner):

    containment_violations = 0

                # Penalize overlap violations
                overlap_violations += 1
                total_penalty += 1e6  # High penalty for overlaps

    # Adaptive penalty adjustment based on violation severity
    # The idea is to make the penalty scheme more sensitive to serious violations
    if containment_violations > 0:
        # If there are containment violations, make penalty even stronger
        total_penalty *= (1 + containment_violations * 0.5)

    if overlap_violations > 0:
        # Apply some scaling to overlap penalties
        total_penalty *= (1 + overlap_violations * 0.2)
