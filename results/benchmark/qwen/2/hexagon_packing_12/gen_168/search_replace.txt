@jit(nopython=True)
def create_spatial_hash(hex_data, cell_size=3.0):
    """Create a spatial hash for efficient neighbor lookup"""
    hash_grid = {}
    for i in range(len(hex_data)):
        x, y, _ = hex_data[i]
        # Calculate grid cell coordinates
        cell_x = int(x / cell_size)
        cell_y = int(y / cell_size)

        # Create key for this cell
        cell_key = (cell_x, cell_y)

        # Add hexagon to this cell
        if cell_key not in hash_grid:
            hash_grid[cell_key] = []
        hash_grid[cell_key].append(i)

    return hash_grid

@jit(nopython=True)
def get_neighbors(hash_grid, hex_id, hex_data, cell_size=3.0):
    """Get nearby hexagons from spatial hash"""
    x, y, _ = hex_data[hex_id]

    # Check neighboring cells for potential overlaps
    neighbors = []
    cell_x = int(x / cell_size)
    cell_y = int(y / cell_size)

    # Check 3x3 surrounding cells
    for dx in range(-1, 2):
        for dy in range(-1, 2):
            cell_key = (cell_x + dx, cell_y + dy)
            if cell_key in hash_grid:
                for neighbor_id in hash_grid[cell_key]:
                    if neighbor_id != hex_id:
                        neighbors.append(neighbor_id)

    return neighbors
