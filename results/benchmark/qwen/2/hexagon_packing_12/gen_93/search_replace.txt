<DESCRIPTION>
Adding spatial hashing optimization to significantly improve the performance of overlap detection in the hexagon packing problem. This addresses the major computational bottleneck in evaluating constraint violations, especially for the 12 hexagon case where we need to check O(n²) pairwise overlaps. The spatial hash will reduce overlap checking from O(n²) to near O(n) average case by grouping hexagons into grid cells and only checking neighbors.

This optimization leverages the fact that hexagons that are far apart cannot overlap, so we can avoid expensive polygon intersection tests by using spatial indexing.

@njit
def check_overlap_hexagons_spatial(x1, y1, angle1, x2, y2, angle2, spatial_hash=None, grid_size=2.0):
    """Efficiently check overlap using spatial hashing"""
    # Simplified check for quick rejection
    dist_sq = (x1 - x2)**2 + (y1 - y2)**2
    # If centers are too far apart, no overlap
    if dist_sq > 4.0:  # 2^2, max possible distance for touching hexagons
        return False

    # If spatial hash provided, check cells
    if spatial_hash is not None:
        # Simple bounding box check
        cell1_x, cell1_y = get_grid_coords(x1, y1, grid_size)
        cell2_x, cell2_y = get_grid_coords(x2, y2, grid_size)

        # If they're in different cells, not necessarily overlapping
        if abs(cell1_x - cell2_x) > 1 or abs(cell1_y - cell2_y) > 1:
            # Still check if they might be close enough
            if dist_sq > 4.0:  # 2 unit distance threshold
                return False

    # For actual overlap check, fall back to full polygon intersection
    vertices1 = generate_hexagon_vertices(x1, y1, angle1, 1.0)
    vertices2 = generate_hexagon_vertices(x2, y2, angle2, 1.0)

    # Create polygons and check intersection
    poly1 = Polygon(vertices1)
    poly2 = Polygon(vertices2)

    # If intersection exists, they overlap
    return poly1.intersects(poly2)

@njit
def build_spatial_hash(hexagon_positions, grid_size=2.0):
    """Build spatial hash for fast overlap detection"""
    hash_table = defaultdict(list)
    for i, (x, y, _) in enumerate(hexagon_positions):
        cell_x, cell_y = get_grid_coords(x, y, grid_size)
        hash_table[(cell_x, cell_y)].append(i)
    return hash_table
