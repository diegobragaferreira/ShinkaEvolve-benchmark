def evaluate_configuration(config, outer_side_length):
    """Evaluate a configuration of 12 hexagons."""
    # Parse configuration into 12 hexagons (x, y, angle)
    hexagons = config.reshape(12, 3)

    # Get vertices for all hexagons
    hexagon_vertices_list = []
    for i in range(12):
        x, y, angle = hexagons[i]
        vertices = hexagon_vertices(x, y, angle)
        hexagon_vertices_list.append(vertices)

    # Initialize penalty
    total_penalty = 0.0

    # Fast containment check first
    containment_violations = 0
    for vertices in hexagon_vertices_list:
        # Check distance from center to each vertex
        for vertex in vertices:
            x, y = vertex
            distance = math.sqrt(x*x + y*y)
            # For outer hexagon with side length R, apothem is R * sqrt(3)/2
            outer_apothem = outer_side_length * math.sqrt(3) / 2
            if distance > outer_apothem:
                containment_violations += 1
                # Adaptive penalty based on how far outside
                excess_distance = distance - outer_apothem
                total_penalty += 1000.0 * (excess_distance + 0.1)**2

    # Fast overlap check using spatial hash
    overlap_violations = 0
    if not check_overlap_spatial_hash(hexagon_vertices_list):
        overlap_violations += 1
        total_penalty += 100000.0  # Large penalty for overlaps

    # Final precise validation
    if not check_overlap_shapely(hexagon_vertices_list):
        overlap_violations += 1
        total_penalty += 100000.0  # Large penalty for overlaps

    if containment_violations > 0 or overlap_violations > 0:
        return False, total_penalty  # Invalid configuration

    return True, total_penalty  # Valid configuration

    """Perform enhanced local refinement to improve configuration."""
    best_config = config.copy()
    best_valid, best_penalty = evaluate_configuration(best_config, outer_side_length)

    if not best_valid:
        return config

    # Multiple passes of local optimization
    for pass_num in range(3):
        # Perform local search with diminishing step sizes
        step_size = 0.1 / (pass_num + 1)

        for iteration in range(max_iter // 3):
            # Try modifying multiple hexagons in each iteration
