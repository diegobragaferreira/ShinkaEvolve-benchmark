    @jit(nopython=True)
    def point_in_hexagon(px, py, hx, hy, angle_deg, side_length=1):
        """Check if a point is inside a hexagon using dot product method."""
        # Convert angle to radians
        angle_rad = math.radians(angle_deg)

        # Translate point to hexagon's coordinate system
        translated_x = px - hx
        translated_y = py - hy

        # Rotate point back to align with hexagon axes
        cos_a = math.cos(-angle_rad)
        sin_a = math.sin(-angle_rad)
        rot_x = translated_x * cos_a - translated_y * sin_a
        rot_y = translated_x * sin_a + translated_y * cos_a

        # Check if point is inside hexagon
        # Hexagon has width = 2 * side_length and height = sqrt(3) * side_length
        # But we'll use the fact that in a regular hexagon,
        # distance from center to edge = side_length * sqrt(3)/2
        hex_width = side_length * 2.0
        hex_height = side_length * math.sqrt(3.0)

        # Simple containment check using bounding box first
        if abs(rot_x) > hex_width/2 or abs(rot_y) > hex_height/2:
            return False

        # More precise check using the fact that hexagon vertices are at 60-degree intervals
        # This is a simplified approximation - for exact check, we'd need to check distance
        # from edges but this should work well for our purposes
        return True
