def hexagon_vertices(center_x, center_y, angle_deg, side_length=1):
    """Return vertices of a regular hexagon with given center, rotation, and side length."""
    angle_rad = np.radians(angle_deg)
    # Vertices of a regular hexagon with side length 1, centered at origin
    base_vertices = np.array([
        [1, 0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1, 0],
        [-0.5, -np.sqrt(3)/2],

    ])

@jit(nopython=True)
def check_overlap_jit(hex1_vertices, hex2_vertices):
    """Check if two hexagon vertices overlap using separating axis theorem."""
    # Get all edges of both hexagons
    def get_edges(vertices):
        edges = np.empty((len(vertices), 2))
        for i in range(len(vertices)):
            edges[i] = vertices[i] - vertices[(i+1)%len(vertices)]
        return edges

    edges1 = get_edges(hex1_vertices)
    edges2 = get_edges(hex2_vertices)

    # Project both hexagons onto each edge direction
    all_axes = np.vstack([edges1, edges2])
    for i in range(len(all_axes)):
        axis = all_axes[i]
        # Normalize axis
        norm = np.sqrt(axis[0]**2 + axis[1]**2)
        if norm > 1e-10:
            norm_axis = axis / norm
        else:
            continue

        # Project both polygons onto this axis
        proj1 = np.empty(len(hex1_vertices))
        proj2 = np.empty(len(hex2_vertices))
        for j in range(len(hex1_vertices)):
            proj1[j] = hex1_vertices[j, 0] * norm_axis[0] + hex1_vertices[j, 1] * norm_axis[1]
        for j in range(len(hex2_vertices)):
            proj2[j] = hex2_vertices[j, 0] * norm_axis[0] + hex2_vertices[j, 1] * norm_axis[1]

        # Check for overlap
        min1, max1 = np.min(proj1), np.max(proj1)
        min2, max2 = np.min(proj2), np.max(proj2)

        # If no overlap, then they don't intersect
        if max1 < min2 or max2 < min1:
            return False

    return True
