import numpy as np

class HexagonGeometry:
    """Handles all geometric computations for hexagon operations"""
    
    @staticmethod
    @jit(nopython=True)
    def vertices(x, y, angle_deg, side_length=1):
        """Generate vertices of a regular hexagon"""
        angle_rad = np.radians(angle_deg)
        angles = np.arange(0, 6) * np.pi / 3
        vertices = np.zeros((6, 2))
        for i in range(6):
            vertices[i, 0] = x + side_length * np.cos(angles[i] + angle_rad)
            vertices[i, 1] = y + side_length * np.sin(angles[i] + angle_rad)
        return vertices

    @staticmethod
    @jit(nopython=True)
    def point_in_polygon(point, polygon):
        """Check if point is inside polygon using ray casting"""
        x, y = point
        n = len(polygon)
        inside = False
        p1x, p1y = polygon[0]
        for i in range(1, n + 1):
            p2x, p2y = polygon[i % n]
            if y > min(p1y, p2y):
                if y <= max(p1y, p2y):
                    if x <= max(p1x, p2x):
                        if p1y != p2y:
                            xinters = (y - p1y) * (p2x - p1x) / (p2y - p1y) + p1x
                        if p1x == p2x or x <= xinters:
                            inside = not inside
            p1x, p1y = p2x, p2y
        return inside

    @staticmethod
    @jit(nopython=True)
    def distance_point_to_line_segment(point, line_start, line_end):
        """Calculate distance from point to line segment"""
        A = point[0] - line_start[0]
        B = point[1] - line_start[1]
        C = line_end[0] - line_start[0]
        D = line_end[1] - line_start[1]
        
        dot = A*C + B*D
        len_sq = C*C + D*D
        if len_sq == 0:
            return np.sqrt(A*A + B*B)
        param = dot / len_sq
        param = max(0, min(1, param))
        xx = line_start[0] + param * C
        yy = line_start[1] + param * D
        dx = point[0] - xx
        dy = point[1] - yy
        return np.sqrt(dx*dx + dy*dy)

    @staticmethod
    @jit(nopython=True)
    def point_to_hexagon_distance(point, hex_vertices):
        """Calculate minimum distance from point to hexagon boundary"""
        min_dist = np.inf
        for i in range(6):
            p1 = hex_vertices[i]
            p2 = hex_vertices[(i+1)%6]
            dist = HexagonGeometry.distance_point_to_line_segment(point, p1, p2)
            min_dist = min(min_dist, dist)
        return min_dist

    @staticmethod
    @jit(nopython=True)
    def check_overlap(hex1_vertices, hex2_vertices):
        """Check if two hexagons overlap"""
        # Check if any vertex of hex1 is inside hex2
        for v in hex1_vertices:
            if HexagonGeometry.point_in_polygon(v, hex2_vertices):
                return True
        # Check if any vertex of hex2 is inside hex1  
        for v in hex2_vertices:
            if HexagonGeometry.point_in_polygon(v, hex1_vertices):
                return True
        return False

class SpatialHashGrid:
    """Implements spatial hashing for efficient overlap detection"""
    
    def __init__(self, cell_size=2.0):
        self.cell_size = cell_size
        self.hash_grid = defaultdict(list)
    
    def clear(self):
        """Clear the hash grid"""
        self.hash_grid.clear()
    
    def add_hexagon(self, index, vertices):
        """Add a hexagon to the spatial hash"""
        # Get bounding box of hexagon
        min_x = min(v[0] for v in vertices)
        max_x = max(v[0] for v in vertices)
        min_y = min(v[1] for v in vertices)
        max_y = max(v[1] for v in vertices)
        
        # Add to all relevant cells
        start_col = int(min_x // self.cell_size)
        end_col = int(max_x // self.cell_size) + 1
        start_row = int(min_y // self.cell_size)
        end_row = int(max_y // self.cell_size) + 1
        
        for col in range(start_col, end_col + 1):
            for row in range(start_row, end_row + 1):
                self.hash_grid[(col, row)].append(index)
    
    def get_overlapping_indices(self, hex_index, vertices):
        """Get indices of potentially overlapping hexagons"""
        overlapping = set()
        # Get bounding box of hexagon
        min_x = min(v[0] for v in vertices)
        max_x = max(v[0] for v in vertices)
        min_y = min(v[1] for v in vertices)
        max_y = max(v[1] for v in vertices)
        
        # Check all relevant cells
        start_col = int(min_x // self.cell_size)
        end_col = int(max_x // self.cell_size) + 1
        start_row = int(min_y // self.cell_size)
        end_row = int(max_y // self.cell_size) + 1
        
        for col in range(start_col, end_col + 1):
            for row in range(start_row, end_row + 1):
                if (col, row) in self.hash_grid:
                    for idx in self.hash_grid[(col, row)]:
                        if idx != hex_index:
                            overlapping.add(idx)
        return overlapping

class HexagonEvaluator:
    """Handles fitness evaluation and constraint checking"""
    
    def __init__(self, outer_radius=None):
        self.outer_radius = outer_radius or 10.0
    
    def compute_outer_radius(self, hex_data):
        """Compute the minimum radius required to contain all hexagons"""
        max_dist = 0
        for i in range(len(hex_data)):
            x, y, angle = hex_data[i]
            vertices = HexagonGeometry.vertices(x, y, angle)
            for vx, vy in vertices:
                dist = np.sqrt(vx*vx + vy*vy)
                max_dist = max(max_dist, dist)
        return max_dist + 1.0  # Add small margin
    
    def calculate_penalty(self, hex_data):
        """Calculate total penalty for all hexagons"""
        n = len(hex_data)
        
        # Precompute vertices for all hexagons
        hex_vertices_list = [HexagonGeometry.vertices(hex_data[i][0], hex_data[i][1], hex_data[i][2]) for i in range(n)]
        
        # Create spatial hash for overlap detection
        spatial_hash = SpatialHashGrid()
        for i, vertices in enumerate(hex_vertices_list):
            spatial_hash.add_hexagon(i, vertices)
        
        total_penalty = 0
        
        # Check containment for each hexagon
        for i in range(n):
            vertices = hex_vertices_list[i]
            outer_hex_vertices = HexagonGeometry.vertices(0, 0, 0, self.outer_radius)
            
            # Check containment penalty
            for vx, vy in vertices:
                point = np.array([vx, vy])
                if not HexagonGeometry.point_in_polygon(point, outer_hex_vertices):
                    dist = np.sqrt(vx*vx + vy*vy)
                    penalty = (dist - self.outer_radius + 0.5)**2
                    total_penalty += penalty
        
        # Check overlaps using spatial hashing
        for i in range(n):
            # Get potentially overlapping hexagons
            overlapping_indices = spatial_hash.get_overlapping_indices(i, hex_vertices_list[i])
            
            # Check actual overlaps
            for j in overlapping_indices:
                if i < j:  # Avoid double counting
                    vertices1 = hex_vertices_list[i]
                    vertices2 = hex_vertices_list[j]
                    
                    if HexagonGeometry.check_overlap(vertices1, vertices2):
                        total_penalty += 1000000  # Large penalty for overlaps
        
        return total_penalty

class PopulationManager:
    """Manages the evolutionary population"""
    
    @staticmethod
    def create_initial_population(population_size):
        """Create diverse initial population"""
        population = []
        
        # Strategy 1: Concentric ring configurations
        for _ in range(population_size // 3):
            positions = []
            # Center hexagon
            positions.append([0.0, 0.0, 0.0])
            # First ring (6 hexagons)
            for i in range(6):
                angle = i * 60
                radius = 1.8  # Tighter than previous
                x = radius * np.cos(np.radians(angle))
                y = radius * np.sin(np.radians(angle))
                positions.append([x, y, random.uniform(0, 360)])
            # Second ring (5 hexagons)
            for i in range(5):
                angle = i * 72
                radius = 3.2  # Slightly smaller than previous
                x = radius * np.cos(np.radians(angle))
                y = radius * np.sin(np.radians(angle))
                positions.append([x, y, random.uniform(0, 360)])
            population.append(np.array(positions))
        
        # Strategy 2: Grid-like configurations
        for _ in range(population_size // 3):
            positions = []
            # Central hexagon
            positions.append([0.0, 0.0, 0.0])
            # Grid layout
            positions.append([-2.0, 0.0, 0.0])   # Left
            positions.append([2.0, 0.0, 0.0])    # Right
            positions.append([0.0, 2.0, 0.0])    # Top
            positions.append([0.0, -2.0, 0.0])   # Bottom
            positions.append([-1.5, 1.5, 0.0])   # Top-left
            positions.append([1.5, 1.5, 0.0])    # Top-right
            positions.append([-1.5, -1.5, 0.0])  # Bottom-left
            positions.append([1.5, -1.5, 0.0])   # Bottom-right
            positions.append([-2.5, 2.5, 0.0])   # Far top-left
            positions.append([2.5, 2.5, 0.0])    # Far top-right
            positions.append([-2.5, -2.5, 0.0])  # Far bottom-left
            positions.append([2.5, -2.5, 0.0])   # Far bottom-right
            population.append(np.array(positions))
        
        # Strategy 3: Random configurations with constraints
        for _ in range(population_size - 2 * (population_size // 3)):
            positions = []
            # Center hexagon
            positions.append([0.0, 0.0, 0.0])
            # Random placement with some constraints
            for i in range(11):
                # Try to avoid very close placements to center
                radius = random.uniform(1.5, 4.0)
                angle = random.uniform(0, 360)
                x = radius * np.cos(np.radians(angle))
                y = radius * np.sin(np.radians(angle))
                positions.append([x, y, random.uniform(0, 360)])
            population.append(np.array(positions))
        
        return population
    
    @staticmethod
    def mutate_individual(individual, mutation_rate=0.3, max_mutation=0.5):
        """Mutate individual hexagon positions and rotations"""
        mutated = individual.copy()
        
        # Mutate positions and rotations
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                # Small random change to position
                mutated[i][0] += random.uniform(-max_mutation, max_mutation)
                mutated[i][1] += random.uniform(-max_mutation, max_mutation)
                # Small random change to rotation
                mutated[i][2] += random.uniform(-max_mutation * 10, max_mutation * 10)
                # Normalize rotation angle to [0, 360)
                mutated[i][2] = mutated[i][2] % 360
        
        return mutated

class EvolutionaryOptimizer:
    """Main evolutionary optimization class"""
    
    def __init__(self, population_size=50, max_generations=100):
        self.population_size = population_size
        self.max_generations = max_generations
    
    def evaluate_population_parallel(self, population):
        """Evaluate fitness of population in parallel"""
        def evaluate_single(individual):
            evaluator = HexagonEvaluator()
            outer_radius = evaluator.compute_outer_radius(individual)
            evaluator.outer_radius = outer_radius
            penalty = evaluator.calculate_penalty(individual)
            # Fitness is negative penalty plus inverse of radius (to maximize 1/R)
            fitness = penalty - 1.0 / (outer_radius + 1e-6)
            return fitness
        
        results = Parallel(n_jobs=-1)(
            delayed(evaluate_single)(individual) for individual in population
        )
        return results
    
    def tournament_selection(self, fitness_scores, tournament_size=3):
        """Perform tournament selection"""
        tournament_indices = random.sample(range(len(fitness_scores)), tournament_size)
        tournament_fitness = [fitness_scores[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        return winner_idx
    
    def crossover(self, parent1, parent2):
        """Single-point crossover between two individuals"""
        crossover_point = random.randint(1, len(parent1) - 1)
        child1 = np.vstack([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.vstack([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def optimize(self):
        """Run the evolutionary optimization"""
        # Create initial population
        population = PopulationManager.create_initial_population(self.population_size)
        
        best_fitness = float('inf')
        best_individual = None
        
        # Evolutionary loop
        for generation in range(self.max_generations):
            # Evaluate fitness of entire population
            fitness_scores = self.evaluate_population_parallel(population)
            
            # Find best individual in this generation
            best_idx = np.argmin(fitness_scores)
            if fitness_scores[best_idx] < best_fitness:
                best_fitness = fitness_scores[best_idx]
                best_individual = population[best_idx].copy()
            
            # Create new population
            new_population = []
            
            # Keep best individual (elitism)
            new_population.append(best_individual)
            
            # Generate rest via selection, crossover, and mutation
            while len(new_population) < self.population_size:
                # Tournament selection
                winner_idx = self.tournament_selection(fitness_scores)
                
                # Select another parent
                other_idx = self.tournament_selection(fitness_scores)
                
                # Crossover
                child1, child2 = self.crossover(population[winner_idx], population[other_idx])
                
                # Mutate children
                child1 = PopulationManager.mutate_individual(child1)
                child2 = PopulationManager.mutate_individual(child2)
                
                new_population.extend([child1, child2])
            
            # Trim population to exact size
            population = new_population[:self.population_size]
        
        return best_individual

def hexagon_packing_12():
    """
    Constructs a packing of 12 disjoint unit regular hexagons inside a larger regular hexagon, maximizing 1/outer_hex_side_length.
    Returns
        inner_hex_data: np.ndarray of shape (12,3), where each row is of the form (x, y, angle_degrees) containing the (x,y) coordinates and angle_degree of the respective inner hexagon.
        outer_hex_data: np.ndarray of shape (3,) of form (x,y,angle_degree) containing the (x,y) coordinates and angle_degree of the outer hexagon.
        outer_hex_side_length: float representing the side length of the outer hexagon.
    """
    start_time = time.time()
    
    try:
        # Run evolutionary optimization
        optimizer = EvolutionaryOptimizer(population_size=50, max_generations=100)
        inner_hex_data = optimizer.optimize()
    except Exception as e:
        warnings.warn(f"Fallback due to optimization error: {e}")
        # Fallback to a simple symmetric configuration
        inner_hex_data = np.array([
            [0, 0, 0],  # center
            [-2.5, 0, 0],  # left
            [2.5, 0, 0],  # right
            [-1.25, 2.17, 0],  # top-left
            [1.25, 2.17, 0],  # top-right
            [-1.25, -2.17, 0],  # bottom-left
            [1.25, -2.17, 0],  # bottom-right
            [-3.75, 2.17, 0],  # far top-left
            [3.75, 2.17, 0],  # far top-right
            [-3.75, -2.17, 0],  # far bottom-left
            [3.75, -2.17, 0],  # far bottom-right,
            [0, -4, 0],  # far bottom-center
        ])
    
    # Compute final outer hexagon size
    outer_hex_side_length = HexagonEvaluator().compute_outer_radius(inner_hex_data)
    
    # Create outer hexagon data (centered, no rotation)
    outer_hex_data = np.array([0, 0, 0])  # centered at origin, no rotation
    
    # Final validation
    final_penalty = HexagonEvaluator(outer_hex_side_length).calculate_penalty(inner_hex_data)
    if final_penalty > 10000:  # If there are major violations
        # Fallback to simple configuration
        inner_hex_data = np.array([
            [0, 0, 0],  # center
            [-2.5, 0, 0],  # left
            [2.5, 0, 0],  # right
            [-1.25, 2.17, 0],  # top-left
            [1.25, 2.17, 0],  # top-right
            [-1.25, -2.17, 0],  # bottom-left
            [1.25, -2.17, 0],  # bottom-right
            [-3.75, 2.17, 0],  # far top-left
            [3.75, 2.17, 0],  # far top-right
            [-3.75, -2.17, 0],  # far bottom-left
            [3.75, -2.17, 0],  # far bottom-right,
            [0, -4, 0],  # far bottom-center
        ])
        outer_hex_side_length = 8
    
    end_time = time.time()
    
    return inner_hex_data, outer_hex_data, outer_hex_side_length
