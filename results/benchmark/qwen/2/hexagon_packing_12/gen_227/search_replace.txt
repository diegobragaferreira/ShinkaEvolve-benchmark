<DESCRIPTION>
This change implements Numba JIT compilation for key geometric operations to significantly improve performance. The existing code already uses Numba for some functions, but we can enhance it by applying the @njit decorator to all geometric computation functions that are called frequently during optimization cycles. This includes functions like `generate_hexagon_vertices`, `check_containment_inner_to_outer`, `check_overlap_hexagons`, and `calculate_penalty`. The performance gains will be substantial during the high-frequency constraint validation required in evolutionary optimization, especially for the 12-hexagon system where thousands of geometric checks occur per evaluation.

The change focuses on these specific functions:
1. `generate_hexagon_vertices` - core hexagon vertex generation
2. `check_containment_inner_to_outer` - containment checking
3. `check_overlap_hexagons` - overlap detection between hexagons
4. `calculate_penalty` - penalty calculation for constraint violations

These are the most computationally intensive functions in the constraint validation pipeline and benefit greatly from JIT compilation.

@njit
def get_grid_coords(x, y, cell_size=2.0):
    """Get grid coordinates for a point"""
    return int(x / cell_size), int(y / cell_size)

@njit
def generate_hexagon_vertices(x, y, angle_degrees, side_length=1):
    """Generate vertices of a regular hexagon given center, rotation, and side length"""
    angle_rad = np.radians(angle_degrees)
    vertices = np.empty((6, 2))
    for i in range(6):
        theta = angle_rad + i * np.pi / 3
        vertices[i, 0] = x + side_length * np.cos(theta)
        vertices[i, 1] = y + side_length * np.sin(theta)
    return vertices

@njit
def create_hexagon_polygon(x, y, angle_degrees, side_length=1):
    """Create shapely polygon representation of hexagon"""
    vertices = generate_hexagon_vertices(x, y, angle_degrees, side_length)
    return Polygon(vertices)

@njit
def point_in_hexagon(point_x, point_y, hex_center_x, hex_center_y, hex_angle, hex_side_length):
    """Check if a point is inside a hexagon using geometric properties"""
    # Transform point to hexagon's coordinate system
    dx = point_x - hex_center_x
    dy = point_y - hex_center_y
    angle_rad = np.radians(hex_angle)

    # Rotate point to align with hexagon axes
    rotated_x = dx * np.cos(-angle_rad) - dy * np.sin(-angle_rad)
    rotated_y = dx * np.sin(-angle_rad) + dy * np.cos(-angle_rad)

    # For unit hexagons, check against the boundary
    # Maximum distance from center in direction of axes
    max_dist = hex_side_length
    if abs(rotated_x) > max_dist or abs(rotated_y) > max_dist:
        return False

    # Check against slanted edges
    # Hexagon is bounded by lines forming 60 degree angles
    # Simplified check for points near center
    if abs(rotated_x) <= max_dist and abs(rotated_y) <= max_dist:
        return True
    return False

@njit
def check_containment_inner_to_outer(inner_x, inner_y, inner_angle, outer_x, outer_y, outer_angle, outer_side_length):
    """Check if inner hexagon is fully contained within outer hexagon"""
    inner_vertices = generate_hexagon_vertices(inner_x, inner_y, inner_angle, 1.0)

    # Create outer hexagon vertices
    outer_vertices = generate_hexagon_vertices(outer_x, outer_y, outer_angle, outer_side_length)
    outer_polygon = Polygon(outer_vertices)

    # Check if all vertices of inner hexagon are inside outer hexagon
    for i in range(6):
        if not outer_polygon.contains(Point(inner_vertices[i, 0], inner_vertices[i, 1])):
            return False

    return True

@njit
def check_overlap_hexagons_spatial(x1, y1, angle1, x2, y2, angle2, spatial_hash=None, grid_size=2.0):
    """Efficiently check overlap using spatial hashing"""
    # Simplified check for quick rejection
    dist_sq = (x1 - x2)**2 + (y1 - y2)**2
    # If centers are too far apart, no overlap
    if dist_sq > 4.0:  # 2^2, max possible distance for touching hexagons
        return False

    # If spatial hash provided, check cells
    if spatial_hash is not None:
        # Simple bounding box check
        cell1_x, cell1_y = get_grid_coords(x1, y1, grid_size)
        cell2_x, cell2_y = get_grid_coords(x2, y2, grid_size)

        # If they're in different cells, not necessarily overlapping
        if abs(cell1_x - cell2_x) > 1 or abs(cell1_y - cell2_y) > 1:
            # Still check if they might be close enough
            if dist_sq > 4.0:  # 2 unit distance threshold
                return False

    # For actual overlap check, fall back to full polygon intersection
    vertices1 = generate_hexagon_vertices(x1, y1, angle1, 1.0)
    vertices2 = generate_hexagon_vertices(x2, y2, angle2, 1.0)

    # Create polygons and check intersection
    poly1 = Polygon(vertices1)
    poly2 = Polygon(vertices2)

    # If intersection exists, they overlap
    return poly1.intersects(poly2)

@njit
def build_spatial_hash(hexagon_positions, grid_size=2.0):

        hash_table[(cell_x, cell_y)].append(i)
    return hash_table

@njit
def calculate_penalty(packed_hexagons, outer_hexagon_params, penalty_weights=(1e6, 1e5)):
    """Calculate penalty based on constraint violations with adaptive weights"""
    penalty = 0.0
    n = len(packed_hexagons)

    outer_x, outer_y, outer_angle, outer_side_length = outer_hexagon_params

    # Check containment penalties (higher priority)
    containment_penalty_weight = penalty_weights[0]
    overlap_penalty_weight = penalty_weights[1]

    for i in range(n):
        if not check_containment_inner_to_outer(
            packed_hexagons[i][0], packed_hexagons[i][1], packed_hexagons[i][2],
            outer_x, outer_y, outer_angle, outer_side_length
        ):
            penalty += containment_penalty_weight

    # Check overlap penalties (lower priority)
    for i in range(n):
        for j in range(i+1, n):
            if check_overlap_hexagons(
                packed_hexagons[i][0], packed_hexagons[i][1], packed_hexagons[i][2],
                packed_hexagons[j][0], packed_hexagons[j][1], packed_hexagons[j][2]
            ):
                penalty += overlap_penalty_weight

    return penalty
