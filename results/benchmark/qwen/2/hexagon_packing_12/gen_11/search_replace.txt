import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import math

def hexagon_vertices(center, side_length, angle_degrees):
    """Generate vertices of a regular hexagon given center, side length, and rotation."""
    angle_rad = math.radians(angle_degrees)
    vertices = []
    for i in range(6):
        theta = angle_rad + i * math.pi / 3
        x = center[0] + side_length * math.cos(theta)
        y = center[1] + side_length * math.sin(theta)
        vertices.append([x, y])
    return np.array(vertices)

def check_containment(hex_center, side_length, angle_degrees, outer_hex_radius):
    """Check if a hexagon is fully contained within the outer hexagon."""
    vertices = hexagon_vertices(hex_center, side_length, angle_degrees)
    # Check that all vertices are within the outer hexagon
    outer_vertices = hexagon_vertices([0, 0], outer_hex_radius, 0)
    # Simplified containment check using distance from center
    max_dist = np.max(np.sqrt(np.sum((vertices - np.array([0, 0]))**2, axis=1)))
    return max_dist <= outer_hex_radius

def check_overlap(hex1_center, hex1_angle, hex2_center, hex2_angle):
    """Check if two hexagons overlap using vertex inclusion test."""
    side_length = 1.0
    hex1_vertices = hexagon_vertices(hex1_center, side_length, hex1_angle)
    hex2_vertices = hexagon_vertices(hex2_center, side_length, hex2_angle)

    # Simple distance-based overlap check for efficiency
    dist = np.linalg.norm(np.array(hex1_center) - np.array(hex2_center))
    # Two unit hexagons overlap if their centers are closer than 2 units
    return dist < 2.0

def objective_function(params):
    """Objective function to maximize inverse of outer hexagon side length."""
    # params: [R, x1, y1, theta1, x2, y2, theta2, ..., x12, y12, theta12]
    R = params[0]
    total_vars = len(params) - 1
    num_hex = (total_vars // 3)

    # Check containment and overlap constraints
    penalty = 0

    # Check if all inner hexagons are contained
    for i in range(num_hex):
        idx = 1 + i * 3
        center = [params[idx], params[idx+1]]
        angle = params[idx+2]
        if not check_containment(center, 1.0, angle, R):
            penalty += 1000

    # Check for overlaps between hexagons
    for i in range(num_hex):
        for j in range(i+1, num_hex):
            idx_i = 1 + i * 3
            idx_j = 1 + j * 3
            center_i = [params[idx_i], params[idx_i+1]]
            angle_i = params[idx_i+2]
            center_j = [params[idx_j], params[idx_j+1]]
            angle_j = params[idx_j+2]

            if check_overlap(center_i, angle_i, center_j, angle_j):
                penalty += 1000

    # Return negative because we want to maximize 1/R, which means minimize -1/R
    return -1.0 / R + penalty
