def hexagon_vertices(center_x, center_y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given center, angle, and side length."""
    angle_rad = math.radians(angle_deg)
    vertices = []
    for i in range(6):
        angle = angle_rad + i * math.pi / 3
        x = center_x + side_length * math.cos(angle)
        y = center_y + side_length * math.sin(angle)
        vertices.append((x, y))
    return vertices

def check_containment_numba(hexagon_vertices_list, outer_side_length):
    """Fast containment check using geometric properties."""
    # For a hexagon with side length 1, the distance from center to vertex is 1
    # For outer hexagon with side length R, distance from center to vertex is R
    # So we just need to check that all inner hexagon centers are within outer hexagon

    # Simplified containment check: verify that all hexagon centers are within outer hexagon
    # Using distance from center check for speed
    outer_radius = outer_side_length  # Distance from center to vertex of outer hexagon

    for vertices in hexagon_vertices_list:
        # Use first vertex as approximation of center (this is okay for our purposes)
        center_x = vertices[0][0]
        center_y = vertices[0][1]
        distance = math.sqrt(center_x*center_x + center_y*center_y)
        if distance > outer_radius:
            return False
    return True

def check_overlap(hexagon_vertices_list):
    """Check if any hexagons overlap."""
    polygons = [Polygon(vertices) for vertices in hexagon_vertices_list]
    try:
        union = unary_union(polygons)
        total_area = sum(polygon.area for polygon in polygons)
        union_area = union.area
        # If areas match, no overlap
        return abs(total_area - union_area) < 1e-10
    except:
        # Fallback for complex cases
        for i in range(len(polygons)):
            for j in range(i+1, len(polygons)):
                if polygons[i].intersects(polygons[j]):
                    return False
        return True

@jit(nopython=True)
def check_overlap_simple_numba(hexagon_vertices_list):
    """Simple overlap check using bounding boxes for speed."""
    n_hexagons = len(hexagon_vertices_list)
    for i in range(n_hexagons):
        vertices_i = hexagon_vertices_list[i]
        # Get bounding box for hexagon i
        min_x = vertices_i[0][0]
        max_x = vertices_i[0][0]
        min_y = vertices_i[0][1]
        max_y = vertices_i[0][1]

        for j in range(1, len(vertices_i)):
            x, y = vertices_i[j]
            if x < min_x: min_x = x
            if x > max_x: max_x = x
            if y < min_y: min_y = y
            if y > max_y: max_y = y

        # Compare with all other hexagons
        for k in range(i+1, n_hexagons):
            vertices_k = hexagon_vertices_list[k]
            # Get bounding box for hexagon k
            min_x_k = vertices_k[0][0]
            max_x_k = vertices_k[0][0]
            min_y_k = vertices_k[0][1]
            max_y_k = vertices_k[0][1]

            for l in range(1, len(vertices_k)):
                x, y = vertices_k[l]
                if x < min_x_k: min_x_k = x
                if x > max_x_k: max_x_k = x
                if y < min_y_k: min_y_k = y
                if y > max_y_k: max_y_k = y

            # Check if bounding boxes intersect
            if max_x >= min_x_k and min_x <= max_x_k and max_y >= min_y_k and min_y <= max_y_k:
                return False  # Overlap detected

    return True  # No overlap
