@jit(nopython=True)
def hexagon_vertices_numba(center_x, center_y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given center, angle, and side length."""
    angle_rad = math.radians(angle_deg)
    vertices = np.empty((6, 2))
    for i in range(6):
        angle = angle_rad + i * math.pi / 3
        x = center_x + side_length * math.cos(angle)
        y = center_y + side_length * math.sin(angle)
        vertices[i] = (x, y)
    return vertices

@jit(nopython=True)
def outer_hexagon_vertices_numba(side_length):
    """Generate vertices of outer hexagon centered at origin."""
    return hexagon_vertices_numba(0, 0, 0, side_length)

@jit(nopython=True)
def check_containment_numba(hexagon_vertices_list, outer_side_length):
    """Fast containment check using geometric properties."""
    # For a hexagon with side length 1, the distance from center to vertex is 1
    # For outer hexagon with side length R, distance from center to vertex is R
    # So we just need to check that all inner hexagon centers are within outer hexagon

    # Simplified containment check: verify that all hexagon centers are within outer hexagon
    # Using distance from center check for speed
    outer_radius = outer_side_length  # Distance from center to vertex of outer hexagon

    for vertices in hexagon_vertices_list:
        # Use first vertex as approximation of center (this is okay for our purposes)
        center_x = vertices[0][0]
        center_y = vertices[0][1]
        distance = math.sqrt(center_x*center_x + center_y*center_y)
        if distance > outer_radius:

    return True

@jit(nopython=True)
def get_hexagon_bounds(vertices):
    """Get bounding box of a hexagon."""
    min_x = vertices[0][0]
    max_x = vertices[0][0]
    min_y = vertices[0][1]
    max_y = vertices[0][1]

    for i in range(1, len(vertices)):
        x, y = vertices[i]
        if x < min_x: min_x = x
        if x > max_x: max_x = x
        if y < min_y: min_y = y
        if y > max_y: max_y = y

    return min_x, max_x, min_y, max_y

@jit(nopython=True)
def hexagon_to_grid_cells(vertices, cell_size):
    """Get all grid cells that a hexagon might occupy."""
    min_x, max_x, min_y, max_y = get_hexagon_bounds(vertices)

    # Get grid indices for bounding box
    min_cell_x = int(min_x // cell_size)
    max_cell_x = int(max_x // cell_size)
    min_cell_y = int(min_y // cell_size)
    max_cell_y = int(max_y // cell_size)

    # Collect all cells
    cells = []
    for x in range(min_cell_x, max_cell_x + 1):
        for y in range(min_cell_y, max_cell_y + 1):
            cells.append((x, y))

    return cells
