<DESCRIPTION>
This improvement implements Numba JIT compilation for the most computationally intensive geometric operations in the hexagon packing problem. The key insight is that the `generate_hexagon_vertices` and `create_hexagon_polygon` functions are called repeatedly during both the evolutionary optimization process and constraint checking, making them prime candidates for JIT compilation. This will significantly speed up the constraint validation and fitness evaluation steps, enabling more iterations within the 180-second time limit and potentially finding better solutions.

The changes involve:
1. Adding `@njit` decorators to the geometric computation functions
2. Using `numba` types for parameter typing to ensure proper JIT compilation
3. Leveraging the existing structure while adding performance enhancements

This follows the pattern established in the prior generations' success with Numba JIT compilation for geometric operations.

class GeometryHandler:
    """Handles all geometric computations for hexagon operations."""

    def __init__(self):
        self.hex_radius = 1.0
        self.hex_apothem = np.sqrt(3) / 2
        self.hex_height = 2 * self.hex_apothem
        self.hex_width = 2 * self.hex_radius

    @staticmethod
    @njit(float64[:,:](float64, float64, float64))
    def generate_hexagon_vertices(center_x: float, center_y: float, angle_deg: float) -> np.ndarray:
        """Generate vertices of a regular hexagon given center and rotation."""
        angle_rad = np.deg2rad(angle_deg)
        # Vertices of a unit hexagon centered at origin
        base_vertices = np.array([
            [1, 0],
            [0.5, np.sqrt(3)/2],
            [-0.5, np.sqrt(3)/2],
            [-1, 0],
            [-0.5, -np.sqrt(3)/2],
            [0.5, -np.sqrt(3)/2]
        ], dtype=np.float64)
        # Rotate and translate
        cos_a = np.cos(angle_rad)
        sin_a = np.sin(angle_rad)
        rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]], dtype=np.float64)
        rotated_vertices = base_vertices @ rotation_matrix.T
        return rotated_vertices + np.array([center_x, center_y], dtype=np.float64)

    @staticmethod
    @njit(float64[:,:](float64, float64, float64))
    def generate_hexagon_vertices_fast(center_x: float, center_y: float, angle_deg: float) -> np.ndarray:
        """Fast version for internal use with pre-computed constants."""
        angle_rad = np.deg2rad(angle_deg)
        cos_a = np.cos(angle_rad)
        sin_a = np.sin(angle_rad)

        # Pre-computed hexagon vertices
        vertices = np.empty((6, 2), dtype=np.float64)
        vertices[0] = [1.0, 0.0]
        vertices[1] = [0.5, np.sqrt(3)/2]
        vertices[2] = [-0.5, np.sqrt(3)/2]
        vertices[3] = [-1.0, 0.0]
        vertices[4] = [-0.5, -np.sqrt(3)/2]
        vertices[5] = [0.5, -np.sqrt(3)/2]

        # Apply rotation and translation
        for i in range(6):
            x, y = vertices[i]
            rotated_x = x * cos_a - y * sin_a
            rotated_y = x * sin_a + y * cos_a
            vertices[i] = [rotated_x + center_x, rotated_y + center_y]

        return vertices

    def create_hexagon_polygon(self, center_x: float, center_y: float, angle_deg: float) -> Polygon:
        """Create Shapely polygon representation of a hexagon."""
        vertices = self.generate_hexagon_vertices_fast(center_x, center_y, angle_deg)
        return Polygon(vertices)
