@jit(nopython=True)
def hexagon_vertices_numba(center_x, center_y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given center, angle, and side length."""
    angle_rad = math.radians(angle_deg)
    vertices = np.empty((6, 2))
    for i in range(6):
        angle = angle_rad + i * math.pi / 3
        x = center_x + side_length * math.cos(angle)
        y = center_y + side_length * math.sin(angle)
        vertices[i] = (x, y)
    return vertices

@jit(nopython=True)
def outer_hexagon_vertices_numba(side_length):
    """Generate vertices of outer hexagon centered at origin."""
    return hexagon_vertices_numba(0, 0, 0, side_length)

@jit(nopython=True)
def point_in_hexagon_numba(px, py, cx, cy, angle_deg, side_length=1):
    """Fast point-in-hexagon check using distance from center."""
    # Simple distance check for containment
    angle_rad = math.radians(angle_deg)
    # Transform point to local coordinate system
    dx = px - cx
    dy = py - cy
    # Rotate point by negative angle
    cos_a = math.cos(-angle_rad)
    sin_a = math.sin(-angle_rad)
    rx = dx * cos_a - dy * sin_a
    ry = dx * sin_a + dy * cos_a
    
    # Check if point is inside hexagon
    # Max distance in x direction for unit hexagon: 1.0
    # Max distance in y direction for unit hexagon: sqrt(3)/2 ≈ 0.866
    max_x = side_length
    max_y = side_length * math.sqrt(3) / 2
    
    return abs(rx) <= max_x and abs(ry) <= max_y

@jit(nopython=True)
def check_containment_fast_numba(hexagon_vertices_list, outer_side_length):
    """Fast containment check using distance from center."""
    outer_radius = outer_side_length * math.sqrt(3) / 2  # Distance to corner of outer hexagon
    
    for vertices in hexagon_vertices_list:
        # Use first vertex as approximate center
        center_x = vertices[0][0]
        center_y = vertices[0][1]
        distance = math.sqrt(center_x*center_x + center_y*center_y)
        if distance > outer_radius:
            return False
    return True

@jit(nopython=True)
def check_overlap_simple_numba(hexagon_vertices_list):
    """Simple overlap check using bounding boxes."""
    n_hexagons = len(hexagon_vertices_list)
    
    for i in range(n_hexagons):
        vertices_i = hexagon_vertices_list[i]
        # Get bounding box for hexagon i
        min_x = vertices_i[0][0]
        max_x = vertices_i[0][0]
        min_y = vertices_i[0][1]
        max_y = vertices_i[0][1]

        for j in range(1, len(vertices_i)):
            x, y = vertices_i[j]
            if x < min_x: min_x = x
            if x > max_x: max_x = x
            if y < min_y: min_y = y
            if y > max_y: max_y = y

        # Compare with all other hexagons
        for k in range(i+1, n_hexagons):
            vertices_k = hexagon_vertices_list[k]
            # Get bounding box for hexagon k
            min_x_k = vertices_k[0][0]
            max_x_k = vertices_k[0][0]
            min_y_k = vertices_k[0][1]
            max_y_k = vertices_k[0][1]

            for l in range(1, len(vertices_k)):
                x, y = vertices_k[l]
                if x < min_x_k: min_x_k = x
                if x > max_x_k: max_x_k = x
                if y < min_y_k: min_y_k = y
                if y > max_y_k: max_y_k = y

            # Check if bounding boxes intersect
            if max_x >= min_x_k and min_x <= max_x_k and max_y >= min_y_k and min_y <= max_y_k:
                # Quick distance check between centers as additional filter
                cx_i = vertices_i[0][0]
                cy_i = vertices_i[0][1]
                cx_k = vertices_k[0][0]
                cy_k = vertices_k[0][1]
                dist_sq = (cx_i - cx_k)**2 + (cy_i - cy_k)**2
                if dist_sq < 4.0:  # Minimum distance for unit hexagons
                    return False

    return True

def estimate_outer_hexagon_side_length(hexagons_data):
    """Estimate the minimum outer hexagon side length needed."""
    centers = np.array([(h[0], h[1]) for h in hexagons_data])
    
    # Find the maximum distance from origin to any center
    distances = np.sqrt(np.sum(centers**2, axis=1))
    max_distance = np.max(distances)
    
    # Add padding for the hexagon radius
    # For unit hexagons, the distance from center to vertex is 1
    return max_distance + 1.0

def hexagon_lattice_objective(params, outer_side_length=None):
    """Objective function for hexagon lattice optimization."""
    # Reshape params into hexagons: x, y, angle for each of 12 hexagons
    hexagons = params.reshape(12, 3)
    
    # Get vertices for all hexagons
    hexagon_vertices_list = []
    for i in range(12):
        x, y, angle = hexagons[i]
        vertices = hexagon_vertices(x, y, angle)
        hexagon_vertices_list.append(vertices)
    
    # Calculate penalty based on constraints
    penalty = 0.0
    
    # Containment penalty (higher weight)
    if outer_side_length is not None:
        if not check_containment_fast_numba(hexagon_vertices_list, outer_side_length):
            penalty += 10000.0
            
        # Check overlap penalty  
        if not check_overlap_simple_numba(hexagon_vertices_list):
            penalty += 5000.0
    
    # Calculate inverse side length as main objective 
    # We want to maximize 1/outer_side_length, so minimize -1/outer_side_length
    if outer_side_length is not None and outer_side_length > 0:
        objective_value = -1.0 / outer_side_length + penalty
    else:
        objective_value = 10000.0 + penalty
    
    return objective_value

def optimize_with_gradient_descent(initial_params, bounds, maxiter=100):
    """Use local optimization to improve the solution."""
    try:
        # Create bounds for optimization
        bnds = [(bounds[i*3], bounds[i*3+1]) for i in range(12)] + \
               [(bounds[36], bounds[37])]  # outer side length bounds
        
        # Optimize using L-BFGS-B which handles bounds well
        result = minimize(
            hexagon_lattice_objective,
            initial_params,
            method='L-BFGS-B',
            bounds=bnds,
            options={'maxiter': maxiter, 'ftol': 1e-8},
            callback=lambda x: None  # Optional callback
        )
        
        return result.x if result.success else initial_params
    except Exception as e:
        return initial_params
