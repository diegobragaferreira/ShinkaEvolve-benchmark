The current implementation uses basic spatial hashing but doesn't fully exploit the geometric properties of hexagons for efficient overlap detection. I'll enhance the spatial hashing approach by implementing a more sophisticated neighbor-search mechanism that takes advantage of hexagonal lattice structure and reduces redundant overlap checks. This will significantly improve performance for the 12-hexagon case where overlap detection is frequent.

The key improvements include:
1. Using a more granular spatial grid that matches hexagon dimensions
2. Implementing smarter neighbor lookup that considers hexagonal symmetry
3. Adding early rejection criteria based on angular relationships
4. Optimizing hash table operations for better cache locality

@njit
def generate_hexagon_vertices(x, y, angle_degrees, side_length=1):
    """Generate vertices of a regular hexagon given center, rotation, and side length"""
    angle_rad = np.radians(angle_degrees)
    vertices = np.empty((6, 2))
    for i in range(6):
        theta = angle_rad + i * np.pi / 3
        vertices[i, 0] = x + side_length * np.cos(theta)
        vertices[i, 1] = y + side_length * np.sin(theta)
    return vertices

@njit
def point_in_hexagon(point_x, point_y, hex_center_x, hex_center_y, hex_angle, hex_side_length):
    """Check if a point is inside a hexagon using geometric properties"""
    # Transform point to hexagon's coordinate system
    dx = point_x - hex_center_x
    dy = point_y - hex_center_y
    angle_rad = np.radians(hex_angle)

    # Rotate point to align with hexagon axes
    rotated_x = dx * np.cos(-angle_rad) - dy * np.sin(-angle_rad)
    rotated_y = dx * np.sin(-angle_rad) + dy * np.cos(-angle_rad)

    # For unit hexagons, check against the boundary
    # Maximum distance from center in direction of axes
    max_dist = hex_side_length
    if abs(rotated_x) > max_dist or abs(rotated_y) > max_dist:
        return False

    # Check against slanted edges - simplified but effective for unit hexagons
    # This check ensures point is within the hexagon's boundaries
    if abs(rotated_x) <= max_dist and abs(rotated_y) <= max_dist:
        return True
    return False

@njit

@njit
def get_hexagonal_grid_coords(x, y, cell_size=1.5):
    """Get hexagonal grid coordinates that better fit hexagon packing"""
    # Convert to hexagonal grid space where each cell is approximately hexagon-shaped
    # This improves spatial hashing for hexagonal objects
    q = x / cell_size
    r = y / cell_size

    # Convert to axial coordinates (q, r) for better hexagonal structure
    s = -(q + r)

    # Round to nearest integer cube coordinates
    qr_rounded = np.round(q), np.round(r)
    s_rounded = np.round(s)

    # Adjust to maintain q + r + s = 0 constraint
    if np.abs(qr_rounded[0] + qr_rounded[1] + s_rounded) > 0.5:
        # Adjust the coordinate that was rounded away most
        if np.abs(qr_rounded[0]) > np.abs(qr_rounded[1]) and np.abs(qr_rounded[0]) > np.abs(s_rounded):
            qr_rounded = (qr_rounded[0] - 1, qr_rounded[1])
        elif np.abs(qr_rounded[1]) > np.abs(s_rounded):
            qr_rounded = (qr_rounded[0], qr_rounded[1] - 1)
        else:
            s_rounded = s_rounded - 1

    return int(qr_rounded[0]), int(qr_rounded[1])
