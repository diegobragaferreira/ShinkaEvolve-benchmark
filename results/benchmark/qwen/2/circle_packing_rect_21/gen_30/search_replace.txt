import numpy as np
from scipy.spatial.distance import cdist


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: perimeter = 4, so width + height = 2
    # Let's use a 1:1 ratio for simplicity (width = height = 1)
    rect_width = 1.0
    rect_height = 1.0

    # Initialize circles in a grid pattern
    circles = np.zeros((21, 3))

    # Place circles in a 5x5 grid (but we only need 21 circles)
    rows, cols = 5, 5
    spacing_x = rect_width / cols
    spacing_y = rect_height / rows

    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= 21:
                break
            x = (j + 0.5) * spacing_x
            y = (i + 0.5) * spacing_y
            # Initial radius based on available space
            max_radius = min(x, rect_width - x, y, rect_height - y)
            r = max_radius * 0.3  # Start with smaller radius to allow for optimization
            circles[idx] = [x, y, r]
            idx += 1

    # Local optimization to improve the configuration
    # Simple iterative improvement: try to increase radii while maintaining constraints
    for iteration in range(100):
        improved = False
        # For each circle, try to increase its radius
        for i in range(21):
            original_radius = circles[i, 2]
            # Get current positions and radii
            positions = circles[:, :2]
            radii = circles[:, 2]

            # Compute minimum distance to other circles
            min_distance = float('inf')
            for j in range(21):
                if i != j:
                    dist = np.sqrt(np.sum((positions[i] - positions[j]) ** 2))
                    min_distance = min(min_distance, dist)

            # Maximum possible radius for this circle
            max_radius = min(
                circles[i, 0],  # Distance to left edge
                rect_width - circles[i, 0],  # Distance to right edge
                circles[i, 1],  # Distance to bottom edge
                rect_height - circles[i, 1]   # Distance to top edge
            )

            # Target radius is limited by:
            # 1. Boundary constraints
            # 2. Non-overlap with other circles
            target_radius = min(max_radius, min_distance / 2.0)

            if target_radius > original_radius:
                circles[i, 2] = target_radius
                improved = True

        if not improved:
            break

    # Final validation - make sure constraints are satisfied
    positions = circles[:, :2]
    radii = circles[:, 2]

    # Check for overlaps and fix them
    for i in range(21):
        for j in range(i+1, 21):
            dist = np.sqrt(np.sum((positions[i] - positions[j]) ** 2))
            required_dist = radii[i] + radii[j]
            if dist < required_dist:
                # Adjust radii to prevent overlap
                adjustment = (required_dist - dist) / 2.0
                circles[i, 2] = max(0.001, circles[i, 2] - adjustment)
                circles[j, 2] = max(0.001, circles[j, 2] - adjustment)

    return circles
