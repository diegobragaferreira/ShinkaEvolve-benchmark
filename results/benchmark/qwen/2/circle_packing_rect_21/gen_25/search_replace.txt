import numpy as np
import random

# Set seed for reproducibility
random.seed(42)
np.random.seed(42)

def check_overlap(circles, i, j):
    """Check if two circles overlap."""
    x1, y1, r1 = circles[i]
    x2, y2, r2 = circles[j]
    distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
    return distance < (r1 + r2)

def check_boundaries(circles):
    """Check if all circles are within the rectangle bounds."""
    for x, y, r in circles:
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return False
    return True

def fitness(circles):
    """Calculate fitness: sum of radii with penalty for violations."""
    # Check for constraint violations
    if not check_boundaries(circles):
        return -1000000  # Heavy penalty for boundary violations

    # Check overlaps
    n = len(circles)
    for i in range(n):
        for j in range(i+1, n):
            if check_overlap(circles, i, j):
                return -1000000  # Heavy penalty for overlap violations

    # Return sum of radii as fitness
    return np.sum(circles[:, 2])

def initialize_population(pop_size=30, n_circles=21):
    """Initialize population with good starting configurations."""
    population = []

    # Strategy 1: Grid-based initialization
    for _ in range(pop_size // 3):
        circles = []
        # Arrange in a rectangular grid pattern
        rows = 5
        cols = 5
        if rows * cols < n_circles:
            rows = 4
            cols = 6

        x_spacing = 0.9 / cols
        y_spacing = 0.9 / rows

        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx >= n_circles:
                    break
                x = 0.05 + (j + 0.5) * x_spacing
                y = 0.05 + (i + 0.5) * y_spacing
                r = min(x_spacing, y_spacing) * 0.2
                circles.append([x, y, r])
                idx += 1
            if idx >= n_circles:
                break

        # Fill remaining slots with random values
        while len(circles) < n_circles:
            x = random.uniform(0.01, 0.99)
            y = random.uniform(0.01, 0.99)
            r = random.uniform(0.005, 0.05)
            circles.append([x, y, r])

        population.append(np.array(circles))

    # Strategy 2: Random initialization with constraints
    for _ in range(pop_size // 3):
        circles = []
        for _ in range(n_circles):
            x = random.uniform(0.01, 0.99)
            y = random.uniform(0.01, 0.99)
            r = random.uniform(0.005, 0.05)
            circles.append([x, y, r])
        population.append(np.array(circles))

    # Strategy 3: Concentrated initialization
    for _ in range(pop_size // 3):
        circles = []
        # Center of rectangle
        center_x, center_y = 0.5, 0.5
        for i in range(n_circles):
            angle = 2 * np.pi * i / n_circles
            distance = 0.2 * (1 + 0.5 * random.random())  # Random distance from center
            x = center_x + distance * np.cos(angle)
            y = center_y + distance * np.sin(angle)
            r = 0.02 + 0.03 * random.random()
            circles.append([x, y, r])
        population.append(np.array(circles))

    return population

def mutate(circles, mutation_rate=0.2):
    """Mutate a circle configuration."""
    mutated = circles.copy()
    n = len(mutated)

    for i in range(n):
        if random.random() < mutation_rate:
            # Mutate x coordinate
            mutated[i][0] = max(0.01, min(0.99, mutated[i][0] + random.gauss(0, 0.01)))
        if random.random() < mutation_rate:
            # Mutate y coordinate
            mutated[i][1] = max(0.01, min(0.99, mutated[i][1] + random.gauss(0, 0.01)))
        if random.random() < mutation_rate:
            # Mutate radius
            mutated[i][2] = max(0.001, min(0.49, mutated[i][2] + random.gauss(0, 0.005)))

    return mutated

def crossover(parent1, parent2):
    """Perform uniform crossover."""
    child1 = parent1.copy()
    child2 = parent2.copy()
    n = len(parent1)

    for i in range(n):
        if random.random() < 0.5:
            child1[i] = parent2[i].copy()
            child2[i] = parent1[i].copy()

    return child1, child2

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Parameters
    pop_size = 30
    generations = 50
    elite_size = 5

    # Initialize population
    population = initialize_population(pop_size)

    # Evolutionary algorithm
    for gen in range(generations):
        # Evaluate fitness
        fitness_scores = [fitness(individual) for individual in population]

        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitness_scores)[::-1]
        sorted_population = [population[i] for i in sorted_indices]

        # Keep track of best
        best_fitness = fitness_scores[sorted_indices[0]]

        # Print progress
        if gen % 10 == 0:
            print(f"Generation {gen}: Best fitness = {best_fitness:.6f}")

        # Create new population
        new_population = []

        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())

        # Generate offspring
        while len(new_population) < pop_size:
            # Selection (tournament)
            tournament_size = 3
            parents = []
            for _ in range(2):
                tournament_indices = random.sample(range(pop_size), tournament_size)
                tournament_fitness = [fitness_scores[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmax(tournament_fitness)]
                parents.append(population[winner_idx])

            # Crossover
            child1, child2 = crossover(parents[0], parents[1])

            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)

            new_population.extend([child1, child2])

        # Trim to exact population size
        population = new_population[:pop_size]

    # Return best solution
    final_fitness_scores = [fitness(individual) for individual in population]
    best_idx = np.argmax(final_fitness_scores)

    return population[best_idx]
