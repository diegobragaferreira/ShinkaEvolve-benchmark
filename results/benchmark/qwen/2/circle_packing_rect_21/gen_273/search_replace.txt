class FitnessEvaluator:
    """Evaluates fitness of circle configurations."""

    def __init__(self, validator: ConstraintValidator, penalty_weight: float = 1000.0):
        self.validator = validator
        self.penalty_weight = penalty_weight

    def evaluate(self, circles: np.ndarray) -> Tuple[float, int]:
        """
        Calculate fitness: sum of radii with penalty for constraint violations

        Returns:
            Tuple of (fitness_score, number_of_violations)
        """
        total_radius = np.sum(circles[:, 2])

        # Validate constraints
        violations, violating_pairs = self.validator.validate_all_constraints(circles)

        # Return fitness score (higher is better)
        # Implement constraint-aware fitness with dynamic penalties
        if violations > 0:
            # Calculate penalty based on violation severity
            # Use a more sophisticated penalty system that considers both
            # the number of violations and their impact
            penalty = 0.0

            # Boundary violations (heavier penalty)
            boundary_violations = 0
            for i in range(len(circles)):
                x, y, r = circles[i]
                if not self.validator.is_valid_circle(x, y, r):
                    boundary_violations += 1

            # Overlap violations (lighter penalty per violation)
            overlap_violations = violations - boundary_violations

            # Combine penalties with different weights
            penalty = (boundary_violations * 1000.0 + overlap_violations * 100.0)

            # Scale penalty based on solution quality
            penalty *= (1.0 + 0.1 * np.log(total_radius + 1))

            return total_radius - penalty, violations
        else:
            # No violations - return pure radius sum
            return total_radius, violations
