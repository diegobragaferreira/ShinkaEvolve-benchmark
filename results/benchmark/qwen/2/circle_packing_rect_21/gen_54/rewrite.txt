import numpy as np
from scipy.spatial import Voronoi
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.
    
    Uses a Voronoi-guided optimization approach starting from a hexagonal lattice pattern.
    
    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Container parameters (perimeter = 4 => width + height = 2)
    container_width = 1.0
    container_height = 1.0
    
    n_circles = 21
    
    # Start with a hexagonal lattice pattern (approximately optimal)
    def generate_hexagonal_pattern(width, height, n):
        # Create a hexagonal lattice pattern that fits within the container
        # Approximate side length of hexagon based on desired number of circles
        # We'll place circles in a hexagonal pattern and then optimize
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        
        # Create a hexagonal grid
        points = []
        row_spacing = 1.0
        col_spacing = 0.866  # sqrt(3)/2
        
        # Generate points in a hexagonal pattern
        for i in range(rows):
            for j in range(cols):
                if len(points) >= n:
                    break
                x = (j + (i % 2) * 0.5) * row_spacing
                y = i * col_spacing
                points.append([x, y])
                
        # Scale to fit container
        if points:
            points_array = np.array(points)[:n]
            x_range = np.max(points_array[:, 0]) - np.min(points_array[:, 0])
            y_range = np.max(points_array[:, 1]) - np.min(points_array[:, 1])
            
            if x_range > 0 and y_range > 0:
                scale_x = width / x_range
                scale_y = height / y_range
                scale = min(scale_x, scale_y) * 0.8  # Leave some margin
                points_array[:, 0] *= scale
                points_array[:, 1] *= scale
                
                # Center in container
                center_x = width/2 - np.mean(points_array[:, 0])
                center_y = height/2 - np.mean(points_array[:, 1])
                points_array[:, 0] += center_x
                points_array[:, 1] += center_y
                
                # Ensure all points are within bounds
                points_array[:, 0] = np.clip(points_array[:, 0], 0.01, width - 0.01)
                points_array[:, 1] = np.clip(points_array[:, 1], 0.01, height - 0.01)
                
                return points_array
        return None
    
    # Generate initial hexagonal pattern
    initial_positions = generate_hexagonal_pattern(container_width, container_height, n_circles)
    
    if initial_positions is None:
        # Fallback to random initialization
        np.random.seed(42)
        initial_positions = np.column_stack([
            np.random.uniform(0.01, container_width - 0.01, n_circles),
            np.random.uniform(0.01, container_height - 0.01, n_circles)
        ])
    
    # Create initial solution vector [x1, y1, r1, x2, y2, r2, ...]
    initial_solution = []
    for i in range(n_circles):
        initial_solution.extend([initial_positions[i, 0], initial_positions[i, 1], 0.05])
    
    # Constraint function to check if circles don't overlap or go out of bounds
    def check_constraints(solution, width, height):
        # Extract positions and radii
        positions = solution[::3][:n_circles]
        radii = solution[2::3][:n_circles]
        
        # Check boundary constraints
        for i in range(n_circles):
            x, y, r = positions[i], positions[i+n_circles], radii[i]
            if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
                return False, "Boundary violation"
        
        # Check overlap constraints with Voronoi-based optimization
        points = np.column_stack([positions, positions[n_circles:]])
        
        # Create Voronoi diagram for better constraint evaluation
        try:
            vor = Voronoi(points)
        except:
            # Fallback to direct checking for problematic cases
            for i in range(n_circles):
                for j in range(i+1, n_circles):
                    dx = points[i, 0] - points[j, 0]
                    dy = points[i, 1] - points[j, 1]
                    distance = np.sqrt(dx*dx + dy*dy)
                    if distance < (radii[i] + radii[j]) and distance > 1e-8:
                        return False, f"Overlap between {i} and {j}"
            return True, ""
        
        # More efficient constraint checking with Voronoi diagram
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                dx = points[i, 0] - points[j, 0]
                dy = points[i, 1] - points[j, 1]
                distance = np.sqrt(dx*dx + dy*dy)
                if distance < (radii[i] + radii[j]) and distance > 1e-8:
                    return False, f"Overlap between {i} and {j}"
        
        return True, ""
    
    # Objective function (negative because we want to maximize sum of radii)
    def objective(solution):
        radii = solution[2::3][:n_circles]
        return -np.sum(radii)
    
    # Constraints for optimization
    def boundary_constraint(solution):
        # Ensures all circles are within bounds
        positions = solution[::3][:n_circles]
        radii = solution[2::3][:n_circles]
        positions_y = solution[1::3][:n_circles]
        
        constraints = []
        for i in range(n_circles):
            x, y, r = positions[i], positions_y[i], radii[i]
            # Distance to left, right, bottom, top boundaries
            constraints.extend([
                x - r,                 # Left boundary
                container_width - x - r,  # Right boundary
                y - r,                 # Bottom boundary
                container_height - y - r  # Top boundary
            ])
        return np.array(constraints)
    
    # Create bounds for variables: [x1, y1, r1, x2, y2, r2, ...]
    bounds = []
    for i in range(n_circles):
        # x coordinates
        bounds.extend([(0.001, container_width - 0.001)])
        # y coordinates  
        bounds.extend([(0.001, container_height - 0.001)])
        # radii (must be positive)
        bounds.extend([(0.001, min(container_width, container_height) * 0.4)])
    
    # Optimization with bounds
    try:
        # Use a few iterations of Nelder-Mead 
        result = minimize(
            objective,
            initial_solution,
            method='Nelder-Mead',
            options={'maxiter': 5000, 'disp': False, 'adaptive': True},
            bounds=bounds
        )
        
        if result.success:
            final_solution = result.x
        else:
            # If optimization fails, use the initial solution
            final_solution = initial_solution
            
    except Exception as e:
        # Fallback to initial solution
        final_solution = initial_solution
    
    # Construct final circles array
    circles = np.zeros((n_circles, 3))
    for i in range(n_circles):
        circles[i, 0] = final_solution[i*3]      # x coordinate
        circles[i, 1] = final_solution[i*3+1]   # y coordinate
        circles[i, 2] = final_solution[i*3+2]   # radius
    
    # Final verification and adjustment
    valid, message = check_constraints(final_solution, container_width, container_height)
    if not valid:
        # If constraints violated, do a simple refinement
        positions = final_solution[::3][:n_circles]
        radii = final_solution[2::3][:n_circles]
        
        # Simple iterative improvement
        for _ in range(100):
            improved = False
            for i in range(n_circles):
                # Try to slightly adjust for constraints
                best_x, best_y, best_r = circles[i, 0], circles[i, 1], circles[i, 2]
                
                # Check if any overlap exists
                for j in range(n_circles):
                    if i != j:
                        dx = circles[i, 0] - circles[j, 0]
                        dy = circles[i, 1] - circles[j, 1]
                        distance = np.sqrt(dx*dx + dy*dy)
                        min_distance = circles[i, 2] + circles[j, 2]
                        
                        if distance < min_distance:
                            # Move away from the other circle
                            if distance > 0:
                                move_x = (dx / distance) * (min_distance - distance) * 0.1
                                move_y = (dy / distance) * (min_distance - distance) * 0.1
                                
                                # Update position with bounds checking
                                new_x = circles[i, 0] + move_x
                                new_y = circles[i, 1] + move_y
                                
                                new_x = np.clip(new_x, circles[i, 2], container_width - circles[i, 2])
                                new_y = np.clip(new_y, circles[i, 2], container_height - circles[i, 2])
                                
                                circles[i, 0] = new_x
                                circles[i, 1] = new_y
                                improved = True
                
                # Try to increase radius if possible
                can_increase = True
                for j in range(n_circles):
                    if i != j:
                        dx = circles[i, 0] - circles[j, 0]
                        dy = circles[i, 1] - circles[j, 1]
                        distance = np.sqrt(dx*dx + dy*dy)
                        if distance < (circles[i, 2] + circles[j, 2]):
                            can_increase = False
                            break
                
                if can_increase and circles[i, 2] < min(container_width, container_height) * 0.3:
                    circles[i, 2] = min(circles[i, 2] * 1.01, min(container_width, container_height) * 0.3)
                    improved = True
                    
            if not improved:
                break
    
    return circles
