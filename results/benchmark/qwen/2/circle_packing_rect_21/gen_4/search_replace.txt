import numpy as np
import math


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    # Use a square for simplicity: width = height = 1
    width, height = 1.0, 1.0

    # Start with a basic grid approach
    # Try to fit circles in a grid pattern
    n = 21

    # Estimate initial radius based on area
    # Total area needed for circles = n * π * r^2
    # Available area = width * height = 1
    # So: n * π * r^2 <= 1 => r <= sqrt(1/(n*π))
    max_radius = math.sqrt(1.0 / (n * math.pi))

    # Use a heuristic grid arrangement
    # Try 5 rows and 4 columns (or similar) to get 20 circles, then adjust
    rows = 5
    cols = 4

    # Calculate spacing
    col_spacing = width / (cols + 1)
    row_spacing = height / (rows + 1)

    circles = []

    # Place circles in grid pattern
    for i in range(rows):
        for j in range(cols):
            if len(circles) >= n:
                break
            x = (j + 1) * col_spacing
            y = (i + 1) * row_spacing
            # Start with a reasonable radius
            r = min(max_radius * 0.8, 0.1)  # Cap at 0.1 to avoid overflow
            circles.append((x, y, r))

    # Add one more circle near center if we haven't reached 21
    while len(circles) < n:
        x = width / 2
        y = height / 2
        r = min(max_radius * 0.5, 0.05)
        circles.append((x, y, r))

    # Trim to exactly 21 circles
    circles = circles[:21]

    # Convert to numpy array
    result = np.array(circles)

    return result
