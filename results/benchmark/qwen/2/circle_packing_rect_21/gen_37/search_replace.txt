import numpy as np
from scipy.spatial.distance import cdist
from scipy.spatial import KDTree
import random
from typing import Tuple, List

# Fixed seed for reproducibility
random.seed(42)
np.random.seed(42)

def check_constraints(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> bool:
    """Check if all circles are within bounds and don't overlap."""
    n = len(circles)

    # Check boundary constraints
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            return False

    # Check overlap constraints using KDTree for efficiency
    points = circles[:, :2]
    radii = circles[:, 2]

    tree = KDTree(points)
    # Find pairs of points that are close enough to potentially overlap
    pairs = tree.query_pairs(r=0.0001)  # Find pairs that might overlap

    for i, j in pairs:
        if i >= j:  # Only check each pair once
            continue
        dist = np.linalg.norm(points[i] - points[j])
        if dist < (radii[i] + radii[j]):
            return False

    return True

def evaluate_fitness(circles: np.ndarray) -> float:
    """Evaluate fitness as sum of radii."""
    return np.sum(circles[:, 2])

def mutate_individual(circles: np.ndarray, mutation_strength: float = 0.05,
                     rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """Mutate a single individual by slightly perturbing positions and radii."""
    mutated = circles.copy()

    # Mutate some random circles
    n = len(mutated)
    mutate_count = max(1, n // 10)  # Mutate about 10% of circles

    indices = np.random.choice(n, size=mutate_count, replace=False)

    for idx in indices:
        # Randomly decide what to mutate
        action = random.randint(0, 2)

        if action == 0:  # Mutate position
            mutated[idx, 0] += np.random.normal(0, mutation_strength * rect_width)
            mutated[idx, 1] += np.random.normal(0, mutation_strength * rect_height)
        elif action == 1:  # Mutate radius
            mutated[idx, 2] *= np.random.normal(1, mutation_strength)
            mutated[idx, 2] = max(0.001, mutated[idx, 2])  # Ensure positive radius
        else:  # Mutate both position and radius
            mutated[idx, 0] += np.random.normal(0, mutation_strength * rect_width)
            mutated[idx, 1] += np.random.normal(0, mutation_strength * rect_height)
            mutated[idx, 2] *= np.random.normal(1, mutation_strength)
            mutated[idx, 2] = max(0.001, mutated[idx, 2])  # Ensure positive radius

    # Keep within bounds
    mutated[:, 0] = np.clip(mutated[:, 0], 0, rect_width)
    mutated[:, 1] = np.clip(mutated[:, 1], 0, rect_height)

    return mutated

def crossover(parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
    """Perform uniform crossover between two parents."""
    child = parent1.copy()
    n = len(parent1)

    # Randomly swap elements between parents
    for i in range(n):
        if random.random() < 0.5:
            child[i] = parent2[i]

    return child

def initialize_population(pop_size: int, n_circles: int, rect_width: float = 1.0, rect_height: float = 1.0) -> List[np.ndarray]:
    """Initialize population with feasible solutions."""
    population = []

    for _ in range(pop_size):
        # Create a simple arrangement of circles
        circles = np.zeros((n_circles, 3))

        # Arrange in a grid
        rows = int(np.ceil(np.sqrt(n_circles)))
        cols = int(np.ceil(n_circles / rows))

        # Calculate spacing
        spacing_x = rect_width / (cols + 1)
        spacing_y = rect_height / (rows + 1)

        # Place circles in a grid pattern
        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx >= n_circles:
                    break
                x = (j + 1) * spacing_x
                y = (i + 1) * spacing_y
                # Set initial radius to small value (will be optimized)
                r = min(spacing_x, spacing_y) / 4
                circles[idx] = [x, y, r]
                idx += 1

        # Add some randomness to initial positions and radii
        for i in range(n_circles):
            if i < n_circles:
                circles[i, 0] += np.random.uniform(-spacing_x/4, spacing_x/4)
                circles[i, 1] += np.random.uniform(-spacing_y/4, spacing_y/4)
                circles[i, 2] *= np.random.uniform(0.5, 1.5)

        # Keep within bounds
        circles[:, 0] = np.clip(circles[:, 0], 0, rect_width)
        circles[:, 1] = np.clip(circles[:, 1], 0, rect_height)

        # Make sure all radii are positive
        circles[:, 2] = np.maximum(circles[:, 2], 0.001)

        # If not valid, generate random valid placement
        if not check_constraints(circles, rect_width, rect_height):
            # Generate random valid placement
            circles = np.zeros((n_circles, 3))
            attempts = 0
            while not check_constraints(circles, rect_width, rect_height) and attempts < 100:
                for i in range(n_circles):
                    circles[i] = [
                        np.random.uniform(0.01, rect_width - 0.01),
                        np.random.uniform(0.01, rect_height - 0.01),
                        np.random.uniform(0.01, min(rect_width, rect_height) / 4)
                    ]
                attempts += 1

        population.append(circles)

    return population

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions - since perimeter = 4, width + height = 2
    # Using square (1x1) for simplicity but could be adjusted
    rect_width = 1.0
    rect_height = 1.0

    # Algorithm parameters
    pop_size = 50
    n_generations = 100
    elite_size = 5
    mutation_rate = 0.1

    # Initialize population
    population = initialize_population(pop_size, 21, rect_width, rect_height)

    # Evolution loop
    for generation in range(n_generations):
        # Evaluate fitness for all individuals in population
        fitness_scores = [evaluate_fitness(individual) for individual in population]

        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores.sort(reverse=True)

        # Print progress
        if generation % 20 == 0:
            print(f"Generation {generation}: Best fitness = {fitness_scores[0]}")

        # Create new population
        new_population = population[:elite_size]  # Keep elite

        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = tournament_selection(population, fitness_scores, 3)
            parent2_idx = tournament_selection(population, fitness_scores, 3)

            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]

            # Crossover
            child = crossover(parent1, parent2)

            # Mutation
            if random.random() < mutation_rate:
                child = mutate_individual(child, mutation_strength=0.05, rect_width=rect_width, rect_height=rect_height)

            # Ensure constraints are satisfied
            if check_constraints(child, rect_width, rect_height):
                new_population.append(child)
            else:
                # If constraints violated, try to repair
                repaired = repair_individual(child, rect_width, rect_height)
                if check_constraints(repaired, rect_width, rect_height):
                    new_population.append(repaired)
                else:
                    # If still invalid, keep original but with repaired constraints
                    new_population.append(child)

        population = new_population[:pop_size]

    # Return best solution
    best_fitness = evaluate_fitness(population[0])
    print(f"Final best fitness: {best_fitness}")

    return population[0]

def tournament_selection(population: List[np.ndarray], fitness_scores: List[float], tournament_size: int) -> int:
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), size=tournament_size, replace=False)
    tournament_fitness = [fitness_scores[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmax(tournament_fitness)]
    return winner_index

def repair_individual(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """Repair an individual to satisfy constraints."""
    repaired = circles.copy()

    # Clamp positions to bounds
    repaired[:, 0] = np.clip(repaired[:, 0], 0, rect_width)
    repaired[:, 1] = np.clip(repaired[:, 1], 0, rect_height)

    return repaired
