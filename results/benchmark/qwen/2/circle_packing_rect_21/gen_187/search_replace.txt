    def generate_adaptive_grid_initialization(self) -> np.ndarray:
        """Generate initial configuration using adaptive grid-based approach"""
        circles = np.zeros((self.num_circles, 3))

        # Calculate optimal grid spacing based on rectangle dimensions and circle count
        # This helps ensure more uniform coverage and better initial packing
        spacing = np.sqrt((self.width * self.height) / self.num_circles)

        # Determine grid dimensions
        cols = max(1, int(np.ceil(self.width / spacing)))
        rows = max(1, int(np.ceil(self.height / spacing)))

        # Ensure we have enough cells for all circles
        if cols * rows < self.num_circles:
            # Adjust dimensions to accommodate all circles
            cols = max(cols, int(np.ceil(np.sqrt(self.num_circles * self.width / self.height))))
            rows = max(rows, int(np.ceil(self.num_circles / cols)))

        # Calculate actual cell dimensions
        cell_width = self.width / cols
        cell_height = self.height / rows

        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx >= self.num_circles:
                    break

                # Position in cell center with slight randomness to avoid perfect symmetry
                x = (j + 0.5) * cell_width + (np.random.random() - 0.5) * cell_width * 0.3
                y = (i + 0.5) * cell_height + (np.random.random() - 0.5) * cell_height * 0.3

                # Ensure valid bounds
                x = np.clip(x, 0.01, self.width - 0.01)
                y = np.clip(y, 0.01, self.height - 0.01)

                # Initial radius based on available space with safety factor
                max_radius = min(x, self.width - x, y, self.height - y) * 0.4
                r = np.clip(max_radius * np.random.uniform(0.2, 0.8), 0.001, 0.15)

                circles[idx] = [x, y, r]
                idx += 1

        # Apply local refinement to improve initial configuration
        circles = self.refine_initial_configuration(circles)

        return circles

    def refine_initial_configuration(self, circles: np.ndarray) -> np.ndarray:
        """Apply simple refinement to initial configuration"""
        # Try to increase radii while respecting constraints
        for _ in range(50):  # Limited iterations for performance
            updated = False
            for i in range(len(circles)):
                x, y, r = circles[i]

                # Calculate maximum possible radius at this location
                max_r = min(x, self.width - x, y, self.height - y)

                # Check for overlap with other circles
                for j in range(len(circles)):
                    if i != j:
                        x2, y2, r2 = circles[j]
                        distance = np.sqrt((x2 - x)**2 + (y2 - y)**2)

                        # Can't get closer than sum of radii
                        if distance < (r + r2):
                            max_r = min(max_r, distance - r2)

                # Increase radius if beneficial and safe
                if max_r > r and max_r > 0.001:
                    # Conservative increase to prevent instability
                    increment = min((max_r - r) * 0.2, 0.02)
                    circles[i][2] = min(max_r, r + increment)
                    updated = True

            if not updated:
                break

        return circles
