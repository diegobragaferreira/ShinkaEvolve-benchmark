    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.

                grid_size = int(np.ceil(np.sqrt(n_circles)))
                spacing_x = width / (grid_size + 1)
                spacing_y = height / (grid_size + 1)
                
                idx = 0
                for i in range(grid_size):
                    for j in range(grid_size):
                        if idx >= n_circles:
                            break

                        y = spacing_y * (j + 1)
                        
                        # Assign random small radius
                        r = min(0.05, min(x, y, width-x, height-y) / 2)
                        circles[idx] = [x, y, r]
                        idx += 1

                        circles[i][1] += (np.random.random() - 0.5) * 0.1
                        circles[i][2] *= (0.9 + np.random.random() * 0.2)
                        
            except:
                # Fallback to pure random initialization
                circles = generate_random_circles(n_circles, width, height)
            
            population.append(circles)
        
        # Pure random initialization for the rest

            circles = generate_random_circles(n_circles, width, height)

    def generate_random_circles(n_circles, w, h):

            x = np.random.uniform(0.05, w - 0.05)

            circles[i] = [x, y, r]

    def check_overlap(circles, i, j):
        """Check if two circles overlap"""
        x1, y1, r1 = circles[i]
        x2, y2, r2 = circles[j]
        distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return distance < (r1 + r2)
    
    def is_valid_placement(circles, index, x, y, r):
        """Check if placing a circle at (x,y) with radius r is valid"""
        # Check boundaries
        if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
            return False
            
        # Check overlaps with existing circles
        for i in range(len(circles)):
            if i == index:

            cx, cy, cr = circles[i]
            distance = np.sqrt((x - cx)**2 + (y - cy)**2)
            if distance < (r + cr):
                return False
                
        return True
    
    def evaluate_fitness(circles):

        # Count constraint violations

        # Check boundary violations

            x, y, r = circles[i]
            if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
                violation_count += 1

            for j in range(i+1, len(circles)):
                if check_overlap(circles, i, j):
                    violation_count += 1
        
        # Apply penalty for violations
        penalty_factor = 0.1
        fitness = radius_sum - (violation_count * penalty_factor)
        
        return fitness
    
    def mutate(circles, generation, max_generations):
        """Mutate a solution with adaptive mutation rate"""
        mutation_rate = initial_mutation_rate + (final_mutation_rate - initial_mutation_rate) * (generation / max_generations)
        
        mutated = circles.copy()
        
        # Mutate each circle

                    mutated[i, 2] = min(r, max_radius)

        child = np.zeros_like(parent1)
        
        for i in range(len(child)):
            # Uniform crossover for each attribute
            if np.random.random() < 0.5:
                child[i] = parent1[i]  # Copy from parent1
            else:
                child[i] = parent2[i]  # Copy from parent2

        selected_indices = np.random.choice(len(population), size=k, replace=False)
        selected_fitnesses = [fitnesses[i] for i in selected_indices]
        winner_idx = selected_indices[np.argmax(selected_fitnesses)]

            fitnesses.append(fitness)
        
        # Track best solution
        current_best_fitness = max(fitnesses)
        current_best_idx = np.argmax(fitnesses)
        current_best = population[current_best_idx]
        
        if current_best_fitness > best_fitness:

            best_solution = current_best.copy()

        sorted_indices = np.argsort(fitnesses)[::-1]

            parent1 = tournament_selection(population, fitnesses, tournament_size)

def ensure_valid_solution(circles, width, height):
    """Ensure solution is valid"""
    # Make a copy to avoid modifying input
    solution = circles.copy()
    
    # First make sure all positions are within bounds
    for i in range(len(solution)):
        x, y, r = solution[i]
        # Clamp to valid range
        solution[i, 0] = np.clip(x, 0.05, width - 0.05)
        solution[i, 1] = np.clip(y, 0.05, height - 0.05)
        # Ensure radius is valid
        max_radius = min(solution[i, 0], solution[i, 1], width - solution[i, 0], height - solution[i, 1])
        solution[i, 2] = min(r, max_radius)
    
    # Simple iterative improvement to resolve overlaps
    improved = True
    iterations = 0
    while improved and iterations < 10:
        improved = False
        for i in range(len(solution)):
            for j in range(i+1, len(solution)):
                # Check if overlapping
                x1, y1, r1 = solution[i]
                x2, y2, r2 = solution[j]
                distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                
                if distance < (r1 + r2):
                    # Move circles apart
                    dx = x2 - x1
                    dy = y2 - y1
                    dist = np.sqrt(dx*dx + dy*dy)
                    
                    if dist > 0:
                        # Normalize and move them apart
                        dx /= dist
                        dy /= dist
                        
                        # Calculate how much they overlap
                        overlap = (r1 + r2) - dist
                        
                        # Move them apart

                        solution[i, 1] -= dy * overlap * 0.1
                        solution[j, 0] += dx * overlap * 0.1
                        solution[j, 1] += dy * overlap * 0.1
                        
                        improved = True
                        
        iterations += 1
    
    return solution
