import numpy as np
from scipy.spatial.distance import cdist
from scipy.spatial import KDTree
from scipy.optimize import differential_evolution, minimize
import time
from itertools import combinations

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: width + height = 2, optimal aspect ratio around 3:1
    width, height = 1.5, 0.5
    
    # Number of circles
    n = 21
    
    def calculate_violations(circles):
        """Calculate total overlap violations"""
        violations = 0
        positions = circles[:, :2]
        radii = circles[:, 2]
        
        # Check all pairs for overlaps
        for i in range(len(circles)):
            for j in range(i+1, len(circles)):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_distance = radii[i] + radii[j]
                
                if dist < min_distance:
                    violations += min_distance - dist
        
        # Check boundary violations
        for i in range(len(circles)):
            x, y, r = circles[i]
            if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
                violations += 100  # Large penalty for boundary violations
                
        return violations
    
    def initialize_config():
        """Initialize configuration with better hexagonal packing"""
        circles = np.zeros((n, 3))
        
        # Hexagonal packing parameters
        max_radius = 0.15  # Start with reasonable max radius
        spacing_factor = 1.8  # Spacing factor for tight packing
        
        # Try different hexagonal arrangements
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows)) + 1
        
        # Try to fit circles in hexagonal pattern
        radius_guess = max_radius
        spacing = spacing_factor * radius_guess
        
        placed = 0
        for i in range(rows):
            for j in range(cols):
                if placed >= n:
                    break
                # Offset every other row
                offset = spacing * 0.5 if i % 2 == 1 else 0
                x = offset + j * spacing + spacing/2
                y = i * spacing * np.sqrt(3)/2 + spacing/2
                
                # Check if within bounds
                if x - radius_guess >= 0 and x + radius_guess <= width and \
                   y - radius_guess >= 0 and y + radius_guess <= height:
                    circles[placed] = [x, y, radius_guess]
                    placed += 1
                else:
                    # Place at random position if boundary check fails
                    x = np.random.uniform(radius_guess, width - radius_guess)
                    y = np.random.uniform(radius_guess, height - radius_guess)
                    circles[placed] = [x, y, radius_guess]
                    placed += 1
                    
        return circles
    
    def refine_radii_positions(circles):
        """Refine positions and radii simultaneously"""
        # First make a copy to work with
        refined = circles.copy()
        
        # Use scipy optimization for better results
        def objective(params):
            # Reshape params back to circles format
            pos_and_rad = params.reshape(-1, 3)
            
            # Calculate negative sum of radii (since we want to maximize)
            return -np.sum(pos_and_rad[:, 2])
        
        def constraint_func(params):
            # Reshape params back to circles format
            pos_and_rad = params.reshape(-1, 3)
            violations = []
            
            # Check all overlaps
            positions = pos_and_rad[:, :2]
            radii = pos_and_rad[:, 2]
            
            # For each pair, check distance vs sum of radii
            for i, j in combinations(range(len(pos_and_rad)), 2):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_distance = radii[i] + radii[j]
                
                if dist < min_distance:
                    violations.append(min_distance - dist)
            
            # Check boundaries
            for i in range(len(pos_and_rad)):
                x, y, r = pos_and_rad[i]
                if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
                    violations.append(100)  # Large penalty
                    
            return violations
        
        # Flatten for optimization
        initial_flat = refined.flatten()
        
        # Optimize
        try:
            # Use differential evolution for global optimization
            bounds = [(0, width), (0, height), (0.001, 0.5)] * n
            result = differential_evolution(
                objective, 
                bounds, 
                constraints={'type': 'ineq', 'fun': constraint_func},
                seed=42,
                maxiter=100,
                popsize=15
            )
            
            if result.success:
                refined = result.x.reshape(-1, 3)
        except:
            pass  # Fall back to simpler approach if optimization fails
            
        return refined
    
    def physics_simulation(circles, iterations=200):
        """Physics-based optimization for rapid convergence"""
        # Create copy to avoid modifying original
        sim_circles = circles.copy()
        
        # Physics parameters
        dt = 0.01
        max_velocity = 0.05
        repulsion_strength = 1000.0
        boundary_strength = 1000.0
        
        for _ in range(iterations):
            # Compute forces between circles
            forces = np.zeros_like(sim_circles)
            
            # For each pair of circles, compute forces
            positions = sim_circles[:, :2]
            radii = sim_circles[:, 2]
            
            # Use KDTree for efficient neighbor search
            tree = KDTree(positions)
            pairs = tree.query_pairs(2 * np.max(radii), output_type='ndarray')
            
            # Process only nearby pairs
            for i, j in pairs:
                if i >= j:
                    continue
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                
                if dist > 0:
                    # Repulsive force based on distance and radii
                    min_distance = radii[i] + radii[j]
                    if dist < min_distance * 1.1:  # Allow slight overlap for physics
                        force_magnitude = repulsion_strength / (dist * dist)
                        forces[i, 0] += force_magnitude * dx / dist
                        forces[i, 1] += force_magnitude * dy / dist
                        forces[j, 0] -= force_magnitude * dx / dist
                        forces[j, 1] -= force_magnitude * dy / dist
            
            # Boundary forces
            for i in range(len(sim_circles)):
                x, y, r = sim_circles[i]
                # Left boundary
                if x - r < 0:
                    forces[i, 0] += boundary_strength * (0 - (x - r))
                # Right boundary  
                if x + r > width:
                    forces[i, 0] -= boundary_strength * ((x + r) - width)
                # Bottom boundary
                if y - r < 0:
                    forces[i, 1] += boundary_strength * (0 - (y - r))
                # Top boundary
                if y + r > height:
                    forces[i, 1] -= boundary_strength * ((y + r) - height)
            
            # Update positions
            for i in range(len(sim_circles)):
                # Limit velocity
                vel_x = np.clip(forces[i, 0], -max_velocity, max_velocity)
                vel_y = np.clip(forces[i, 1], -max_velocity, max_velocity)
                
                sim_circles[i, 0] += vel_x * dt
                sim_circles[i, 1] += vel_y * dt
                
                # Clamp to boundaries
                sim_circles[i, 0] = np.clip(sim_circles[i, 0], radii[i], width - radii[i])
                sim_circles[i, 1] = np.clip(sim_circles[i, 1], radii[i], height - radii[i])
        
        return sim_circles
    
    def optimize_radii(circles):
        """Optimize radii while keeping positions fixed"""
        # Make safe copy
        optimized = circles.copy()
        
        # Simple iterative approach to increase radii
        for iter_num in range(50):
            improved = False
            original_sum = np.sum(optimized[:, 2])
            
            # Try to increase each radius one by one
            for i in range(len(optimized)):
                # Save original
                original_r = optimized[i, 2]
                
                # Try increasing radius
                test_r = min(original_r + 0.005, 0.2)
                optimized[i, 2] = test_r
                
                # Check constraints
                valid = True
                positions = optimized[:, :2]
                radii = optimized[:, 2]
                
                # Check overlaps
                for j in range(len(optimized)):
                    if i != j:
                        dx = positions[i, 0] - positions[j, 0]
                        dy = positions[i, 1] - positions[j, 1]
                        dist = np.sqrt(dx*dx + dy*dy)
                        min_distance = radii[i] + radii[j]
                        
                        if dist < min_distance * 0.99:
                            valid = False
                            break
                
                # Check boundaries
                x, y, r = optimized[i]
                if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
                    valid = False
                
                if not valid:
                    optimized[i, 2] = original_r
                else:
                    improved = True
                    
            if not improved:
                break
                
        return optimized
    
    # Multi-start approach
    best_sum = 0
    best_circles = None
    
    # Run multiple initializations to avoid local optima
    for start_iter in range(3):
        # Initialize
        circles = initialize_config()
        
        # Adjust radii to fit better
        for i in range(len(circles)):
            circles[i, 2] = min(0.1, 0.5 * width / np.sqrt(n))
        
        # Physics simulation for fast convergence
        circles = physics_simulation(circles, iterations=200)
        
        # Refine with optimization
        circles = refine_radii_positions(circles)
        
        # Final radius optimization
        circles = optimize_radii(circles)
        
        # Check final quality
        current_sum = np.sum(circles[:, 2])
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_circles = circles.copy()
    
    return best_circles

# EVOLVE-BLOCK-END

if __name__ == "__main__":
