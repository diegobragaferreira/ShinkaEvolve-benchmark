        """Generate initial population of circle configurations"""

        for _ in range(pop_size):
            circles = np.zeros((self.num_circles, 3))

            # Generate initial configuration using adaptive grid-based approach
            # Calculate optimal grid dimensions based on aspect ratio and circle count
            aspect_ratio = self.width / self.height

            # For 21 circles, we want a grid that approximates good packing
            # Adjust grid size based on aspect ratio
            if aspect_ratio >= 1:  # Width >= height
                cols = int(np.ceil(np.sqrt(self.num_circles * aspect_ratio)))
                rows = int(np.ceil(self.num_circles / cols))
            else:  # Height > width
                rows = int(np.ceil(np.sqrt(self.num_circles / aspect_ratio)))
                cols = int(np.ceil(self.num_circles / rows))

            # Ensure we have enough cells
            while cols * rows < self.num_circles:
                if aspect_ratio >= 1:
                    cols += 1
                else:
                    rows += 1

            # Calculate spacing
            spacing_x = self.width / (cols + 1)
            spacing_y = self.height / (rows + 1)

            # Create hexagonal-like packing pattern for better initial coverage
            placed_count = 0

                    if placed_count >= self.num_circles:

                    base_y = (i + 1) * spacing_y

                    perturbation = 0.15 * min(spacing_x, spacing_y)

                               0.01, self.width - 0.01)
