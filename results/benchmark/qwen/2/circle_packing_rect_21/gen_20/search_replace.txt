import numpy as np
from scipy.spatial.distance import cdist
import random


def calculate_overlap_penalty(circles):
    """Calculate penalty for overlapping circles"""
    penalty = 0
    positions = circles[:, :2]
    radii = circles[:, 2]

    # Compute pairwise distances
    distances = cdist(positions, positions)

    # Check all pairs
    for i in range(len(circles)):
        for j in range(i + 1, len(circles)):
            dist = distances[i, j]
            sum_radii = radii[i] + radii[j]
            if dist < sum_radii:
                penalty += (sum_radii - dist) ** 2  # Quadratic penalty

    return penalty


def calculate_boundary_penalty(circles):
    """Calculate penalty for circles going outside boundary"""
    penalty = 0
    for i in range(len(circles)):
        x, y, r = circles[i]
        # Check if circle extends beyond rectangle boundaries [0,1] x [0,1]
        if x - r < 0:
            penalty += (0 - (x - r)) ** 2
        if x + r > 1:
            penalty += ((x + r) - 1) ** 2
        if y - r < 0:
            penalty += (0 - (y - r)) ** 2
        if y + r > 1:
            penalty += ((y + r) - 1) ** 2
    return penalty


def evaluate_fitness(circles):
    """Evaluate fitness of circle configuration"""
    # Objective: maximize sum of radii
    total_radius = np.sum(circles[:, 2])

    # Constraints: penalize violations
    overlap_penalty = calculate_overlap_penalty(circles)
    boundary_penalty = calculate_boundary_penalty(circles)

    # Combine with penalties
    penalty = overlap_penalty + boundary_penalty

    # Fitness = total radius - penalty (higher is better)
    return total_radius - penalty * 1000.0


def mutate(circles, mutation_strength=0.02):
    """Mutate circle configuration"""
    mutated = circles.copy()

    # Mutate some circles
    num_mutations = max(1, int(len(circles) * 0.3))
    indices = random.sample(range(len(circles)), num_mutations)

    for i in indices:
        # Small random changes
        mutated[i, 0] += np.random.normal(0, mutation_strength * 0.5)
        mutated[i, 1] += np.random.normal(0, mutation_strength * 0.5)
        mutated[i, 2] += np.random.normal(0, mutation_strength * 0.1)

        # Enforce bounds
        mutated[i, 0] = np.clip(mutated[i, 0], 0.01, 0.99)
        mutated[i, 1] = np.clip(mutated[i, 1], 0.01, 0.99)
        mutated[i, 2] = np.clip(mutated[i, 2], 0.001, 0.49)

    return mutated


def crossover(parent1, parent2):
    """Create child from two parents"""
    child = parent1.copy()

    # Single-point crossover
    crossover_point = random.randint(1, len(parent1) - 1)
    child[crossover_point:, :] = parent2[crossover_point:, :]

    return child


def create_initial_population(pop_size):
    """Create initial population with better starting configuration"""
    population = []

    # Grid-based initialization for good spread
    n_circles = 21
    grid_size = int(np.ceil(np.sqrt(n_circles)))
    spacing_x = 1.0 / (grid_size + 1)
    spacing_y = 1.0 / (grid_size + 1)

    for _ in range(pop_size):
        circles = np.zeros((n_circles, 3))
        idx = 0

        # Place circles in grid with small noise
        for row in range(grid_size):
            for col in range(grid_size):
                if idx >= n_circles:
                    break
                x = (col + 1) * spacing_x + np.random.normal(0, 0.005)
                y = (row + 1) * spacing_y + np.random.normal(0, 0.005)
                r = min(0.05, 0.5 * spacing_x) + np.random.normal(0, 0.002)

                # Ensure valid bounds
                x = np.clip(x, 0.01, 0.99)
                y = np.clip(y, 0.01, 0.99)
                r = np.clip(r, 0.001, 0.49)

                circles[idx] = [x, y, r]
                idx += 1

        # Fill remaining with random valid positions
        while idx < n_circles:
            x = np.random.uniform(0.01, 0.99)
            y = np.random.uniform(0.01, 0.99)
            r = np.random.uniform(0.001, 0.1)
            circles[idx] = [x, y, r]
            idx += 1

        population.append(circles)

    return population


def is_valid(circles):
    """Check if circles are valid"""
    # Check boundary violations
    for i in range(len(circles)):
        x, y, r = circles[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return False

    # Check overlap violations
    positions = circles[:, :2]
    radii = circles[:, 2]
    distances = cdist(positions, positions)

    for i in range(len(circles)):
        for j in range(i + 1, len(circles)):
            if distances[i, j] < (radii[i] + radii[j]):
                return False

    return True


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Parameters
    pop_size = 30
    generations = 100
    elite_size = 5
    tournament_size = 3

    # Create initial population
    population = create_initial_population(pop_size)

    # Evolution loop
    for generation in range(generations):
        # Evaluate fitness
        fitness_scores = [evaluate_fitness(individual) for individual in population]

        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]

        if generation % 20 == 0:
            print(f"Generation {generation}: Best fitness = {fitness_scores[0]:.6f}")
            print(f"Best sum of radii = {np.sum(population[0][:, 2]):.6f}")

        # Elitism: keep best individuals
        new_population = population[:elite_size]

        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = random.sample(range(elite_size), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmax(tournament_fitness)]

            tournament_indices = random.sample(range(elite_size), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmax(tournament_fitness)]

            # Crossover
            child = crossover(population[parent1_idx], population[parent2_idx])

            # Mutation
            child = mutate(child)

            new_population.append(child)

        population = new_population

    # Return best solution
    best_solution = population[0]

    # Final validation
    if not is_valid(best_solution):
        # If invalid, use fallback
        print("Warning: Best solution is invalid, using fallback")
        circles = np.zeros((21, 3))
        # Simple random initialization with bounds
        for i in range(21):
            circles[i, 0] = np.random.uniform(0.01, 0.99)
            circles[i, 1] = np.random.uniform(0.01, 0.99)
            circles[i, 2] = np.random.uniform(0.001, 0.1)
        return circles

    return best_solution
