import numpy as np
from deap import base, creator, tools, algorithms
import random
from scipy.spatial.distance import cdist
import time


def check_collision(circle1, circle2):
    """Check if two circles collide."""
    x1, y1, r1 = circle1
    x2, y2, r2 = circle2
    distance = np.sqrt((x2 - x1)**2 + (y2 - y1)**2)
    return distance < (r1 + r2)


def check_boundaries(circle, width, height):
    """Check if circle is within boundaries."""
    x, y, r = circle
    return (r <= x <= width - r) and (r <= y <= height - r)


def evaluate_fitness(individual, width=1.0, height=1.0):
    """Evaluate fitness of individual (circle configuration)."""
    # Convert individual to circles array
    circles = individual.reshape(-1, 3)

    # Check boundary constraints
    for circle in circles:
        if not check_boundaries(circle, width, height):
            return (0,)  # Invalid solution

    # Check overlap constraints
    for i in range(len(circles)):
        for j in range(i+1, len(circles)):
            if check_collision(circles[i], circles[j]):
                return (0,)  # Invalid solution

    # Return sum of radii as fitness
    return (np.sum(circles[:, 2]),)


def create_individual(width=1.0, height=1.0, n_circles=21):
    """Create a random valid individual."""
    # Try multiple times to generate a valid individual
    for _ in range(1000):  # Prevent infinite loop
        individual = []
        for _ in range(n_circles):
            # Randomly place circle with small radius
            x = random.uniform(0.01, width - 0.01)
            y = random.uniform(0.01, height - 0.01)
            r = random.uniform(0.01, min(x, y, width-x, height-y) * 0.9)
            individual.extend([x, y, r])

        # Check if this individual is valid
        circles = np.array(individual).reshape(-1, 3)
        valid = True
        for i in range(len(circles)):
            if not check_boundaries(circles[i], width, height):
                valid = False
                break
            for j in range(i+1, len(circles)):
                if check_collision(circles[i], circles[j]):
                    valid = False
                    break
            if not valid:
                break

        if valid:
            return individual

    # If couldn't find valid individual, return minimal valid one
    individual = []
    for _ in range(n_circles):
        individual.extend([0.1, 0.1, 0.05])
    return individual


def mutate_individual(individual, width=1.0, height=1.0, indpb=0.1):
    """Mutate an individual."""
    mutated = individual[:]
    for i in range(0, len(mutated), 3):
        if random.random() < indpb:
            # Mutate position and radius
            mutated[i] = max(0.01, min(width - 0.01, mutated[i] + random.gauss(0, 0.05)))
            mutated[i+1] = max(0.01, min(height - 0.01, mutated[i+1] + random.gauss(0, 0.05)))
            mutated[i+2] = max(0.001, mutated[i+2] + random.gauss(0, 0.02))
    return tuple(mutated)


def initialize_toolbox(width=1.0, height=1.0, n_circles=21):
    """Initialize DEAP toolbox."""
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    toolbox = base.Toolbox()
    toolbox.register("individual", create_individual, width, height, n_circles)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate_fitness, width=width, height=height)
    toolbox.register("mate", tools.cxUniform, indpb=0.5)
    toolbox.register("mutate", mutate_individual, width=width, height=height)
    toolbox.register("select", tools.selTournament, tournsize=3)
    return toolbox


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Set random seed for reproducibility
    random.seed(42)
    np.random.seed(42)

    # Initialize parameters
    width = 1.0  # Since perimeter = 4, width + height = 2, we'll use width=1.0, height=1.0 for simplicity
    height = 1.0
    n_circles = 21
    pop_size = 50
    n_generations = 100
    cxpb = 0.7  # Crossover probability
    mutpb = 0.3  # Mutation probability

    # Initialize toolbox
    toolbox = initialize_toolbox(width, height, n_circles)

    # Create initial population
    population = toolbox.population(n=pop_size)

    # Evaluate initial population
    fitnesses = list(map(toolbox.evaluate, population))
    for ind, fit in zip(population, fitnesses):
        ind.fitness.values = fit

    # Begin evolution
    start_time = time.time()
    for generation in range(n_generations):
        # Select the next generation individuals
        offspring = toolbox.select(population, len(population))
        offspring = list(map(toolbox.clone, offspring))

        # Apply crossover and mutation on the offspring
        for child1, child2 in zip(offspring[::2], offspring[1::2]):
            if random.random() < cxpb:
                toolbox.mate(child1, child2)
                del child1.fitness.values
                del child2.fitness.values

        for mutant in offspring:
            if random.random() < mutpb:
                toolbox.mutate(mutant)
                del mutant.fitness.values

        # Evaluate the individuals with an invalid fitness
        invalid_ind = [ind for ind in offspring if not ind.fitness.valid]
        fitnesses = map(toolbox.evaluate, invalid_ind)
        for ind, fit in zip(invalid_ind, fitnesses):
            ind.fitness.values = fit

        # Replace population with offspring
        population[:] = offspring

        # Print progress
        if generation % 20 == 0:
            fit_values = [ind.fitness.values[0] for ind in population if ind.fitness.values[0] > 0]
            if fit_values:
                best_fit = max(fit_values)
                print(f"Generation {generation}: Best fitness = {best_fit}")

    # Get the best individual
    best_individual = tools.selBest(population, 1)[0]
    circles = np.array(best_individual).reshape(-1, 3)

    # Ensure we have exactly 21 circles and they are valid
    if len(circles) < 21:
        # Fill with default circles if needed
        circles = np.pad(circles, ((0, 21-len(circles)), (0, 0)), mode='constant', constant_values=0.05)

    return circles
