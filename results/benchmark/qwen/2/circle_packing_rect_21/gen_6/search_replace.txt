def calculate_distance(circle1, circle2):
    """Calculate Euclidean distance between two circles' centers"""
    return np.sqrt((circle1[0] - circle2[0])**2 + (circle1[1] - circle2[1])**2)

def check_collision(circle1, circle2):
    """Check if two circles collide"""
    distance = calculate_distance(circle1, circle2)
    return distance < (circle1[2] + circle2[2])

def check_boundaries(circle, width, height):
    """Check if circle is within boundaries"""
    x, y, r = circle
    return (r <= x <= width - r) and (r <= y <= height - r)

def is_valid_placement(circles, new_circle, width, height):
    """Check if new circle placement is valid"""
    if not check_boundaries(new_circle, width, height):
        return False

    for circle in circles:
        if check_collision(circle, new_circle):
            return False

    return True

def compute_total_radius(circles):
    """Compute sum of all circle radii"""

def circle_packing21() -> np.ndarray:

    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:

    """
    # Rectangle dimensions: perimeter = 4, so width + height = 2
    # For square: width = height = 1
    width = 1.0
    height = 1.0

    # Start with hexagonal packing approximation
    n = 21
    circles = np.zeros((n, 3))

    # Initialize with a systematic approach
    # Place circles in a grid-like pattern, then optimize
    rows = int(math.ceil(math.sqrt(n)))
    cols = int(math.ceil(n / rows))

    # Create initial configuration with equal-sized circles
    base_radius = 0.1

    placed_count = 0
    for i in range(rows):
        for j in range(cols):
            if placed_count >= n:
                break

            x = (j + 0.5) * (width / cols)
            y = (i + 0.5) * (height / rows)
            r = base_radius

            # Adjust position to be within bounds
            x = max(r, min(width - r, x))
            y = max(r, min(height - r, y))

            circles[placed_count] = [x, y, r]
            placed_count += 1

    # Now optimize the configuration
    # First, try to increase radii as much as possible
    for i in range(n):
        # Try to grow each circle as much as possible
        max_radius = min(
            circles[i][0],
            circles[i][1],
            width - circles[i][0],
            height - circles[i][1]
        )

        # Binary search for maximum radius
        low, high = circles[i][2], max_radius
        best_radius = circles[i][2]

        while high - low > 0.0001:
            mid = (low + high) / 2
            test_circle = np.array([circles[i][0], circles[i][1], mid])

            for j, circle in enumerate(circles):
                if i != j and check_collision(circle, test_circle):
                    valid = False
                    break

            if valid:
                best_radius = mid
                low = mid

                high = mid

        circles[i][2] = best_radius

    # Apply local optimization to fine-tune positions

        circles[i][1] = np.clip(circles[i][1], circles[i][2], height - circles[i][2])

    return circles
