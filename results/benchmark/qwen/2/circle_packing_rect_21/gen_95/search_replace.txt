    def initialize_population(self):
        """Generate diverse initial population with adaptive grid initialization"""
        population = []
        for _ in range(self.population_size):
            circles = np.zeros((self.n_circles, 3))

            # Adaptive grid-based initialization
            # Calculate grid dimensions based on circle count
            grid_size = int(np.ceil(np.sqrt(self.n_circles)))
            cell_width = self.container_width / grid_size
            cell_height = self.container_height / grid_size

            # Place circles on a grid with slight randomization
            for i in range(self.n_circles):
                row = i // grid_size
                col = i % grid_size

                # Grid position
                x_base = (col + 0.5) * cell_width
                y_base = (row + 0.5) * cell_height

                # Add slight randomization to avoid perfect grid
                x = np.clip(x_base + np.random.normal(0, cell_width * 0.1),
                           0.01, self.container_width - 0.01)
                y = np.clip(y_base + np.random.normal(0, cell_height * 0.1),
                           0.01, self.container_height - 0.01)

                # Set initial radius (small but scalable)
                r = np.random.uniform(0.01, 0.05)

                circles[i, 0] = x
                circles[i, 1] = y
                circles[i, 2] = r

            # Normalize total radius to reasonable value
            total_radius = np.sum(circles[:, 2])
            if total_radius > 0:
                scaling_factor = 0.5 / total_radius
                circles[:, 2] *= scaling_factor

            population.append(circles.copy())
        return population
