# EVOLVE-BLOCK-START
import numpy as np


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 21
    # Use a simple grid-based initialization
    # Rectangle has perimeter 4, so width + height = 2
    # Let's use a 1x1 rectangle (width=1, height=1) for simplicity
    rect_width = 1.0
    rect_height = 1.0

    # Create grid arrangement
    rows = 5
    cols = 5
    circles = []

    # Calculate spacing
    x_spacing = rect_width / (cols + 1)
    y_spacing = rect_height / (rows + 1)

    # Fill grid with circles
    count = 0
    for i in range(min(rows, n)):
        for j in range(min(cols, n-count)):
            x = (j + 1) * x_spacing
            y = (i + 1) * y_spacing
            # Initial radius - small but reasonable
            r = min(x_spacing, y_spacing) * 0.3
            circles.append([x, y, r])
            count += 1
            if count >= n:
                break
        if count >= n:
            break

    # Fill remaining circles in center with small radii
    while len(circles) < n:
        circles.append([rect_width/2, rect_height/2, 0.05])

    return np.array(circles[:n])
