import numpy as np
import random
from scipy.spatial.distance import cdist

def check_overlap(circles):
    """Check if any circles overlap"""
    n = len(circles)
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            distance = np.sqrt((x1-x2)**2 + (y1-y2)**2)
            if distance < (r1 + r2):
                return True
    return False

def check_boundaries(circles, width, height):
    """Check if all circles are within boundaries"""
    for x, y, r in circles:
        if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
            return False
    return True

def fitness_function(circles, width, height):
    """Calculate fitness as sum of radii, penalize constraint violations"""
    if check_overlap(circles) or not check_boundaries(circles, width, height):
        return 0.0
    return np.sum(circles[:, 2])

def create_random_solution(width, height, n_circles):
    """Create a random valid solution"""
    circles = []
    max_attempts = 1000
    attempts = 0

    while len(circles) < n_circles and attempts < max_attempts:
        # Generate random circle
        x = np.random.uniform(0, width)
        y = np.random.uniform(0, height)
        r = np.random.uniform(0.01, min(0.5, width/2, height/2))

        new_circle = [x, y, r]

        # Check if it conflicts with existing circles
        valid = True
        for existing_circle in circles:
            ex, ey, er = existing_circle
            distance = np.sqrt((x-ex)**2 + (y-ey)**2)
            if distance < (r + er):  # Overlap detected
                valid = False
                break

        if valid:
            circles.append(new_circle)

        attempts += 1

    # If we couldn't generate enough circles, fill with minimal valid ones
    while len(circles) < n_circles:
        x = np.random.uniform(0, width)
        y = np.random.uniform(0, height)
        r = 0.01
        circles.append([x, y, r])

    return np.array(circles)

def mutate_solution(circles, width, height, mutation_rate=0.2):
    """Mutate a solution by slightly changing positions and radii"""
    mutated = circles.copy()
    n = len(mutated)

    for i in range(n):
        if random.random() < mutation_rate:
            # Randomly decide what to mutate
            if random.random() < 0.7:  # Mutate position
                mutated[i][0] = np.clip(mutated[i][0] + np.random.normal(0, 0.05), 0, width)
                mutated[i][1] = np.clip(mutated[i][1] + np.random.normal(0, 0.05), 0, height)
            else:  # Mutate radius
                mutated[i][2] = np.clip(mutated[i][2] + np.random.normal(0, 0.02), 0.01, min(0.5, width/2, height/2))

    return mutated

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle with perimeter 4 => width + height = 2
    # For simplicity, use square of side length 1
    width = 1.0
    height = 1.0

    # Parameters
    population_size = 50
    generations = 500
    elite_size = 5

    # Initialize population
    population = [create_random_solution(width, height, 21) for _ in range(population_size)]

    # Evolution loop
    for generation in range(generations):
        # Evaluate fitness for all individuals
        fitness_scores = [fitness_function(individual, width, height) for individual in population]

        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores.sort(reverse=True)

        # Print progress
        if generation % 100 == 0:
            print(f"Generation {generation}: Best fitness = {fitness_scores[0]}")

        # Create new population with elitism
        new_population = population[:elite_size]

        # Fill rest with offspring
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]
            parent = population[winner_index]

            # Mutate parent
            child = mutate_solution(parent, width, height)
            new_population.append(child)

        population = new_population

    # Return best solution
    best_fitness_scores = [fitness_function(individual, width, height) for individual in population]
    best_index = np.argmax(best_fitness_scores)

    return population[best_index]
