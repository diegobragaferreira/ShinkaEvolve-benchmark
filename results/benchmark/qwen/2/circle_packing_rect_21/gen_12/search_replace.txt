import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize
import math


def validate_and_score_circles(circles, width=1.0, height=1.0):
    """
    Validates circle configuration and returns negative sum of radii (for minimization).
    Returns very large negative number if invalid.
    """
    if len(circles) == 0:
        return -1e10

    # Check boundary constraints
    for x, y, r in circles:
        if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
            return -1e10

    # Check overlap constraints using distance matrix
    positions = circles[:, :2]
    radii = circles[:, 2]

    if len(positions) < 2:
        return np.sum(radii)

    # Compute pairwise distances
    distances = cdist(positions, positions)

    # Check overlap constraints
    n = len(circles)
    for i in range(n):
        for j in range(i+1, n):
            dist = distances[i, j]
            sum_radii = radii[i] + radii[j]
            if dist < sum_radii:
                return -1e10  # Penalty for overlap

    return -np.sum(radii)  # Negative because we want to maximize (minimize negative)


def initialize_grid_placement(n_circles, width=1.0, height=1.0):
    """Initialize circles in a grid pattern."""
    # Try to arrange in a square-like pattern
    rows = int(math.ceil(math.sqrt(n_circles)))
    cols = int(math.ceil(n_circles / rows))

    # Calculate spacing
    spacing_x = width / (cols + 1)
    spacing_y = height / (rows + 1)

    circles = []
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n_circles:
                break
            x = (col + 1) * spacing_x
            y = (row + 1) * spacing_y
            # Radius based on minimum spacing
            max_radius = min(spacing_x, spacing_y) * 0.4
            circles.append([x, y, max_radius])
            idx += 1

    return np.array(circles)


def optimize_circle_placement(initial_circles, width=1.0, height=1.0):
    """Use local optimization to improve the initial placement."""
    n_circles = len(initial_circles)

    # Flatten initial positions and radii for optimization
    initial_flat = initial_circles.flatten()

    def objective(params):
        # Reshape parameters back to circles
        circles = params.reshape(n_circles, 3)
        return validate_and_score_circles(circles, width, height)

    # Simple optimization using scipy minimize
    try:
        # Use L-BFGS-B which works well for bound-constrained problems
        result = minimize(objective, initial_flat, method='L-BFGS-B',
                         bounds=[(0, width), (0, height), (0.001, width/2)] * n_circles,
                         options={'maxiter': 500})
        if result.success:
            optimized_circles = result.x.reshape(n_circles, 3)
            return optimized_circles
    except Exception:
        pass

    return initial_circles


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    np.random.seed(42)  # For reproducibility

    n = 21
    width = 1.0  # Since perimeter = 4, width + height = 2, using 1x1 rectangle
    height = 1.0

    # Initialize with grid placement
    circles = initialize_grid_placement(n, width, height)

    # Apply local optimization
    circles = optimize_circle_placement(circles, width, height)

    # Fine-tune by improving the arrangement further
    best_score = validate_and_score_circles(circles, width, height)
    best_circles = circles.copy()

    # Try multiple random restarts to find better solution
    for _ in range(5):
        # Perturb the current solution slightly
        perturbed = circles.copy()
        for i in range(len(perturbed)):
            # Add small random perturbation to position and radius
            perturbed[i, 0] += np.random.normal(0, 0.02 * width)
            perturbed[i, 1] += np.random.normal(0, 0.02 * height)
            perturbed[i, 2] *= (1 + np.random.normal(0, 0.05))  # Small relative change to radius

        # Keep within bounds
        perturbed[:, 0] = np.clip(perturbed[:, 0], 0, width)
        perturbed[:, 1] = np.clip(perturbed[:, 1], 0, height)
        perturbed[:, 2] = np.maximum(perturbed[:, 2], 0.001)

        score = validate_and_score_circles(perturbed, width, height)
        if score < best_score:
            best_score = score
            best_circles = perturbed.copy()

    # Final validation
    final_score = validate_and_score_circles(best_circles, width, height)
    if final_score == -1e10:
        # If we still have invalid solution, fall back to a simple good configuration
        circles = initialize_grid_placement(n, width, height)
        # Try to slightly adjust to avoid strict overlaps
        for i in range(len(circles)):
            circles[i, 2] *= 0.8  # Reduce all radii slightly to ensure validity

    return best_circles
