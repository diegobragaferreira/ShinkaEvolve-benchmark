import numpy as np
import random
from scipy.spatial.distance import cdist

# Set random seed for reproducibility
random.seed(42)
np.random.seed(42)

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: width=1, height=1 (perimeter = 2, which satisfies perimeter=4 constraint when we consider width+height=2)
    width, height = 1.0, 1.0

    # Problem parameters
    n_circles = 21
    max_radius = min(width, height) / 2.0  # Maximum possible radius for any circle to fit

    def evaluate_fitness(circles):
        """Evaluate fitness of a circle configuration"""
        # Calculate total radius sum (this is our objective to maximize)
        total_radius = np.sum(circles[:, 2])

        # Check boundary constraints
        penalty = 0
        for i, (x, y, r) in enumerate(circles):
            if (x - r < 0 or x + r > width or
                y - r < 0 or y + r > height):
                penalty += 1000  # Heavy penalty for boundary violations

        # Check overlap constraints
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]

                # Calculate Euclidean distance between centers
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)

                # Penalty for overlaps (distance < sum of radii)
                if dist < (r1 + r2):
                    # Penalty based on how much they overlap
                    overlap = (r1 + r2) - dist
                    penalty += overlap * 10000

        return total_radius - penalty

    def mutate_individual(circles, mutation_rate=0.1):
        """Mutate a single circle configuration"""
        mutated = circles.copy()
        for i in range(n_circles):
            if random.random() < mutation_rate:
                # Randomly modify x, y, or r
                param_idx = random.randint(0, 2)
                if param_idx == 0:  # x coordinate
                    mutated[i, 0] = np.random.uniform(0 + max_radius, width - max_radius)
                elif param_idx == 1:  # y coordinate
                    mutated[i, 1] = np.random.uniform(0 + max_radius, height - max_radius)
                else:  # radius
                    mutated[i, 2] = np.random.uniform(0.001, max_radius)
        return mutated

    def crossover(parent1, parent2):
        """Create offspring from two parents"""
        child1 = parent1.copy()
        child2 = parent2.copy()

        # Single point crossover for each circle
        crossover_point = random.randint(1, n_circles - 1)
        for i in range(crossover_point, n_circles):
            child1[i] = parent2[i].copy()
            child2[i] = parent1[i].copy()

        return child1, child2

    # Initialize population with random valid configurations
    def create_random_individual():
        individual = np.zeros((n_circles, 3))
        for i in range(n_circles):
            individual[i, 0] = np.random.uniform(0 + max_radius, width - max_radius)  # x
            individual[i, 1] = np.random.uniform(0 + max_radius, height - max_radius)  # y
            individual[i, 2] = np.random.uniform(0.001, max_radius)  # radius
        return individual

    # Create initial population
    population_size = 50
    population = [create_random_individual() for _ in range(population_size)]

    # Evolution loop
    best_fitness = float('-inf')
    best_individual = None

    for generation in range(100):  # Limit generations
        # Evaluate fitness for all individuals
        fitness_scores = [evaluate_fitness(ind) for ind in population]

        # Track best individual
        max_fitness_idx = np.argmax(fitness_scores)
        if fitness_scores[max_fitness_idx] > best_fitness:
            best_fitness = fitness_scores[max_fitness_idx]
            best_individual = population[max_fitness_idx].copy()

        # Selection: keep top 30%
        sorted_indices = np.argsort(fitness_scores)[::-1][:population_size//3]
        selected_population = [population[i] for i in sorted_indices]

        # Generate new population through crossover and mutation
        new_population = selected_population.copy()

        # Fill up the rest with offspring
        while len(new_population) < population_size:
            parent1 = random.choice(selected_population)
            parent2 = random.choice(selected_population)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate_individual(child1)
            child2 = mutate_individual(child2)
            new_population.extend([child1, child2])

        # Trim to exact population size
        population = new_population[:population_size]

    # Final check to ensure we return the best solution
    if best_individual is not None:
        return best_individual
    else:
        # Fallback to a simple heuristic approach
        circles = np.zeros((n_circles, 3))
        rows = int(np.ceil(np.sqrt(n_circles)))
        cols = int(np.ceil(n_circles / rows))
        cell_width = width / cols
        cell_height = height / rows
        max_radius = min(cell_width, cell_height) / 2.0

        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx >= n_circles:
                    break
                x = (j + 0.5) * cell_width
                y = (i + 0.5) * cell_height
                # Add some randomness to avoid perfect grid
                x += np.random.normal(0, cell_width * 0.1)
                y += np.random.normal(0, cell_height * 0.1)
                r = np.random.uniform(0.01, max_radius * 0.8)

                # Ensure circle fits in bounds
                x = np.clip(x, r, width - r)
                y = np.clip(y, r, height - r)

                circles[idx] = [x, y, r]
                idx += 1

        return circles
