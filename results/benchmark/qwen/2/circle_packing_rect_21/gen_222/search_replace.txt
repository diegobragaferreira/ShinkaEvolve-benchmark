The modification adds explicit penalties for both boundary violations and overlap violations to the fitness calculation, which guides the optimizer toward valid solutions more consistently. This prevents the optimization from getting trapped in regions with severe constraint violations and improves overall convergence reliability. The penalty weights are dynamically adjusted based on the severity of constraint violations to maintain a balance between optimality and feasibility.

    # Efficient constraint validation with spatial indexing
    def calculate_fitness_with_spatial_indexing(circles_array):
        total_radius = np.sum(circles_array[:, 2])

        penalty = 0

        # Boundary penalties (quadratic)
        for i in range(n):
            cx, cy, r = circles_array[i]
            # Calculate distance to nearest boundary
            dist_to_left = cx - r
            dist_to_right = rect_width - cx - r
            dist_to_bottom = cy - r
            dist_to_top = rect_height - cy - r

            # Penalize if any boundary distance is negative (violation)
            if dist_to_left < 0:
                penalty += 10000 * (-dist_to_left)**2
            if dist_to_right < 0:
                penalty += 10000 * (-dist_to_right)**2
            if dist_to_bottom < 0:
                penalty += 10000 * (-dist_to_bottom)**2
            if dist_to_top < 0:
                penalty += 10000 * (-dist_to_top)**2

        # Overlap penalties using spatial indexing for efficiency
        points = circles_array[:, :2]
        tree = KDTree(points)

        # Query for all possible overlapping pairs efficiently
        # This reduces from O(n^2) to near O(n log n) for overlap checking
        for i in range(n):
            cx, cy, r = circles_array[i]

            # Find neighbors within 2*(r + safety_margin) distance
            neighbor_indices = tree.query_ball_point([cx, cy], 2*(r + 0.01) + 0.001)

            for j in neighbor_indices:
                if i != j:
                    other_cx, other_cy, other_r = circles_array[j]
                    dist = np.sqrt((cx - other_cx)**2 + (cy - other_cy)**2)
                    overlap = (r + other_r) - dist

                    if overlap > 0:
                        penalty += 100000 * overlap**2

        # Add a small penalty for very small radii to promote meaningful circle sizes
        for i in range(n):
            r = circles_array[i, 2]
            if r < 0.001:
                penalty += 1000 * (0.001 - r)**2

        return total_radius - penalty
