    def initialize_population(self):
        """Generate diverse initial population with improved adaptive grid initialization"""
        population = []
        for _ in range(self.population_size):
            circles = np.zeros((self.n_circles, 3))

            # Improved adaptive grid-based initialization
            # Calculate more optimal grid dimensions based on container aspect ratio
            aspect_ratio = self.container_width / self.container_height
            if aspect_ratio > 1:  # Width > Height
                cols = int(np.ceil(np.sqrt(self.n_circles * aspect_ratio)))
                rows = int(np.ceil(self.n_circles / cols))
            else:  # Height >= Width
                rows = int(np.ceil(np.sqrt(self.n_circles / aspect_ratio)))
                cols = int(np.ceil(self.n_circles / rows))

            # Ensure reasonable grid dimensions
            cols = max(1, min(cols, 10))
            rows = max(1, min(rows, 10))

            cell_width = self.container_width / cols
            cell_height = self.container_height / rows

            # Place circles on a grid with better distribution
            circle_idx = 0
            for i in range(rows):
                for j in range(cols):
                    if circle_idx >= self.n_circles:
                        break

                    # Grid position with more controlled randomness
                    x_base = (j + 0.5) * cell_width
                    y_base = (i + 0.5) * cell_height

                    # Add more systematic jitter based on position to avoid symmetry issues
                    x_jitter = np.random.normal(0, cell_width * 0.15)
                    y_jitter = np.random.normal(0, cell_height * 0.15)
