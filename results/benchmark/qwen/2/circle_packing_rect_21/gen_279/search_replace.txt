<DESCRIPTION>
The current implementation uses nested loops for overlap detection which has O(n²) complexity. This is inefficient for our 21-circle problem. By replacing the brute-force overlap checking with KDTree-based spatial indexing, we can reduce the overlap validation time from O(n²) to near O(n log n), enabling faster optimization iterations and potentially better convergence to higher radii sums.

This change involves:
1. Replacing the brute-force overlap detection in `calculate_fitness` with KDTree queries
2. Using spatial indexing for both boundary and overlap constraint validation
3. Maintaining the same penalty structure but with much faster execution
</DESCRIPTION>

# Constraint validation and penalty calculation with spatial indexing
    from scipy.spatial import KDTree

    def calculate_fitness(circles_array):
        total_radius = np.sum(circles_array[:, 2])

        penalty = 0

        # Boundary penalties (quadratic for smooth gradient)
        for i in range(n):
            cx, cy, r = circles_array[i]
            if cx - r < 0.01:
                penalty += 10000 * (r - cx)**2
            if cx + r > rect_width - 0.01:
                penalty += 10000 * (cx + r - rect_width)**2
            if cy - r < 0.01:
                penalty += 10000 * (r - cy)**2
            if cy + r > rect_height - 0.01:
                penalty += 10000 * (cy + r - rect_height)**2

        # Overlap penalties using spatial indexing for efficiency
        points = circles_array[:, :2]
        tree = KDTree(points)

        # Query for all possible overlapping pairs efficiently
        # This reduces from O(n^2) to near O(n log n) for overlap checking
        for i in range(n):
            cx, cy, r = circles_array[i]

            # Find neighbors within 2*(r + safety_margin) distance
            neighbor_indices = tree.query_ball_point([cx, cy], 2*(r + 0.01) + 0.001)

            for j in neighbor_indices:
                if i != j:
                    other_cx, other_cy, other_r = circles_array[j]
                    dist = np.sqrt((cx - other_cx)**2 + (cy - other_cy)**2)
                    overlap = (r + other_r) - dist

                    if overlap > 0:
                        penalty += 100000 * overlap**2

        return total_radius - penalty
