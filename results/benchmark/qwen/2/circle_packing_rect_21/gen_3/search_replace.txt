import numpy as np
from scipy.spatial.distance import cdist
import random


# Constants for rectangle dimensions
RECT_PERIMETER = 4.0
RECT_WIDTH = RECT_PERIMETER / 2.0  # Assuming square for simplicity
RECT_HEIGHT = RECT_PERIMETER / 2.0


def is_valid_placement(circles, x, y, r):
    """Check if a circle placement is valid (within bounds and non-overlapping)"""
    # Check bounds
    if x - r < 0 or x + r > RECT_WIDTH or y - r < 0 or y + r > RECT_HEIGHT:
        return False

    # Check overlap with existing circles
    if len(circles) > 0:
        centers = circles[:, :2]
        radii = circles[:, 2]
        distances = cdist([[x, y]], centers)[0]
        # Circle overlaps if distance between centers is less than sum of radii
        if np.any(distances <= (radii + r)):
            return False

    return True
