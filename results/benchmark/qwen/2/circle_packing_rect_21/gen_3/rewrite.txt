# EVOLVE-BLOCK-START
import numpy as np
from scipy.spatial import cKDTree
from deap import base, creator, tools, algorithms
import random
from functools import partial
import time

# Set random seed for reproducibility
random.seed(42)
np.random.seed(42)

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    # Using 1:1 ratio for simplicity (square-like)
    container_width = 1.0
    container_height = 1.0
    
    # Problem setup
    n_circles = 21
    max_radius = min(container_width, container_height) / 2.0
    
    def evaluate_individual(individual):
        """Evaluate fitness of an individual (set of circles)"""
        # Extract positions and radii from individual
        positions = np.array(individual[::2]).reshape(-1, 2)
        radii = np.array(individual[1::2])
        
        # Check if all radii are valid
        if np.any(radii <= 0) or np.any(radii > max_radius):
            return (-np.inf,)  # Invalid solution
        
        # Check bounds
        if np.any(positions[:, 0] - radii < 0) or np.any(positions[:, 0] + radii > container_width):
            return (-np.inf,)
        if np.any(positions[:, 1] - radii < 0) or np.any(positions[:, 1] + radii > container_height):
            return (-np.inf,)
            
        # Calculate total radius sum
        total_radius_sum = np.sum(radii)
        
        # Penalty for overlaps
        penalty = 0
        
        # Check for overlaps using KDTree for efficiency
        try:
            tree = cKDTree(positions)
            pairs = tree.query_pairs(radii.sum() + 1e-8)  # Small epsilon for neighbor check
            
            for i, j in pairs:
                dist_sq = np.sum((positions[i] - positions[j]) ** 2)
                min_dist_sq = (radii[i] + radii[j]) ** 2
                
                if dist_sq < min_dist_sq:
                    # Overlap penalty - proportional to how much they overlap
                    overlap = min_dist_sq - dist_sq
                    penalty += overlap
                    
        except Exception:
            penalty = np.inf
            
        # Return negative because DEAP minimizes
        return (-penalty - 1e6 * (penalty > 0), total_radius_sum)
    
    def create_individual():
        """Create a random valid individual"""
        individual = []
        for _ in range(n_circles):
            # Random position within bounds
            x = np.random.uniform(0, container_width)
            y = np.random.uniform(0, container_height)
            # Random radius within limits
            r = np.random.uniform(0.001, max_radius * 0.9)
            
            individual.extend([x, y, r])
        
        return individual
    
    def mutate_individual(individual, indpb=0.1):
        """Mutate an individual"""
        for i in range(len(individual)):
            if random.random() < indpb:
                if i % 3 == 0:  # x coordinate
                    individual[i] = np.clip(individual[i] + np.random.normal(0, 0.05), 0, container_width)
                elif i % 3 == 1:  # y coordinate
                    individual[i] = np.clip(individual[i] + np.random.normal(0, 0.05), 0, container_height)
                else:  # radius
                    individual[i] = np.clip(individual[i] + np.random.normal(0, 0.02), 0.001, max_radius * 0.9)
        
        return individual,
    
    def cx_uniform(ind1, ind2, indpb=0.3):
        """Uniform crossover"""
        for i in range(len(ind1)):
            if random.random() < indpb:
                ind1[i], ind2[i] = ind2[i], ind1[i]
        return ind1, ind2
    
    # Setup DEAP framework
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)
    
    toolbox = base.Toolbox()
    toolbox.register("individual", create_individual)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate_individual)
    toolbox.register("mate", cx_uniform)
    toolbox.register("mutate", mutate_individual)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Multi-start approach
    best_individual = None
    best_fitness = -np.inf
    best_total_radius = 0
    
    # Run multiple optimization processes
    for start_seed in range(5):
        random.seed(start_seed)
        np.random.seed(start_seed)
        
        population_size = 50
        generations = 50
        
        # Create initial population
        pop = toolbox.population(n=population_size)
        
        # Evaluate initial population
        fitnesses = list(map(toolbox.evaluate, pop))
        for ind, fit in zip(pop, fitnesses):
            ind.fitness.values = fit
        
        # Evolution loop
        for gen in range(generations):
            # Select the next generation individuals
            offspring = toolbox.select(pop, len(pop))
            offspring = list(map(toolbox.clone, offspring))
            
            # Apply crossover and mutation
            for child1, child2 in zip(offspring[::2], offspring[1::2]):
                if random.random() < 0.5:
                    toolbox.mate(child1, child2)
                    del child1.fitness.values
                    del child2.fitness.values

            for mutant in offspring:
                if random.random() < 0.2:
                    toolbox.mutate(mutant)
                    del mutant.fitness.values
            
            # Evaluate the individuals with an invalid fitness
            invalid_ind = [ind for ind in offspring if not ind.fitness.valid]
            fitnesses = map(toolbox.evaluate, invalid_ind)
            for ind, fit in zip(invalid_ind, fitnesses):
                ind.fitness.values = fit
            
            # Replace the old population with the new population
            pop[:] = offspring
            
            # Find best individual in this run
            best_in_run = tools.selBest(pop, 1)[0]
            if best_in_run.fitness.values[0] > best_fitness:
                best_fitness = best_in_run.fitness.values[0]
                best_total_radius = best_in_run.fitness.values[1]
                best_individual = best_in_run[:]
    
    # If no good solution found, use a deterministic approach
    if best_individual is None or best_fitness == -np.inf:
        # Use a simple grid-based approach for fallback
        best_individual = []
        # Create a regular grid pattern
        rows = int(np.ceil(np.sqrt(n_circles)))
        cols = int(np.ceil(n_circles / rows))
        
        cell_width = container_width / cols
        cell_height = container_height / rows
        max_radius = min(cell_width, cell_height) / 2.0
        
        count = 0
        for i in range(rows):
            for j in range(cols):
                if count >= n_circles:
                    break
                x = (j + 0.5) * cell_width
                y = (i + 0.5) * cell_height
                r = max_radius * 0.8  # Slightly reduce to allow for spacing
                best_individual.extend([x, y, r])
                count += 1
        
        # Ensure we have exactly n_circles
        while len(best_individual) < n_circles * 3:
            best_individual.append(max_radius * 0.8)
    
    # Convert final solution to numpy array
    circles = np.array(best_individual).reshape(-1, 3)
    return circles

# EVOLVE-BLOCK-END

if __name__ == "__main__":
    start_time = time.time()
    circles = circle_packing21()
    end_time = time.time()
    
    sum_radii = np.sum(circles[:, 2])
    print(f"Radii sum: {sum_radii}")
    print(f"Execution time: {end_time - start_time:.4f} seconds")
    print(f"Benchmark ratio: {sum_radii / 2.3658321334167627:.6f}")
