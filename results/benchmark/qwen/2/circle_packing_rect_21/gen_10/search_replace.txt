def evaluate_circles(circles, container_width=1.0, container_height=1.0):
    """
    Evaluate the fitness of a configuration of circles.

    Args:
        circles: array of shape (n, 3) where each row is (x, y, r)
        container_width: width of the container rectangle
        container_height: height of the container rectangle

    Returns:
        fitness: tuple (sum_of_radii, penalty)
    """
    n = len(circles)
    if n == 0:
        return 0.0, float('inf')

    # Extract positions and radii
    positions = circles[:, :2]
    radii = circles[:, 2]

    # Calculate sum of radii (maximize this)
    sum_radii = np.sum(radii)

    # Penalty for circles outside container bounds
    penalty = 0.0

    # Check boundary constraints
    for i in range(n):
        x, y, r = circles[i]
        # Circle must fit within container
        if x - r < 0 or x + r > container_width or y - r < 0 or y + r > container_height:
            penalty += 1000.0  # Large penalty for out-of-bounds circles

    # Penalty for overlapping circles
    distances = cdist(positions, positions)
    for i in range(n):
        for j in range(i+1, n):
            if i != j:
                dist = distances[i, j]
                r_i, r_j = radii[i], radii[j]
                # Circles overlap if distance < sum of radii
                if dist < r_i + r_j:
                    overlap = (r_i + r_j - dist)  # Overlap amount
                    penalty += overlap * 100.0  # Penalty based on overlap amount

    # Return negative sum of radii (since we minimize in NSGA-II) plus penalty
    return -sum_radii, penalty
