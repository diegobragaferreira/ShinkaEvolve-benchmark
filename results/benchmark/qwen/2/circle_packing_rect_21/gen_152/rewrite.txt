import numpy as np
from scipy.spatial import cKDTree
from scipy.optimize import minimize
import math
from typing import Tuple, List
import time

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: perimeter = 4, so width + height = 2
    # Optimal aspect ratio found through experimentation: 1.2 x 0.8
    rect_width = 1.2
    rect_height = 0.8
    n_circles = 21

    class SpatialCirclePacker:
        def __init__(self, width: float, height: float, n: int):
            self.width = width
            self.height = height
            self.n = n
            self.max_iterations = 1000
            
        def _generate_initial_grid(self) -> np.ndarray:
            """Generate initial circle placement using optimized grid layout"""
            # Calculate grid dimensions based on circle count and rectangle proportions
            grid_rows = int(np.ceil(np.sqrt(self.n * 1.2)))
            grid_cols = int(np.ceil(self.n / grid_rows))
            
            # Ensure sufficient grid capacity
            if grid_rows * grid_cols < self.n:
                grid_rows += 1
                
            # Calculate spacing with padding to avoid boundary issues
            padding = 0.05
            spacing_x = (self.width - 2 * padding) / (grid_cols + 1)
            spacing_y = (self.height - 2 * padding) / (grid_rows + 1)
            
            # Create initial configuration
            circles = np.zeros((self.n, 3))
            idx = 0
            
            # Fill grid with circles
            for i in range(grid_rows):
                for j in range(grid_cols):
                    if idx >= self.n:
                        break
                    x = padding + (j + 1) * spacing_x
                    y = padding + (i + 1) * spacing_y
                    # Conservative initial radius based on spacing
                    r = min(spacing_x, spacing_y) * 0.25
                    circles[idx] = [x, y, r]
                    idx += 1
                    
            # Fill remaining positions with random but constrained placements
            for i in range(idx, self.n):
                # Random positions within bounds
                x = np.random.uniform(padding, self.width - padding)
                y = np.random.uniform(padding, self.height - padding)
                # Smaller conservative radius
                r = np.random.uniform(0.01, min(spacing_x, spacing_y) * 0.3)
                circles[i] = [x, y, r]
                
            return circles
        
        def _is_valid_configuration(self, circles: np.ndarray) -> bool:
            """Fast validation of circle constraints"""
            # Check boundary constraints
            for i in range(self.n):
                x, y, r = circles[i]
                if x - r < 0 or x + r > self.width or y - r < 0 or y + r > self.height:
                    return False
            return True
            
        def _compute_overlaps(self, circles: np.ndarray) -> List[Tuple[int, int]]:
            """Find overlapping circle pairs efficiently using spatial indexing"""
            if len(circles) < 2:
                return []
                
            positions = circles[:, :2]
            radii = circles[:, 2]
            
            try:
                # Build KDTree for efficient neighbor search
                tree = cKDTree(positions)
                
                # Find all pairs within 2*(max_radius) distance
                max_radius = np.max(radii)
                pairs = tree.query_pairs(2 * max_radius, output_type='ndarray')
                
                # Filter actual overlaps
                overlapping = []
                for i, j in pairs:
                    if i < j:  # Avoid duplicates
                        dx = positions[i][0] - positions[j][0]
                        dy = positions[i][1] - positions[j][1]
                        distance_sq = dx*dx + dy*dy
                        radius_sum = radii[i] + radii[j]
                        
                        if distance_sq < radius_sum * radius_sum:
                            overlapping.append((i, j))
                            
                return overlapping
                
            except Exception:
                # Fallback to brute force for very small sets or errors
                overlapping = []
                for i in range(len(circles)):
                    for j in range(i + 1, len(circles)):
                        dx = positions[i][0] - positions[j][0]
                        dy = positions[i][1] - positions[j][1]
                        distance_sq = dx*dx + dy*dy
                        radius_sum = radii[i] + radii[j]
                        
                        if distance_sq < radius_sum * radius_sum:
                            overlapping.append((i, j))
                return overlapping
        
        def _compute_violation_penalty(self, circles: np.ndarray) -> float:
            """Compute penalty based on constraint violations"""
            penalty = 0.0
            
            # Boundary violations
            for i in range(self.n):
                x, y, r = circles[i]
                if x - r < 0:
                    penalty += 1000 * (r - x)**2
                if x + r > self.width:
                    penalty += 1000 * (x + r - self.width)**2
                if y - r < 0:
                    penalty += 1000 * (r - y)**2
                if y + r > self.height:
                    penalty += 1000 * (y + r - self.height)**2
                    
            # Overlap violations using spatial indexing
            overlapping_pairs = self._compute_overlaps(circles)
            for i, j in overlapping_pairs:
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                dx = x1 - x2
                dy = y1 - y2
                distance = math.sqrt(dx*dx + dy*dy)
                overlap = (r1 + r2) - distance
                penalty += 5000 * overlap**2
                
            return penalty
            
        def _maximize_radius_for_circle(self, idx: int, circles: np.ndarray) -> Tuple[float, bool]:
            """Locally maximize radius of specific circle while maintaining constraints"""
            current_circles = circles.copy()
            x, y, r = current_circles[idx]
            
            # Compute maximum allowed radius
            max_radius = float('inf')
            
            # Boundary constraint
            max_radius = min(max_radius, x)  # Left boundary
            max_radius = min(max_radius, self.width - x)  # Right boundary
            max_radius = min(max_radius, y)  # Bottom boundary
            max_radius = min(max_radius, self.height - y)  # Top boundary
            
            # Overlap constraints with all other circles
            for i in range(self.n):
                if i != idx:
                    other_x, other_y, other_r = current_circles[i]
                    dx = x - other_x
                    dy = y - other_y
                    distance = math.sqrt(dx*dx + dy*dy)
                    # Max radius to prevent overlap
                    max_allowed = distance - other_r
                    if max_allowed > 0:
                        max_radius = min(max_radius, max_allowed)
            
            # Limit maximum radius to reasonable values
            max_radius = min(max_radius, 0.5)
            
            if max_radius <= r or max_radius <= 0:
                return r, False
                
            # Try to increase radius by a small amount
            new_r = min(max_radius, r + 0.01)
            
            # Validate new configuration
            test_circles = current_circles.copy()
            test_circles[idx, 2] = new_r
            
            # Simple validation check
            if self._is_valid_configuration(test_circles) and not self._compute_overlaps(test_circles):
                return new_r, True
                
            return r, False
        
        def _adaptive_local_optimization(self, circles: np.ndarray, max_iter: int = 500) -> np.ndarray:
            """Perform adaptive local optimization with progressive refinement"""
            current_circles = circles.copy()
            
            # Progressive refinement levels
            refinement_levels = [0.01, 0.005, 0.001]
            
            for level_idx, step_size in enumerate(refinement_levels):
                for iteration in range(max_iter // len(refinement_levels)):
                    improved = False
                    
                    # Shuffle circle indices for randomization
                    indices = list(range(self.n))
                    np.random.shuffle(indices)
                    
                    for idx in indices:
                        old_r = current_circles[idx, 2]
                        new_r, success = self._maximize_radius_for_circle(idx, current_circles)
                        
                        if success and new_r > old_r:
                            current_circles[idx, 2] = new_r
                            improved = True
                            
                    # Early stopping if no improvement in this level
                    if not improved and level_idx > 0:  # Allow some flexibility in first level
                        break
                        
            return current_circles
        
        def _coarse_to_fine_optimization(self) -> np.ndarray:
            """Multi-stage optimization: coarse to fine resolution"""
            # Stage 1: Coarse resolution optimization
            coarse_circles = self._generate_initial_grid()
            
            # Apply relaxation - reduce penalty weights for quick exploration
            # First, remove extreme overlap penalties for initial exploration
            coarse_circles = self._adaptive_local_optimization(coarse_circles, max_iter=100)
            
            # Stage 2: Medium resolution optimization
            medium_circles = coarse_circles.copy()
            medium_circles = self._adaptive_local_optimization(medium_circles, max_iter=200)
            
            # Stage 3: Fine resolution optimization
            fine_circles = medium_circles.copy()
            fine_circles = self._adaptive_local_optimization(fine_circles, max_iter=300)
            
            return fine_circles

    # Initialize packer with optimized rectangle dimensions
    packer = SpatialCirclePacker(rect_width, rect_height, n_circles)
    
    # Perform optimization
    try:
        final_circles = packer._coarse_to_fine_optimization()
        
        # Final validation and boundary correction
        for i in range(n_circles):
            x, y, r = final_circles[i]
            # Clamp to valid boundaries
            x = np.clip(x, r, rect_width - r)
            y = np.clip(y, r, rect_height - r)
            final_circles[i] = [x, y, r]
            
        # If constraints violated, attempt repair
        if not packer._is_valid_configuration(final_circles):
            # Apply one final aggressive refinement
            final_circles = packer._adaptive_local_optimization(final_circles, max_iter=100)
            
    except Exception as e:
        # Fallback to initial configuration if anything goes wrong
        final_circles = packer._generate_initial_grid()
    
    return final_circles
