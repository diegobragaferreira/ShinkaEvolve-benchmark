        grid_size = int(math.ceil(math.sqrt(self.n_circles)))

        # Adjust grid size based on container aspect ratio for better packing
        aspect_ratio = self.container_width / self.container_height
        if aspect_ratio > 1:
            # Wide container - more columns
            grid_cols = max(1, int(math.ceil(math.sqrt(self.n_circles * aspect_ratio))))
            grid_rows = max(1, int(math.ceil(self.n_circles / grid_cols)))

            # Tall container - more rows
