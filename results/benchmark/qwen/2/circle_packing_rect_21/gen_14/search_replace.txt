import numpy as np
import math


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    # Using a 1:1 square ratio for simplicity
    rect_width = 1.0
    rect_height = 1.0

    # Set seed for reproducibility
    np.random.seed(42)

    # Initialize circles array
    circles = np.zeros((21, 3))

    # Use a grid-based approach for initial placement
    # Try to place circles in a roughly 5x5 grid with some randomness
    rows = 5
    cols = 5

    # Adjust grid size to fit exactly 21 circles
    actual_rows = min(rows, 21)
    actual_cols = math.ceil(21 / actual_rows)

    # Calculate spacing to avoid edge issues
    margin = 0.05
    cell_width = (rect_width - 2*margin) / actual_cols
    cell_height = (rect_height - 2*margin) / actual_rows

    # Place circles with increasing radii
    placed_count = 0

    for i in range(actual_rows):
        for j in range(actual_cols):
            if placed_count >= 21:
                break

            # Position with some jitter
            x = margin + (j + 0.5) * cell_width + np.random.uniform(-cell_width*0.1, cell_width*0.1)
            y = margin + (i + 0.5) * cell_height + np.random.uniform(-cell_height*0.1, cell_height*0.1)

            # Radius initially set to a reasonable fraction of cell size
            max_radius = min(cell_width, cell_height) * 0.4
            r = max_radius * np.random.uniform(0.8, 1.0)

            # Ensure circles stay within bounds (especially near edges)
            r = min(r, x, rect_width - x, y, rect_height - y)

            circles[placed_count] = [x, y, r]
            placed_count += 1

        if placed_count >= 21:
            break

    # Refine the solution with a simple greedy approach
    # This is a basic improvement step - we could do much better with proper optimization
    # But for now this gives a reasonable starting point that beats zero initialization
    for _ in range(100):  # Limited iterations for quick response
        # Try small adjustments to increase radii where possible
        for idx in range(21):
            x, y, r = circles[idx]

            # Try to increase radius while keeping it valid
            # Don't allow radius to exceed boundaries
            max_possible_radius = min(x, rect_width - x, y, rect_height - y)

            if max_possible_radius > r and r < max_possible_radius:
                # Add a small increment to radius (but don't violate other constraints)
                new_r = min(max_possible_radius, r + 0.01)

                # Check if we can actually increase radius without violating others
                valid = True
                for other_idx in range(21):
                    if other_idx != idx:
                        other_x, other_y, other_r = circles[other_idx]
                        dist = math.sqrt((x - other_x)**2 + (y - other_y)**2)
                        if dist < (new_r + other_r):
                            valid = False
                            break

                if valid:
                    circles[idx][2] = new_r

    return circles
