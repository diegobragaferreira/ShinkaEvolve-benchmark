    rect_width, rect_height = 1.0, 1.0

    # Initialize to a good hexagonal packing pattern as starting point
    # This will give us a reasonable initial configuration
    n = 21
    circles = np.zeros((n, 3))

    # We'll place circles in a grid-like fashion that approximates hexagonal packing
    rows = 5
    cols = 5

        rows, cols = 3, 3
    elif n <= 16:
        rows, cols = 4, 4
    else:
        rows, cols = 5, 5

    spacing_x = rect_width / (cols + 1)
    spacing_y = rect_height / (rows + 1)

    idx = 0

        for j in range(cols):
            if idx >= n:
                break
            x = (j + 1) * spacing_x
            y = (i + 1) * spacing_y
            # Initial radius - small but within bounds

        y = np.random.uniform(0.1, rect_height - 0.1)
        r = np.random.uniform(0.01, 0.1)
        circles[i] = [x, y, r]

    # Simple local optimization: try to increase radii while respecting constraints
    # Use a greedy approach - for each circle, try to increase its radius as much as possible
    max_iterations = 100

            max_radius = current_r
