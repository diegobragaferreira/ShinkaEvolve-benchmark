    def crossover(self, parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
        """Perform probability-based crossover between two parents, favoring better performers"""
        child = parent1.copy()

        # Calculate fitness difference to determine crossover probabilities
        fitness1, _ = self.calculate_fitness(parent1)
        fitness2, _ = self.calculate_fitness(parent2)

        # Use softmax to convert fitness to probabilities (higher fitness = higher probability)
        # This ensures that better solutions contribute more to offspring
        fit_diff = fitness1 - fitness2
        # Use a temperature parameter to control how much difference matters
        temp = 0.5
        prob_parent1 = 1 / (1 + np.exp(-fit_diff / temp))

        # For each circle, decide inheritance based on fitness-based probability
        for i in range(self.num_circles):
            # Probability of selecting from parent1 increases with fitness difference
            if np.random.rand() < prob_parent1:
                child[i] = parent1[i].copy()
            else:
                child[i] = parent2[i].copy()

        return child
