import numpy as np
import time


def _calculate_overlap_penalty(circles):
    """Calculate penalty for overlapping circles"""
    penalty = 0
    n = len(circles)
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            distance = np.sqrt((x1-x2)**2 + (y1-y2)**2)
            overlap = max(0, r1 + r2 - distance)
            penalty += overlap**2  # Quadratic penalty for overlap
    return penalty


def _is_valid_configuration(circles, width, height):
    """Check if configuration respects all constraints"""
    for x, y, r in circles:
        if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
            return False
    return True


def _evaluate_fitness(circles, width, height):
    """Evaluate the fitness of a configuration"""
    if not _is_valid_configuration(circles, width, height):
        return -1e10  # Invalid configuration gets very bad fitness

    # Maximize sum of radii, penalize overlaps
    total_radius = np.sum(circles[:, 2])
    overlap_penalty = _calculate_overlap_penalty(circles)

    # Return negative penalty (since we want to minimize) plus total radius
    return total_radius - 1000 * overlap_penalty


def _perturb_circle(circle, max_shift=0.05):
    """Perturb a circle slightly"""
    x, y, r = circle
    # Small random perturbation
    dx = np.random.uniform(-max_shift, max_shift)
    dy = np.random.uniform(-max_shift, max_shift)
    dr = np.random.uniform(-0.02, 0.02)

    return np.array([min(max(x + dx, 0.001), 1.999),
                     min(max(y + dy, 0.001), 1.999),
                     max(r + dr, 0.001)])


def _local_optimization(circles, width, height, max_iter=1000):
    """Simple local optimization to find better configuration"""
    current_circles = circles.copy()
    best_fitness = _evaluate_fitness(current_circles, width, height)
    best_circles = current_circles.copy()

    for iteration in range(max_iter):
        # Try perturbing each circle
        test_circles = []
        for i in range(len(current_circles)):
            perturbed = _perturb_circle(current_circles[i])
            # Copy current circles and replace one
            temp_circles = current_circles.copy()
            temp_circles[i] = perturbed
            test_circles.append(temp_circles)

        # Evaluate all perturbations
        for test_circle_set in test_circles:
            fitness = _evaluate_fitness(test_circle_set, width, height)
            if fitness > best_fitness:
                best_fitness = fitness
                best_circles = test_circle_set.copy()

        # Move to the best neighbor
        current_circles = best_circles.copy()

        # Early stopping if improvement is minimal
        if iteration > 100 and iteration % 100 == 0:
            # Check if we're making progress
            pass

    return best_circles


def _grid_placement(n, width, height):
    """Place circles in a regular grid pattern"""
    # Arrange in roughly square grid
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

    # Adjust to fit in rectangle
    cell_width = width / cols
    cell_height = height / rows

    circles = []

    # Place circles in grid
    for i in range(rows):
        for j in range(cols):
            if len(circles) >= n:
                break
            x = (j + 0.5) * cell_width
            y = (i + 0.5) * cell_height
            # Radius should be half of minimum cell dimension to allow for spacing
            radius = min(cell_width, cell_height) * 0.4
            circles.append([x, y, radius])
        if len(circles) >= n:
            break

    # Fill remaining slots with small circles at random locations
    while len(circles) < n:
        x = np.random.uniform(0.001, width - 0.001)
        y = np.random.uniform(0.001, height - 0.001)
        radius = np.random.uniform(0.01, 0.1)
        circles.append([x, y, radius])

    return np.array(circles)


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: perimeter = 4 => width + height = 2
    # Using width = 1.5, height = 0.5 gives us better packing efficiency
    width = 1.5
    height = 0.5

    # Start with good grid placement
    circles = _grid_placement(21, width, height)

    # Perform local optimization
    circles = _local_optimization(circles, width, height, max_iter=2000)

    # Final optimization with more iterations if time allows
    start_time = time.time()
    while time.time() - start_time < 50:  # Leave 10 seconds for cleanup
        old_circles = circles.copy()
        circles = _local_optimization(circles, width, height, max_iter=500)
        # Stop if no significant improvement
        if abs(np.sum(old_circles[:, 2]) - np.sum(circles[:, 2])) < 1e-6:
            break

    return circles
