import numpy as np
from scipy.spatial import cKDTree
import random
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.
    Uses physics-based optimization with simulated annealing.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions - perimeter = 4, so width + height = 2
    # Optimized rectangle dimensions for maximum packing efficiency
    rect_width = 1.3
    rect_height = 0.7
    
    # Fixed seed for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    # Physics simulation parameters
    n_circles = 21
    max_iterations = 1000
    temp_start = 1.0
    temp_end = 0.01
    cooling_rate = 0.995
    max_force_iterations = 50
    
    # Initialize circles with grid-based approach
    def initialize_circles():
        # Create a grid layout with random offsets for better distribution
        sqrt_n = np.sqrt(n_circles)
        rows = int(np.ceil(sqrt_n))
        cols = int(np.ceil(n_circles / rows))
        
        cell_width = rect_width / cols
        cell_height = rect_height / rows
        
        # Set initial radius to be proportional to cell size
        max_radius = min(cell_width, cell_height) * 0.4
        
        circles = []
        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx >= n_circles:
                    break
                # Add slight random offset to break symmetry
                x = (j + 0.5 + np.random.uniform(-0.15, 0.15)) * cell_width
                y = (i + 0.5 + np.random.uniform(-0.15, 0.15)) * cell_height
                
                # Ensure positions stay within bounds
                x = np.clip(x, max_radius, rect_width - max_radius)
                y = np.clip(y, max_radius, rect_height - max_radius)
                
                circles.append([x, y, max_radius])
                idx += 1
                
        return np.array(circles)
    
    # Fast collision checking using KDTree
    def fast_collision_check(circles):
        coords = circles[:, :2]
        radii = circles[:, 2]
        
        # Early boundary check
        if np.any((coords[:, 0] - radii < 0) | (coords[:, 0] + radii > rect_width) |
                  (coords[:, 1] - radii < 0) | (coords[:, 1] + radii > rect_height)):
            return False
        
        # Use KDTree for efficient neighbor search
        tree = cKDTree(coords)
        pairs = tree.query_pairs(2 * np.max(radii), p=2)
        
        if len(pairs) == 0:
            return True
            
        # Check actual distances for overlap
        for i, j in pairs:
            if i >= len(circles) or j >= len(circles):
                continue
            x1, y1 = coords[i]
            x2, y2 = coords[j]
            r1, r2 = radii[i], radii[j]
            dx = x1 - x2
            dy = y1 - y2
            dist_sq = dx*dx + dy*dy
            min_dist_sq = (r1 + r2) * (r1 + r2)
            
            if dist_sq < min_dist_sq:
                return False
                
        return True
    
    # Physics-based energy calculation
    def calculate_energy(circles):
        """
        Calculate total energy of the system:
        - Positive energy component: sum of radii (we want to maximize)
        - Negative energy component: penalty for constraint violations
        """
        coords = circles[:, :2]
        radii = circles[:, 2]
        
        # Energy from radii (maximize this)
        radius_energy = np.sum(radii)
        
        # Penalty for boundary violations
        boundary_penalty = 0
        boundary_violations = (coords[:, 0] - radii < 0) | (coords[:, 0] + radii > rect_width) | \
                              (coords[:, 1] - radii < 0) | (coords[:, 1] + radii > rect_height)
        boundary_penalty = -1000 * np.sum(boundary_violations.astype(float))
        
        # Penalty for overlaps using spatial indexing
        overlap_penalty = 0
        if len(circles) > 1:
            tree = cKDTree(coords)
            pairs = tree.query_pairs(2 * np.max(radii), p=2)
            
            for i, j in pairs:
                x1, y1 = coords[i]
                x2, y2 = coords[j]
                r1, r2 = radii[i], radii[j]
                dx = x1 - x2
                dy = y1 - y2
                dist_sq = dx*dx + dy*dy
                min_dist_sq = (r1 + r2) * (r1 + r2)
                
                if dist_sq < min_dist_sq:
                    overlap_penalty -= 1000 * (min_dist_sq - dist_sq)
        
        total_energy = radius_energy + boundary_penalty + overlap_penalty
        return total_energy
    
    # Calculate forces between circles
    def calculate_forces(circles, temp):
        coords = circles[:, :2]
        radii = circles[:, 2]
        
        # Initialize force array
        forces = np.zeros_like(coords)
        
        # Repulsion forces between overlapping circles
        tree = cKDTree(coords)
        pairs = tree.query_pairs(2 * np.max(radii), p=2)
        
        for i, j in pairs:
            x1, y1 = coords[i]
            x2, y2 = coords[j]
            r1, r2 = radii[i], radii[j]
            
            dx = x1 - x2
            dy = y1 - y2
            dist_sq = dx*dx + dy*dy
            
            # Only apply forces if circles are overlapping or nearly overlapping
            min_dist_sq = (r1 + r2) * (r1 + r2)
            if dist_sq < min_dist_sq * 1.1:  # Allow some buffer
                dist = np.sqrt(dist_sq)
                if dist > 0.001:  # Avoid division by zero
                    # Repulsive force (inversely proportional to distance squared)
                    force_magnitude = (min_dist_sq - dist_sq) / (dist_sq + 0.001)
                    force_magnitude *= 0.1  # Scale factor
                    # Apply temperature scaling
                    force_magnitude *= temp
                    fx = force_magnitude * dx / dist
                    fy = force_magnitude * dy / dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    # Apply equal and opposite force to second circle
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Attractive forces to center (encourage spreading out)
        center_x, center_y = rect_width / 2, rect_height / 2
        for i in range(len(coords)):
            x, y = coords[i]
            dx = center_x - x
            dy = center_y - y
            dist = np.sqrt(dx*dx + dy*dy)
            
            if dist > 0.001:
                # Attractive force (towards center)
                force_magnitude = 0.001 * dist * temp
                forces[i, 0] += force_magnitude * dx / dist
                forces[i, 1] += force_magnitude * dy / dist
        
        return forces
    
    # Apply movement with boundary constraints
    def apply_movement(circles, forces, step_size=0.1):
        new_circles = circles.copy()
        
        for i in range(len(circles)):
            x, y, r = new_circles[i]
            fx, fy = forces[i]
            
            # Apply force to position
            new_x = x + fx * step_size
            new_y = y + fy * step_size
            
            # Apply boundary constraints
            new_x = np.clip(new_x, r, rect_width - r)
            new_y = np.clip(new_y, r, rect_height - r)
            
            new_circles[i] = [new_x, new_y, r]
        
        return new_circles
    
    # Main physics optimization loop
    def physics_optimization():
        # Initialize circles
        circles = initialize_circles()
        best_circles = circles.copy()
        best_energy = calculate_energy(circles)
        
        # Temperature schedule
        temperature = temp_start
        
        # Main optimization loop
        for iteration in range(max_iterations):
            # Calculate forces
            forces = calculate_forces(circles, temperature)
            
            # Apply movement
            circles = apply_movement(circles, forces)
            
            # Check for collisions and fix if needed
            if not fast_collision_check(circles):
                # Apply simple corrections for boundary and overlap violations
                for i in range(len(circles)):
                    x, y, r = circles[i]
                    
                    # Fix boundary violations
                    if x - r < 0:
                        x = r
                    elif x + r > rect_width:
                        x = rect_width - r
                        
                    if y - r < 0:
                        y = r
                    elif y + r > rect_height:
                        y = rect_height - r
                        
                    circles[i] = [x, y, r]
            
            # Evaluate energy
            current_energy = calculate_energy(circles)
            
            # Update best solution
            if current_energy > best_energy:
                best_energy = current_energy
                best_circles = circles.copy()
            
            # Cool down temperature
            temperature = max(temp_end, temperature * cooling_rate)
            
            # Occasionally reset with better initialization if stuck
            if iteration % 50 == 0 and iteration > 0:
                # Random restart with better spread
                restart_circles = initialize_circles()
                restart_energy = calculate_energy(restart_circles)
                if restart_energy > best_energy:
                    best_energy = restart_energy
                    best_circles = restart_circles.copy()
                    circles = restart_circles.copy()
        
        return best_circles
    
    # Run physics-based optimization
    final_circles = physics_optimization()
    
    # Final validation and cleanup
    if not fast_collision_check(final_circles):
        # Try one final refinement pass with simple adjustment
        coords = final_circles[:, :2]
        radii = final_circles[:, 2]
        
        # Simple iterative improvement
        for _ in range(100):
            improved = False
            for i in range(len(final_circles)):
                x, y, r = final_circles[i]
                
                # Try to slightly adjust position to resolve overlaps
                for j in range(len(final_circles)):
                    if i != j:
                        x2, y2, r2 = final_circles[j]
                        dx = x - x2
                        dy = y - y2
                        dist = np.sqrt(dx*dx + dy*dy)
                        
                        if dist < r + r2:
                            # Move away from overlapping circle
                            if dist > 0.001:
                                scale = 0.01 * (r + r2 - dist) / dist
                                x += dx * scale
                                y += dy * scale
                                improved = True
                            
                            # Also reduce radius if necessary
                            new_r = max(0.001, (dist - 0.001) / 2)
                            if new_r < r:
                                r = new_r
                                improved = True
                
                # Apply boundary constraints
                x = np.clip(x, r, rect_width - r)
                y = np.clip(y, r, rect_height - r)
                final_circles[i] = [x, y, r]
                
                if improved:
                    break
        
        # Final collision check
        if not fast_collision_check(final_circles):
            # If still invalid, return the best valid solution so far
            return best_circles
    
    # Ensure we have valid circles
    coords = final_circles[:, :2]
    radii = final_circles[:, 2]
    # Fix boundary issues that might remain
    for i in range(len(final_circles)):
        x, y, r = final_circles[i]
        x = np.clip(x, r, rect_width - r)
        y = np.clip(y, r, rect_height - r)
        final_circles[i] = [x, y, r]
    
    return final_circles

if __name__ == "__main__":
