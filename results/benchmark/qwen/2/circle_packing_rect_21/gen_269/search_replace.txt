    # Generate grid-based initial placement using adaptive hexagonal packing pattern
    def generate_adaptive_hexagonal_grid(num_circles, width, height):
        # Calculate adaptive spacing based on rectangle area and circle count
        # Formula: spacing = sqrt((width * height) / n) - proven effective approach
        total_area = width * height
        spacing = np.sqrt(total_area / num_circles) * 0.8  # Scale factor for safety

        # Ensure minimum spacing
        spacing = max(spacing, 0.05)

        # Calculate grid dimensions
        cols = max(1, int(np.ceil(width / spacing)))
        rows = max(1, int(np.ceil(height / spacing)))

        # Adjust grid to roughly fit the required number of circles
        actual_circles = cols * rows
        if actual_circles < num_circles:
            # Increase grid size to accommodate more circles
            cols = max(cols, int(np.ceil(np.sqrt(num_circles * width / height))))
            rows = max(rows, int(np.ceil(num_circles / cols)))

        # Calculate actual spacing to fit within bounds
        actual_spacing_x = width / cols if cols > 0 else spacing
        actual_spacing_y = height / rows if rows > 0 else spacing

        # For hexagonal packing, adjust y spacing
        hex_spacing_y = actual_spacing_y * np.sqrt(3) / 2

        circles = []

        # Fill grid with circles
        for i in range(rows):
            for j in range(cols):
                if len(circles) >= num_circles:
                    break

                # Hexagonal offset for odd rows
                x_offset = (i % 2) * (actual_spacing_x / 2)
                x = (j * actual_spacing_x) + x_offset + spacing/2
                y = (i * hex_spacing_y) + spacing/2

                # Ensure circle fits within bounds with margin
                if (x >= spacing/2 and x <= width - spacing/2 and
                    y >= spacing/2 and y <= height - spacing/2):

                    # Use adaptive radius based on spacing
                    r = min(spacing/3, width/10, height/10)
                    # Ensure minimum radius
                    r = max(r, 0.01)
                    circles.append([x, y, r])

            if len(circles) >= num_circles:
                break

        # If we didn't place enough circles, fill remaining space randomly
        while len(circles) < num_circles:
            x = np.random.uniform(spacing/2, width - spacing/2)
            y = np.random.uniform(spacing/2, height - spacing/2)

            max_radius = min(x, width-x, y, height-y) * 0.3
            r = max(0.01, min(max_radius, np.random.uniform(0.02, spacing/3)))
            circles.append([x, y, r])

        return np.array(circles)
