<DESCRIPTION>
Implementing an adaptive local search mechanism that automatically adjusts its intensity based on the solution quality and constraint violation levels. This approach will enhance the optimization by applying different refinement strategies:
1. Light refinement for good solutions with few violations
2. Medium refinement for moderate solutions with some violations
3. Heavy refinement for poor solutions with many violations

This builds upon the existing local_optimize function but makes it context-aware and more aggressive when needed, which should help escape local optima and improve overall convergence towards the benchmark.

def local_optimize(circles: np.ndarray, max_iterations: int = 20) -> np.ndarray:
    """Apply local optimization to improve a solution by making small adjustments."""
    optimized = circles.copy()

    # Try to increase radii where possible
    for _ in range(max_iterations):
        improved = False

        # For each circle, try to increase its radius
        for i in range(len(optimized)):
            current_r = optimized[i, 2]

            # Try increasing radius slightly
            new_r = min(current_r * 1.05, 0.5)  # Cap at reasonable size

            # Check if we can increase radius without violating constraints
            valid = True
            for j in range(len(optimized)):
                if i != j:
                    dist = np.sqrt((optimized[i, 0] - optimized[j, 0])**2 +
                                 (optimized[i, 1] - optimized[j, 1])**2)
                    if dist < new_r + optimized[j, 2]:
                        valid = False
                        break

            # Update if improvement is possible and within bounds
            if valid and new_r <= 1 - max(optimized[i, 0], 1 - optimized[i, 0]) and \
               new_r <= 1 - max(optimized[i, 1], 1 - optimized[i, 1]):
                optimized[i, 2] = new_r
                improved = True

        # If no improvement made, stop early
        if not improved:
            break

    return optimized

def count_overlaps(circles: np.ndarray) -> int:
    """Count total number of overlap violations in the current configuration."""
    n = len(circles)
    overlap_count = 0

    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if distance < r1 + r2:
                overlap_count += 1

    return overlap_count
