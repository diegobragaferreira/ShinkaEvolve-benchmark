    Apply local refinement to improve the circle packing solution using constrained optimization.
    This helps fine-tune the solution after evolutionary search.
    """
    n_circles = len(circles)

    # Flatten the circles array for optimization (x, y, r for each circle)
    initial_params = circles.flatten()

    def objective(params):
        # Reshape parameters back to circles
        circles_test = params.reshape(-1, 3)
        return -np.sum(circles_test[:, 2])  # Negative because we want to maximize

    def constraint_func(params):
        # This constraint ensures circles don't overlap and stay within bounds
        circles_test = params.reshape(-1, 3)

        # Check containment constraints
        containment = []
        for i in range(n_circles):
            x, y, r = circles_test[i]
            containment.extend([
                r - x,  # x must be >= r
                r - y,  # y must be >= r
                x + r - 1,  # x + r must be <= 1
                y + r - 1   # y + r must be <= 1

                min_distance_sq = (r1 + r2)**2
                overlap.append(min_distance_sq - distance_sq)  # Should be >= 0

    bounds = []
