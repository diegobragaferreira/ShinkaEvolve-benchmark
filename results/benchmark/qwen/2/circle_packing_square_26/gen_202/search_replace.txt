    """Apply adaptive mutation to an individual with three-phase scheduling"""
    mutated = individual.copy()
    n = len(mutated)

    # Three-phase adaptive mutation rate scheduling
    if generation < GENERATIONS * 0.4:
        # Phase 1: High exploration (generations 1-80)
        mutation_rate = 0.15
    elif generation < GENERATIONS * 0.7:
        # Phase 2: Balanced refinement (generations 81-140)
        mutation_rate = 0.05
    else:
        # Phase 3: Exploitation (generations 141-200)
        mutation_rate = 0.015

    # Increase mutation rate when diversity is low (for exploration)
    if diversity < 0.3:
        mutation_rate *= 1.5

                    step_size = 0.1

                    # Balanced phase - moderate moves
                    step_size = 0.05

                    step_size = 0.01
