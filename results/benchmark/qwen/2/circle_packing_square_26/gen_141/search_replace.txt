def adaptive_local_search(circles, max_iterations=50):
    """
    Apply adaptive local search to improve circle packing quality.
    Uses different refinement intensities based on solution characteristics.
    """
    improved = circles.copy()
    n = len(improved)

    # Count overlap violations
    overlap_count = 0
    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = improved[i]
            x2, y2, r2 = improved[j]
            distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if distance < r1 + r2:
                overlap_count += 1

    # Determine refinement intensity based on overlap severity
    if overlap_count == 0:
        # No overlaps - focus on radius maximization
        refinement_intensity = 3
    elif overlap_count <= 5:
        # Few overlaps - moderate refinement
        refinement_intensity = 2
    else:
        # Many overlaps - aggressive refinement
        refinement_intensity = 1

    # Apply refinement iterations
    for iteration in range(refinement_intensity * max_iterations):
        improved_any = False

        # Try to increase radii first (with limited iterations)
        if iteration < max_iterations // 2:
            for i in range(n):
                x, y, r = improved[i]
                max_r = min(x, 1-x, y, 1-y)

                if max_r > r:
                    # Binary search for maximum possible radius
                    left, right = r, max_r
                    best_r = r

                    for _ in range(8):  # Limit binary search steps
                        mid = (left + right) / 2
                        valid = True
                        for j in range(n):
                            if i != j:
                                x2, y2, r2 = improved[j]
                                dist = np.sqrt((x - x2)**2 + (y - y2)**2)
                                if dist < mid + r2:
                                    valid = False
                                    break

                        if valid:
                            best_r = mid
                            left = mid
                        else:
                            right = mid

                    if best_r > r:
                        improved[i, 2] = best_r
                        improved_any = True

        # Then fix overlap issues with positional adjustment
        else:
            for i in range(n):
                x, y, r = improved[i]
                original_x, original_y = x, y
                moved = False

                # Check overlaps and adjust position
                for j in range(n):

                            # Move away from overlapping circle
