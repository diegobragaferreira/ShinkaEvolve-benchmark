def create_initial_population(pop_size: int, n_circles: int) -> list:
    """Create initial population of valid circle arrangements."""
    population = []

    # First, try to create valid individuals with structured grid initialization
    for _ in range(pop_size):
        # Create circles in a grid-like pattern and then apply small random perturbations
        circles = np.zeros((n_circles, 3))

        # Determine grid dimensions
        grid_size = int(np.ceil(np.sqrt(n_circles)))
        spacing_x = 1.0 / (grid_size + 1)
        spacing_y = 1.0 / (grid_size + 1)

        # Initial grid placement with reduced radius
        radius = spacing_x / 3.0

        count = 0
        for i in range(grid_size):
            for j in range(grid_size):
                if count >= n_circles:
                    break
                x = (i + 1) * spacing_x
                y = (j + 1) * spacing_y

                # Apply small random perturbation to position
                x += np.random.uniform(-spacing_x/6, spacing_x/6)
                y += np.random.uniform(-spacing_y/6, spacing_y/6)

                # Ensure it stays within bounds
                x = np.clip(x, radius, 1 - radius)
                y = np.clip(y, radius, 1 - radius)

                circles[count] = [x, y, radius]
                count += 1
            if count >= n_circles:
                break

        # Refine positions to improve validity and fitness
        for i in range(n_circles):
            # Try to increase radius if possible without violating constraints
            original_radius = circles[i, 2]
            test_radius = original_radius * 1.1  # Try slightly larger radius

            # Check if we can increase the radius
            valid = True
            for j in range(n_circles):
                if i != j:
                    dist = np.sqrt((circles[i, 0] - circles[j, 0])**2 + (circles[i, 1] - circles[j, 1])**2)
                    if dist < test_radius + circles[j, 2]:
                        valid = False
                        break

                valid_circles.append(circles[i].copy())

                # If invalid, reset to basic configuration

                y = np.random.uniform(0.01, 0.99)
                r = np.random.uniform(0.001, 0.05)

            valid_circles.append([0.5, 0.5, 0.05])

        circles = np.array(valid_circles[:n_circles])

    return population
