    def refine_configuration(config):
        """
        Enhanced local optimization using physics-inspired repulsion forces
        and geometric constraint solving.
        """
        max_iterations = 200
        improved = True
        tolerance = 1e-6

        for iteration in range(max_iterations):
            if not improved:
                break
            improved = False

            # Store original configuration to detect changes
            original_config = config.copy()

            # Phase 1: Repulsion force simulation
            # Calculate repulsion forces between overlapping circles
            forces = np.zeros((n, 2))  # Force vectors (dx, dy) for each circle

            for i in range(n):
                for j in range(i+1, n):  # Only check unique pairs
                    x1, y1, r1 = config[i]
                    x2, y2, r2 = config[j]

                    dx = x2 - x1
                    dy = y2 - y1
                    distance = np.sqrt(dx*dx + dy*dy)

                    # If circles overlap or are too close
                    if distance > 0 and distance < (r1 + r2):
                        # Repulsion force (inverse distance law)
                        force_magnitude = 0.01 / (distance * distance + 1e-10)

                        # Normalize and apply force
                        fx = force_magnitude * dx / distance
                        fy = force_magnitude * dy / distance

                        # Apply opposite forces to both circles
                        forces[i, 0] -= fx
                        forces[i, 1] -= fy
                        forces[j, 0] += fx
                        forces[j, 1] += fy

            # Apply forces to move circles apart
            for i in range(n):
                # Limit force magnitude to prevent large jumps
                force_magnitude = np.sqrt(forces[i, 0]**2 + forces[i, 1]**2)
                if force_magnitude > 0.01:
                    forces[i] *= 0.01 / force_magnitude

                # Apply force with damping
                damping = 0.5
                config[i, 0] += forces[i, 0] * damping
                config[i, 1] += forces[i, 1] * damping

                # Keep within bounds
                config[i, 0] = np.clip(config[i, 0], config[i, 2], 1 - config[i, 2])
                config[i, 1] = np.clip(config[i, 1], config[i, 2], 1 - config[i, 2])

            # Phase 2: Radius optimization
            # Try to increase radii while keeping constraints
            for i in range(n):
                orig_r = config[i, 2]
                # Calculate maximum possible radius at current position
                max_radius = min(
                    config[i, 0],
                    1 - config[i, 0],
                    config[i, 1],
                    1 - config[i, 1]
                )

                # Try to increase radius based on available space
                test_r = min(orig_r + 0.005, max_radius)

                # Check if we can actually increase the radius without conflicts
                valid = True
                for j in range(n):
                    if i != j:
                        dx = config[i, 0] - config[j, 0]
                        dy = config[i, 1] - config[j, 1]
                        dist_sq = dx*dx + dy*dy
                        min_dist_sq = (test_r + config[j, 2])**2

                        if dist_sq < min_dist_sq:
                            valid = False
                            break

                if valid and test_r > orig_r:
                    config[i, 2] = test_r
                    improved = True

            # Phase 3: Boundary constraint correction
            for i in range(n):
                # Ensure boundaries are respected
                r = config[i, 2]
                config[i, 0] = np.clip(config[i, 0], r, 1 - r)
                config[i, 1] = np.clip(config[i, 1], r, 1 - r)

            # Check if significant improvement was made
            new_total = np.sum(config[:, 2])
            old_total = np.sum(original_config[:, 2])
            if new_total - old_total < tolerance:
                improved = False

        return config
