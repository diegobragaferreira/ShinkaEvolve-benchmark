    def cx_constraint_aware(ind1, ind2):
        # Constraint-aware crossover that maintains validity
        ind1 = ind1.reshape(n, 3)
        ind2 = ind2.reshape(n, 3)

        # Perform uniform crossover first
        for i in range(n):
            if random.random() < 0.6:
                ind1[i] = ind2[i].copy()

        # Repair any constraint violations
        repaired_ind1 = repair_individual(ind1)
        repaired_ind2 = repair_individual(ind2)

        return (repaired_ind1.flatten(), repaired_ind2.flatten())

    def repair_individual(individual):
        # Simple repair: reduce radii of overlapping circles
        config = individual.copy()
        max_repair_attempts = 100

        for attempt in range(max_repair_attempts):
            valid = True
            # Check all pairs for overlaps
            for i in range(n):
                for j in range(i+1, n):
                    dx = config[i, 0] - config[j, 0]
                    dy = config[i, 1] - config[j, 1]
                    dist_sq = dx*dx + dy*dy
                    min_dist_sq = (config[i, 2] + config[j, 2])**2

                    if dist_sq < min_dist_sq:
                        # Found overlap, reduce radius of one of them
                        # Prefer reducing the smaller radius
                        if config[i, 2] <= config[j, 2]:
                            config[i, 2] *= 0.99
                        else:
                            config[j, 2] *= 0.99
                        valid = False
                        break
                if not valid:
                    break

            if valid:
                break
