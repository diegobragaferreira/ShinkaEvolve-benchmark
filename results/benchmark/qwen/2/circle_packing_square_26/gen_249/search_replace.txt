def analyze_overlap_severity(circles: np.ndarray) -> int:
    """Calculate the number of overlap violations in the current configuration."""
    n = len(circles)
    overlaps = 0

    for i in range(n):
        for j in range(i+1, n):
            x1, y1, r1 = circles[i]
            x2, y2, r2 = circles[j]
            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if dist < r1 + r2:
                overlaps += 1

    return overlaps
