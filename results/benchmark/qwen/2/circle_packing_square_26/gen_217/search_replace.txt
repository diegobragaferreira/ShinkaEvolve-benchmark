    @staticmethod
    def constraint_aware_crossover(parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
        """Perform crossover with constraint awareness based on overlap probability"""
        n = len(parent1)
        child = np.zeros_like(parent1)

        # Calculate overlap risk between corresponding circles in parents
        overlap_risks = np.zeros(n)
        for i in range(n):
            x1, y1, r1 = parent1[i]
            x2, y2, r2 = parent2[i]
            distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            # Risk increases with proximity and decreases with radius difference
            overlap_risks[i] = min(1.0, distance / (r1 + r2 + 1e-8))

        # Perform crossover with weighted probability based on overlap risk
        for i in range(n):
            # Higher overlap risk = higher probability of inheriting from parent with better fit
            overlap_risk = overlap_risks[i]
            crossover_prob = 0.3 + 0.5 * (1 - overlap_risk)  # 0.3 to 0.8 crossover probability

            if np.random.random() < crossover_prob:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]

        return child
