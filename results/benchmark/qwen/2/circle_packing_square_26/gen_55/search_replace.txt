def initialize_population(pop_size, n_circles):
    """Initialize population with improved grid-based valid configurations"""
    population = []

    # Grid-based initialization for better spatial distribution
    grid_size = int(np.ceil(np.sqrt(n_circles)))
    grid_spacing_x = 1.0 / (grid_size + 1)
    grid_spacing_y = 1.0 / (grid_size + 1)

    for _ in range(pop_size):
        # Start with grid-based placement
        circles = np.zeros((n_circles, 3))

        # Place circles in a grid pattern with slight randomness
        placed_count = 0
        for i in range(grid_size):
            for j in range(grid_size):
                if placed_count >= n_circles:
                    break
                x = (i + 1) * grid_spacing_x + np.random.uniform(-0.02, 0.02)
                y = (j + 1) * grid_spacing_y + np.random.uniform(-0.02, 0.02)

                # Initial radius estimate based on proximity to neighbors
                r = min(x, 1-x, y, 1-y) * 0.4

                # Ensure reasonable minimum radius
                r = max(r, 0.01)

                circles[placed_count] = [x, y, r]
                placed_count += 1

            if placed_count >= n_circles:
                break

        # Apply greedy insertion for remaining circles
        for i in range(placed_count, n_circles):
            best_x, best_y, best_r = 0.5, 0.5, 0.05

            # Try several random placements and pick the best valid one
            best_radius = 0
            for _ in range(50):
                x = np.random.uniform(0.05, 0.95)
                y = np.random.uniform(0.05, 0.95)
                r = min(x, 1-x, y, 1-y) * 0.4
                r = max(r, 0.01)

                # Check if this circle would fit
                valid = True
                for k in range(i):
                    cx, cy, cr = circles[k]
                    distance = np.sqrt((x - cx)**2 + (y - cy)**2)
                    if distance < r + cr:
                        valid = False
                        break

                if valid and r > best_radius:
                    best_radius = r
                    best_x, best_y = x, y

            circles[i] = [best_x, best_y, best_radius]

        # Apply local optimization to improve initial solution
        circles = optimize_local(circles)
        population.append(circles)

    return population

def optimize_local(circles, max_iter=100):
    """Apply simple local optimization to improve circle packing"""
    circles = circles.copy()
    n = len(circles)

    # Simple optimization: try to increase radii while maintaining constraints
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            x, y, r = circles[i]

            # Try to increase radius slightly
            new_r = min(r * 1.05,
                       min(x, 1-x, y, 1-y))  # Don't exceed boundary limits

            # Check if we can safely increase radius
            valid = True
            for j in range(n):
                if i != j:
                    xj, yj, rj = circles[j]
                    distance = np.sqrt((x - xj)**2 + (y - yj)**2)
                    if distance < new_r + rj:
                        valid = False
                        break

            if valid and new_r > r + 0.0001:
                circles[i, 2] = new_r
                improved = True

        if not improved:
            break

    return circles
