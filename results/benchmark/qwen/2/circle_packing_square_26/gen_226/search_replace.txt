    @staticmethod
    def _calculate_overlap_severity(circles: np.ndarray) -> int:
        """Calculate number of overlap violations."""
        n = len(circles)
        overlaps = 0

        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2:
                    overlaps += 1

        return overlaps

    @staticmethod
    def analyze_overlap_pattern(circles: np.ndarray) -> dict:
        """Analyze the overlap pattern to determine optimization approach."""
        n = len(circles)
        if n <= 1:
            return {"type": "none", "severity": 0, "conflict_count": 0}

        # Count overlaps and identify dense regions
        overlaps = 0
        conflict_counts = np.zeros(n)
        avg_distances = np.zeros(n)

        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2:
                    overlaps += 1
                    conflict_counts[i] += 1
                    conflict_counts[j] += 1

        # Identify patterns
        avg_conflicts = np.mean(conflict_counts) if len(conflict_counts) > 0 else 0
        max_conflicts = np.max(conflict_counts) if len(conflict_counts) > 0 else 0

        # Determine pattern type
        if overlaps == 0:
            pattern_type = "none"
        elif avg_conflicts < 2:
            pattern_type = "sparse"
        elif avg_conflicts < 5:
            pattern_type = "moderate"
        else:
            pattern_type = "dense"

        return {
            "type": pattern_type,
            "severity": overlaps,
            "avg_conflicts": avg_conflicts,
            "max_conflicts": max_conflicts,
            "conflict_counts": conflict_counts
        }

    @staticmethod
    def optimize_placement(circles: np.ndarray, max_iter: int = 100) -> np.ndarray:
        """Apply adaptive local optimization based on overlap analysis."""
        circles = circles.copy()
        n = len(circles)

        # Analyze current state
        analysis = LocalOptimizer.analyze_overlap_pattern(circles)
        overlap_type = analysis["type"]
        overlap_severity = analysis["severity"]

        # Choose optimization strategy based on analysis
        if overlap_severity == 0:
            # No overlaps - focus on radius expansion
            return LocalOptimizer._radius_expansion_only(circles, max_iter)
        elif overlap_type == "sparse":
            # Light overlaps - combine radius expansion with gentle adjustments
            return LocalOptimizer._sparse_overlap_optimization(circles, max_iter)
        elif overlap_type == "moderate":
            # Moderate overlaps - focused conflict resolution
            return LocalOptimizer._moderate_overlap_optimization(circles, max_iter)
        else:  # dense
            # Heavy overlaps - aggressive conflict resolution
            return LocalOptimizer._dense_overlap_optimization(circles, max_iter)
