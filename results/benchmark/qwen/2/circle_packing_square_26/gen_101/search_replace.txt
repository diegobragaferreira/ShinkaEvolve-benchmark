def local_optimization_refinement(circles: np.ndarray) -> np.ndarray:
    """
    Apply local optimization to refine the solution by adjusting positions
    to reduce overlaps and increase total radius
    """
    result = circles.copy()
    n = len(result)

    # Try multiple local optimization passes
    for pass_num in range(3):  # Reduced number of passes for speed
        improved = False

        # For each circle, try to improve its position and radius
        for i in range(n):
            original_pos = result[i].copy()
            original_radius = result[i][2]
            original_total = np.sum(result[:, 2])

            # Track best configuration found
            best_config = original_pos.copy()
            best_radius = original_radius
            best_total = original_total
            best_valid = True

            # Test various adjustments
            test_positions = []
            test_radii = []

            # Generate test positions around current location
            step_size = 0.015
            for dx in [-step_size, 0, step_size]:
                for dy in [-step_size, 0, step_size]:
                    test_positions.append((original_pos[0] + dx, original_pos[1] + dy))

            # Generate test radii
            radius_step = 0.01
            for dr in [-radius_step, 0, radius_step]:
                test_radii.append(max(0.001, original_radius + dr))

            # Test combinations of position and radius changes
            for new_x, new_y in test_positions:
                for new_r in test_radii:
                    # Check if new configuration is valid
                    if (new_x - new_r >= 0 and new_x + new_r <= 1 and
                        new_y - new_r >= 0 and new_y + new_r <= 1):

                        # Temporarily update
