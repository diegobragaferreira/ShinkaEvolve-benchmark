class EvolutionaryOperator:
    """Handles evolutionary operators with constraint awareness."""

    def __init__(self, n_circles: int):
        self.n_circles = n_circles

    def _calculate_overlap_severity(self, circles: np.ndarray) -> int:
        """Calculate total overlap violations."""
        n = len(circles)
        overlaps = 0

        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2:
                    overlaps += 1

        return overlaps

    def _get_circle_overlap_info(self, circles: np.ndarray, index: int) -> dict:
        """Get detailed overlap information for a specific circle."""
        n = len(circles)
        info = {"overlaps": [], "total_overlap": 0.0}

        x1, y1, r1 = circles[index]
        for j in range(n):
            if j != index:
                x2, y2, r2 = circles[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2:
                    overlap_amount = (r1 + r2 - dist)
                    info["overlaps"].append({
                        "index": j,
                        "distance": dist,
                        "overlap": overlap_amount
                    })
                    info["total_overlap"] += overlap_amount

        return info

    def crossover(self, parent1: np.ndarray, parent2: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
        """Perform constraint-aware crossover with overlap severity consideration."""
