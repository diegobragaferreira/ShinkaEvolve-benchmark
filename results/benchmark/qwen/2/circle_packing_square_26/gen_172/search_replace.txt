    def _find_overlaps(self, circles: np.ndarray) -> List[Tuple[int, int]]:
        """Find pairs of overlapping circles."""
        overlaps = []
        n = len(circles)

        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if distance < r1 + r2:
                    overlaps.append((i, j))
        return overlaps

    def _apply_repulsion_force(self, circles: np.ndarray, target_idx: int,
                              overlaps: List[Tuple[int, int]]) -> Optional[Tuple[float, float]]:
        """Apply physics-based repulsion force to resolve overlaps."""
        # Collect all overlapping circles that affect the target
        affected_circles = [j for i, j in overlaps if i == target_idx] + [i for i, j in overlaps if j == target_idx]

        if not affected_circles:
            return None

        # Calculate repulsion force vector
        repulsion_x, repulsion_y = 0.0, 0.0
        target_x, target_y, target_r = circles[target_idx]

        for other_idx in affected_circles:
            if other_idx != target_idx:
                other_x, other_y, other_r = circles[other_idx]
                distance = np.sqrt((target_x - other_x)**2 + (target_y - other_y)**2)

                if distance > 0:
                    # Repulsion force magnitude inversely proportional to distance
                    force_magnitude = (target_r + other_r - distance) / (distance + 1e-8)

                    # Direction from other circle to target
                    dx = target_x - other_x
                    dy = target_y - other_y

                    # Normalize and scale force
                    norm = np.sqrt(dx*dx + dy*dy) + 1e-8
                    force_x = force_magnitude * dx / norm
                    force_y = force_magnitude * dy / norm

                    repulsion_x += force_x
                    repulsion_y += force_y

        # Apply force to move the circle away from overlaps
        if abs(repulsion_x) > 1e-6 or abs(repulsion_y) > 1e-6:
            # Scale down the force to prevent large jumps that might cause new overlaps
            scale_factor = min(0.1, 1.0 / (abs(repulsion_x) + abs(repulsion_y) + 1e-8))
            new_x = target_x + repulsion_x * scale_factor
            new_y = target_y + repulsion_y * scale_factor

            # Clamp to valid region considering the radius
            new_x = max(target_r, min(1 - target_r, new_x))
            new_y = max(target_r, min(1 - target_r, new_y))

            return (new_x, new_y)

        return None
