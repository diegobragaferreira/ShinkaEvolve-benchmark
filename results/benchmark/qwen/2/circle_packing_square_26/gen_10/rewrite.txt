import numpy as np
from scipy.spatial import Voronoi, distance
from scipy.spatial.distance import cdist
import random
from typing import Tuple
import time

def circle_packing26() -> np.ndarray:
    """
    Places 26 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (26,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    np.random.seed(42)
    random.seed(42)
    
    # Parameters
    n_circles = 26
    max_iterations = 1000
    population_size = 100
    mutation_rate = 0.1
    elite_size = 10
    
    def create_initial_population(n_pop: int, n_circles: int) -> np.ndarray:
        """Create initial population using Voronoi-based seeding"""
        population = []
        for _ in range(n_pop):
            # Start with random points
            points = np.random.rand(n_circles, 2)
            
            # Create Voronoi diagram and get cell centers
            try:
                vor = Voronoi(points)
                # Get Voronoi vertices as potential circle centers
                if len(vor.vertices) > 0:
                    # Select valid vertices that are inside the unit square
                    valid_vertices = vor.vertices[
                        (vor.vertices[:, 0] >= 0) & (vor.vertices[:, 0] <= 1) &
                        (vor.vertices[:, 1] >= 0) & (vor.vertices[:, 1] <= 1)
                    ]
                    if len(valid_vertices) >= n_circles:
                        # Take first n_circles points from valid vertices
                        selected_points = valid_vertices[:n_circles]
                    else:
                        # Fall back to random points for remaining
                        selected_points = np.random.rand(n_circles, 2)
                else:
                    selected_points = np.random.rand(n_circles, 2)
            except:
                selected_points = np.random.rand(n_circles, 2)
                
            # Initialize with small radii
            radii = np.full(n_circles, 0.01)
            population.append(np.column_stack([selected_points, radii]))
        
        return np.array(population)
    
    def calculate_fitness(circles: np.ndarray) -> float:
        """Calculate fitness based on sum of radii with penalty for overlaps"""
        total_radius = np.sum(circles[:, 2])
        
        # Penalty for overlaps
        penalty = 0.0
        positions = circles[:, :2]
        radii = circles[:, 2]
        
        # Use scipy's distance matrix for efficiency
        dist_matrix = cdist(positions, positions)
        
        # Check overlaps
        for i in range(len(circles)):
            for j in range(i+1, len(circles)):
                d = dist_matrix[i, j]
                r_sum = radii[i] + radii[j]
                if d < r_sum:
                    # Overlap penalty proportional to how much they overlap
                    overlap = r_sum - d
                    penalty += overlap ** 2
        
        # Penalty for going outside bounds
        for i in range(len(circles)):
            x, y, r = circles[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                penalty += 1000000
        
        return total_radius - penalty * 0.1
    
    def mutate(individual: np.ndarray, mutation_rate: float) -> np.ndarray:
        """Mutate an individual"""
        mutated = individual.copy()
        
        for i in range(len(individual)):
            if random.random() < mutation_rate:
                # Mutate position
                mutated[i, 0] += np.random.normal(0, 0.01)
                mutated[i, 1] += np.random.normal(0, 0.01)
                
                # Keep within bounds
                mutated[i, 0] = np.clip(mutated[i, 0], 0, 1)
                mutated[i, 1] = np.clip(mutated[i, 1], 0, 1)
                
                # Mutate radius
                mutated[i, 2] += np.random.normal(0, 0.005)
                mutated[i, 2] = max(0.001, mutated[i, 2])  # Minimum radius
                
        return mutated
    
    def crossover(parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
        """Simple uniform crossover"""
        child = parent1.copy()
        for i in range(len(parent1)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child
    
    def check_feasible(circles: np.ndarray) -> bool:
        """Check if configuration is feasible"""
        positions = circles[:, :2]
        radii = circles[:, 2]
        
        # Check bounds
        for i in range(len(circles)):
            x, y, r = circles[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        
        # Check overlaps
        dist_matrix = cdist(positions, positions)
        for i in range(len(circles)):
            for j in range(i+1, len(circles)):
                d = dist_matrix[i, j]
                r_sum = radii[i] + radii[j]
                if d < r_sum:
                    return False
        return True
    
    def optimize_individual(individual: np.ndarray, max_iter: int = 50) -> np.ndarray:
        """Local optimization for an individual"""
        best = individual.copy()
        best_fitness = calculate_fitness(best)
        
        for _ in range(max_iter):
            # Try small mutations
            mutated = mutate(best, 0.3)
            mutated_fitness = calculate_fitness(mutated)
            
            if mutated_fitness > best_fitness:
                best = mutated
                best_fitness = mutated_fitness
                
        return best
    
    # Main evolution loop
    population = create_initial_population(population_size, n_circles)
    
    best_solution = None
    best_fitness = float('-inf')
    
    for generation in range(max_iterations):
        # Calculate fitness for all individuals
        fitness_scores = []
        for individual in population:
            # Local optimization
            optimized = optimize_individual(individual)
            fitness = calculate_fitness(optimized)
            fitness_scores.append(fitness)
            
            if fitness > best_fitness:
                best_fitness = fitness
                best_solution = optimized.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = population[sorted_indices]
        
        # Keep elite
        elite = population[:elite_size].copy()
        
        # Generate new population
        new_population = [elite[i] for i in range(elite_size)]
        
        # Fill rest with crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = tournament_selection(population, fitness_scores, 3)
            parent2_idx = tournament_selection(population, fitness_scores, 3)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        population = np.array(new_population)
    
    # Final optimization of best solution
    if best_solution is not None:
        final_solution = optimize_individual(best_solution, 200)
        if calculate_fitness(final_solution) > best_fitness:
            best_solution = final_solution
    
    # If the final solution has issues, use a fallback approach
    if best_solution is None or not check_feasible(best_solution):
        print("Using fallback approach")
        # Fallback: Start with regular grid and optimize
        grid_size = int(np.ceil(np.sqrt(n_circles)))
        points = []
        for i in range(grid_size):
            for j in range(grid_size):
                if len(points) >= n_circles:
                    break
                x = (i + 0.5) / grid_size
                y = (j + 0.5) / grid_size
                points.append([x, y])
        points = np.array(points[:n_circles])
        
        # Initialize with small radii
        circles = np.column_stack([points, np.full(n_circles, 0.01)])
        
        # Simple greedy optimization
        for _ in range(1000):
            for i in range(n_circles):
                # Try to increase radius
                old_radius = circles[i, 2]
                test_radius = min(old_radius + 0.01, 0.5)
                
                # Check feasibility
                circles[i, 2] = test_radius
                if not check_feasible(circles):
                    circles[i, 2] = old_radius
            
            # Random perturbations
            for i in range(n_circles):
                if random.random() < 0.1:
                    circles[i, 0] += np.random.normal(0, 0.001)
                    circles[i, 1] += np.random.normal(0, 0.001)
                    circles[i, 0] = np.clip(circles[i, 0], 0, 1)
                    circles[i, 1] = np.clip(circles[i, 1], 0, 1)
        
        best_solution = circles
    
    return best_solution

def tournament_selection(population: np.ndarray, fitness_scores: list, k: int) -> int:
    """Select an individual using tournament selection"""
    tournament_indices = random.sample(range(len(population)), k)
    tournament_fitness = [fitness_scores[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmax(tournament_fitness)]
    return winner_index
