class VoronoiAnalyzer:
    @staticmethod
    def analyze_constraint_regions(points: np.ndarray, threshold_ratio: float = 0.3) -> np.ndarray:
        """Analyze current configuration using spherical Voronoi to identify constraints"""
        try:
            # Create spherical Voronoi diagram
            sv = SphericalVoronoi(points)

            # Compute Voronoi cell areas properly
            cell_areas = sv.calculate_areas()
            mean_area = np.mean(cell_areas) if len(cell_areas) > 0 else 1.0

            # Find points in cells that are below 20% of average area (dense regions)
            dense_point_indices = np.where(cell_areas < 0.2 * mean_area)[0] if len(cell_areas) > 0 else np.array([])

            # Also identify points that have many close neighbors
            distances = pdist(points)
            if len(distances) > 0:
                distance_matrix = squareform(distances)
                # Count neighbors within 10th percentile distance
                threshold_dist = np.percentile(distances, 10)
                close_neighbor_counts = np.sum(distance_matrix < threshold_dist, axis=1)
            else:
                close_neighbor_counts = np.zeros(len(points))

            # Combine criteria for constraint identification with proper normalization
            constraint_scores = np.zeros(len(points))

            # Score based on close neighbors (normalized)
            if len(close_neighbor_counts) > 0:
                max_neighbors = np.max(close_neighbor_counts) if np.max(close_neighbor_counts) > 0 else 1
                constraint_scores += close_neighbor_counts / max_neighbors * 0.5

            # Bonus for being in a dense region
            if len(dense_point_indices) > 0:
                for idx in dense_point_indices:
                    if idx < len(constraint_scores):  # Safety check
                        constraint_scores[idx] += 2.0

            # Ensure scores are non-negative
            constraint_scores = np.maximum(constraint_scores, 0)

            return constraint_scores

        except Exception as e:
            # Fallback if Voronoi computation fails
            return np.ones(len(points)) * 0.5
