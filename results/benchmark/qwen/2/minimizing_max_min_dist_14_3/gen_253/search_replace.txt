The current multi-start approach only uses a fixed set of seeds with Fibonacci initialization. To better explore the solution space and potentially find configurations that beat the benchmark of 0.4898, I'll enhance the initialization strategies by incorporating multiple diverse approaches (Fibonacci, icosahedron, random uniform, and Latin hypercube sampling) combined with different seeds. This multi-faceted initialization strategy increases the chances of finding better local optima.

The key insight is that different initialization methods can lead to different basin of attractions, and by systematically exploring these, we can improve our chances of finding a configuration with a higher min/max ratio.

    def initialize_points_icosahedron(seed=42):
        """Initialize points based on perturbed icosahedron vertices."""
        np.random.seed(seed)
        # Regular icosahedron vertices (normalized)
        phi = (1 + math.sqrt(5)) / 2  # golden ratio
        vertices = [
            [0, 1, phi], [0, -1, phi], [0, 1, -phi], [0, -1, -phi],
            [1, phi, 0], [-1, phi, 0], [1, -phi, 0], [-1, -phi, 0],
            [phi, 0, 1], [phi, 0, -1], [-phi, 0, 1], [-phi, 0, -1]
        ]

        # Normalize vertices
        vertices = np.array(vertices)
        vertices = vertices / np.linalg.norm(vertices, axis=1, keepdims=True)

        # Perturb vertices slightly to break symmetry
        points = vertices + np.random.normal(0, 0.02, vertices.shape)
        points = points / np.linalg.norm(points, axis=1, keepdims=True)

        return points

    def initialize_points_random_uniform(seed=42):
        """Initialize points using uniform random sampling on sphere."""
        np.random.seed(seed)
        points = []
        while len(points) < 14:
            # Generate random point in cube [-1,1]^3
            p = np.random.uniform(-1, 1, 3)
            # If inside unit sphere, project to surface
            norm = np.linalg.norm(p)
            if norm <= 1:
                p = p / norm
                points.append(p)
        return np.array(points[:14])

    def initialize_points_latin_hypercube(seed=42):
        """Initialize points using spherical Latin Hypercube Sampling."""
        np.random.seed(seed)
        points = []

        # Generate in spherical coordinates first
        # Divide into bins for each coordinate
        theta_bins = np.linspace(0, np.pi, 15)[:-1] + np.pi/30
        phi_bins = np.linspace(0, 2*np.pi, 15)[:-1] + np.pi/14

        # Place one point in each bin
        for i in range(14):
            theta = np.random.uniform(theta_bins[i%len(theta_bins)], theta_bins[i%len(theta_bins)] + np.pi/30)
            phi = np.random.uniform(phi_bins[i%len(phi_bins)], phi_bins[i%len(phi_bins)] + 2*np.pi/14)

            x = np.sin(theta) * np.cos(phi)
            y = np.sin(theta) * np.sin(phi)
            z = np.cos(theta)

            points.append([x, y, z])

        return np.array(points)

    def multi_start_optimization():
        """Run adaptive simulated annealing multiple times with diverse initializations."""
        best_points = None
        best_ratio = 0.0
        best_config = ""

        # Different initialization methods with different seeds for diversity
        initialization_configs = [
            ("fibonacci", 42), ("fibonacci", 123), ("fibonacci", 456),
            ("icosahedron", 42), ("icosahedron", 123), ("icosahedron", 456),
            ("random_uniform", 42), ("random_uniform", 123), ("random_uniform", 456),
            ("latin_hypercube", 42), ("latin_hypercube", 123), ("latin_hypercube", 456)
        ]

        for method, seed in initialization_configs:
            np.random.seed(seed)
            if method == "fibonacci":
                initial_points = initialize_points_fibonacci(seed)
            elif method == "icosahedron":
                initial_points = initialize_points_icosahedron(seed)
            elif method == "random_uniform":
                initial_points = initialize_points_random_uniform(seed)
            elif method == "latin_hypercube":
                initial_points = initialize_points_latin_hypercube(seed)

            points, ratio = adaptive_simulated_annealing(initial_points, max_iterations=100000)

            if ratio > best_ratio:
                best_ratio = ratio
                best_points = points.copy()
                best_config = f"{method}_seed_{seed}"

        return best_points, best_ratio
