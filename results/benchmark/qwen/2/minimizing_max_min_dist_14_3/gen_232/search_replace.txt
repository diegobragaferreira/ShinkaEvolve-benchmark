The current adaptive perturbation mechanism is overly complex and doesn't provide significant benefits. I'll simplify and improve it by implementing a more effective targeted perturbation strategy that focuses on points that are too close together, which is the primary bottleneck in achieving higher min/max ratios. The key insight is that we should prioritize perturbing points that form the smallest distances, as these are the limiting factors in our optimization.

    def _compute_targeted_perturbation(self, points: np.ndarray, point_idx: int) -> np.ndarray:
        """Compute a targeted perturbation that should improve the configuration"""
        try:
            # Analyze distances to all other points
            distances = cdist([points[point_idx]], points)[0]
            distances = distances[distances > 0]  # Remove self-distance

            if len(distances) > 0:
                min_distance = np.min(distances)
                avg_distance = np.mean(distances)

                # Focus on points that are too close (below 30% of average distance)
                if min_distance < avg_distance * 0.3:
                    # Find the closest point(s) to repel from
                    closest_indices = np.where(distances < avg_distance * 0.5)[0]

                    # Repel away from the closest points
                    repulsion = np.zeros(3)
                    for idx in closest_indices:
                        diff = points[point_idx] - points[idx]
                        dist = np.linalg.norm(diff)
                        if dist > 0:
                            # Weight repulsion by inverse distance (closer points have stronger effect)
                            repulsion += diff / dist / dist

                    if np.linalg.norm(repulsion) > 0:
                        repulsion = repulsion / np.linalg.norm(repulsion)
                        # Stronger repulsion for very close points
                        magnitude = 0.05 * (1.0 - min_distance / avg_distance)
                        return repulsion * magnitude
                    else:
                        # Fallback to random perturbation if no clear repulsion vector
                        direction = np.random.randn(3)
                        direction /= np.linalg.norm(direction)
                        return direction * 0.02

                    # For well-spaced points, make a small random perturbation
                    direction = np.random.randn(3)
                    direction /= np.linalg.norm(direction)

                    return direction * 0.01

                return direction * 0.02
