class PointInitializer:
    @staticmethod
    def fibonacci_sphere(n: int, seed: int = 42) -> np.ndarray:
        """Generate points on sphere using Fibonacci spiral method"""
        np.random.seed(seed)
        points = []
        phi = math.pi * (3.0 - math.sqrt(5.0))  # golden angle

        for i in range(n):
            y = 1 - (i / float(n - 1)) * 2  # y goes from 1 to -1
            radius = math.sqrt(1 - y * y)  # radius at y

            theta = phi * i  # golden angle increment

            x = math.cos(theta) * radius
            z = math.sin(theta) * radius

            points.append([x, y, z])

        return np.array(points)

    @staticmethod
    def icosahedral_initialization(n: int) -> np.ndarray:
        """Initialize points using icosahedral symmetry for better spread"""
        points = np.zeros((n, 3))

        # Add poles for better symmetry
        points[0] = [0, 0, 1]       # North pole
        points[1] = [0, 0, -1]      # South pole

        # Add equatorial points in two rings
        angle_step = 2 * math.pi / 5
        for i in range(5):
            angle1 = i * angle_step
            angle2 = angle1 + angle_step / 2

            # First ring
            points[2+i] = [math.cos(angle1), math.sin(angle1), 0.0]
            # Second ring offset
            points[7+i] = [math.cos(angle2), math.sin(angle2), 0.0]

        # Add additional points near poles
        points[12] = [0, 0, 0.7]
        points[13] = [0, 0, -0.7]

        # Add small random perturbations
        points += np.random.normal(0, 0.01, points.shape)

        return points

    @staticmethod
    def spherical_cap_distribution(n: int, num_caps: int = 4, cap_radius: float = 0.3) -> np.ndarray:
        """Create points distributed in spherical caps to enhance diversity"""
        points = []
        # Distribute points in multiple caps
        for cap_idx in range(num_caps):
            # Generate points in a circular cap
            angle_offset = cap_idx * 2 * math.pi / num_caps
            for i in range(n // num_caps):
                # Generate points in spherical cap
                r = np.random.random() * cap_radius
                theta = np.random.random() * 2 * math.pi + angle_offset
                phi = np.random.random() * math.pi / 2  # Only upper hemisphere

                x = r * math.sin(phi) * math.cos(theta)
                y = r * math.sin(phi) * math.sin(theta)
                z = r * math.cos(phi)
                points.append([x, y, z])

        # Fill remaining points randomly on sphere
        remaining = n - len(points)
        for i in range(remaining):
            # Generate random point on sphere
            theta = np.random.random() * 2 * math.pi
            phi = np.random.random() * math.pi
            x = math.sin(phi) * math.cos(theta)
            y = math.sin(phi) * math.sin(theta)
            z = math.cos(phi)
            points.append([x, y, z])

        return np.array(points)

    @staticmethod
    def perturbed_fibonacci_sphere(n: int, perturbation_strength: float = 0.02, seed: int = 42) -> np.ndarray:
        """Generate fibonacci sphere with custom perturbation"""
        points = PointInitializer.fibonacci_sphere(n, seed)
        # Add controlled random perturbation
        noise = np.random.normal(0, perturbation_strength, points.shape)
        points += noise

        # Normalize to unit sphere
        norms = np.linalg.norm(points, axis=1)
        points = points / norms[:, np.newaxis]

        return points

    """
    Creates 14 points in 3 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (14,3) containing the (x,y,z) coordinates of the 14 points.
    """
    np.random.seed(42)

    # Multi-start optimization with diverse initializations
    best_points = None
    best_ratio = 0.0

    # Collect all initialization strategies
    initializations = []

    # 1. Multiple Fibonacci sphere configurations with different seeds
    for i in range(8):  # Increase from 5 to 8 for more diversity
        initializations.append(("fibonacci", lambda seed=i: PointInitializer.fibonacci_sphere(14, seed=42 + seed * 100)))

    # 2. Perturbed Fibonacci spheres with different strengths
    for i, strength in enumerate([0.01, 0.015, 0.02]):
        initializations.append(("perturbed_fibonacci", lambda s=strength, seed=i: PointInitializer.perturbed_fibonacci_sphere(14, s, seed=1000 + seed * 100)))

    # 3. Icosahedral initialization variants
    initializations.append(("icosahedral", lambda: PointInitializer.icosahedral_initialization(14)))

    # 4. Spherical cap distributions
    initializations.append(("spherical_cap", lambda: PointInitializer.spherical_cap_distribution(14)))

    # 5. Additional random initialization from different seeds
    for i in range(5):
        initializations.append(("random", lambda seed=i: np.random.uniform(-1, 1, (14, 3))))

    # Run optimizations from all initializations
    for name, initializer_func in initializations:
        try:
            # Execute initialization
            points = initializer_func()

            # Normalize to unit sphere if needed
            if name != "random":  # Random starts might not be normalized
                norms = np.linalg.norm(points, axis=1)
                if np.max(norms) > 0:
                    points = points / norms[:, np.newaxis]

            # Run optimization
            optimizer = AdaptiveSimulatedAnnealing(max_time=300)  # Reduce time to leave room for multiple runs
            optimized_points, optimized_ratio = optimizer.optimize(points)

            if optimized_ratio > best_ratio:
                best_ratio = optimized_ratio
                best_points = optimized_points.copy()
        except Exception as e:
            # Skip failed initializations gracefully
            continue

    # Run one final optimization from the best found solution to fine-tune
    if best_points is not None:
        np.random.seed(42)  # Reset seed for reproducibility
        optimizer = AdaptiveSimulatedAnnealing(max_time=60)  # Shorter time for final refinement
        final_points, final_ratio = optimizer.optimize(best_points)
        if final_ratio > best_ratio:
            best_points = final_points
            best_ratio = final_ratio

    return best_points
