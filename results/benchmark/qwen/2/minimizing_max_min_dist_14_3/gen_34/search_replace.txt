<DESCRIPTION>
Adding multi-objective optimization to simultaneously maximize both the minimum/maximum distance ratio and the uniformity of point distribution. This approach will use the spherical Voronoi entropy score as a secondary objective to guide the optimization towards configurations that are not only good for distance ratio but also well-distributed on the sphere. This addresses the limitation of single-objective optimization that can get stuck in local optima where distances are maximized but points are clustered in non-uniform ways.

    def voronoi_entropy_score(points):
        """
        Calculate entropy-based score of Voronoi cell distribution.
        High entropy indicates more uniform cell distribution.
        """
        # Only import when needed to avoid dependency issues
        from scipy.spatial import SphericalVoronoi
        try:
            sv = SphericalVoronoi(points)
            areas = sv.calculate_areas()
            # Normalize areas
            areas = areas / np.sum(areas)
            # Entropy calculation
            entropy = -np.sum(areas * np.log(areas + 1e-10))
            return entropy
        except:
            # Fallback for cases where SphericalVoronoi fails
            return 0.0

    def multi_objective_fitness(points, ratio_weight=1.0, uniformity_weight=0.1):
        """Fitness function combining distance ratio and uniformity."""
        ratio = calculate_min_max_ratio(points)
        uniformity = voronoi_entropy_score(points)
        # Combined fitness: prioritize both uniformity and distance ratio
        return ratio_weight * ratio + uniformity_weight * uniformity
