        def adaptive_perturb_point(points, point_idx, temp):
            """More sophisticated perturbation that adapts based on local configuration."""
            # Get distances to all other points
            distances = []
            for i in range(14):
                if i != point_idx:
                    dist = np.linalg.norm(points[point_idx] - points[i])
                    distances.append(dist)

            # Determine if point is part of a small-distance pair
            if len(distances) > 0:
                avg_dist = np.mean(distances)
                min_dist = np.min(distances)

                # If point is too close to others, make more aggressive perturbations
                if min_dist < avg_dist * 0.3:
                    perturbation_scale = temp * 0.03
                elif min_dist > avg_dist * 1.5:
                    # If point is too far from others, make smaller perturbations
                    perturbation_scale = temp * 0.005
                else:
                    # Normal case
                    perturbation_scale = temp * 0.015
            else:
                perturbation_scale = temp * 0.01

            # Generate perturbation in tangent plane
            delta = np.random.randn(3)
            # Project to tangent plane (orthogonal to point)
            delta = delta - np.dot(delta, points[point_idx]) * points[point_idx]
            # Normalize the tangent vector
            delta_norm = np.linalg.norm(delta)
            if delta_norm > 1e-10:
                delta = delta / delta_norm
            else:
                # Fallback to orthogonal vector if needed
                perpendicular = np.array([1, 0, 0]) if abs(points[point_idx][0]) < 0.9 else np.array([0, 1, 0])
                delta = np.cross(perpendicular, points[point_idx])
                delta = delta / np.linalg.norm(delta)

            # Apply perturbation
            new_point = points[point_idx] + perturbation_scale * delta
            # Project back to sphere
            new_point = new_point / np.linalg.norm(new_point)
            return new_point
