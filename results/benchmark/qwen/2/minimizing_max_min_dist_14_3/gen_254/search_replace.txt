def min_max_dist_dim3_14() -> np.ndarray:
    """
    Creates 14 points in 3 dimensions in order to maximize the ratio of minimum to maximum distance.

    Uses a spherical Voronoi-based evolution approach that combines geometric insights with
    adaptive optimization strategies to improve upon traditional simulated annealing methods.

    Returns:
        points: np.ndarray of shape (14,3) containing the (x,y,z) coordinates of the 14 points.
    """

    # Set seed for reproducibility
    np.random.seed(42)

    # Create configuration
    config = OptimizationConfig(
        n_points=14,
        n_dimensions=3,
        max_iterations=100000,
        initial_temp=1.0,
        final_temp=1e-6,
        cooling_rate=0.9995,
        log_interval=1000,
        num_seeds=12,  # Increased number of seeds for better exploration
        voronoi_threshold=0.1,
        local_refinement_iters=500,
        early_stopping_patience=10000,
        stagnation_limit=5000
    )

    best_overall_points = None
    best_overall_ratio = 0.0

    # Multi-start approach: run optimization from multiple Fibonacci seeds
    for seed_offset in range(config.num_seeds):
        # Create fresh optimizer for each seed
        optimizer = VoronoiOptimizer(config)

        # Initialize with specific seed
        optimizer._initialize_population()

        # Run optimization
        optimized_points, best_ratio = optimizer._run_optimization_loop()

        # Keep track of the best overall solution
        if best_ratio > best_overall_ratio:
            best_overall_ratio = best_ratio
            best_overall_points = optimized_points.copy()

    # Final validation and fallback
    if best_overall_points is not None:
        final_min, final_max = optimizer._calculate_distances(best_overall_points)
        if final_max <= 0:
            warnings.warn("Final validation failed, returning Fibonacci sphere initialization")
            best_overall_points = optimizer._fibonacci_sphere(14)
    else:
        # Fallback to Fibonacci initialization if optimization failed
        warnings.warn("All optimizations failed, using Fibonacci sphere initialization")
        best_overall_points = optimizer._fibonacci_sphere(14)

    return best_overall_points
