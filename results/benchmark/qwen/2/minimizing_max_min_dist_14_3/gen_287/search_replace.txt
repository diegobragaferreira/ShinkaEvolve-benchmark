    def _select_and_apply_perturbation(self, current_points: np.ndarray, temp: float,
                                     iteration_count: int, phase: str) -> np.ndarray:
        """Select and apply the most appropriate perturbation based on context."""

        # Determine which perturbation strategy to use
        if phase == "exploration":
            # In early phases, use more diverse strategies to explore space
            if iteration_count % 4 == 0:
                return self.perturbation_engine.voronoi_guided_perturbation(
                    current_points, temp, self.voronoi_computer, self.config.voronoi_threshold
                )
            else:
                return self.perturbation_engine.distance_weighted_perturbation(
                    current_points, temp, self.distance_calculator
                )
        elif phase == "exploitation":
            # In middle phases, focus on exploitation with careful balance
            if iteration_count % 3 == 0:
                return self.perturbation_engine.voronoi_guided_perturbation(
                    current_points, temp, self.voronoi_computer, self.config.voronoi_threshold
                )
            else:
                return self.perturbation_engine.distance_weighted_perturbation(
                    current_points, temp, self.distance_calculator
                )
        else:  # refinement phase
            # In late phases, be more selective and aggressive
            return self.perturbation_engine.voronoi_guided_perturbation(
                current_points, temp, self.voronoi_computer, self.config.voronoi_threshold
            )
