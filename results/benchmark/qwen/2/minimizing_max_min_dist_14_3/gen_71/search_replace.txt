This enhancement improves the simulated annealing optimization by implementing adaptive perturbation sizes based on the current optimization progress and introducing a more sophisticated cooling schedule that adjusts dynamically based on convergence behavior. The key improvements include:
1. Using adaptive perturbation sizes that decrease as optimization progresses
2. Implementing a more aggressive cooling schedule when improvement is slow
3. Adding a diversity maintenance mechanism to prevent premature convergence
4. Improving the restart strategy with better seeding

These changes should help escape local optima more effectively and achieve better convergence towards the global optimum, potentially beating the benchmark ratio of 0.4898.

    def simulated_annealing_optimization(initial_points, max_iter=10000):
        """Optimize points using simulated annealing with adaptive cooling and perturbation."""
        current_points = initial_points.copy()
        current_points = project_to_sphere(current_points)

        best_points = current_points.copy()
        best_ratio = compute_min_max_ratio(current_points)

        # Enhanced adaptive cooling schedule
        temperature = 0.1
        cooling_rate = 0.9995
        min_temp = 1e-6

        # Track convergence for adaptive cooling
        last_improvement = 0
        stagnation_counter = 0
        max_stagnation = 500

        for iteration in range(max_iter):
            # Store current configuration
            old_points = current_points.copy()
            old_ratio = compute_min_max_ratio(current_points)

            # Select random point to perturb
            idx = np.random.randint(len(current_points))

            # Adaptive perturbation size based on iteration progress
            # Decrease perturbation size as optimization progresses
            perturbation_size = 0.01 * (1 - iteration / max_iter) + 0.001
            perturbation = np.random.normal(0, perturbation_size, 3)

            # Project perturbation to tangent plane of the sphere at current point
            current_point = current_points[idx]
            # Tangent plane projection: remove component parallel to radius vector
            projection_factor = np.dot(perturbation, current_point)
            perturbation_tangent = perturbation - projection_factor * current_point

            # Apply perturbation
            current_points[idx] += perturbation_tangent

            # Project back to sphere
            current_points[idx] = project_to_sphere(current_points[idx:idx+1])[0]

            # Compute new ratio
            new_ratio = compute_min_max_ratio(current_points)

            # Accept or reject based on Metropolis criterion
            if new_ratio > best_ratio:
                best_ratio = new_ratio
                best_points = current_points.copy()
                last_improvement = iteration
                stagnation_counter = 0
            elif np.random.random() < np.exp((new_ratio - old_ratio) / temperature):
                # Accept worse solution with probability
                pass  # Keep the new configuration
            else:
                # Revert to previous configuration
                current_points = old_points

            # Adaptive cooling based on improvement rate
            stagnation_counter += 1
            if stagnation_counter > max_stagnation:
                # If no improvement for a while, cool faster
                temperature = max(min_temp, temperature * 0.95)
                stagnation_counter = 0
            else:
                # Normal cooling
                temperature = max(min_temp, temperature * cooling_rate)

            # Diversity maintenance - occasionally add more randomness
            if iteration % 2000 == 0 and iteration > 0:
                # Introduce some random perturbations to maintain diversity
                for i in range(len(current_points)):

                        current_points[i] = project_to_sphere(current_points[i:i+1])[0]

        return best_points, best_ratio
