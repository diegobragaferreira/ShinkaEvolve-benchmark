import numpy as np
from scipy.spatial.distance import pdist, squareform


def fibonacci_sphere(n: int) -> np.ndarray:
    """
    Generate n points distributed as evenly as possible on a unit sphere
    using Fibonacci spiral method.
    """
    points = []
    phi = np.pi * (3. - np.sqrt(5.))  # golden angle

    for i in range(n):
        y = 1 - (i / float(n - 1)) * 2  # y goes from 1 to -1
        radius = np.sqrt(1 - y * y)  # radius at y

        theta = phi * i  # golden angle increment

        x = np.cos(theta) * radius
        z = np.sin(theta) * radius

        points.append([x, y, z])

    return np.array(points)


def compute_min_max_ratio(points: np.ndarray) -> tuple:
    """
    Compute the minimum and maximum distances between all pairs of points,
    then return their ratio.
    """
    # Compute pairwise distances
    distances = pdist(points)

    # Get minimum and maximum distances
    dmin = np.min(distances)
    dmax = np.max(distances)

    # Return ratio and individual values for debugging
    return dmin / dmax, dmin, dmax


def min_max_dist_dim3_14() -> np.ndarray:
    """
    Creates 14 points in 3 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (14,3) containing the (x,y,z) coordinates of the 14 points.

    """
    n = 14
    np.random.seed(42)

    # Initialize points using Fibonacci sphere method for better distribution
    points = fibonacci_sphere(n)

    # Normalize points to unit sphere (ensuring they're on the surface)
    norms = np.linalg.norm(points, axis=1, keepdims=True)
    points = points / norms

    # Simulated Annealing parameters
    initial_temp = 1.0
    final_temp = 1e-6
    alpha = 0.9995  # Cooling rate
    max_iter = 100000

    # Current solution
    current_points = points.copy()
    current_ratio, _, _ = compute_min_max_ratio(current_points)

    # Best solution found
    best_points = current_points.copy()
    best_ratio, _, _ = current_ratio

    # Temperature schedule
    temp = initial_temp

    # Main optimization loop
    for iteration in range(max_iter):
        # Cool down temperature
        temp *= alpha

        # If temperature gets too low, break
        if temp < final_temp:
            break

        # Create a neighbor solution by perturbing one point
        # Select a random point to perturb
        idx = np.random.randint(0, n)

        # Create a small random perturbation
        perturbation = np.random.normal(0, 0.01, 3)

        # Apply perturbation to selected point
        new_points = current_points.copy()
        new_points[idx] += perturbation

        # Project back onto unit sphere
        norm = np.linalg.norm(new_points[idx])
        if norm > 0:
            new_points[idx] = new_points[idx] / norm

        # Compute new ratio
        new_ratio, _, _ = compute_min_max_ratio(new_points)

        # Accept or reject the new solution
        if new_ratio > current_ratio:
            # Always accept if better
            current_points = new_points
            current_ratio = new_ratio
        else:
            # Accept with probability based on temperature
            delta = new_ratio - current_ratio
            if np.random.rand() < np.exp(delta / temp):
                current_points = new_points
                current_ratio = new_ratio

        # Update best solution if needed
        if current_ratio > best_ratio:
            best_points = current_points.copy()
            best_ratio = current_ratio

    return best_points
