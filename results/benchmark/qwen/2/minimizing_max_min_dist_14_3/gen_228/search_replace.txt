    avg_distances = np.mean(distances, axis=1)

    # Points with much smaller min distances than average are candidates for aggressive perturbation
    clustering_indicator = (avg_distances - min_distances) / (avg_distances + 1e-10)

    global_clustered_fraction = np.sum(clustering_indicator > 0.5) / len(points)

    # If few points show clustering, perturb fewer points
