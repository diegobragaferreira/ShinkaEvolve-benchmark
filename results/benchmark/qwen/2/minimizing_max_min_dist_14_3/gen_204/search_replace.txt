    def adaptive_cooling_schedule(iteration, max_iterations, last_improvement, patience, base_cooling_rate=0.9995):
        """Dynamic cooling schedule that adapts based on optimization progress"""
        # Calculate cooling rate based on iteration progress
        progress = iteration / max_iterations

        # Early stages: aggressive cooling for rapid exploration
        if progress < 0.3:
            cooling_rate = 0.9998  # Very slow cooling in early stages
        elif progress < 0.7:
            cooling_rate = 0.9995  # Moderate cooling in middle stages
        else:
            cooling_rate = 0.9992  # Slow cooling in late stages for fine-tuning

        # Adaptive cooling based on recent improvement rate
        # When we haven't improved for a while, apply more aggressive cooling
        if iteration - last_improvement > patience:
            cooling_rate = min(0.9999, cooling_rate * 1.05)

        return cooling_rate
