def evolutionary_search(max_time_seconds: int = 180) -> List[float]:
    """
    Run evolutionary optimization guided by spectral properties.
    """
    start_time = time.time()

    # Parameters
    population_size = 40  # Increased for better diversity
    generations = 60  # More generations to allow for better convergence
    elite_size = 8  # More elite individuals preserved
    initial_mutation_rate = 0.15  # Higher initial mutation rate for more exploration

    # Initialize diverse population with varied strategies
    population = []
    for _ in range(population_size):
        # Varying sequence lengths and initialization methods
        n = random.randint(80, 1000)
        init_strategy = random.choice(['spectral', 'uniform', 'exponential'])
        if init_strategy == 'spectral':
            sequence = create_spectral_guided_sequence(n)
        elif init_strategy == 'uniform':
            sequence = [random.uniform(0.1, 1.0) for _ in range(n)]
        else:  # exponential
            sequence = [np.exp(-i / (n/5)) for i in range(n)]
            sequence = [x / sum(sequence) * 10 for x in sequence]
        population.append(sequence)

    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > max_time_seconds - 2:
            break

        # Evaluate fitness
        fitnesses = [compute_c1_constant(seq) for seq in population]

        # Adaptive mutation rate that decreases over generations
        mutation_rate = initial_mutation_rate * np.exp(-gen / (generations / 2))

        # Preserve elite
        elite_indices = np.argsort(fitnesses)[-elite_size:]
        elite_individuals = [population[i] for i in elite_indices]

        # Create new population
        new_population = elite_individuals[:]

        # Fill rest with offspring
        while len(new_population) < population_size:
            # Parent selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)

            # Crossover
            child = crossover_sequences(parent1, parent2)

            # Apply spectral filtering to child
            filtered_child = spectral_filter(child)

            # Mutation with adaptive rate
            mutated_child = mutate_sequence(filtered_child, mutation_rate)

            # Ensure minimum sum
            if sum(mutated_child) < 0.01:
                mutated_child[0] = 0.1

            new_population.append(mutated_child)

        population = new_population

    # Return best individual
    final_fitnesses = [compute_c1_constant(seq) for seq in population]
    best_idx = np.argmax(final_fitnesses)
    return population[best_idx]
