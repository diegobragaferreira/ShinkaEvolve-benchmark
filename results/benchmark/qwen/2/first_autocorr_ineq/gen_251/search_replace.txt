    """Solves the convolution LP for a given sequence and RHS."""
    try:
        n = len(f_sequence)
        if n < 1:
            return None

        # Sample key convolution indices with weighted sampling towards peak
        # Compute full convolution to determine peak location
        full_conv = fftconvolve(f_sequence, f_sequence, mode='full')
        peak_location = np.argmax(full_conv)

        max_constraints = min(5000, 2 * n)  # Cap at reasonable number

        # Use weighted sampling: prioritize indices near the peak
        if n > 100:  # Only apply weighting for larger sequences
            # Create a Gaussian weight centered on peak
            weights = np.exp(-0.5 * ((np.arange(2*n-1) - peak_location) / (n//4))**2)
            # Sample with replacement based on weights
            constraint_indices = np.random.choice(2*n-1, size=max_constraints, p=weights/weights.sum())
        else:
            # For smaller sequences, use uniform sampling
            constraint_indices = np.random.choice(2*n-1, size=max_constraints, replace=False)
