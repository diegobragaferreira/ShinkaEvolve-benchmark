This edit introduces elite preservation mechanism to maintain high-performing sequences during evolution. The idea is to keep the top performing sequences from each generation and inject them into the next generation to preserve good solutions and avoid losing valuable progress due to random mutations. This builds upon successful techniques used in evolutionary algorithms that maintain diversity while preserving quality.

def search_for_best_sequence():
    """Main search function using hybrid optimization."""
    start_time = time.time()
    max_time_seconds = 170  # Leave time for cleanup

    best_sequence = None
    best_inv_c1 = 0.0
    history = deque(maxlen=10)  # Track recent improvements
    elite_sequences = []  # Store top performing sequences
    elite_size = 5  # Number of elite sequences to preserve

    # Multi-start strategy
    num_starts = 10

    for start_idx in range(num_starts):
        if time.time() - start_time > max_time_seconds:
            break

        # Generate diverse initial sequences
        if random.random() < 0.5:
            initial_sequence = generate_random_sequence()
        else:
            length = random.randint(100, 1000)
            initial_sequence = generate_structured_sequence(length)

        current_sequence = initial_sequence.copy()
        current_inv_c1 = compute_inverse_c1(current_sequence)

        # Local optimization with gradient ascent
        for iter_idx in range(100):  # Local iterations
            if time.time() - start_time > max_time_seconds:
                break

            # Occasionally use DE for global refinement
            if iter_idx % 10 == 0:
                current_sequence = optimize_with_differential_evolution(current_sequence)

            new_sequence = gradient_ascent_step(current_sequence, step_size=0.01)
            new_inv_c1 = compute_inverse_c1(new_sequence)

            if new_inv_c1 > current_inv_c1:
                current_sequence = new_sequence
                current_inv_c1 = new_inv_c1
            else:
                # Occasionally escape local minima with random perturbation
                if random.random() < 0.05:
                    current_sequence = generate_random_sequence(len(current_sequence))
                    current_inv_c1 = compute_inverse_c1(current_sequence)

            # Track history for stagnation detection
            history.append(current_inv_c1)

            # Detect stagnation and reset
            if len(history) == history.maxlen:
                if abs(history[-1] - history[0]) < 1e-6:
                    current_sequence = generate_random_sequence(len(current_sequence))
                    current_inv_c1 = compute_inverse_c1(current_sequence)

            # Update elite sequences (keep top performers)
            if len(elite_sequences) < elite_size:
                elite_sequences.append((current_sequence.copy(), current_inv_c1))
            else:
                # Replace worst elite if current is better
                worst_idx = min(range(len(elite_sequences)), key=lambda i: elite_sequences[i][1])
                if current_inv_c1 > elite_sequences[worst_idx][1]:
                    elite_sequences[worst_idx] = (current_sequence.copy(), current_inv_c1)

            # Update global best
            if current_inv_c1 > best_inv_c1:
                best_inv_c1 = current_inv_c1
                best_sequence = current_sequence.copy()

    # Incorporate elite sequences into final optimization
    if elite_sequences:
        # Sort elites by performance
        elite_sequences.sort(key=lambda x: x[1], reverse=True)
        # Select the best elite as starting point for final optimization
        best_elite_sequence = elite_sequences[0][0]
        if best_elite_sequence is not None:
            best_sequence = optimize_with_differential_evolution(best_elite_sequence)

    # Final refinement with DE
    if best_sequence is not None:
        best_sequence = optimize_with_differential_evolution(best_sequence)

    # Fallback if no valid sequence found
    if best_sequence is None:
        best_sequence = generate_random_sequence()

    return best_sequence
