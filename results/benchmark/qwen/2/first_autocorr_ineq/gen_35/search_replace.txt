def get_good_direction_to_move_into(
    sequence: list[float],
) -> list[float] | None:
    """Returns the direction to move into the sequence."""
    n = len(sequence)
    sum_sequence = np.sum(sequence)

    # Avoid division by zero
    if sum_sequence < 1e-10:
        return None

    # Normalize sequence properly
    normalized_sequence = np.array(sequence) * np.sqrt(2 * n) / sum_sequence

    # Compute convolution to find maximum
    convolved = fftconvolve(normalized_sequence, normalized_sequence, mode='full')
    max_conv_value = np.max(convolved)

    # For a better gradient estimate, we compute the gradient of the objective function
    # The key insight is to find how to perturb the sequence to reduce max_conv_value
    try:
        # Use a simple but effective gradient estimate
        # Find the position where the maximum occurs
        max_positions = np.where(convolved == max_conv_value)[0]
        if len(max_positions) == 0:
            return None
        max_pos = max_positions[0]  # Take first occurrence

        # Create gradient direction that tries to reduce the maximum convolution value
        # This is a heuristic: we try to decrease the values around the peak position
        grad_dir = np.zeros_like(normalized_sequence)

        # Focus on the region around the maximum convolution index
        # The convolution at position k is sum_{i+j=k} sequence[i] * sequence[j]
        # To reduce max_conv_value, we should reduce values around indices that contribute significantly

        # Simple approach: reduce the elements that contribute most to the maximum convolution
        # Let's take a broader neighborhood
        window_size = min(5, n // 2)
        for i in range(max(0, max_pos - window_size), min(n, max_pos + window_size)):
            # Reduce values that contribute to maximum convolution
            if i < n:  # Ensure within bounds
                grad_dir[i] = -1.0  # Negative gradient to decrease values

        # Normalize the gradient direction
        grad_norm = np.linalg.norm(grad_dir)
        if grad_norm > 1e-10:
            grad_dir /= grad_norm
        else:
            # Fallback to simple gradient if normalization fails
            grad_dir = np.zeros_like(normalized_sequence)
            if max_pos < n:
                grad_dir[max_pos] = -1.0
                grad_norm = np.linalg.norm(grad_dir)
                if grad_norm > 1e-10:
                    grad_dir /= grad_norm

        t = 0.005

            (1 - t) * x + t * y for x, y in zip(normalized_sequence, grad_dir)

        scaled_new_sequence = [x * sum_sequence / np.sqrt(2 * n) for x in new_sequence]
