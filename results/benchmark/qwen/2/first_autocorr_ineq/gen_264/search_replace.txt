def compute_convolution_fft(seq):
    """Compute the autoconvolution using FFT for efficiency."""
    n = len(seq)
    # Pad to avoid circular convolution effects
    padded_len = 2 * n - 1
    padded_seq = np.pad(seq, (0, n-1), mode='constant')
    # Use FFT for efficient convolution
    fft_seq = np.fft.fft(padded_seq, padded_len)
    conv_result = np.fft.ifft(fft_seq * np.conj(fft_seq))
    return np.real(conv_result[:n])
