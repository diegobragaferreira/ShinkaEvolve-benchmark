def compute_c1_constant(sequence: List[float]) -> float:
    """
    Computes the C1 constant for a given sequence.
    C1 = 2n * max(convolution) / (sum(sequence))^2
    Returns 1/C1 (objective to maximize).
    """
    a = np.array(sequence)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return 0.0  # Invalid sequence

    # Efficiently compute autoconvolution using FFT
    n = len(a)
    if n > FFT_THRESHOLD:
        # Use FFT for large sequences
        padded_len = 2 * n - 1
        seq_fft = fft(a, padded_len)
        conv_fft = seq_fft * seq_fft.conj()
        conv = ifft(conv_fft).real
        max_conv = max(conv[:n])
    else:
        # Direct convolution for small sequences for accuracy
        conv = signal.convolve(a, a, mode='full')
        max_conv = max(conv[:n])

    c1 = (2 * n * max_conv) / (sum_a ** 2)
    if c1 == 0:
        return 0.0
    return 1.0 / c1
