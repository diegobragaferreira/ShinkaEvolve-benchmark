    """Solves the convolution LP for a given sequence and RHS using scipy.optimize.linprog."""
    try:
        n = len(f_sequence)
        if n < 1:
            return None

        # For each convolution index k, we generate constraint coefficients
        num_constraints = 2 * n - 1
        a_ub = np.zeros((num_constraints, n))
        b_ub = np.zeros(num_constraints)

        # Fill constraint matrix
        for k in range(num_constraints):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    a_ub[k, j] = f_sequence[i]
            b_ub[k] = rhs

        # Add non-negativity constraints (x_i >= 0)
        a_ub_nonneg = -np.eye(n)
        b_ub_nonneg = np.zeros(n)

        # Combine all constraints
        a_ub = np.vstack([a_ub, a_ub_nonneg])

            return result.x

def solve_convolution_lp(f_sequence, rhs):
    """Solves the convolution LP for a given sequence and RHS using scipy.optimize.linprog."""
    try:
        n = len(f_sequence)
        if n < 1:
            return None

        # Create constraint matrix A_ub such that A_ub * x <= b_ub
        # Vectorized approach for better performance

        # Generate constraint matrix using broadcasting
        num_constraints = 2 * n - 1
        a_ub = np.zeros((num_constraints, n))
        b_ub = np.full(num_constraints, rhs)

        # Efficiently create all constraint rows using broadcasting
        k_vals = np.arange(num_constraints)[:, np.newaxis]  # Shape (2*n-1, 1)
        i_vals = np.arange(n)  # Shape (n,)

        # Calculate j indices for all combinations at once
        j_vals = k_vals - i_vals  # Shape (2*n-1, n)

        # Create mask for valid j indices (0 <= j < n)
        valid_mask = (j_vals >= 0) & (j_vals < n)

        # Fill valid positions with f_sequence values
        a_ub[valid_mask] = f_sequence[j_vals[valid_mask]]

        # Add non-negativity constraints (x_i >= 0)
        a_ub_nonneg = -np.eye(n)
        b_ub_nonneg = np.zeros(n)

        # Combine all constraints
        a_ub = np.vstack([a_ub, a_ub_nonneg])
        b_ub = np.hstack([b_ub, b_ub_nonneg])

        # Define objective function (minimize -sum x, i.e., maximize sum x)
        c = -np.ones(n)

        # Solve linear programming problem with presolve turned off for faster convergence
        result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'presolve': False, 'time_limit': 10})

        if result.success:
            return result.x
        else:
            return None

    except Exception as e:
        return None
