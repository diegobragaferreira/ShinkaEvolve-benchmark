def get_spectral_guidance(sequence: List[float]) -> List[float]:
    """
    Generate a spectral-guided direction for sequence modification.
    This helps in guiding evolution towards better frequency characteristics.
    """
    try:
        n = len(sequence)
        if n < 1:
            return sequence

        # Get spectral information
        freq_indices, power_spectrum = spectral_analysis(sequence)

        threshold = np.percentile(power_spectrum, 70)  # Top 30% of power

        guided_sequence = np.array(sequence).copy()

        # This encourages smoothness in regions that contribute most to convolution
        if len(dominant_freqs) > 0:

            mask = np.zeros(n)

            seq_fft = fft(guided_sequence)

            filtered_fft = seq_fft * mask

            # Transform back to time domain
            guided_sequence = np.real(ifft(filtered_fft))

            # Ensure non-negativity
            guided_sequence = np.maximum(guided_sequence, 0)

        # Fallback to original sequence if spectral guidance fails
        return sequence

def get_spectral_guidance(sequence: List[float]) -> List[float]:
    """
    Generate a spectral-guided direction for sequence modification.
    This helps in guiding evolution towards better frequency characteristics.
    """
    try:
        n = len(sequence)
        if n < 1:
            return sequence

        # Get spectral information
        freq_indices, power_spectrum = spectral_analysis(sequence)

        # Identify frequencies with significant power, adjusted for better convolution properties
        # Focus on lower frequencies that typically contribute to smoother convolution shapes
        threshold = np.percentile(power_spectrum, 60)  # Adjusted threshold to capture more structure
        dominant_freqs = freq_indices[power_spectrum > threshold]

        # Create a modified sequence based on spectral features
        guided_sequence = np.array(sequence).copy()

        # Apply enhanced filtering to shape the spectrum for favorable autoconvolution
        # We emphasize frequencies that are less likely to produce large convolution peaks
        if len(dominant_freqs) > 0:
            # Create a custom filter that modifies the frequency response
            # This filter targets reducing high-power spikes in the spectrum
            mask = np.ones(n)

            # Apply attenuation to high-power regions to promote smoother sequences
            for i in range(n):
                if power_spectrum[i] > threshold * 1.5:  # Strong peaks
                    mask[i] = 0.5  # Reduce their influence
                elif power_spectrum[i] > threshold * 0.5:  # Moderate peaks
                    mask[i] = 0.8  # Partially dampen

            # Transform to frequency domain
            seq_fft = fft(guided_sequence)

            # Apply custom mask
            filtered_fft = seq_fft * mask

            # Transform back to time domain
            guided_sequence = np.real(ifft(filtered_fft))

            # Ensure non-negativity
            guided_sequence = np.maximum(guided_sequence, 0)

        return guided_sequence.tolist()

    except Exception as e:
        # Fallback to original sequence if spectral guidance fails
        return sequence
