    def solve_convolution_lp(self, f_sequence, rhs):
        """Solves the convolution LP for a given sequence and RHS."""
        try:
            n = len(f_sequence)
            c = -np.ones(n)
            a_ub = []
            b_ub = []

            if self.use_fft_constraints:
                # Use FFT-based convolution for more accurate constraint generation
                # Compute the full convolution using FFT
                padded_length = 2 * n - 1
                fa = fft(f_sequence, padded_length)
                # We create a matrix representing convolution constraints
                # Each row corresponds to one element of the convolution result
                for k in range(padded_length):
                    row = np.zeros(n)
                    # Inverse convolution relationship
                    for i in range(n):
                        j = k - i
                        if 0 <= j < n:
                            row[j] = f_sequence[i]
                    a_ub.append(row)
                    b_ub.append(rhs)
            else:
                # Traditional constraint building approach
                # Build convolution constraints
                for k in range(2 * n - 1):
                    row = np.zeros(n)

                        j = k - i
                        if 0 <= j < n:
                            row[j] = f_sequence[i]
