    n = len(sequence)
    if n == 0:
        return float('inf')

    # Use FFT for fast convolution with better numerical stability
    # Using fftconvolve for more reliable results
    try:
        # Compute convolution using fftconvolve
        autocorr = fftconvolve(sequence, sequence, mode='full')[:2*n-1]
        # Take only the first n elements for correlation
        autocorr = autocorr[:n]

        # Normalize and compute C1
        sum_seq = np.sum(sequence)
        if sum_seq < 0.01:
            return float('inf')  # Reject invalid sequences
