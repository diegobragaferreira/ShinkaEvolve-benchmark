def gradient_based_improvement(sequence, max_iter=50):
    """Improve sequence using simple gradient estimation."""
    current_sequence = np.array(sequence, dtype=float)
    current_fitness = evaluate_sequence(current_sequence)
    step_size = 0.01
    eps = 1e-6

    for iteration in range(max_iter):
        # Estimate gradient using finite differences
        grad = np.zeros_like(current_sequence)
        for i in range(len(current_sequence)):
            # Compute numerical gradient
            delta = np.zeros_like(current_sequence)
            delta[i] = eps
            f_plus = evaluate_sequence(current_sequence + delta)
            f_minus = evaluate_sequence(current_sequence - delta)
            grad[i] = (f_plus - f_minus) / (2 * eps)

        # Update step
        new_sequence = current_sequence + step_size * grad

        # Ensure non-negativity and reasonable bounds
        new_sequence = np.maximum(0, new_sequence)
        new_sequence = np.minimum(1000, new_sequence)

        # Check if update improved fitness
        new_fitness = evaluate_sequence(new_sequence)
        if new_fitness > current_fitness:
            current_sequence = new_sequence
            current_fitness = new_fitness
        else:
            # Reduce step size if no improvement
            step_size *= 0.95
            if step_size < 1e-6:
                break

    return current_sequence.tolist(), current_fitness

def gradient_based_improvement(sequence, max_iter=50):
    """Improve sequence using enhanced gradient estimation with adaptive step sizing."""
    current_sequence = np.array(sequence, dtype=float)
    current_fitness = evaluate_sequence(current_sequence)
    eps = 1e-6

    # Track elite sequences for better convergence
    elite_sequences = []
    elite_fitness = []

    for iteration in range(max_iter):
        # Estimate gradient using finite differences with adaptive epsilon
        grad = np.zeros_like(current_sequence)
        for i in range(len(current_sequence)):
            # Determine adaptive epsilon based on element magnitude
            elem_mag = abs(current_sequence[i])
            if elem_mag < 1e-6:
                adaptive_eps = eps
            else:
                adaptive_eps = eps * elem_mag

            # Compute numerical gradient
            delta = np.zeros_like(current_sequence)
            delta[i] = adaptive_eps
            f_plus = evaluate_sequence(current_sequence + delta)
            f_minus = evaluate_sequence(current_sequence - delta)
            grad[i] = (f_plus - f_minus) / (2 * adaptive_eps)

        # Adaptive step size with exponential decay
        step_size = 0.01 * np.exp(-iteration / 50.0)

        # Update step
        new_sequence = current_sequence + step_size * grad

        # Ensure non-negativity and reasonable bounds
        new_sequence = np.maximum(0, new_sequence)
        new_sequence = np.minimum(1000, new_sequence)

        # Check if update improved fitness
        new_fitness = evaluate_sequence(new_sequence)
        if new_fitness > current_fitness:
            current_sequence = new_sequence
            current_fitness = new_fitness

            # Preserve elite sequences
            if len(elite_sequences) < 5:
                elite_sequences.append(current_sequence.copy())
                elite_fitness.append(current_fitness)
            else:
                # Replace worst elite if new is better
                worst_idx = np.argmin(elite_fitness)
                if current_fitness > elite_fitness[worst_idx]:
                    elite_sequences[worst_idx] = current_sequence.copy()
                    elite_fitness[worst_idx] = current_fitness
        else:
            # Early stopping if no significant improvement
            if iteration > 10 and abs(new_fitness - current_fitness) < 1e-10:
                break

    # Return best sequence among elites or current if none preserved
    if elite_sequences:
        best_elite_idx = np.argmax(elite_fitness)
        return elite_sequences[best_elite_idx].tolist(), elite_fitness[best_elite_idx]
    else:
        return current_sequence.tolist(), current_fitness
