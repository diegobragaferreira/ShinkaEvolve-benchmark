        """Solves the convolution LP for a given sequence and RHS with warm-start."""
        try:
            n = len(f_sequence)
            c = -np.ones(n)

            # Efficiently build convolution constraints using FFT-based approach
            # Create Toeplitz-like convolution matrix using FFT
            # We'll construct the constraints more efficiently by leveraging FFT properties

            # Instead of looping through each k, we can precompute the convolution matrix
            # in a more efficient manner

            # Precompute the convolution constraint matrix directly
            a_ub = np.zeros((2 * n - 1, n))
            for k in range(2 * n - 1):
                # Fill the row for constraint k
                for i in range(n):
