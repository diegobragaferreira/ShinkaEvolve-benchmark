import numpy as np
from scipy import signal, optimize
from scipy.fft import fft, ifft
import random
import time
from typing import List, Optional, Tuple

# Constants
MAX_TIME_SECONDS = 180
MIN_SEQ_LENGTH = 10
MAX_SEQ_LENGTH = 1000
FFT_THRESHOLD = 100
CONVERGENCE_TOLERANCE = 1e-6
MAX_STAGNANT_ITERATIONS = 50
RANDOM_SEED = 42

# Set seed for reproducibility
np.random.seed(RANDOM_SEED)
random.seed(RANDOM_SEED)

def autocorrelation_constant(sequence: List[float]) -> float:
    """
    Calculates C₁ = 2n * max(b) / (sum(a))^2 where b = a * a (autoconvolution).
    Returns the inverse 1/C₁ which we want to maximize.
    """
    n = len(sequence)
    if n == 0:
        return 0.0
    
    sum_a = sum(sequence)
    if sum_a < 0.01:
        return 0.0
    
    # Compute autoconvolution using FFT for efficiency
    if n > FFT_THRESHOLD:
        # Use FFT for fast convolution
        padded_len = 2 * n - 1
        seq_fft = fft(sequence, padded_len)
        conv_fft = seq_fft * seq_fft.conj()  # Element-wise multiplication
        autoconv = ifft(conv_fft).real
        max_conv = max(autoconv)
    else:
        # Direct convolution for small sequences
        autoconv = signal.convolve(sequence, sequence, mode='full')
        max_conv = max(autoconv)
    
    # Calculate C₁
    c1 = (2 * n * max_conv) / (sum_a ** 2)
    if c1 == 0:
        return 0.0
    return 1.0 / c1

def compute_gradient_approximation(sequence: List[float], epsilon: float = 1e-4) -> List[float]:
    """
    Approximate gradient using finite differences.
    """
    n = len(sequence)
    grad = []
    for i in range(n):
        # Perturb dimension i
        perturbed_plus = sequence[:]
        perturbed_minus = sequence[:]
        perturbed_plus[i] += epsilon
        perturbed_minus[i] -= epsilon
        
        # Ensure non-negativity
        perturbed_plus[i] = max(0, perturbed_plus[i])
        perturbed_minus[i] = max(0, perturbed_minus[i])
        
        # Evaluate function
        f_plus = autocorrelation_constant(perturbed_plus)
        f_minus = autocorrelation_constant(perturbed_minus)
        
        grad_i = (f_plus - f_minus) / (2 * epsilon)
        grad.append(grad_i)
    
    return grad

def adaptive_step_size(current_inv_c1: float, iteration: int, prev_grad_norm: float = 1.0) -> float:
    """
    Adaptive step size based on convergence behavior.
    """
    # Start with a base step size
    base_step = 0.01
    
    # Reduce step size over time to stabilize near optimum
    if iteration > 100:
        base_step *= 0.5
    
    # Adjust based on gradient magnitude
    if prev_grad_norm > 1e-3:
        base_step *= min(1.0, 1.0 / prev_grad_norm)
    
    return base_step

def initialize_sequence(length: int) -> List[float]:
    """Initialize a sequence using a combination of random and heuristic values."""
    # Generate a base sequence with some structure
    seq = [random.uniform(0.1, 1.0) for _ in range(length)]
    
    # Apply a simple symmetry pattern to encourage good convolution behavior
    for i in range(length // 2):
        seq[length - 1 - i] = seq[i]
        
    return seq

def local_search(sequence: List[float], max_iterations: int = 100) -> List[float]:
    """
    Performs local search by slightly perturbing the sequence to improve the score.
    """
    current_seq = sequence[:]
    current_score = autocorrelation_constant(current_seq)
    
    for _ in range(max_iterations):
        if time.time() > start_time + MAX_TIME_SECONDS - 2:
            break
            
        # Make a small perturbation
        idx = random.randint(0, len(current_seq) - 1)
        delta = random.uniform(-0.05, 0.05)
        new_seq = current_seq[:]
        new_seq[idx] = max(0, new_seq[idx] + delta)
        
        new_score = autocorrelation_constant(new_seq)
        
        if new_score > current_score:
            current_seq = new_seq
            current_score = new_score
            
    return current_seq

def hybrid_optimization(sequence: List[float], max_iter: int = 500) -> List[float]:
    """
    A hybrid optimization approach using gradient ascent and local search.
    """
    current_seq = sequence[:]
    current_score = autocorrelation_constant(current_seq)
    prev_grad_norm = 1.0
    stagnant_count = 0
    
    for iteration in range(max_iter):
        if time.time() > start_time + MAX_TIME_SECONDS - 2:
            break
            
        # Compute gradient
        try:
            gradient = compute_gradient_approximation(current_seq)
        except Exception:
            # Fallback to local search
            current_seq = local_search(current_seq, 10)
            continue
            
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-10:
            break
            
        # Compute adaptive step size
        step_size = adaptive_step_size(current_score, iteration, prev_grad_norm)
        
        # Update sequence
        new_seq = []
        for i in range(len(current_seq)):
            new_val = current_seq[i] + step_size * gradient[i]
            new_seq.append(max(0, new_val))
        
        # Evaluate new sequence
        new_score = autocorrelation_constant(new_seq)
        
        if new_score > current_score:
            current_seq = new_seq
            current_score = new_score
            stagnant_count = 0
        else:
            stagnant_count += 1
            if stagnant_count > MAX_STAGNANT_ITERATIONS:
                # If stuck, perform local search
                current_seq = local_search(current_seq, 50)
                current_score = autocorrelation_constant(current_seq)
                stagnant_count = 0
                
        prev_grad_norm = grad_norm
    
    return current_seq

def get_good_direction_to_move_into(
    sequence: List[float],
) -> Optional[List[float]]:
    """
    Returns the direction to move into the sequence using a hybrid strategy.
    """
    n = len(sequence)
    if n == 0:
        return None
    
    # Check if we have enough elements
    if n < MIN_SEQ_LENGTH:
        # Expand sequence
        extended_seq = sequence + [0.0] * (MIN_SEQ_LENGTH - n)
        sequence = extended_seq
    
    # Use hybrid optimization
    try:
        sequence = hybrid_optimization(sequence, 300)
    except Exception:
        # Fallback to basic improvement
        sequence = local_search(sequence, 100)
    
    return sequence

def search_for_best_sequence() -> List[float]:
    """
    Function to search for the best coefficient sequence.
    """
    global start_time
    start_time = time.time()
    
    # Start with a well-structured sequence
    n = random.randint(MIN_SEQ_LENGTH, MAX_SEQ_LENGTH)
    sequence = initialize_sequence(n)
    
    # Try to improve it
    improved_sequence = get_good_direction_to_move_into(sequence)
    
    if improved_sequence is not None and len(improved_sequence) > 0:
        return improved_sequence
    else:
        # Return original if nothing works
        return sequence
