def search_for_best_sequence() -> list[float]:
    """Main search function to find the best coefficient sequence."""
    start_time = time.time()

    # Initialize a random sequence with varying length
    n = np.random.randint(MIN_SEQ_LENGTH, MAX_SEQ_LENGTH)
    best_sequence = [np.random.random() for _ in range(n)]
    best_inv_c1 = compute_inv_c1(best_sequence)

    # Keep track of elite sequences
    elites = [(best_sequence[:], best_inv_c1)]

    # Iteratively improve the sequence
    for iteration in range(100):  # Limited iterations to respect time budget
        if time.time() - start_time > MAX_TIME_SECONDS - 2:
            break

        improved_seq = get_good_direction_to_move_into(best_sequence)
        if improved_seq is not None:
            improved_inv_c1 = compute_inv_c1(improved_seq)
            if improved_inv_c1 > best_inv_c1:
                best_sequence = improved_seq
                best_inv_c1 = improved_inv_c1

                # Update elites
                if len(elites) < ELITE_SIZE:
                    elites.append((best_sequence[:], best_inv_c1))
                else:
                    # Replace the worst elite
                    worst_idx = min(range(len(elites)), key=lambda i: elites[i][1])
                    elites[worst_idx] = (best_sequence[:], best_inv_c1)
        else:
            # Fallback: modify the sequence slightly
            idx = np.random.randint(0, len(best_sequence))
            best_sequence[idx] = (best_sequence[idx] + np.random.rand()) % 1

        # Occasionally restart from elite if stuck
        if iteration % 10 == 0 and len(elites) > 0:
            # Pick a random elite to restart from
            restart_seq, _ = random.choice(elites)
            best_sequence = restart_seq[:]
            best_inv_c1 = compute_inv_c1(best_sequence)

    return best_sequence
