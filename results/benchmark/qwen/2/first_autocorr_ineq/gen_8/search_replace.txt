import numpy as np
from scipy import optimize
from scipy.fft import convolve as fft_convolve
import random
import time

def compute_autocorrelation_constant(sequence):
    """Computes C1 for a given sequence."""
    if len(sequence) == 0:
        return float('inf')
    sequence = np.array(sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return float('inf')  # Reject sequences with too small sum

    # Use FFT for faster convolution
    autocorr = fft_convolve(sequence, sequence[::-1], mode='full')
    max_b = np.max(autocorr)

    # Compute C1 = 2n * max(b) / (sum(a))^2
    n = len(sequence)
    if max_b == 0:
        return float('inf')

    c1 = 2 * n * max_b / (sum_a ** 2)
    return c1

def compute_inv_c1(sequence):
    """Computes 1/C1 for a given sequence."""
    c1 = compute_autocorrelation_constant(sequence)
    if c1 == float('inf'):
        return 0  # Invalid sequences get penalty
    return 1.0 / c1

def solve_convolution_lp(f_sequence, rhs):
    """Solves the convolution LP for a given sequence and RHS."""
    try:
        n = len(f_sequence)
        c = -np.ones(n)
        a_ub = []
        b_ub = []

        # Build constraint matrix for convolution constraints
        for k in range(2 * n - 1):
            row = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    row[j] = f_sequence[i]
            a_ub.append(row)
            b_ub.append(rhs)

        # Non-negativity constraints
        a_ub_nonneg = -np.eye(n)
        b_ub_nonneg = np.zeros(n)

        a_ub = np.vstack([a_ub, a_ub_nonneg])
        b_ub = np.hstack([b_ub, b_ub_nonneg])

        # Add small regularization to avoid numerical issues
        result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs')

        if result.success:
            g_sequence = result.x
            # Ensure non-negativity (sometimes LP solver can return small negatives)
            g_sequence = np.maximum(g_sequence, 0)
            return g_sequence
        else:
            return None
    except Exception as e:
        return None

def get_good_direction_to_move_into(
    sequence: list[float],
) -> list[float] | None:
    """Returns the direction to move into the sequence."""
    if len(sequence) < 2:
        return None

    n = len(sequence)
    sum_sequence = np.sum(sequence)
    if sum_sequence < 0.01:
        return None

    # Normalize sequence
    normalized_sequence = np.array(sequence) * np.sqrt(2 * n) / sum_sequence
    rhs = np.max(fft_convolve(normalized_sequence, normalized_sequence[::-1], mode='full'))

    g_fun = solve_convolution_lp(normalized_sequence, rhs)
    if g_fun is None or np.isnan(g_fun).any():
        return None

    sum_g = np.sum(g_fun)
    if sum_g < 0.01:
        return None

    normalized_g_fun = g_fun * np.sqrt(2 * n) / sum_g

    # Move toward the solution
    t = 0.05  # Slightly larger step size
    new_sequence = [
        (1 - t) * x + t * y for x, y in zip(sequence, normalized_g_fun)
    ]

    # Ensure non-negativity and clip extreme values
    new_sequence = [max(0, min(1000, x)) for x in new_sequence]

    return new_sequence

def initialize_sequence(length=None):
    """Initialize a good starting sequence."""
    if length is None:
        length = random.randint(100, 500)  # Randomly choose length

    # Create a mix of high and low values to explore diverse regions
    sequence = []
    for _ in range(length):
        # Mix of uniform and exponential distributions to get good variety
        val = random.expovariate(1.0) if random.random() < 0.7 else random.random()
        sequence.append(min(1000, max(0, val)))

    return sequence

def search_for_best_sequence(max_time=170) -> list[float]:
    """Function to search for the best coefficient sequence."""
    start_time = time.time()

    # Start with several initializations to explore the space
    best_inv_c1 = 0
    best_sequence = None

    # Try multiple starting points
    for attempt in range(10):
        if time.time() - start_time > max_time:
            break

        # Initialize with different strategies
        init_length = random.randint(100, 800)
        sequence = initialize_sequence(init_length)

        # Multiple iterations of local search
        for iteration in range(50):
            if time.time() - start_time > max_time:
                break

            h_function = get_good_direction_to_move_into(sequence)
            if h_function is not None:
                sequence = h_function
            else:
                # If we can't move, perturb slightly
                sequence[random.randint(0, len(sequence)-1)] = random.random() * 100

        # Evaluate this sequence
        inv_c1 = compute_inv_c1(sequence)
        if inv_c1 > best_inv_c1:
            best_inv_c1 = inv_c1
            best_sequence = sequence.copy()

    # Final refinement with better convergence
    if best_sequence is not None:
        refined_sequence = best_sequence.copy()
        for _ in range(20):
            if time.time() - start_time > max_time:
                break
            h_function = get_good_direction_to_move_into(refined_sequence)
            if h_function is not None:
                refined_sequence = h_function
            else:
                break
        final_inv_c1 = compute_inv_c1(refined_sequence)
        if final_inv_c1 > best_inv_c1:
            best_sequence = refined_sequence

    return best_sequence if best_sequence is not None else []
