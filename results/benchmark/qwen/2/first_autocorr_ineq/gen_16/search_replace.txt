import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import fftconvolve
import time

def compute_c1(sequence):
    """
    Compute C1 for a given sequence.
    C1 = 2*n*max(convolution) / (sum(sequence))^2
    We want to maximize 1/C1, which means minimizing C1.
    """
    if len(sequence) == 0 or abs(sum(sequence)) < 1e-10:
        return float('inf')

    # Use FFT-based convolution for efficiency
    convolved = fftconvolve(sequence, sequence, mode='full')
    max_conv = np.max(convolved)
    sum_seq = sum(sequence)

    # Return C1 value
    return (2 * len(sequence) * max_conv) / (sum_seq * sum_seq)

def evaluate_sequence(sequence):
    """
    Evaluate a sequence by computing 1/C1.
    Returns negative because differential_evolution minimizes.
    """
    c1 = compute_c1(sequence)
    if c1 == float('inf'):
        return float('inf')
    return -1.0 / c1  # We want to maximize 1/C1, so minimize -1/C1

import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import fftconvolve
import time
from collections import deque

# Configuration parameters for adaptive optimization
ADAPTIVE_THRESHOLD = 0.01  # Threshold for switching to gradient-based updates
LOCAL_SEARCH_ITERATIONS = 20  # Number of iterations for local search
MAX_LOCAL_SEARCHES = 3  # Maximum number of local searches per run
MIN_SEQUENCE_LENGTH = 50
MAX_SEQUENCE_LENGTH = 500

def compute_c1(sequence):
    """
    Compute C1 for a given sequence.
    C1 = 2*n*max(convolution) / (sum(sequence))^2
    We want to maximize 1/C1, which means minimizing C1.
    """
    if len(sequence) == 0 or abs(sum(sequence)) < 1e-10:
        return float('inf')

    # Use FFT-based convolution for efficiency
    convolved = fftconvolve(sequence, sequence, mode='full')
    max_conv = np.max(convolved)
    sum_seq = sum(sequence)

    # Return C1 value
    return (2 * len(sequence) * max_conv) / (sum_seq * sum_seq)

def evaluate_sequence(sequence):
    """
    Evaluate a sequence by computing 1/C1.
    Returns negative because differential_evolution minimizes.
    """
    c1 = compute_c1(sequence)
    if c1 == float('inf') or c1 > 1e10:  # Prevent overflow
        return float('inf')
    return -1.0 / c1  # We want to maximize 1/C1, so minimize -1/C1

def get_good_direction_to_move_into(
    sequence: list[float],
) -> list[float] | None:
    """Returns the direction to move into the sequence."""
    n = len(sequence)
    sum_sequence = np.sum(sequence)

    # Avoid division by zero
    if sum_sequence < 1e-10:
        return None

    normalized_sequence = [x * np.sqrt(2 * n) / sum_sequence for x in sequence]
    rhs = np.max(np.convolve(normalized_sequence, normalized_sequence))

    # Simplified LP solver attempt - if this fails, use basic gradient descent
    try:
        # Compute approximate gradient direction
        convolved = fftconvolve(normalized_sequence, normalized_sequence, mode='full')
        max_pos = np.argmax(convolved)

        # Simple gradient update based on position of maximum
        grad_dir = np.zeros_like(normalized_sequence)
        if max_pos < n:
            # Gradient points toward reducing the maximum
            grad_dir[max_pos] = 1.0

        # Normalize gradient direction
        grad_norm = np.linalg.norm(grad_dir)
        if grad_norm > 1e-10:
            grad_dir /= grad_norm

        # Apply small step in gradient direction
        t = 0.01
        new_sequence = [
            (1 - t) * x + t * y for x, y in zip(normalized_sequence, grad_dir)
        ]

        # Convert back to original scale
        scaled_new_sequence = [x * sum_sequence / np.sqrt(2 * n) for x in new_sequence]
        return scaled_new_sequence

    except Exception:
        return None

def adaptive_search(current_seq, current_score, history):
    """
    Perform adaptive search combining global and local optimization.
    """
    n = len(current_seq)
    # Check if we should perform local search
    if len(history) > 1 and abs(history[-1] - history[-2]) < ADAPTIVE_THRESHOLD:
        # Try local gradient-based update
        direction = get_good_direction_to_move_into(current_seq)
        if direction is not None:
            # Apply gradient update
            updated_seq = direction
            updated_score = evaluate_sequence(updated_seq)

            if updated_score > current_score:
                return updated_seq, updated_score, True  # Success with local update

    # Default to global optimization approach
    return current_seq, current_score, False
