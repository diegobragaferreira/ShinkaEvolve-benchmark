def get_spectral_guidance(sequence: List[float]) -> List[float]:
    """
    Generate a spectral-guided direction for sequence modification.
    This helps in guiding evolution towards better frequency characteristics.
    """
    try:
        n = len(sequence)
        if n < 1:
            return sequence

        # Get spectral information
        freq_indices, power_spectrum = spectral_analysis(sequence)

        # Identify dominant frequencies (those with significant power)
        threshold = np.percentile(power_spectrum, 70)  # Top 30% of power
        dominant_freqs = freq_indices[power_spectrum > threshold]

        # Create a modified sequence based on spectral features
        guided_sequence = np.array(sequence).copy()

        # Apply a smoothing filter focused on dominant frequencies
        # This encourages smoothness in regions that contribute most to convolution
        if len(dominant_freqs) > 0:
            # Create a frequency domain mask
            mask = np.zeros(n)
            mask[dominant_freqs] = 1.0

            # Transform to frequency domain
            seq_fft = fft(guided_sequence)

            # Apply mask (low-pass filter based on dominant frequencies)
            filtered_fft = seq_fft * mask

            # Transform back to time domain
            guided_sequence = np.real(ifft(filtered_fft))

            # Ensure non-negativity
            guided_sequence = np.maximum(guided_sequence, 0)

        return guided_sequence.tolist()

    except Exception as e:
        # Fallback to original sequence if spectral guidance fails
        return sequence

def calculate_diversity(population):
    """Calculate diversity of the population."""
    if len(population) < 2:
        return 0.0
    # Flatten all sequences into one big array
    flat_array = np.concatenate(population)
    std_dev = np.std(flat_array)
    # Normalize by the mean
    mean_val = np.mean(flat_array)
    diversity = std_dev / (mean_val + 1e-10) if mean_val != 0 else 0.0

    # Also consider sequence length diversity
    lengths = [len(seq) for seq in population]
    length_std = np.std(lengths)
    length_mean = np.mean(lengths)
    length_diversity = length_std / (length_mean + 1e-10) if length_mean != 0 else 0.0

    # Combine both diversities
    combined_diversity = (diversity + length_diversity) / 2.0
    return combined_diversity
