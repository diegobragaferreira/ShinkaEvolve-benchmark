    def solve_convolution_lp(self, f_sequence, rhs, prev_solution=None):
        """Solves the convolution LP for a given sequence and RHS with warm-start."""
        try:
            n = len(f_sequence)
            c = -np.ones(n)

            # Use FFT-based convolution for constraint generation to improve efficiency
            # Instead of generating convolution constraints manually, we precompute
            # the constraint matrix using FFT-based approach

            # For convolution constraints, we can build them more efficiently:
            # Each constraint corresponds to a convolution coefficient
            # The convolution constraint matrix can be built using properties of circulant matrices

            # However, for now, let's stick to the standard approach but with better error handling
            # and ensure we don't create overly large constraint matrices

            # Limit the constraint size for very large sequences to prevent memory issues
            max_constraints = 10000
            num_constraints = min(2 * n - 1, max_constraints)

            a_ub = []
            b_ub = []

            # Build convolution constraints efficiently
            # We'll just take a subset of constraints if the sequence is very large
            constraint_indices = np.linspace(0, 2 * n - 2, num_constraints, dtype=int)

            for k in constraint_indices:
                row = np.zeros(n)
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        row[j] = f_sequence[i]
                a_ub.append(row)
                b_ub.append(rhs)

            # Add non-negativity constraints
            a_ub_nonneg = -np.eye(n)
            b_ub_nonneg = np.zeros(n)

            a_ub = np.vstack([a_ub, a_ub_nonneg])
            b_ub = np.hstack([b_ub, b_ub_nonneg])

            # Solve linear program with warm start if available
            if prev_solution is not None:
                result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', x0=prev_solution)
            else:
                result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs')

            if result.success:
                return result.x
            else:
                return None

        except Exception:
            # If there's an issue, fall back to basic approach
            try:
                # Try with alternative method
                n = len(f_sequence)
                c = -np.ones(n)
                a_ub = []
                b_ub = []
