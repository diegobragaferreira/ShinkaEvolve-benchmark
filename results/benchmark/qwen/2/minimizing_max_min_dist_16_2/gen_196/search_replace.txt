def _adaptive_perturbed_grid_init():
    """Initialize with grid points plus adaptive random perturbations based on current distribution."""
    # Create a denser and more strategically placed grid pattern
    # Using a 5x5 pattern with central point removed for better distribution
    grid_points = []
    for i in range(5):
        for j in range(5):
            # Skip the center to create more uniform spacing
            if i == 2 and j == 2:
                continue
            grid_points.append([i, j])

    # Convert to numpy array and normalize to [0,1] range
    points = np.array(grid_points, dtype=float) / 4.0

    # Add some randomness to break perfect symmetry
    np.random.seed(42)
    points += np.random.uniform(-0.01, 0.01, points.shape)

    # Ensure points stay within bounds
    points = np.clip(points, 0, 1)

    # If we have too many points, sample down to 16
    if len(points) > 16:
        indices = np.random.choice(len(points), 16, replace=False)
        points = points[indices]
    elif len(points) < 16:
        # If we have too few points, add some random points
        additional_points = np.random.rand(16 - len(points), 2)
        points = np.vstack([points, additional_points])

    # Ensure we have exactly 16 points
    points = points[:16]

    return points
