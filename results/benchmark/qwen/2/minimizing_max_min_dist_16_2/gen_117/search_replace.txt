    def analyze_grid_quality(points):
        """Analyze the quality of a grid configuration based on distance distribution"""
        if len(points) < 2:
            return 0.0

        distances = pdist(points)
        if len(distances) == 0:
            return 0.0

        min_dist = np.min(distances)
        max_dist = np.max(distances)
        avg_dist = np.mean(distances)
        std_dist = np.std(distances)

        # Quality metric: consider both minimum distance and distribution uniformity
        if max_dist == 0:
            return 0.0

        # Ratio of min distance to max distance (what we want to maximize)
        ratio = min_dist / max_dist

        # Bonus for uniform distribution (lower standard deviation)
        uniformity_bonus = 1.0 / (1.0 + std_dist / avg_dist)

        # Combined quality score
        quality = ratio * uniformity_bonus
        return quality

    def multi_start_optimization(initial_points_list):
        """Perform multi-start optimization with progressive refinement"""
        best_ratio = -np.inf
