This addresses the problem's need for both global search capability (to escape local optima) and precise local optimization (to refine good solutions).     def optimize_with_hybrid_approach(initial_configs, max_time=170):
        """Run hybrid optimization using evolutionary algorithms and local refinement"""
        best_ratio = -np.inf
        best_points = None
        start_time = time.time()

        # First, run a population-based evolutionary optimization to find promising regions
        # This helps avoid getting stuck in poor local optima
        evolutionary_configs = []

        # Try a few different starting configurations for evolutionary approach
        for i in range(min(3, len(initial_configs))):
            init_points = initial_configs[i].copy()
            try:
                bounds = [(0, 1) for _ in range(32)]

                # Run differential evolution on a subset of initial configurations
                de_timeout = min(max_time * 0.3, 50)
                if de_timeout > 20:
                    result_de = differential_evolution(
                        objective_with_regularization,
                        bounds,
                        seed=42,
                        maxiter=int(de_timeout/2),
                        popsize=min(30, 16 * 2),
                        mutation=(0.7, 1),
                        recombination=0.8,
                        tol=1e-10
                    )

                    if result_de.success:
                        de_points = result_de.x.reshape(-1, 2)
                        evolutionary_configs.append(de_points)
            except:
                pass

        # Combine evolutionary results with initial configurations
        all_configs = initial_configs + evolutionary_configs

        # Sort all configurations by quality
        all_ratios = [compute_min_max_ratio(cfg) for cfg in all_configs]

        # Now apply adaptive local optimization to top candidates
