import numpy as np
from scipy.optimize import minimize


def min_max_dist_dim2_16() -> np.ndarray:
    """
    Creates 16 points in 2 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (16,2) containing the (x,y) coordinates of the 16 points.

    """

    n = 16
    d = 2

    # Initialize with a structured 4x4 grid (more balanced than random)
    grid_points = np.array([[i, j] for i in range(4) for j in range(4)])
    points = grid_points.astype(float) / 3.0  # Normalize to [0,1] range

    # Define objective function: negative ratio (we'll minimize this)
    def objective(x):
        # Reshape flat array back to points
        pts = x.reshape(n, d)

        # Calculate all pairwise distances
        distances = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((pts[i] - pts[j])**2))
                distances.append(dist)

        distances = np.array(distances)
        min_dist = np.min(distances)
        max_dist = np.max(distances)

        # Return negative ratio to maximize the ratio
        if max_dist == 0:
            return 0
        return -min_dist / max_dist

    # Define bounds (points must be in [0,1] x [0,1])
    bounds = [(0, 1) for _ in range(n * d)]

    # Optimize using L-BFGS-B algorithm
    result = minimize(objective, points.flatten(), method='L-BFGS-B', bounds=bounds,
                      options={'ftol': 1e-12, 'gtol': 1e-12})

    # Extract optimized points
    optimized_points = result.x.reshape(n, d)

    return optimized_points

import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform


def min_max_dist_dim2_16() -> np.ndarray:
    """
    Creates 16 points in 2 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (16,2) containing the (x,y) coordinates of the 16 points.

    """

    n = 16
    d = 2
    num_restarts = 5
    best_ratio = -np.inf
    best_points = None

    # Try multiple initializations with different perturbations
    for restart in range(num_restarts):
        # Create initial configuration - regular grid with small perturbations
        np.random.seed(42 + restart)  # Different seed for each restart
        base_grid = np.array([[i, j] for i in range(4) for j in range(4)])
        points = base_grid.astype(float) / 3.0  # Normalize to [0,1] range

        # Add small random perturbations
        perturbation_magnitude = 0.05 + 0.02 * restart  # Increasing perturbation with restart
        points += np.random.uniform(-perturbation_magnitude, perturbation_magnitude, points.shape)

        # Ensure points are within bounds
        points = np.clip(points, 0, 1)

        # Define objective function: negative ratio (we'll minimize this)
        def objective(x):
            # Reshape flat array back to points
            pts = x.reshape(n, d)

            # Calculate all pairwise distances efficiently using scipy
            distances = pdist(pts)

            # Handle case where there are no distances (shouldn't happen)
            if len(distances) == 0:
                return 0

            min_dist = np.min(distances)
            max_dist = np.max(distances)

            # Return negative ratio to maximize the ratio
            if max_dist <= 1e-12:  # Avoid division by zero
                return 0
            return -min_dist / max_dist

        # Define bounds (points must be in [0,1] x [0,1])
        bounds = [(0, 1) for _ in range(n * d)]

        # Optimize using L-BFGS-B algorithm
        try:
            result = minimize(objective, points.flatten(), method='L-BFGS-B', bounds=bounds,
                              options={'ftol': 1e-10, 'gtol': 1e-10})

            # Extract optimized points
            optimized_points = result.x.reshape(n, d)

            # Calculate the actual ratio for this solution
            distances = pdist(optimized_points)
            if len(distances) > 0:
                min_dist = np.min(distances)
                max_dist = np.max(distances)
                if max_dist > 0:
                    ratio = min_dist / max_dist
                    if ratio > best_ratio:
                        best_ratio = ratio
                        best_points = optimized_points.copy()

        except Exception as e:
            continue  # Skip this restart if optimization fails

    # If we didn't find any good solution, return the original grid
    if best_points is None:
        grid_points = np.array([[i, j] for i in range(4) for j in range(4)])
        best_points = grid_points.astype(float) / 3.0

    return best_points
