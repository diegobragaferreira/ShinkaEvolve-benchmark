    def analyze_distance_distribution(points):
        """Analyze the distance distribution to inform perturbation strategy"""
        distances = pdist(points)
        if len(distances) == 0:
            return 0, 1, 0.0  # default values if no distances exist

        min_dist = np.min(distances)
        max_dist = np.max(distances)
        ratio = min_dist / max_dist if max_dist > 0 else 0.0
        mean_dist = np.mean(distances) if len(distances) > 0 else 0.0

        return min_dist, max_dist, ratio

    def generate_adaptive_perturbation(base_points, iteration=0):
        """Generate perturbed points with adaptive magnitude based on current distribution quality"""
        # Analyze current configuration
        min_dist, max_dist, ratio = analyze_distance_distribution(base_points)

        # Base perturbation magnitude based on how far we are from optimal
        # If ratio is low (bad), use larger perturbations; if high (good), use smaller ones
        base_magnitude = max(0.005, 0.05 * (1.0 - ratio * 2))

        # Ensure we don't get too aggressive with perturbations
        base_magnitude = min(0.05, base_magnitude)

        # Add random perturbations
        perturbation = np.random.normal(0, base_magnitude, base_points.shape)
        perturbed_points = base_points + perturbation

        # Clip to valid range
        perturbed_points = np.clip(perturbed_points, 0.01, 0.99)
        return perturbed_points
