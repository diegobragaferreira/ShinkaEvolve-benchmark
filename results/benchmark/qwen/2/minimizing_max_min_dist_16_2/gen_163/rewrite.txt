import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
import math
from typing import Tuple, List, Optional

class AdaptiveHexagonalEvolver:
    """Advanced optimizer using adaptive hexagonal patterns and intelligent optimization."""
    
    def __init__(self, num_points: int = 16, dimension: int = 2):
        self.num_points = num_points
        self.dimension = dimension
        self.bounds = [(0.001, 0.999) for _ in range(num_points * dimension)]
        
    def calculate_ratio(self, points: np.ndarray) -> Tuple[float, float, float]:
        """Calculate min/max distance ratio along with actual values."""
        if len(points) < 2:
            return 0.0, 0.0, 0.0
        
        distances = pdist(points)
        if len(distances) == 0:
            return 0.0, 0.0, 0.0
            
        min_dist = np.min(distances)
        max_dist = np.max(distances)
        
        if max_dist == 0:
            return 0.0, min_dist, max_dist
            
        ratio = min_dist / max_dist
        return ratio, min_dist, max_dist
    
    def objective_function(self, x: np.ndarray) -> float:
        """Objective function to minimize (negative ratio)."""
        points = x.reshape(-1, self.dimension)
        ratio, _, _ = self.calculate_ratio(points)
        return -ratio
    
    def generate_adaptive_hexagonal_grid(self) -> np.ndarray:
        """Generate enhanced hexagonal lattice with adaptive spacing for better distribution."""
        points = []
        rows = 4
        cols = 4
        
        # Use more optimal spacing ratios
        spacing_x = 0.9 / (cols - 1) if cols > 1 else 0.9
        spacing_y = 0.9 / (rows - 1) if rows > 1 else 0.9
        
        # Apply golden ratio-inspired adjustment
        phi = (1 + math.sqrt(5)) / 2
        spacing_x *= 0.85
        spacing_y *= 0.85
        
        for i in range(rows):
            for j in range(cols):
                if len(points) >= self.num_points:
                    break
                x_offset = spacing_x * 0.25 if i % 2 == 1 else 0.0
                x = 0.05 + (j * spacing_x) + x_offset
                y = 0.05 + i * spacing_y
                
                # Ensure bounds are maintained explicitly
                x = max(0.001, min(0.999, x))
                y = max(0.001, min(0.999, y))
                
                points.append([x, y])
        
        # Trim to exact number of points
        return np.array(points[:self.num_points])
    
    def generate_advanced_fibonacci(self) -> np.ndarray:
        """Generate improved Fibonacci spiral with better point distribution."""
        points = []
        # Use a more sophisticated golden ratio variant
        phi = (1 + math.sqrt(5)) / 2
        
        # Add randomness to avoid perfect patterns
        np.random.seed(42)
        
        for i in range(self.num_points):
            # Slightly randomized Fibonacci pattern
            theta = math.acos(-1 + (2 * i) / (self.num_points - 1))
            phi_angle = (i * 2 * math.pi) / (phi * phi)
            
            # Add small perturbations to avoid regularity
            theta += np.random.normal(0, 0.05)
            phi_angle += np.random.normal(0, 0.03)
            
            x = math.sin(theta) * math.cos(phi_angle)
            y = math.sin(theta) * math.sin(phi_angle)
            
            # More careful mapping to avoid boundary issues
            x = 0.05 + 0.9 * (x + 1) / 2
            y = 0.05 + 0.9 * (y + 1) / 2
            
            # Ensure bounds
            x = max(0.001, min(0.999, x))
            y = max(0.001, min(0.999, y))
            
            points.append([x, y])
        
        return np.array(points)
    
    def generate_improved_grid(self) -> np.ndarray:
        """Generate improved grid with better centering and spacing."""
        points = []
        side_length = 4  # Fixed for 16 points
        
        for i in range(side_length):
            for j in range(side_length):
                if len(points) >= self.num_points:
                    break
                # Better positioned grid points
                x = 0.05 + (i + 0.5) * 0.9 / side_length
                y = 0.05 + (j + 0.5) * 0.9 / side_length
                points.append([x, y])
        
        return np.array(points[:self.num_points])
    
    def generate_initial_configurations(self) -> List[np.ndarray]:
        """Generate diverse and high-quality initial configurations."""
        configs = []
        
        # High-quality base configurations
        configs.append(self.generate_adaptive_hexagonal_grid())
        configs.append(self.generate_advanced_fibonacci())
        configs.append(self.generate_improved_grid())
        
        # Create multiple variants with different perturbation levels
        np.random.seed(42)
        for base_config in configs:
            # Multiple perturbation levels to increase diversity
            for perturbation_level in [0.01, 0.015, 0.02, 0.025]:
                perturbed = base_config + np.random.normal(0, perturbation_level, base_config.shape)
                # Ensure bounds are maintained after perturbation
                perturbed[:, 0] = np.clip(perturbed[:, 0], 0.001, 0.999)
                perturbed[:, 1] = np.clip(perturbed[:, 1], 0.001, 0.999)
                configs.append(perturbed)
        
        # Add one more pattern - circle arrangement with radial distribution
        circle_points = []
        for i in range(self.num_points):
            angle = (i * 2 * math.pi) / self.num_points
            # Distribute radially with some randomness
            radius = 0.3 + 0.4 * np.random.random()
            x = 0.5 + radius * np.cos(angle)
            y = 0.5 + radius * np.sin(angle)
            circle_points.append([x, y])
        
        circle_config = np.array(circle_points)
        circle_config[:, 0] = np.clip(circle_config[:, 0], 0.001, 0.999)
        circle_config[:, 1] = np.clip(circle_config[:, 1], 0.001, 0.999)
        configs.append(circle_config)
        
        return configs
    
    def adaptive_optimization(self, x0: np.ndarray, max_iter: int = 200) -> Optional[np.ndarray]:
        """Perform adaptive optimization with intelligent method switching."""
        try:
            # First attempt with L-BFGS-B for quick convergence
            result1 = minimize(
                self.objective_function,
                x0,
                method='L-BFGS-B',
                bounds=self.bounds,
                options={'maxiter': max_iter // 2, 'ftol': 1e-8, 'gtol': 1e-5}
            )
            
            if result1.success:
                # If L-BFGS-B succeeded, try refining with SLSQP
                result2 = minimize(
                    self.objective_function,
                    result1.x,
                    method='SLSQP',
                    bounds=self.bounds,
                    options={'maxiter': max_iter // 2, 'ftol': 1e-10, 'gtol': 1e-8}
                )
                
                if result2.success:
                    return result2.x.reshape(-1, self.dimension)
                else:
                    # Return L-BFGS-B result if SLSQP fails
                    return result1.x.reshape(-1, self.dimension)
            else:
                # Fallback to TNC if L-BFGS-B fails
                try:
                    result3 = minimize(
                        self.objective_function,
                        x0,
                        method='TNC',
                        bounds=self.bounds,
                        options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-5}
                    )
                    
                    if result3.success:
                        return result3.x.reshape(-1, self.dimension)
                except Exception:
                    pass
                    
        except Exception:
            pass
            
        # Last resort - return original points
        return x0.reshape(-1, self.dimension)
    
    def get_best_solution(self, configs: List[np.ndarray]) -> np.ndarray:
        """Find best solution among all starting configurations with enhanced strategy."""
        best_ratio = -np.inf
        best_points = None
        
        # Evaluate each configuration in sequence
        for i, config in enumerate(configs):
            # Perform adaptive optimization
            optimized_points = self.adaptive_optimization(config.flatten(), max_iter=150)
            
            if optimized_points is not None:
                ratio, _, _ = self.calculate_ratio(optimized_points)
                
                if ratio > best_ratio:
                    best_ratio = ratio
                    best_points = optimized_points.copy()
        
        return best_points if best_points is not None else configs[0]

def min_max_dist_dim2_16() -> np.ndarray:
    """
    Creates 16 points in 2 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (16,2) containing the (x,y) coordinates of the 16 points.
    """
    
    # Initialize advanced evolver
    evolver = AdaptiveHexagonalEvolver(16, 2)
    
    # Generate high-quality initial configurations
    initial_configs = evolver.generate_initial_configurations()
    
    # Find best solution with improved optimization strategy
    best_points = evolver.get_best_solution(initial_configs)
    
    # Final refinement of the best result
    final_points = evolver.adaptive_optimization(best_points.flatten(), max_iter=200)
    
    # If final refinement failed, use the previously best result
    if final_points is None:
        return best_points
    
    return final_points
