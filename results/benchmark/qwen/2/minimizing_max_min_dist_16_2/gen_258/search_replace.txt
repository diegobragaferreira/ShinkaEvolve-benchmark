        if std_dist / avg_dist < 0.1:

        min_dist_factor = 1.0

            min_dist_factor = 1.5  # Small min distance - moderate changes

        perturbation_magnitude = base_perturbation * quality_factor * uniformity_factor * min_dist_factor
