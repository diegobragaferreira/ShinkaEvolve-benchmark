        """Create initial 4x4 grid points"""

    def perturb_points_adaptive(points):
        """Add adaptive random perturbations based on current distance distribution"""
        # Calculate current distance distribution
        distances = pdist(points)
        if len(distances) > 0:
            current_ratio = np.min(distances) / np.max(distances) if np.max(distances) > 0 else 0

            # Scale perturbation magnitude inversely to the current ratio
            # If ratio is low (poor distribution), use larger perturbations to explore
            # If ratio is high (good distribution), use smaller perturbations to refine
            if current_ratio > 0:
                # Perturbation magnitude inversely proportional to current ratio
                perturbation_magnitude = max(0.005, 0.05 * (1.0 - current_ratio * 2.0))
            else:
                perturbation_magnitude = 0.03
        else:
            perturbation_magnitude = 0.03

        np.random.seed(42)  # For reproducibility
        noise = np.random.normal(0, perturbation_magnitude, points.shape)
        perturbed = points + noise
        return np.clip(perturbed, 0, 1)
