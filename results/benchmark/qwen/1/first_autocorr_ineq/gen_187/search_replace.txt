def refine_with_gradient_descent(sequence, max_iter=100, learning_rate=0.01, tolerance=1e-6):
    """Improved gradient descent with early stopping and dynamic learning rate."""
    current_seq = sequence.copy()
    prev_fitness = objective_function(current_seq)

    for iteration in range(max_iter):
        grad = get_gradient_estimate(current_seq)
        if grad is None:
            break

        # Adaptive learning rate
        current_learning_rate = learning_rate * (1.0 - 0.9 * (iteration / max_iter))

        # Gradient ascent (since we're maximizing)
        current_seq = [max(0, x + current_learning_rate * g) for x, g in zip(current_seq, grad)]

        # Normalize to keep sum reasonable
        sum_seq = np.sum(current_seq)
        if sum_seq > 0.01:
            current_seq = [x * 0.01 / sum_seq for x in current_seq]

        # Early stopping based on improvement
        current_fitness = objective_function(current_seq)
        if abs(current_fitness - prev_fitness) < tolerance:
            break
        prev_fitness = current_fitness

    return current_seq

def refine_with_local_search(sequence, max_iter=100):
    """Refine the sequence using a multi-start local search approach with cached results."""
    from scipy.optimize import minimize

    # Define the objective function for local optimization
    def objective(x):
        # Ensure non-negativity
        x = np.maximum(x, 0)
        # Normalize for fair comparison
        sum_x = np.sum(x)
        if sum_x < 1e-10:
            return float('inf')
        x_norm = x / sum_x
        # Return negative because we are minimizing
        return -objective_function(x_norm)

    # Multi-start approach
    best_x = np.array(sequence)
    best_fitness = objective_function(sequence)

    for _ in range(5):  # Multiple random starts
        # Start from a perturbed version of the sequence
        x_start = np.array(sequence) * (0.9 + 0.2 * np.random.rand(len(sequence)))
        # Ensure non-negativity
        x_start = np.maximum(x_start, 0)

        # Local optimization using L-BFGS
        try:
            res = minimize(objective, x_start, method='L-BFGS-B', bounds=[(0, 1000)] * len(x_start))
            if res.success:
                x_opt = res.x
                # Normalize
                sum_x = np.sum(x_opt)
                if sum_x > 0:
                    x_opt = x_opt / sum_x
                    fitness_opt = objective_function(x_opt)
                    if fitness_opt > best_fitness:
                        best_fitness = fitness_opt
                        best_x = x_opt
        except:
            pass  # Continue if optimization fails

    # Return best result
    return best_x.tolist()
