def generate_structured_step_sequence(length: int, num_steps: int) -> List[float]:
    """Generate a structured step function sequence designed to improve autocorrelation properties."""
    # Create steps that are more evenly distributed and have strategic heights
    # Use fewer steps with larger heights to reduce peak convolution

    # Ensure we don't have more steps than positions
    num_steps = min(num_steps, length)

    # Place steps at regular intervals to spread out the mass
    step_positions = [int(i * length / num_steps) for i in range(num_steps)]
    # Ensure unique positions and handle edge cases
    step_positions = sorted(set(step_positions))
    if len(step_positions) < num_steps:
        # Fill missing positions
        while len(step_positions) < num_steps:
            # Insert a position where there's maximum gap
            gaps = [(step_positions[i+1] - step_positions[i], i)
                   for i in range(len(step_positions)-1)]
            if gaps:
                max_gap, idx = max(gaps)
                insert_pos = (step_positions[idx] + step_positions[idx+1]) // 2
                step_positions.append(insert_pos)
            else:
                step_positions.append(length//2)
            step_positions = sorted(set(step_positions))

    step_heights = []

    sequence = [0.0] * length

def mutate_sequence(sequence: List[float], generation: int, population_size: int, mutation_strength=0.3) -> List[float]:
    """Apply adaptive mutation to a sequence with decreasing rate over generations."""
    mutated = sequence.copy()
    # Adaptive mutation rate that decreases with generation and increases with diversity
    mutation_rate = max(0.05, 0.3 * (1 - generation / (population_size * 2)))

    # Determine if sequence looks like a step function for tailored mutation
    is_step_like = False
    if len(sequence) > 10:
        # Check for few distinct values indicating step-like nature
        unique_vals = set(sequence)

            is_step_like = True

            if is_step_like:

    return mutated
