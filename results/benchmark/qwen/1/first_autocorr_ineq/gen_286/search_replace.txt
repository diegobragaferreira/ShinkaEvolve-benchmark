def compute_autocorrelation_constant(sequence):
    """Computes the autocorrelation constant C₁ for a given sequence."""
    if len(sequence) == 0 or np.sum(sequence) < 0.01:
        return float('inf')

    n = len(sequence)

    # Use direct computation for small sequences (under 500) to avoid FFT artifacts
    if n < 500:
        # Direct computation for better numerical stability
        autocorr = np.zeros(2 * n - 1)
        for i in range(n):
            for j in range(n):
                autocorr[i + j] += sequence[i] * sequence[j]
        # Extract the autocorrelation values
        autocorr = autocorr[n-1:]  # Only take the relevant part
    else:
        # Use FFT for larger sequences
        autocorr_full = fftconvolve(sequence, sequence[::-1], mode='full')
        # Take the second half which corresponds to the actual autocorrelation
        autocorr = autocorr_full[n-1:]

    # Early termination: check if we can skip further computation
    max_autocorr = np.max(autocorr)
    if max_autocorr == 0:
        return float('inf')  # Avoid division by zero

    sum_sq = np.sum(sequence)**2
    if sum_sq == 0:
        return float('inf')

    # C₁ = 2n * max(autocorr) / (sum(sequence))^2
    C1 = 2 * n * max_autocorr / sum_sq

    return C1
