<DESCRIPTION>
This edit introduces adaptive mutation and crossover rates that dynamically adjust based on the optimization progress. The idea is to use higher mutation rates in early generations to explore the search space widely, and decrease them in later generations to exploit promising areas. Similarly, crossover rate is adjusted to maintain a balance between exploration and exploitation. This approach is inspired by successful metaheuristic techniques that adapt parameters based on performance history.

def adaptive_mutation_rate(population_fitnesses: List[float], generation: int, max_generations: int) -> float:
    """Calculate adaptive mutation rate based on population diversity and generation."""
    if len(population_fitnesses) < 2:
        return 0.1

    std_dev = np.std(population_fitnesses)
    avg_fitness = np.mean(population_fitnesses)

    # Higher diversity = higher mutation rate
    if avg_fitness > 0:
        base_mutation = 0.1 + std_dev / avg_fitness
    else:
        base_mutation = 0.1

    # Decrease mutation rate over generations
    decay_factor = 1.0 - (generation / max_generations)
    mutation_rate = min(0.3, max(0.01, base_mutation * decay_factor))

    return mutation_rate

def adaptive_crossover_rate(generation: int, max_generations: int) -> float:
    """Adaptively adjust crossover rate over generations."""
    # Start with high crossover rate and decrease it over time
    return 0.8 * (1.0 - (generation / max_generations))
