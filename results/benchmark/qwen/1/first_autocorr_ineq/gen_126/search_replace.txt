def multi_start_local_search(sequence, num_starts=5, max_iter_per_start=20):
    """Perform multi-start local search to find a better sequence."""
    best_seq = sequence.copy()
    best_score = evaluate_sequence(best_seq)

    for _ in range(num_starts):
        # Slightly perturb the current sequence
        perturbed_seq = [x * (0.9 + 0.2 * np.random.random()) for x in sequence]

        improved_seq = local_search_improvement(perturbed_seq, max_iter_per_start)
        improved_score = evaluate_sequence(improved_seq)

        # Keep the better of the two
