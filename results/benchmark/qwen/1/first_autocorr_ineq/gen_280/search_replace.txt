def compute_autocorrelation_variance(sequence: List[float]) -> float:
    """Compute the variance of the autocorrelation profile to encourage structural consistency."""
    if not sequence or sum(sequence) < 0.01:
        return float('inf')

    n = len(sequence)

    # Compute the full autocorrelation (not just the max)
    try:
        # Use FFT-based convolution for efficiency
        conv = np.convolve(sequence, sequence, mode='full')
        # Only take the first n elements for the autocorrelation profile
        autocorr_profile = conv[:n]

        # Compute variance of the autocorrelation profile
        variance = np.var(autocorr_profile)
        return variance
    except Exception:
        return float('inf')

def evaluate_fitness_single(sequence: List[float]) -> float:
    """Evaluate fitness for a single sequence with secondary regularization."""
    _, inv_c1 = compute_autocorrelation_constant(sequence)
    variance = compute_autocorrelation_variance(sequence)

    # Apply penalty based on autocorrelation variance to encourage structural consistency
    # The penalty is proportional to the variance, scaled by a factor
    penalty_factor = 0.001
    regularized_fitness = inv_c1 - penalty_factor * variance

    # Ensure we don't return negative fitness
    return max(0.0, regularized_fitness)
