These modifications aim to expand the search space exploration and refine solutions more effectively, potentially leading to better upper bounds on C₁.

def generate_pattern_based_sequence(n: int) -> List[float]:
    """Generate a sequence with a custom pattern to encourage structure."""
    # Create a combination of peaks and valleys
    sequence = [0.0] * n
    num_peaks = max(2, min(10, n // 50))
    peak_positions = sorted(random.sample(range(n), num_peaks))

    # Assign increasing heights to peaks
    peak_heights = [random.uniform(10, 100) for _ in range(num_peaks)]
    for i, (pos, height) in enumerate(zip(peak_positions, peak_heights)):
        # Set the area around the peak to the height
        radius = max(1, n // 20)
        start = max(0, pos - radius)
        end = min(n, pos + radius)
        for j in range(start, end):
            sequence[j] = height
    return sequence

def generate_peak_sequence(n: int, num_peaks: int = None) -> List[float]:
    """Generate a sequence with specified peaks to encourage structured exploration."""
    if num_peaks is None:
        num_peaks = max(3, min(15, n // 20))
    peak_positions = sorted(random.sample(range(n), num_peaks))
    peak_heights = [random.uniform(50.0, 150.0) for _ in range(num_peaks)]

    sequence = [0.0] * n
    for i, (pos, height) in enumerate(zip(peak_positions, peak_heights)):
        # Distribute height across a wider range for more structure
        radius = max(1, n // 30)
        start = max(0, pos - radius)
        end = min(n, pos + radius)
        for j in range(start, end):
            # Apply smooth tapering for peak shapes
            distance = abs(j - pos)
            taper_factor = max(0, 1 - distance / radius)
            sequence[j] += height * taper_factor
    return sequence

def local_refinement(sequence: List[float], iterations: int = 10, learning_rate: float = 0.01) -> List[float]:
    """Perform enhanced local refinement using L-BFGS for better convergence."""
    try:
        # Convert to numpy array for vectorized operations
        refined = np.array(sequence, dtype=float)
        n = len(refined)

        # Define objective function: negative of 1/C1 (since we maximize 1/C1)
        def objective(x):
            _, inv_c1 = compute_autocorrelation_constant(x.tolist())
            return -inv_c1 if inv_c1 > 0 else 1e10

        # Define gradient function using finite differences
        def gradient(x):
            eps = 1e-6
            grad = np.zeros_like(x)
            base_obj = objective(x)
            for i in range(len(x)):
                perturbed = x.copy()
                perturbed[i] += eps
                perturbed = np.maximum(perturbed, 0.01)
                perturbed_obj = objective(perturbed)
                grad[i] = (perturbed_obj - base_obj) / eps
            return grad

        # Perform L-BFGS optimization
        from scipy.optimize import minimize
        result = minimize(
            objective,
            refined,
            method='L-BFGS-B',
            jac=gradient,
            bounds=[(0.01, None)] * n,
            options={'maxiter': iterations}
        )

        if result.success:
            refined = result.x
        else:
            # Fallback to gradient descent if L-BFGS fails
            for _ in range(iterations):
                grad = gradient(refined)
                refined += learning_rate * grad
                refined = np.maximum(refined, 0.01)

        return refined.tolist()
    except Exception:
        # Fallback to simple gradient descent if any issue occurs
