random.seed(42)
np.random.seed(42)

class AutocorrelationEvaluator:
    """Handles evaluation of sequences for autocorrelation properties."""
    
    def __init__(self):
        self.cache = {}
        self.eval_count = 0
    
    def evaluate_sequence(self, sequence: List[float]) -> Tuple[float, float, float, float]:
        """Evaluate sequence and return performance metrics."""
        seq_tuple = tuple(sequence)
        if seq_tuple in self.cache:
            return self.cache[seq_tuple]

        try:
            a = np.array(sequence, dtype=np.float64)
            sum_a = np.sum(a)

            if sum_a < 1e-10:
                result = (float('inf'), 0.0, 0.0, sum_a)
                self.cache[seq_tuple] = result
                self.eval_count += 1
                return result

            b = fftconvolve(a, a, mode='full')
            b = b[len(a)-1:2*len(a)-1]

            max_b = np.max(b)
            n = len(a)
            c1 = 2 * n * max_b / (sum_a ** 2)

            inv_c1 = 1.0 / c1 if c1 > 0 else 0.0
            
            result = (c1, inv_c1, max_b, sum_a)
            self.cache[seq_tuple] = result
            self.eval_count += 1
            return result
        except Exception as e:
            warnings.warn(f"Evaluation error: {e}")
            result = (float('inf'), 0.0, 0.0, 0.0)
            self.cache[seq_tuple] = result
            self.eval_count += 1
            return result

    def evaluate_population(self, population: List[List[float]]) -> List[float]:
        """Evaluate a list of sequences."""
        return [self.evaluate_sequence(seq)[1] for seq in population]

class SequenceGenerator:
    """Generates diverse initial sequences for optimization."""
    
    @staticmethod
    def generate_multiscale_sequence() -> List[float]:
        """Generate structured sequence using multi-scale pattern."""
        length = random.choice([50, 100, 150, 200, 250, 300, 350, 400, 450, 500])
        sequence = [abs(np.sin(i * np.pi / length)) * 500 for i in range(length)]
        
        # Add noise for exploration
        for i in range(length):
            if random.random() < 0.2:
                sequence[i] += random.uniform(-100, 100)
                
        # Ensure non-negative
        sequence = [max(0, x) for x in sequence]
        
        # Ensure meaningful sum
        if sum(sequence) < 0.01:
            sequence[random.randint(0, length-1)] += 0.01
            
        return sequence

    @staticmethod
    def generate_structured_sequence(length: int = 100) -> List[float]:
        """Generate a structured sequence for better initialization."""
        # Create base pattern (e.g., exponential decay)
        sequence = [max(0.01, 100 * np.exp(-i * 0.05)) for i in range(length)]
        
        # Add random noise
        noise = [random.uniform(0.9, 1.1) for _ in range(length)]
        sequence = [base * noise[i] for i, base in enumerate(sequence)]
        
        # Ensure sum is meaningful
        if sum(sequence) < 0.01:
            sequence[random.randint(0, length-1)] += 0.01
            
        return sequence

class GradientRefiner:
    """Provides gradient-based refinement to improve sequences."""
    
    @staticmethod
    def refine(sequence: List[float], steps: int = 50, lr: float = 1e-4) -> List[float]:
        """Apply gradient-based refinement to improve sequence."""
        seq = np.array(sequence, dtype=float)
        n = len(seq)
        
        # Normalize
        sum_seq = np.sum(seq)
        if sum_seq < 1e-10:
            seq += 1e-5
            sum_seq = np.sum(seq)
        seq /= sum_seq

        prev_max_conv = float('inf')

        for step in range(steps):
            # Compute convolution
            conv = fftconvolve(seq, seq, mode='full')
            conv = conv[n-1:2*n-1]
            max_conv = np.max(conv)

            # Early stopping
            if abs(prev_max_conv - max_conv) < 1e-10:
                break
            prev_max_conv = max_conv

            # Gradient estimation using central differences
            grad = np.zeros_like(seq)
            epsilon = 1e-6
            for i in range(n):
                # Perturb forward and backward
                seq_forward = seq.copy()
                seq_forward[i] += epsilon
                seq_forward /= np.sum(seq_forward)

                seq_backward = seq.copy()
                seq_backward[i] -= epsilon
                seq_backward /= np.sum(seq_backward)

                # Convolution for perturbed sequences
                conv_forward = fftconvolve(seq_forward, seq_forward, mode='full')
                conv_forward = conv_forward[n-1:2*n-1]
                max_forward = np.max(conv_forward)

                conv_backward = fftconvolve(seq_backward, seq_backward, mode='full')
                conv_backward = conv_backward[n-1:2*n-1]
                max_backward = np.max(conv_backward)

                # Central difference gradient estimate
                grad[i] = (max_forward - max_backward) / (2 * epsilon)

            # Adaptive learning rate
            adaptive_lr = lr * (1.0 - 0.9 * (step / steps))

            # Update with gradient ascent
            seq += adaptive_lr * grad
            seq = np.maximum(seq, 0)  # Ensure non-negative

            # Renormalize
            sum_seq = np.sum(seq)
            if sum_seq > 0:
                seq /= sum_seq

        return seq.tolist()

class PopulationManager:
    """Manages populations and evolutionary operations."""
    
    @staticmethod
    def tournament_selection(population: List[List[float]],
                           fitnesses: List[float],
                           tournament_size: int = 3) -> List[List[float]]:
        """Select individuals using tournament selection."""
        selected = []
        for _ in range(len(population)):
            tournament_indices = random.sample(range(len(population)), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitnesses)]
            selected.append(copy.deepcopy(population[winner_index]))
        return selected

    @staticmethod
    def crossover_sequences(seq1: List[float], seq2: List[float], cxpb: float = 0.5) -> Tuple[List[float], List[float]]:
        """Perform uniform crossover."""
        child1, child2 = copy.deepcopy(seq1), copy.deepcopy(seq2)
        for i in range(len(child1)):
            if random.random() < cxpb:
                child1[i], child2[i] = child2[i], child1[i]
        return child1, child2

    @staticmethod
    def mutate_sequence(sequence: List[float], mu: float = 0, sigma: float = 100, indpb: float = 0.1) -> List[float]:
        """Mutate sequence with Gaussian noise."""
        mutated = copy.deepcopy(sequence)
        for i in range(len(mutated)):
            if random.random() < indpb:
                mutated[i] += random.gauss(mu, sigma)
                mutated[i] = max(0, mutated[i])
        return mutated

class AutocorrelationOptimizer:
    """Main optimizer class orchestrating the search process."""
    
    def __init__(self, pop_size: int = 50, generations: int = 50):
        self.pop_size = pop_size
        self.generations = generations
        self.best_score = 0.0
        self.best_sequence = None
        self.evaluator = AutocorrelationEvaluator()
        self.generator = SequenceGenerator()
        self.refiner = GradientRefiner()
        self.manager = PopulationManager()
        self.timeout = 170  # seconds

    def optimize(self) -> List[float]:
        """Run the optimization process."""
        start_time = time.time()
        
        # Initialize population
        population = [self.generator.generate_multiscale_sequence() for _ in range(self.pop_size)]
        
        # Initial evaluation
        fitnesses = self.evaluator.evaluate_population(population)
        
        # Track best
        best_idx = np.argmax(fitnesses)
        self.best_score = fitnesses[best_idx]
        self.best_sequence = copy.deepcopy(population[best_idx])
        
        print(f"Initial best score: {self.best_score:.6f}")
        
        # History tracking
        fitness_history = [self.best_score]
        stagnation_counter = 0
        
        # Evolution loop
        for gen in range(self.generations):
            if time.time() - start_time > self.timeout:
                break
                
            # Selection
            selected = self.manager.tournament_selection(population, fitnesses)
            
            # Create offspring
            offspring = []
            for i in range(0, len(selected), 2):
                if i + 1 < len(selected):
                    child1, child2 = self.manager.crossover_sequences(selected[i], selected[i+1])
                    child1 = self.manager.mutate_sequence(child1)
                    child2 = self.manager.mutate_sequence(child2)
                    offspring.extend([child1, child2])
                else:
                    offspring.append(self.manager.mutate_sequence(selected[i]))
                    
            offspring = offspring[:self.pop_size]
            
            # Refinement
            top_indices = np.argsort(fitnesses)[-min(10, len(offspring)):][::-1]
            refine_count = max(1, int(0.3 * len(top_indices)))
            for i in range(refine_count):
                idx = top_indices[i]
                offspring[idx] = self.refiner.refine(offspring[idx])

            # Evaluate offspring
            offspring_fitnesses = self.evaluator.evaluate_population(offspring)
            
            # Replace population
            population = offspring
            fitnesses = offspring_fitnesses
            
            # Update best
            best_idx = np.argmax(fitnesses)
            if fitnesses[best_idx] > self.best_score:
                self.best_score = fitnesses[best_idx]
                self.best_sequence = copy.deepcopy(population[best_idx])
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                
            # Track history
            fitness_history.append(self.best_score)
            if len(fitness_history) > 10:
                fitness_history.pop(0)
                
            print(f"Gen {gen}: Best = {self.best_score:.6f}")

            # Termination checks
            if gen > 10:
                recent_improvement = max(fitness_history) - min(fitness_history)
                if recent_improvement < 1e-8:
                    print("Fitness plateau detected, stopping early")
                    break
                    
                if stagnation_counter > 5:
                    print("Stagnation detected, stopping early")
                    break
                    
                if time.time() - start_time > self.timeout - 5:
                    print("Approaching time limit, stopping early")
                    break

        return self.best_sequence

def search_for_best_sequence() -> List[float]:
    """Main function to search for the best coefficient sequence."""
    optimizer = AutocorrelationOptimizer(pop_size=50, generations=50)
    try:
        best_sequence = optimizer.optimize()
        return best_sequence
    except Exception as e:
        print(f"Optimization failed: {e}")
        return [1.0] * 100
