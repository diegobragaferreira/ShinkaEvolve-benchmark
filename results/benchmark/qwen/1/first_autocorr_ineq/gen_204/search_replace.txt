def multi_start_lbfgs_optimization(sequence: list[float], max_iter: int = 50) -> list[float]:
    """Multi-start L-BFGS optimization to refine a sequence."""
    best_sequence = sequence.copy()
    best_inv_c1 = 0.0

    # Try multiple starting points
    for _ in range(5):  # 5 different start points
        # Slightly perturb the sequence
        perturbed = [x * random.uniform(0.9, 1.1) for x in sequence]
        # Ensure all values are valid
        perturbed = [max(0.01, x) for x in perturbed]

        try:
            # Use scipy.optimize.minimize with L-BFGS
            def objective(x):
                # Compute 1/C1 (we want to maximize this, so minimize -1/C1)
                _, inv_c1 = compute_autocorrelation_constant(x)
                return -inv_c1

            # Constraints: all values must be non-negative
            constraints = [{'type': 'ineq', 'fun': lambda x: x}]

            # Bounds: ensure all values remain reasonable
            bounds = [(0.01, 1000.0) for _ in range(len(perturbed))]

            result = optimize.minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds,
                                     constraints=constraints, options={'maxiter': max_iter})

            if result.success:
                _, inv_c1 = compute_autocorrelation_constant(result.x)
                if inv_c1 > best_inv_c1:
                    best_inv_c1 = inv_c1
                    best_sequence = result.x.tolist()
        except Exception:
            continue  # Skip if optimization fails

    return best_sequence
