def local_refinement(sequence: List[float], iterations: int = 3, learning_rate: float = 0.01) -> List[float]:
    """Perform local refinement using gradient-based updates."""
    refined = np.array(sequence, dtype=float)
    for _ in range(iterations):
        # Compute finite difference gradient
        grad = np.zeros_like(refined)
        epsilon = 1e-6
        base_inv_c1 = compute_autocorrelation_constant(refined.tolist())[1]

        for i in range(len(refined)):
            perturbed = refined.copy()
            perturbed[i] += epsilon
            perturbed = np.maximum(perturbed, 0.01)  # Keep non-negative
            perturbed_inv_c1 = compute_autocorrelation_constant(perturbed.tolist())[1]
            grad[i] = (perturbed_inv_c1 - base_inv_c1) / epsilon

        # Update with gradient ascent (since we're maximizing)
        refined += learning_rate * grad
        refined = np.maximum(refined, 0.01)  # Keep non-negative

    return refined.tolist()
