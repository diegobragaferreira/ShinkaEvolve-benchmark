def evaluate_fitness(sequence: List[float]) -> tuple[float, float]:
    """
    Evaluates the fitness of a sequence by computing C₁.
    Returns (inverse_C1, C1) tuple.
    """
    if len(sequence) == 0:
        return 0.0, float('inf')

    a = np.array(sequence, dtype=np.float64)
    sum_a = np.sum(a)

    # Avoid division by zero
    if sum_a < 1e-10:
        return 0.0, float('inf')

    # Compute autoconvolution using more efficient FFT approach
    n = len(sequence)
    padded_len = 2 * n - 1
    padded_seq = np.pad(a, (0, padded_len - n), 'constant')
    fft_seq = fft(padded_seq, axis=0)
    conv_result = ifft(fft_seq * np.conj(fft_seq), axis=0)
    b = np.real(conv_result[:padded_len])

    max_b = np.max(b)

    # Compute C1
    c1 = (2 * n * max_b) / (sum_a ** 2)

    # Return inverse for maximization
    inv_c1 = 1.0 / c1 if c1 > 0 else 0.0

    return inv_c1, c1
