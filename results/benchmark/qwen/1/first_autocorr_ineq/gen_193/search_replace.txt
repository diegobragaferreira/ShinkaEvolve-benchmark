    @staticmethod
    def mutate_sequence_adaptive(sequence: List[float], generation: int,
                                diversity: float, base_scale: float = 100.0) -> List[float]:
        """Apply adaptive mutation based on generation and population diversity."""
        mutated = copy.deepcopy(sequence)
        n = len(mutated)

        # Dynamic scale adjustment based on generation
        # More exploration in early generations, less in later ones
        generation_factor = max(0.1, 1.0 - generation / 100.0)

        # Diversity-based adjustment
        # High diversity = more exploration, low diversity = more exploitation
        diversity_factor = min(1.0, diversity / 0.1) if diversity > 0 else 1.0

        # Combined scale factor
        scale = base_scale * generation_factor * diversity_factor

        for i in range(n):
            # Use mixture of Cauchy and Gaussian for adaptive behavior
            # Early stages: prefer Cauchy for heavy-tailed exploration
            # Later stages: prefer Gaussian for fine-tuning
            if generation < 20 or np.random.random() < 0.3:
                # Heavy-tailed Cauchy mutation for exploration
                mutated[i] += np.random.standard_cauchy() * scale
            else:
                # Gaussian mutation for exploitation
                mutated[i] += np.random.normal(0, scale / 10.0)

            mutated[i] = max(0, mutated[i])  # Ensure non-negative
        return mutated

    def gradient_refine(sequence: List[float], steps: int = 50, lr: float = 1e-4) -> List[float]:
        """Apply adaptive gradient-based refinement to improve the sequence."""
        seq = np.array(sequence, dtype=float)
        n = len(seq)
        # Normalize to avoid numerical issues
        sum_seq = np.sum(seq)
        if sum_seq < 1e-10:
            seq += 1e-5
            sum_seq = np.sum(seq)
        seq /= sum_seq

        # Initialize momentum for smoother updates
        momentum = np.zeros_like(seq)
        beta = 0.9  # Momentum factor

        prev_max_conv = float('inf')
        improvement_threshold = 1e-8
        no_improvement_count = 0

            conv = fftconvolve(seq, seq, mode='full')

            if abs(prev_max_conv - max_conv) < improvement_threshold:
                no_improvement_count += 1
                if no_improvement_count > 3:
                    break
            else:
                no_improvement_count = 0
            prev_max_conv = max_conv

                seq_forward[i] += epsilon

                seq_backward[i] -= epsilon
