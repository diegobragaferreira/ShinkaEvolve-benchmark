class LocalSearchStrategy:
    """Handles local search strategies including gradient-based improvements."""

    @staticmethod
    def gradient_ascent_step(sequence: List[float], learning_rate: float = 0.1,
                           epsilon: float = 1e-4) -> List[float] | None:
        """Perform a single gradient ascent step with adaptive learning rate."""
        n = len(sequence)
        sum_sequence = np.sum(sequence)

        # Avoid division by zero
        if sum_sequence < 1e-10:
            return None

        try:
            base_fitness = 1.0 / _evaluator.compute(sequence)[0] if _evaluator.compute(sequence)[0] != float('inf') else 0.0

            # Compute gradient per dimension using central differences for better accuracy
            grad = np.zeros(n)
            for i in range(n):
                # Perturb forward and backward
                perturbed_forward = sequence[:]
                perturbed_backward = sequence[:]

                perturbed_forward[i] = max(0.01, perturbed_forward[i] + epsilon)
                perturbed_backward[i] = max(0.01, perturbed_backward[i] - epsilon)

                forward_fitness = 1.0 / _evaluator.compute(perturbed_forward)[0] if _evaluator.compute(perturbed_forward)[0] != float('inf') else 0.0
                backward_fitness = 1.0 / _evaluator.compute(perturbed_backward)[0] if _evaluator.compute(perturbed_backward)[0] != float('inf') else 0.0

                grad[i] = (forward_fitness - backward_fitness) / (2 * epsilon)

            # Adaptive learning rate based on gradient magnitude
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                adaptive_lr = learning_rate / (1.0 + grad_norm)
            else:
                adaptive_lr = learning_rate

            # Update in the direction of gradient ascent
            updated = [max(0.01, sequence[i] + adaptive_lr * grad[i]) for i in range(n)]

            return updated
        except Exception as e:
            print(f"Gradient ascent failed: {e}")
            return None

    @staticmethod
    def multi_start_gradient_ascent(sequence: List[float], max_starts: int = 10,

        # Store the best performing sequence

            for iter_idx in range(max_iterations):

        return best_performing[0]
