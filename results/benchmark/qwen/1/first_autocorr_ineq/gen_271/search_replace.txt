def compute_c1(sequence):
    """Compute C₁ for a given sequence (needed for gradient calculation)."""
    if len(sequence) == 0:
        return float('inf')
    a = np.array(sequence)
    conv = fftconvolve(a, a, mode='full')
    conv = conv[len(a)-1:]  # Take the relevant part
    max_conv = np.max(conv)
    sum_sq = np.sum(a)**2
    if sum_sq == 0:
        return float('inf')
    return 2 * len(a) * max_conv / sum_sq
