def get_good_direction_to_move_into(
    sequence: list[float],
) -> list[float] | None:
    """Returns the direction to move into the sequence using enhanced gradient-based refinement."""
    n = len(sequence)
    sum_sequence = np.sum(sequence)

    # Avoid division by zero
    if sum_sequence < 1e-10:
        return None

    # Normalize the sequence to avoid numerical issues
    normalized_sequence = np.array(sequence) * np.sqrt(2 * n) / sum_sequence

    # Enhanced gradient refinement using second-order finite differences
    steps = 100  # Increased steps for better convergence
    lr = 1e-4   # Initial learning rate
    epsilon = 1e-6  # Step size for finite differences
    prev_max_conv = float('inf')

    # Momentum term for accelerated convergence
    velocity = np.zeros_like(normalized_sequence)
    beta = 0.9  # Momentum coefficient

    for step in range(steps):
        # Compute convolution
        conv = fftconvolve(normalized_sequence, normalized_sequence, mode='full')
        conv = conv[n-1:2*n-1]
        max_conv = np.max(conv)

        # Early stopping condition
        if abs(prev_max_conv - max_conv) < 1e-12:
            break
        prev_max_conv = max_conv

        # Compute gradient estimate using second-order central differences
        grad = np.zeros_like(normalized_sequence)
        for i in range(n):
            # Forward perturbation
            seq_forward = normalized_sequence.copy()
            seq_forward[i] += epsilon
            seq_forward /= np.sum(seq_forward)

            # Backward perturbation
            seq_backward = normalized_sequence.copy()
            seq_backward[i] -= epsilon
            seq_backward /= np.sum(seq_backward)

            # Compute max convolution values
            conv_forward = fftconvolve(seq_forward, seq_forward, mode='full')
            conv_forward = conv_forward[n-1:2*n-1]
            max_forward = np.max(conv_forward)

            conv_backward = fftconvolve(seq_backward, seq_backward, mode='full')
            conv_backward = conv_backward[n-1:2*n-1]
            max_backward = np.max(conv_backward)

            # Second-order central difference gradient estimate
            grad[i] = (max_forward - max_backward) / (2 * epsilon)

        # Update with momentum for accelerated convergence
        velocity = beta * velocity + (1 - beta) * grad

        # Adaptive learning rate
        adaptive_lr = lr * (1.0 - 0.95 * (step / steps))

        # Gradient ascent update
        normalized_sequence += adaptive_lr * velocity
        normalized_sequence = np.maximum(normalized_sequence, 0)  # Ensure non-negativity

        # Renormalize
        sum_seq = np.sum(normalized_sequence)
        if sum_seq > 0:
            normalized_sequence /= sum_seq

    # Convert back to list and clip values
    refined_sequence = normalized_sequence.tolist()
    refined_sequence = [max(0, min(x, 1000)) for x in refined_sequence]
    return refined_sequence
