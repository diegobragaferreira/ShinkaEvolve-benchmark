def refine_solution(sequence, iterations=5):
    """Apply gradient-based refinement to improve the solution."""
    # Simple gradient ascent with small steps
    for _ in range(iterations):
        # Compute current score
        current_score = compute_inv_c1(sequence)

        # Estimate gradient via finite differences
        epsilon = 1e-4
        gradient = []

        for i in range(len(sequence)):
            # Perturb sequence at index i
            perturbed_sequence = sequence[:]
            perturbed_sequence[i] += epsilon

            # Compute new score
            new_score = compute_inv_c1(perturbed_sequence)

            # Estimate gradient
            grad = (new_score - current_score) / epsilon
            gradient.append(grad)

        # Update sequence using gradient ascent
        learning_rate = 0.01
        updated_sequence = [
            max(0, seq_val + learning_rate * grad_val)
            for seq_val, grad_val in zip(sequence, gradient)
        ]

        # Clamp values to [0, 1000]
        updated_sequence = [min(1000, max(0, val)) for val in updated_sequence]

        sequence = updated_sequence

    return sequence

def search_for_best_sequence():
    """Function to search for the best coefficient sequence using improved evolutionary approach."""
    # Set seed for reproducibility
    random.seed(42)
    np.random.seed(42)

    best_sequence, best_score = adaptive_evolutionary_optimization(
