The changes include:

2. Improving the gradient estimation and local refinement approach

def solve_convolution_lp(f_sequence, rhs):
    """Solves the convolution LP for a given sequence and RHS."""
    n = len(f_sequence)
    if n == 0:
        return None

    c = -np.ones(n)
    a_ub = []
    b_ub = []

    # Build constraint matrix efficiently
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Add non-negativity constraints
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # Add bounds to avoid numerical issues
    bounds = [(0, 1000) for _ in range(n)]  # Clips heights to [0, 1000]

        if result.success:
            g_sequence = result.x
            return g_sequence
        else:
            return None
    except Exception:
        return None

def solve_convolution_lp(f_sequence, rhs):
    """Solves the convolution LP for a given sequence and RHS with early termination."""
    n = len(f_sequence)
    if n == 0:
        return None

    c = -np.ones(n)
    a_ub = []
    b_ub = []

    # Optimized constraint building with early termination for large sequences
    if n > 1000:
        # For large sequences, limit constraints to key indices
        constraint_indices = list(range(0, 2*n-1, max(1, (2*n-1)//50)))  # Sample every ~50th index
        for k in constraint_indices:
            row = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    row[j] = f_sequence[i]
            a_ub.append(row)
            b_ub.append(rhs)
    else:
        # For smaller sequences, include all convolution constraints
        for k in range(2 * n - 1):
            row = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    row[j] = f_sequence[i]
            a_ub.append(row)
            b_ub.append(rhs)

    # Add non-negativity constraints
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # Add bounds to avoid numerical issues
    bounds = [(0, 1000) for _ in range(n)]  # Clips heights to [0, 1000]
