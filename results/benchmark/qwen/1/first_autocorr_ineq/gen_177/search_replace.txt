        length = random.randint(100, 1000)
    if num_steps is None:
        num_steps = max(2, min(50, length // 20))  # Adaptive number of steps

    # Place steps at regular intervals to distribute mass evenly
    step_positions = [i * length // num_steps for i in range(num_steps)] + [length]

    # Assign heights that decrease exponentially to reduce convolution peaks
    step_heights = [1000 * np.exp(-i/5) for i in range(num_steps)]

    # Ensure all heights are positive and normalize
    step_heights = [max(1.0, h) for h in step_heights]
    total_height = sum(step_heights)

    return sequence
