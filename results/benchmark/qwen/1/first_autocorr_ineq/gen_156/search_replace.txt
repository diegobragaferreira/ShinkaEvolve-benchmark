    # Set up evolutionary algorithm parameters
    initial_pop_size = 50
    n_generations = 200

    # Initialize population
    population = toolbox.population(initial_pop_size)

    # Track fitness history for dynamic population adjustment
    fitness_history = []
    stagnation_count = 0
    max_stagnation = 10

    # Evolution loop
    for gen in range(n_generations):
        # Select next generation
        offspring = toolbox.select(population, len(population))
        offspring = list(map(toolbox.clone, offspring))

        for child1, child2 in zip(offspring[::2], offspring[1::2]):

        fitnesses = map(toolbox.evaluate, invalid_ind)
        for ind, fit in zip(invalid_ind, fitnesses):
            ind.fitness.values = fit

        # Replace population
        population[:] = offspring

        # Dynamic population size adjustment based on performance
        current_fitnesses = [ind.fitness.values[0] for ind in population if ind.fitness.valid]
        if len(current_fitnesses) > 0:
            avg_fitness = np.mean(current_fitnesses)
            fitness_history.append(avg_fitness)

            # Check for stagnation
            if len(fitness_history) >= 2:
                recent_change = abs(fitness_history[-1] - fitness_history[-2])
                if recent_change < 1e-5:
                    stagnation_count += 1
                else:
                    stagnation_count = 0

            # Adjust population size dynamically
            if stagnation_count > max_stagnation:
                # Increase diversity by increasing population size
                new_pop_size = min(100, len(population) + 10)
                if new_pop_size > len(population):
                    # Add new random individuals to maintain diversity
                    additional_individuals = toolbox.population(new_pop_size - len(population))
                    population.extend(additional_individuals)
                    stagnation_count = 0  # Reset stagnation counter
            elif len(fitness_history) > 5 and np.std(fitness_history[-5:]) < 1e-3:
                # Reduce population size if converging too quickly
                new_pop_size = max(20, len(population) - 5)
                if new_pop_size < len(population):
                    # Keep individuals with highest fitness
                    fitness_scores = [(ind, ind.fitness.values[0]) for ind in population if ind.fitness.valid]
                    sorted_fitness = sorted(fitness_scores, key=lambda x: x[1], reverse=True)
                    kept_individuals = [ind for ind, _ in sorted_fitness[:new_pop_size]]
                    population[:] = kept_individuals

    best_individual = tools.selBest(population, 1)[0]
    return list(best_individual)
