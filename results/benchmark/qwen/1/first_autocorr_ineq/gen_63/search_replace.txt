    @staticmethod
    def compute_adaptive_mutation_rate(diversity: float, generation: int,
                                     max_generations: int) -> float:
        """
        Compute adaptive mutation rate based on population diversity and generation.

        Args:
            diversity: Standard deviation of fitness values
            generation: Current generation number
            max_generations: Total number of generations

        Returns:
            float: Adaptive mutation probability
        """
        # Base mutation rate
        base_mutation_rate = 0.1

        # Adjust based on diversity (higher diversity = higher mutation rate)
        diversity_factor = 1.0 if diversity > 1e-3 else 0.5

        # Adjust based on generation (later generations = lower mutation rate)
        progress = generation / max_generations
        generation_factor = 1.0 - 0.7 * progress  # Decrease mutation rate over time

        # Combined adaptive rate
        adaptive_rate = base_mutation_rate * diversity_factor * generation_factor
        return max(0.01, min(0.5, adaptive_rate))  # Clamp between 0.01 and 0.5
