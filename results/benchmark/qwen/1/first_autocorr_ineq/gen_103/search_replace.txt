def optimize_step_function(initial_sequence, max_iter=100):
    """Specialized optimizer for step function construction"""
    sequence = initial_sequence.copy()
    current_score = evaluate_sequence(sequence)

    # Adaptive step sizes
    step_sizes = [0.1, 0.05, 0.01]

    for iteration in range(max_iter):
        # Try different step sizes to find best improvement
        best_seq = sequence.copy()
        best_score = current_score

        for step_size in step_sizes:
            # Generate neighbor sequence with small perturbations
            neighbor = []
            for i, val in enumerate(sequence):
                # Randomly adjust with bounded noise
                noise = np.random.normal(0, step_size * val if val > 0 else 0.1)
                new_val = max(0, val + noise)
                # Apply hard bounds
                new_val = min(1000, new_val)
                neighbor.append(new_val)

            # Evaluate neighbor
            neighbor_score = evaluate_sequence(neighbor)
            if neighbor_score > best_score:
                best_score = neighbor_score
                best_seq = neighbor

        # Accept improvement
        if best_score > current_score:
            sequence = best_seq
            current_score = best_score
        else:
            # Reduce step size if no improvement
            step_sizes = [s * 0.8 for s in step_sizes]
            if all(s < 1e-6 for s in step_sizes):
                break

    return sequence
