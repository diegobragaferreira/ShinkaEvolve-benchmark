def multi_start_gradient_descent(initial_seq, max_iter=100, num_starts=5):
    """
    Perform multi-start gradient descent to refine a sequence.
    Uses multiple random starting points to escape local optima.
    """
    best_seq = initial_seq.copy()
    best_score = evaluate_sequence(best_seq)

    for start in range(num_starts):
        # Random perturbation of initial sequence
        current_seq = [x * (0.9 + 0.2 * np.random.random()) for x in initial_seq]
        current_score = evaluate_sequence(current_seq)

        for _ in range(max_iter):
            # Estimate gradient using finite differences
            grad = get_gradient_estimate(current_seq)
            if grad is None:
                break

            # Take a step in the negative gradient direction
            step_size = 0.01
            new_seq = [max(0, min(x - step_size * grad[i], 1000))
                      for i, x in enumerate(current_seq)]

            new_score = evaluate_sequence(new_seq)
            if new_score > current_score:
                current_seq = new_seq
                current_score = new_score
            else:
                # Reduce step size if no improvement
                step_size *= 0.5
                if step_size < 1e-6:
                    break

        if current_score > best_score:
            best_seq = current_seq
            best_score = current_score

    return best_seq
