def generate_adaptive_step_function(n: int) -> List[float]:
    """Generate an adaptive step function based on sequence length and optimization insights."""
    # Determine number of steps based on sequence length
    # For shorter sequences, use fewer steps; for longer sequences, use more
    num_steps = max(2, min(n // 10, 20))  # Between 2 and 20 steps

    # Create step positions evenly distributed
    step_positions = []
    step_width = n / num_steps
    for i in range(num_steps):
        start_pos = int(i * step_width)
        end_pos = int((i + 1) * step_width)
        # Ensure we don't go out of bounds
        if i == num_steps - 1:
            end_pos = n
        step_positions.append((start_pos, end_pos))

    # Generate step heights based on autocorrelation considerations
    # Higher heights at beginning, decreasing to reduce peak convolution
    step_heights = []
    for i in range(num_steps):
        # Exponential decay with some randomness for diversification
        base_height = 100 * np.exp(-i * 0.3)
        # Add randomness to avoid perfect symmetry
        noise_factor = random.uniform(0.8, 1.2)
        step_heights.append(max(0.01, base_height * noise_factor))

    # Construct the sequence
    sequence = [0.0] * n
    for i, (start_pos, end_pos) in enumerate(step_positions):
        height = step_heights[i]
        for j in range(start_pos, end_pos):
            sequence[j] = height

    return sequence

    return generate_adaptive_step_function(n)
