def is_valid_solution(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> bool:
    """
    Fast validity check using spatial indexing for collision detection.

    Args:
        circles: Array of shape (n, 3) with (x, y, radius) for each circle
        rect_width: Width of rectangle container
        rect_height: Height of rectangle container

    Returns:
        True if solution is valid, False otherwise
    """
    n = len(circles)

    # Check boundary constraints first
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            return False

    if n <= 1:
        return True

    # Use spatial distance matrix for quick check
    coords = circles[:, :2]
    radii = circles[:, 2]

    # Simple pairwise check for small numbers
    if n <= 10:
        for i in range(n):
            for j in range(i+1, n):
                distance = np.linalg.norm(coords[i] - coords[j])
                min_distance = radii[i] + radii[j]

                if distance < min_distance:
                    return False
        return True

    # For larger sets, use a more efficient approach
    # Compute distance matrix
    dist_matrix = cdist(coords, coords)

    # Check for overlaps
    for i in range(n):
        for j in range(i+1, n):
            if dist_matrix[i, j] < (radii[i] + radii[j]):
                return False

    return True

    Fast validity check using spatial indexing for collision detection.

    Args:
        circles: Array of shape (n, 3) with (x, y, radius) for each circle

    for i in range(n):

        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            return False

    if n <= 1:
        return True

    coords = circles[:, :2]
    radii = circles[:, 2]

    # Build KDTree for efficient neighbor queries
    try:
        tree = cKDTree(coords)

        # For each circle, find neighbors within sum of radii distance

                        return False

            for j in range(i+1, n):
                distance = np.linalg.norm(coords[i] - coords[j])
                min_distance = radii[i] + radii[j]

def calculate_fitness(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> Tuple[float, float]:
    """
    Calculate fitness of circle configuration with penalty for constraint violations.

    Args:
        circles: Array of shape (n, 3) with (x, y, radius) for each circle
        rect_width: Width of rectangle container
        rect_height: Height of rectangle container

    Returns:
        tuple: (fitness_value, overlap_penalty)
    """
    n = len(circles)

    # Check boundary constraints
    penalty = 0.0
    for i in range(n):
        x, y, r = circles[i]
        # Circle must be fully contained within rectangle
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            # Apply penalty based on how much it violates boundaries
            overlap = 0.0
            if x - r < 0:
                overlap += abs(x - r)
            if x + r > rect_width:
                overlap += abs(x + r - rect_width)
            if y - r < 0:
                overlap += abs(y - r)
            if y + r > rect_height:
                overlap += abs(y + r - rect_height)
            penalty += overlap * 1000

    # Check overlap constraints
    overlap_penalty = 0.0
    if n > 1:
        # Compute pairwise distances using optimized approach
        coords = circles[:, :2]
        radii = circles[:, 2]

        # Use more efficient distance calculation for many circles
        if n <= 10:  # Small case, direct check
            for i in range(n):
                for j in range(i+1, n):
                    distance = np.linalg.norm(coords[i] - coords[j])
                    min_distance = radii[i] + radii[j]

                    if distance < min_distance:
                        # Overlap exists
                        overlap_amount = min_distance - distance
                        overlap_penalty += overlap_amount * 1000  # Heavy penalty for overlaps
        else:  # Larger case, use distance matrix
            dist_matrix = cdist(coords, coords)
            for i in range(n):
                for j in range(i+1, n):
                    distance = dist_matrix[i, j]
                    min_distance = radii[i] + radii[j]

                    if distance < min_distance:
                        # Overlap exists
                        overlap_amount = min_distance - distance
                        overlap_penalty += overlap_amount * 1000  # Heavy penalty for overlaps

    # Fitness is sum of radii minus penalties
    total_radius = np.sum(circles[:, 2])
    fitness = total_radius - penalty - overlap_penalty

    return fitness, overlap_penalty
