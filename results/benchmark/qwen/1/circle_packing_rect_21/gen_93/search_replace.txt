The current implementation uses O(n²) distance calculations for collision detection in both the constraint checking and local optimization phases. This becomes increasingly expensive as the number of circles increases. The proposed improvement implements a spatial hashing approach to reduce collision detection complexity from O(n²) to O(n) by partitioning the space into grid cells and only checking collisions between circles in the same or adjacent cells. This will significantly speed up both the constraint validation and local optimization phases.

This replaces the O(n²) pairwise distance calculations with a grid-based approach that only checks collisions between circles in adjacent grid cells, reducing complexity to O(n). This is particularly beneficial for the constraint checking and repair solution phases in the evolutionary algorithm.

    def _get_grid_cell(self, x: float, y: float, cell_size: float) -> Tuple[int, int]:
        """Get grid cell coordinates for a given position."""
        return (int(x / cell_size), int(y / cell_size))

    def _get_adjacent_cells(self, cell: Tuple[int, int]) -> List[Tuple[int, int]]:
        """Get all adjacent cells including diagonals."""
        return [(cell[0]+dx, cell[1]+dy) for dx in [-1, 0, 1] for dy in [-1, 0, 1]]

    def _build_spatial_hash(self, circles: np.ndarray, cell_size: float) -> dict:
        """Build spatial hash grid for efficient collision detection."""
        grid = defaultdict(list)
        for i, (x, y, r) in enumerate(circles):
            cell = self._get_grid_cell(x, y, cell_size)
            grid[cell].append(i)
        return grid

    def repair_solution(self, circles: np.ndarray) -> np.ndarray:

        repaired = circles.copy()

        # Ensure positive radii
        repaired[:, 2] = np.maximum(repaired[:, 2], 0.001)

        # Enforce bounds
        for i in range(len(repaired)):
            x, y, r = repaired[i]
            x = np.clip(x, r, self.rect_width - r)
            y = np.clip(y, r, self.rect_height - r)
            repaired[i] = [x, y, r]

        # Resolve overlaps iteratively with better algorithm
        for iteration in range(50):
            # Calculate pairwise distances
            positions = repaired[:, :2]
            radii = repaired[:, 2]
            distances = cdist(positions, positions)

            conflicts = []
            for i in range(len(repaired)):
                for j in range(i+1, len(repaired)):
                    if distances[i, j] < (radii[i] + radii[j]):
                        conflicts.append((i, j))

            if not conflicts:
                break

            try:
                vor = Voronoi(positions)

                cell_areas = np.zeros(len(positions))
