    def calculate_density_constraints(circles_list: List[Tuple[float, float, float]],
                                    grid_resolution: float = 0.2) -> dict:
        """Calculate density-based constraint information using grid-based approach"""
        if len(circles_list) < 2:
            return {}

        # Create a grid covering the container
        grid_width = int(width / grid_resolution) + 1
        grid_height = int(height / grid_resolution) + 1

        # Count circles per grid cell
        grid_counts = defaultdict(int)
        grid_centers = {}  # Store center of each grid cell

        for x, y, r in circles_list:
            grid_x = int(x / grid_resolution)
            grid_y = int(y / grid_resolution)
            grid_counts[(grid_x, grid_y)] += 1
            if (grid_x, grid_y) not in grid_centers:
                grid_centers[(grid_x, grid_y)] = (
                    (grid_x + 0.5) * grid_resolution,
                    (grid_y + 0.5) * grid_resolution
                )

        # For each circle, compute density in surrounding cells
        density_info = {}
        for i, (x, y, r) in enumerate(circles_list):
            # Define neighborhood around the circle based on its radius
            neighborhood_radius = r * 2.0  # Consider 2x circle radius for neighbor consideration
            neighbors = 0

            # Check nearby grid cells
            grid_x = int(x / grid_resolution)
            grid_y = int(y / grid_resolution)

            # Check a 5x5 grid around this circle (centered at grid_x, grid_y)
            for dx in range(-2, 3):
                for dy in range(-2, 3):
                    nx, ny = grid_x + dx, grid_y + dy
                    if (nx, ny) in grid_counts:
                        # Check if the grid cell's center is in the neighborhood
                        gx, gy = grid_centers[(nx, ny)]
                        dist = np.sqrt((x - gx)**2 + (y - gy)**2)
                        if dist <= neighborhood_radius:
                            neighbors += grid_counts[(nx, ny)]

            # Normalize by the circle's own contribution (subtract 1 to exclude itself)
            effective_neighbors = max(0, neighbors - 1)
            # Higher density = higher constraint value
            density_info[i] = 1.0 + effective_neighbors * 0.5

        return density_info

        """Mutate circles with adaptive step sizes based on density"""
        mutated = circles.copy()
        mutation_rate = 0.2 + 0.1 * np.exp(-generation/50)  # Decrease over generations

        for i in range(len(mutated)):
            # Get constraint density for this circle (higher means tighter constraints)
            density = constraint_density.get(i, 1.0)

            step_size = 0.02 * np.exp(-density) * (0.5 + 0.5 * np.exp(-generation/100))

    def adaptive_mutation(circles: np.ndarray, constraint_density: dict, generation: int) -> np.ndarray:
        """Mutate circles with adaptive step sizes based on density"""
        mutated = circles.copy()
        mutation_rate = 0.2 + 0.1 * np.exp(-generation/50)  # Decrease over generations

        for i in range(len(mutated)):
            # Get constraint density for this circle (higher means tighter constraints)
            density = constraint_density.get(i, 1.0)

            # Adaptive step size based on density and generation
            # Lower density (sparse regions) allow larger steps
            # Higher density (crowded regions) require smaller steps
            base_step = 0.02
            density_factor = 1.0 / (1.0 + density * 0.5)
            step_size = base_step * density_factor * (0.5 + 0.5 * np.exp(-generation/100))

            if random.random() < mutation_rate:
                # Mutate position
                mutated[i, 0] += np.random.normal(0, step_size)
                mutated[i, 1] += np.random.normal(0, step_size)

                # Mutate radius with different behavior based on density
                if density > 2.0:  # High constraint area - be more conservative
                    radius_step = np.random.normal(0, step_size * 0.3)
                elif density < 1.0:  # Low constraint area - be more aggressive
                    radius_step = np.random.normal(0, step_size * 1.5)
                else:  # Medium constraint area
                    radius_step = np.random.normal(0, step_size * 0.8)

                mutated[i, 2] = max(0.001, mutated[i, 2] + radius_step)

                # Boundary checking and correction
                mutated[i, 0] = max(mutated[i, 2], min(width - mutated[i, 2], mutated[i, 0]))
                mutated[i, 1] = max(mutated[i, 2], min(height - mutated[i, 2], mutated[i, 1]))

                # Ensure radius remains feasible given position
                max_radius_x = min(mutated[i, 0], width - mutated[i, 0])
                max_radius_y = min(mutated[i, 1], height - mutated[i, 1])
                mutated[i, 2] = min(mutated[i, 2], max_radius_x, max_radius_y)

        return mutated
