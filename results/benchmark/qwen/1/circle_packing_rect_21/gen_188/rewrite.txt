import numpy as np
from scipy.spatial import Voronoi, distance_matrix
from scipy.spatial.distance import cdist
import random
import time
from typing import Tuple, List

# Set seed for reproducibility
np.random.seed(42)
random.seed(42)

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    
    # Rectangle dimensions: optimized width=1.5, height=0.5 (perimeter = 4)
    rect_width = 1.5
    rect_height = 0.5

    def compute_voronoi_areas(points: np.ndarray) -> np.ndarray:
        """Compute Voronoi cell areas for each point"""
        try:
            vor = Voronoi(points)
            areas = []
            for i in range(len(points)):
                region_idx = vor.point_region[i]
                if region_idx < len(vor.regions) and -1 not in vor.regions[region_idx]:
                    vertices = np.array([vor.vertices[j] for j in vor.regions[region_idx] if j >= 0])
                    if len(vertices) >= 3:
                        # Calculate area using shoelace formula
                        x = vertices[:, 0]
                        y = vertices[:, 1]
                        area = 0.5 * np.abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))
                        areas.append(area)
                    else:
                        areas.append(1.0)
                else:
                    areas.append(1.0)
            return np.array(areas)
        except:
            return np.ones(len(points))

    def compute_constraint_density(circles: np.ndarray) -> np.ndarray:
        """Compute constraint density based on neighbor distances"""
        n = len(circles)
        densities = np.zeros(n)
        
        if n <= 1:
            return densities
            
        # For each circle, count nearby neighbors
        positions = circles[:, :2]
        radii = circles[:, 2]
        
        # Compute distance matrix
        distances = cdist(positions, positions)
        
        # For each circle, count how many other circles are close enough to potentially conflict
        for i in range(n):
            nearby_count = 0
            for j in range(n):
                if i != j:
                    # Check if potentially overlapping (distance less than sum of radii)
                    if distances[i, j] < (radii[i] + radii[j]):
                        nearby_count += 1
            densities[i] = nearby_count
            
        return densities

    def create_hexagonal_initial_layout(width: float, height: float, n: int) -> np.ndarray:
        """Create initial hexagonal layout for better packing"""
        circles = np.zeros((n, 3))
        
        # Arrange in hexagonal grid
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        
        # Cell dimensions
        margin = 0.1
        cell_width = (width - 2 * margin) / cols
        cell_height = (height - 2 * margin) / rows
        cell_diag = min(cell_width, cell_height) * 0.8
        
        # Hexagonal spacing
        hex_radius = cell_diag * 0.5
        x_spacing = hex_radius * 2.0
        y_spacing = hex_radius * np.sqrt(3)
        
        placed = 0
        for row in range(rows):
            for col in range(cols):
                if placed >= n:
                    break
                    
                # Offset odd rows
                offset = (row % 2) * (x_spacing / 2)
                x = margin + col * x_spacing + offset
                y = margin + row * y_spacing + y_spacing / 2
                
                # Clamp to bounds
                x = np.clip(x, hex_radius, width - hex_radius)
                y = np.clip(y, hex_radius, height - hex_radius)
                
                # Adjust radius to stay away from boundaries
                max_radius = min(x, y, width - x, height - y) * 0.9
                r = min(hex_radius, max_radius)
                
                circles[placed] = [x, y, r]
                placed += 1
                
                if placed >= n:
                    break
                    
        return circles

    def apply_forces_and_refine(circles: np.ndarray, width: float, height: float, 
                               max_iterations: int = 100, learning_rate: float = 0.05) -> np.ndarray:
        """Apply Voronoi-based force optimization"""
        optimized = circles.copy()
        
        for iteration in range(max_iterations):
            # Compute Voronoi-based constraint information
            positions = optimized[:, :2]
            radii = optimized[:, 2]
            
            # Calculate Voronoi cell areas for constraint density estimation
            voronoi_areas = compute_voronoi_areas(positions)
            
            # Calculate constraint density (number of nearby circles)
            constraint_densities = compute_constraint_density(optimized)
            
            # Calculate forces
            forces = np.zeros_like(positions)
            
            # Compute all pairwise distances
            distances = cdist(positions, positions)
            
            # Apply repulsion forces for overlapping circles
            for i in range(len(optimized)):
                for j in range(i+1, len(optimized)):
                    dx = positions[j, 0] - positions[i, 0]
                    dy = positions[j, 1] - positions[i, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    
                    if dist > 1e-8 and dist < (radii[i] + radii[j]):
                        # Repulsion force (inverse square law)
                        force_magnitude = 1.0 / (dist * dist + 1e-8)
                        
                        # Weight by Voronoi area and constraint density
                        # Smaller Voronoi cells (more constrained) move less
                        # Higher constraint density = more important to resolve
                        weight = 0.5 + 0.5 * (1.0 / (1.0 + voronoi_areas[i] + voronoi_areas[j]))
                        weight *= (1.0 + constraint_densities[i] * 0.1)
                        
                        forces[i, 0] -= force_magnitude * dx / dist * weight * learning_rate
                        forces[i, 1] -= force_magnitude * dy / dist * weight * learning_rate
                        forces[j, 0] += force_magnitude * dx / dist * weight * learning_rate
                        forces[j, 1] += force_magnitude * dy / dist * weight * learning_rate
            
            # Apply forces with boundary constraints
            for i in range(len(optimized)):
                optimized[i, 0] += forces[i, 0]
                optimized[i, 1] += forces[i, 1]
                
                # Boundary constraints
                optimized[i, 0] = np.clip(optimized[i, 0], radii[i], width - radii[i])
                optimized[i, 1] = np.clip(optimized[i, 1], radii[i], height - radii[i])
        
        return optimized

    def refine_with_boundary_attention(circles: np.ndarray, width: float, height: float) -> np.ndarray:
        """Specialized refinement that focuses on boundary-constrained circles"""
        refined = circles.copy()
        positions = refined[:, :2]
        radii = refined[:, 2]
        
        # Identify boundary-constrained circles (close to edges)
        boundary_threshold = 0.1
        is_near_boundary = (
            (positions[:, 0] < boundary_threshold + radii) |
            (positions[:, 0] > width - boundary_threshold - radii) |
            (positions[:, 1] < boundary_threshold + radii) |
            (positions[:, 1] > height - boundary_threshold - radii)
        )
        
        # Apply focused refinement to boundary circles
        if np.any(is_near_boundary):
            # Create a more intensive refinement for boundary circles
            for _ in range(20):
                forces = np.zeros_like(positions)
                
                # Only compute forces for boundary circles with all others
                for i in range(len(refined)):
                    if is_near_boundary[i]:
                        for j in range(len(refined)):
                            if i != j:
                                dx = positions[j, 0] - positions[i, 0]
                                dy = positions[j, 1] - positions[i, 1]
                                dist = np.sqrt(dx*dx + dy*dy)
                                
                                if dist > 1e-8:
                                    # Apply stronger force to boundary circles
                                    force_magnitude = 1.0 / (dist * dist + 1e-8)
                                    weight = 2.0 if is_near_boundary[i] else 1.0
                                    
                                    forces[i, 0] -= force_magnitude * dx / dist * weight * 0.02
                                    forces[i, 1] -= force_magnitude * dy / dist * weight * 0.02
                
                # Apply boundary-constrained movement
                for i in range(len(refined)):
                    refined[i, 0] += forces[i, 0]
                    refined[i, 1] += forces[i, 1]
                    
                    # Ensure boundaries are respected
                    refined[i, 0] = np.clip(refined[i, 0], radii[i], width - radii[i])
                    refined[i, 1] = np.clip(refined[i, 1], radii[i], height - radii[i])
        
        return refined

    def smart_radius_optimization(circles: np.ndarray, width: float, height: float) -> np.ndarray:
        """Optimize radii by adjusting them based on Voronoi density"""
        optimized = circles.copy()
        positions = optimized[:, :2]
        radii = optimized[:, 2]
        
        # Compute Voronoi areas and constraint densities
        voronoi_areas = compute_voronoi_areas(positions)
        constraint_densities = compute_constraint_density(optimized)
        
        # Adjust radii based on local constraints
        for i in range(len(optimized)):
            # Larger Voronoi areas (more space) = opportunity to increase radius
            # Higher constraint densities = need to be more conservative
            area_factor = 1.0 / (1.0 + voronoi_areas[i])
            density_factor = 1.0 / (1.0 + constraint_densities[i] * 0.5)
            
            # Prefer to increase radius if space is available and not too constrained
            if area_factor > 0.7 and density_factor > 0.7:
                # Try to increase radius by small amount
                max_increase = 0.01
                current_radius = radii[i]
                # Don't allow too large increases that might violate constraints
                safe_increase = min(max_increase, current_radius * 0.1)
                optimized[i, 2] = current_radius + safe_increase * area_factor * density_factor
            
            # Ensure radii stay positive and reasonable
            optimized[i, 2] = max(0.001, optimized[i, 2])
            optimized[i, 2] = min(0.5, optimized[i, 2])
        
        return optimized

    def validate_solution(circles: np.ndarray, width: float, height: float) -> bool:
        """Validate that solution is physically feasible"""
        # Check boundary constraints
        for i in range(len(circles)):
            x, y, r = circles[i]
            if x - r < 0 or x + r > width or y - r < 0 or y + r > height:
                return False
                
        # Check overlap constraints
        positions = circles[:, :2]
        radii = circles[:, 2]
        distances = cdist(positions, positions)
        
        for i in range(len(circles)):
            for j in range(i+1, len(circles)):
                if distances[i, j] < (radii[i] + radii[j]):
                    return False
                    
        return True

    def objective_function(circles: np.ndarray) -> float:
        """Return sum of radii"""
        return np.sum(circles[:, 2])

    # Main optimization routine
    start_time = time.time()
    
    # Phase 1: Initial hexagonal layout
    circles = create_hexagonal_initial_layout(rect_width, rect_height, 21)
    
    # Phase 2: Multi-stage optimization
    # Stage 1: Basic force-based optimization
    circles = apply_forces_and_refine(circles, rect_width, rect_height, max_iterations=50)
    
    # Stage 2: Boundary attention refinement
    circles = refine_with_boundary_attention(circles, rect_width, rect_height)
    
    # Stage 3: Radius optimization guided by Voronoi structure
    circles = smart_radius_optimization(circles, rect_width, rect_height)
    
    # Stage 4: Final force optimization with stricter convergence
    circles = apply_forces_and_refine(circles, rect_width, rect_height, max_iterations=100, learning_rate=0.02)
    
    # Final validation and cleanup
    if not validate_solution(circles, rect_width, rect_height):
        # Fallback to hexagonal pattern if validation fails
        circles = create_hexagonal_initial_layout(rect_width, rect_height, 21)
        
        # Apply quick refinement if needed
        circles = apply_forces_and_refine(circles, rect_width, rect_height, max_iterations=30)
    
    # Final optimization pass
    circles = apply_forces_and_refine(circles, rect_width, rect_height, max_iterations=20, learning_rate=0.01)
    
    # Make sure all radii are positive and reasonable
    circles[:, 2] = np.maximum(circles[:, 2], 0.001)
    
    return circles
