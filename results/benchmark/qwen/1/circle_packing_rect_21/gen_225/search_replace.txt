    def refine_solution(self, circles: np.ndarray, iterations: int = 150) -> np.ndarray:
        """Apply local refinement to improve final solution with simulated annealing approach."""
        refined = circles.copy()

        # Simulated annealing-inspired refinement
        current_fitness, _ = self.calculate_fitness(refined)

        # Start with relatively high temperature for accepting worse solutions
        temperature = 0.1
        cooling_rate = 0.995

        # Track best solution found
        best_solution = refined.copy()
        best_fitness = current_fitness

        for iteration in range(iterations):
            # Reduce temperature gradually
            if iteration > 0 and iteration % 10 == 0:
                temperature *= cooling_rate

            # Try random perturbations on a random circle
            i = random.randint(0, len(refined) - 1)

            # Save current state
            old_x, old_y, old_r = refined[i]

            # Adapt mutation strength based on iteration and temperature
            pos_mutation_strength = 0.005 * (1.0 - iteration / iterations) + 0.001
            rad_mutation_strength = 0.001 * (1.0 - iteration / iterations) + 0.0001

            # Try random moves
            new_x = old_x + np.random.normal(0, pos_mutation_strength)
            new_y = old_y + np.random.normal(0, pos_mutation_strength)
            new_r = old_r + np.random.normal(0, rad_mutation_strength)

            # Clip to bounds
            new_x = np.clip(new_x, new_r, self.rect_width - new_r)
            new_y = np.clip(new_y, new_r, self.rect_height - new_r)
            new_r = max(0.001, new_r)

            # Test if this change improves fitness
            test_config = refined.copy()
            test_config[i] = [new_x, new_y, new_r]

            # Check if this move improves fitness or is accepted based on temperature
            test_fitness, _ = self.calculate_fitness(test_config)

            # Accept the move if it's better, or with probability based on temperature if worse
            if test_fitness > current_fitness:
                refined = test_config
                current_fitness = test_fitness
            else:
                # Probability of accepting worse move decreases with temperature
                if random.random() < np.exp((test_fitness - current_fitness) / max(temperature, 1e-10)):
                    refined = test_config
                    current_fitness = test_fitness

            # Update best solution found
            if current_fitness > best_fitness:
                best_fitness = current_fitness
                best_solution = refined.copy()

        return best_solution
