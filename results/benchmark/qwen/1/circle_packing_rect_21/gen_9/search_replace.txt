import numpy as np
import math


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: perimeter = 4, so width + height = 2
    # Let's use 1.2 x 0.8 as a reasonable aspect ratio
    rect_width = 1.2
    rect_height = 0.8

    # Initial placement in a hexagonal-like pattern
    circles = np.zeros((21, 3))

    # Try a structured approach
    # Create a grid pattern with slight offset for better packing
    rows = 4
    cols = 6

    # Place circles in a structured way with some randomness for variety
    for i in range(21):
        row = i // cols
        col = i % cols

        # Use offset for better packing
        offset = 0.5 if row % 2 == 0 else 0.0
        x = (col + 0.5 + offset) * (rect_width / (cols + 1))
        y = (row + 1) * (rect_height / (rows + 1))

        # Set initial radius to a small value
        r = 0.05

        # Ensure we don't go out of bounds
        x = max(r, min(rect_width - r, x))
        y = max(r, min(rect_height - r, y))

        circles[i] = [x, y, r]

    # Simple local optimization to improve the configuration
    # We'll try to increase radii while maintaining non-overlapping constraints
    max_iterations = 100
    for iteration in range(max_iterations):
        improved = False

        # Try to increase each circle's radius
        for i in range(21):
            original_radius = circles[i, 2]
            original_x, original_y = circles[i, 0], circles[i, 1]

            # Try to increase radius up to a limit
            max_radius_increase = 0.05

            # Check current neighbors to determine how much we can safely increase
            max_safe_radius = original_radius + max_radius_increase

            for j in range(21):
                if i != j:
                    dx = circles[i, 0] - circles[j, 0]
                    dy = circles[i, 1] - circles[j, 1]
                    distance = math.sqrt(dx*dx + dy*dy)

                    # We want distance >= (radius_i + radius_j)
                    # So: radius_j <= distance - radius_i
                    min_radius_for_this_j = distance - original_radius

                    if min_radius_for_this_j > 0 and min_radius_for_this_j < max_safe_radius:
                        max_safe_radius = min(max_safe_radius, min_radius_for_this_j)

            # Check boundary constraints
            boundary_safe_radius = min(
                original_x,
                rect_width - original_x,
                original_y,
                rect_height - original_y
            )

            max_safe_radius = min(max_safe_radius, boundary_safe_radius)

            if max_safe_radius > original_radius:
                # Increase radius
                new_radius = max_safe_radius - 0.001  # Small safety margin
                if new_radius > original_radius:
                    circles[i, 2] = new_radius
                    improved = True

        # If no improvement was made, break early
        if not improved:
            break

    # Final check to ensure all constraints are met
    # If there are still overlaps, reduce radii until valid
    def check_overlap():
        for i in range(21):
            for j in range(i+1, 21):
                dx = circles[i, 0] - circles[j, 0]
                dy = circles[i, 1] - circles[j, 1]
                distance = math.sqrt(dx*dx + dy*dy)
                radii_sum = circles[i, 2] + circles[j, 2]
                if distance < radii_sum:
                    return True
        return False

    # If overlaps exist, reduce all radii proportionally
    while check_overlap():
        for i in range(21):
            circles[i, 2] *= 0.99

    return circles
