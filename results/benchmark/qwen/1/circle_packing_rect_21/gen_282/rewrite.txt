random.seed(42)
np.random.seed(42)

def compute_voronoi_cell_areas(centers: np.ndarray) -> np.ndarray:
    """
    Compute Voronoi cell areas for each generator point.
    Returns array of areas corresponding to each center.
    """
    try:
        vor = Voronoi(centers)
        areas = []
        for i in range(len(centers)):
            # Get vertices of Voronoi cell for point i
            region = vor.regions[vor.point_region[i]]
            if -1 in region:
                # Infinite region - skip for area calculation
                areas.append(float('inf'))
                continue
                
            # Extract vertices and compute area using shoelace formula
            vertices = np.array([vor.vertices[j] for j in region if j >= 0])
            if len(vertices) < 3:
                areas.append(0.0)
                continue
                
            # Close the polygon
            vertices = np.vstack([vertices, vertices[0]])
            
            # Shoelace formula for polygon area
            area = 0.5 * abs(sum(vertices[i][0]*vertices[i+1][1] - vertices[i+1][0]*vertices[i][1] 
                                for i in range(len(vertices)-1)))
            areas.append(area)
        
        return np.array(areas)
    except:
        # Fallback: return uniform areas
        return np.ones(len(centers)) * 10.0

def check_constraints(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> bool:
    """
    Fast constraint checking using efficient spatial data structures.
    """
    n = len(circles)
    
    # Boundary checks
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            return False
    
    if n <= 1:
        return True
    
    # Grid-based spatial hashing for overlap detection
    avg_radius = np.mean(circles[:, 2]) if n > 0 else 0.1
    cell_size = max(0.05, avg_radius * 1.5)
    
    grid_width = max(1, int(np.ceil(rect_width / cell_size)))
    grid_height = max(1, int(np.ceil(rect_height / cell_size)))
    
    grid_width = min(grid_width, 1000)
    grid_height = min(grid_height, 1000)
    
    grid = [[[] for _ in range(grid_height)] for _ in range(grid_width)]
    
    # Assign circles to grid cells
    for i in range(n):
        x, y, r = circles[i]
        min_col = max(0, int((x - r) / cell_size))
        max_col = min(grid_width - 1, int((x + r) / cell_size))
        min_row = max(0, int((y - r) / cell_size))
        max_row = min(grid_height - 1, int((y + r) / cell_size))
        
        for row in range(min_row, max_row + 1):
            for col in range(min_col, max_col + 1):
                grid[col][row].append(i)
    
    # Check overlaps within grid cells
    for i in range(n):
        x1, y1, r1 = circles[i]
        min_col = max(0, int((x1 - r1) / cell_size))
        max_col = min(grid_width - 1, int((x1 + r1) / cell_size))
        min_row = max(0, int((y1 - r1) / cell_size))
        max_row = min(grid_height - 1, int((y1 + r1) / cell_size))
        
        for row in range(min_row, max_row + 1):
            for col in range(min_col, max_col + 1):
                for j in grid[col][row]:
                    if i != j:
                        x2, y2, r2 = circles[j]
                        distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                        if distance < (r1 + r2):
                            return False
    
    return True

def compute_geometric_constraint_info(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> Tuple[np.ndarray, np.ndarray]:
    """
    Compute geometric constraint indicators for each circle.
    Returns (voronoi_areas, angular_distributions).
    """
    n = len(circles)
    if n < 2:
        return np.ones(n) * 10.0, np.ones(n) * 0.0
    
    centers = circles[:, :2]
    
    # Compute Voronoi cell areas
    voronoi_areas = compute_voronoi_cell_areas(centers)
    
    # Compute angular distributions (how evenly neighbors are distributed)
    angular_dists = np.zeros(n)
    for i in range(n):
        center_i = centers[i]
        angles = []
        
        # Collect angles to neighbors
        for j in range(n):
            if i != j:
                dx = centers[j][0] - center_i[0]
                dy = centers[j][1] - center_i[1]
                angle = np.arctan2(dy, dx)
                angles.append(angle)
        
        if len(angles) > 1:
            angles = np.array(angles)
            # Measure spread of angles (lower variance = more clustered)
            angular_dists[i] = np.var(angles)
        else:
            angular_dists[i] = 0.0
    
    return voronoi_areas, angular_dists

def calculate_fitness(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> Tuple[float, float]:
    """
    Enhanced fitness calculation with geometric constraint awareness.
    """
    n = len(circles)
    
    # Penalty for boundary violations
    penalty = 0.0
    for i in range(n):
        x, y, r = circles[i]
        boundary_penalty = 0.0
        if x - r < 0:
            boundary_penalty += abs(x - r)
        if x + r > rect_width:
            boundary_penalty += abs(x + r - rect_width)
        if y - r < 0:
            boundary_penalty += abs(y - r)
        if y + r > rect_height:
            boundary_penalty += abs(y + r - rect_height)
        penalty += boundary_penalty * 1000.0
    
    # Overlap penalty using efficient spatial indexing
    overlap_penalty = 0.0
    if n > 1:
        coords = circles[:, :2]
        radii = circles[:, 2]
        
        try:
            # Use brute-force but optimized for small number of circles
            for i in range(n):
                for j in range(i+1, n):
                    distance = np.linalg.norm(coords[i] - coords[j])
                    min_distance = radii[i] + radii[j]
                    if distance < min_distance:
                        overlap_amount = min_distance - distance
                        overlap_penalty += overlap_amount * 1000.0
                        
        except:
            pass  # Fall back to simple method if anything goes wrong
    
    # Base fitness is sum of radii
    total_radius = np.sum(circles[:, 2])
    
    # Geometric bonus based on Voronoi cell areas - larger areas lead to better fitness
    if n >= 2:
        try:
            centers = circles[:, :2]
            voronoi_areas = compute_voronoi_cell_areas(centers)
            # Normalize by total area and penalize very small areas
            normalized_areas = voronoi_areas / (rect_width * rect_height)
            geometric_bonus = np.sum(np.maximum(normalized_areas, 0.0001)) * 10.0
            total_radius += geometric_bonus
        except:
            pass  # Fallback to basic sum
    
    fitness = total_radius - penalty - overlap_penalty
    
    return fitness, overlap_penalty

def initialize_hexagonal_spiral(n_circles: int, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Initialize using a combination of hexagonal lattice and spiral pattern.
    """
    # Start with hexagonal pattern
    total_area = rect_width * rect_height
    circle_area = total_area / n_circles * 0.9
    estimated_radius = np.sqrt(circle_area / np.pi)
    side_length = 2 * estimated_radius
    
    cols = max(1, int(rect_width / side_length) + 1)
    rows = max(1, int(rect_height / (side_length * np.sqrt(3) / 2)) + 1)
    
    points = []
    for i in range(rows):
        for j in range(cols):
            x = (j + (i % 2) * 0.5) * side_length
            y = i * side_length * np.sqrt(3) / 2
            
            if x >= estimated_radius and x <= rect_width - estimated_radius and \
               y >= estimated_radius and y <= rect_height - estimated_radius:
                points.append([x, y])
    
    # Add extra points using spiral to fill gaps
    while len(points) < n_circles:
        # Spiral pattern starting from center
        angle = len(points) * 0.5
        radius = 0.2 + (len(points) * 0.02)
        x = rect_width/2 + radius * np.cos(angle)
        y = rect_height/2 + radius * np.sin(angle)
        
        if x >= estimated_radius and x <= rect_width - estimated_radius and \
           y >= estimated_radius and y <= rect_height - estimated_radius:
            points.append([x, y])
    
    points = points[:n_circles]
    
    circles = np.zeros((n_circles, 3))
    for i, (x, y) in enumerate(points):
        circles[i] = [x, y, estimated_radius * 0.8]
    
    return circles

def adaptive_mutate(circles: np.ndarray, constraint_info: Tuple[np.ndarray, np.ndarray],
                   rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Advanced mutation with adaptive parameters based on geometric constraints.
    """
    mutated = circles.copy()
    n = len(mutated)
    
    voronoi_areas, angular_dists = constraint_info
    
    # Normalize constraint info for scaling
    normalized_voronoi = (voronoi_areas - np.min(voronoi_areas)) / (np.max(voronoi_areas) - np.min(voronoi_areas) + 1e-8)
    normalized_angular = (angular_dists - np.min(angular_dists)) / (np.max(angular_dists) - np.min(angular_dists) + 1e-8)
    
    for i in range(n):
        x, y, r = mutated[i]
        
        # Adaptive mutation: low constraint areas get more aggressive mutations
        # High Voronoi area means space is abundant, so can take bigger steps
        # Low angular variance means neighbors are clustered, less safe for big moves
        
        # Base mutation strengths
        base_pos_strength = 0.02
        base_rad_strength = 0.01
        
        # Adjust based on Voronoi area (larger areas = more room for change)
        voronoi_factor = 1.0 + 0.5 * (1.0 - normalized_voronoi[i])
        
        # Adjust based on angular distribution (more uniform = more stable)
        angular_factor = 1.0 + 0.5 * normalized_angular[i]
        
        # Combined adaptive factors
        pos_strength = base_pos_strength * voronoi_factor * angular_factor
        rad_strength = base_rad_strength * voronoi_factor * angular_factor
        
        # Mutate position
        x += np.random.normal(0, pos_strength)
        y += np.random.normal(0, pos_strength)
        
        # Mutate radius
        r += np.random.normal(0, rad_strength)
        
        # Ensure bounds and positivity
        x = np.clip(x, r, rect_width - r)
        y = np.clip(y, r, rect_height - r)
        r = np.clip(r, 0.001, rect_width / 2)
        
        mutated[i] = [x, y, r]
        
    return mutated

def tournament_selection(population: List[np.ndarray], fitness_scores: List[float], 
                        tournament_size: int = 5) -> np.ndarray:
    """
    Tournament selection with adaptive pressure.
    """
    tournament_indices = random.sample(range(len(population)), min(tournament_size, len(population)))
    tournament_fitness = [fitness_scores[i] for i in tournament_indices]
    
    # Selection pressure increasing with generation to favor exploitation
    selected_idx = tournament_indices[np.argmax(tournament_fitness)]
    
    return population[selected_idx]

def global_local_refinement(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0,
                          iterations: int = 100) -> np.ndarray:
    """
    Multi-scale refinement combining global and local optimization.
    """
    refined = circles.copy()
    n = len(refined)
    
    if n <= 1:
        return refined
    
    # Global refinement (larger steps)
    for iter_num in range(iterations // 2):
        for i in range(n):
            old_x, old_y, old_r = refined[i]
            
            # Larger global steps
            new_x = old_x + np.random.normal(0, 0.05)
            new_y = old_y + np.random.normal(0, 0.05)
            new_r = old_r + np.random.normal(0, 0.02)
            
            new_x = np.clip(new_x, new_r, rect_width - new_r)
            new_y = np.clip(new_y, new_r, rect_height - new_r)
            new_r = np.clip(new_r, 0.001, rect_width / 2)
            
            test_config = refined.copy()
            test_config[i] = [new_x, new_y, new_r]
            
            if check_constraints(test_config, rect_width, rect_height):
                current_fitness, _ = calculate_fitness(refined, rect_width, rect_height)
                test_fitness, _ = calculate_fitness(test_config, rect_width, rect_height)
                
                if test_fitness > current_fitness:
                    refined = test_config
    
    # Local refinement (smaller steps)
    for iter_num in range(iterations // 2):
        for i in range(n):
            old_x, old_y, old_r = refined[i]
            
            # Smaller local steps
            new_x = old_x + np.random.normal(0, 0.01)
            new_y = old_y + np.random.normal(0, 0.01)
            new_r = old_r + np.random.normal(0, 0.005)
            
            new_x = np.clip(new_x, new_r, rect_width - new_r)
            new_y = np.clip(new_y, new_r, rect_height - new_r)
            new_r = np.clip(new_r, 0.001, rect_width / 2)
            
            test_config = refined.copy()
            test_config[i] = [new_x, new_y, new_r]
            
            if check_constraints(test_config, rect_width, rect_height):
                current_fitness, _ = calculate_fitness(refined, rect_width, rect_height)
                test_fitness, _ = calculate_fitness(test_config, rect_width, rect_height)
                
                if test_fitness > current_fitness:
                    refined = test_config
                    
    return refined

def voronoi_guided_evolution(n_circles: int = 21,
                          rect_width: float = 1.0,
                          rect_height: float = 1.0,
                          population_size: int = 100,
                          generations: int = 100) -> np.ndarray:
    """
    Main optimization function with Voronoi-guided evolutionary approach.
    """
    # Initialize population
    population = []
    for _ in range(population_size):
        circles = initialize_hexagonal_spiral(n_circles, rect_width, rect_height)
        # Add noise to initial positions
        for i in range(n_circles):
            circles[i][0] += random.uniform(-0.03, 0.03)
            circles[i][1] += random.uniform(-0.03, 0.03)
            circles[i][0] = np.clip(circles[i][0], circles[i][2], rect_width - circles[i][2])
            circles[i][1] = np.clip(circles[i][1], circles[i][2], rect_height - circles[i][2])
        population.append(circles)
    
    # Evolutionary loop
    best_fitness_history = []
    
    for gen in range(generations):
        # Evaluate fitness of population
        fitness_scores = []
        for circles in population:
            fitness, _ = calculate_fitness(circles, rect_width, rect_height)
            fitness_scores.append(fitness)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]  # Descending order
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        best_fitness_history.append(fitness_scores[0])
        
        # Print progress
        if gen % 20 == 0:
            print(f"Generation {gen}, Best fitness: {fitness_scores[0]:.6f}")
        
        # Create new generation
        new_population = [population[0]]  # Elitism - keep best individual
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness_scores)
            
            # Select second parent
            parent2 = tournament_selection(population, fitness_scores)
            
            # Simple crossover (blend)
            offspring = parent1.copy()
            for i in range(n_circles):
                alpha = np.random.random()
                offspring[i] = [
                    alpha * parent1[i][0] + (1 - alpha) * parent2[i][0],
                    alpha * parent1[i][1] + (1 - alpha) * parent2[i][1],
                    alpha * parent1[i][2] + (1 - alpha) * parent2[i][2]
                ]
            
            # Compute constraint info and mutate adaptively
            constraint_info = compute_geometric_constraint_info(offspring, rect_width, rect_height)
            offspring = adaptive_mutate(offspring, constraint_info, rect_width, rect_height)
            
            new_population.append(offspring)
        
        population = new_population[:population_size]
        
        # Early stopping if no improvement
        if len(best_fitness_history) >= 5:
            recent_improvement = best_fitness_history[-1] - best_fitness_history[-5]
            if recent_improvement < 1e-6 and gen > 30:
                break
    
    # Return best solution
    best_index = np.argmax([calculate_fitness(ind, rect_width, rect_height)[0] for ind in population])
    return population[best_index]

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.
    """
    # Rectangle dimensions: perimeter = 4 => width + height = 2
    # Optimized aspect ratio
    rect_width = 1.3
    rect_height = 0.7
    
    # Run Voronoi-guided optimization
    best_solution = voronoi_guided_evolution(
        n_circles=21,
        rect_width=rect_width,
        rect_height=rect_height,
        population_size=100,
        generations=100
    )
    
    # Apply multi-scale refinement
    refined_solution = global_local_refinement(best_solution, rect_width, rect_height)
    
    return refined_solution
