<DESCRIPTION>
I'll implement spatial indexing to significantly improve the efficiency of collision detection. The current implementation uses O(n²) pairwise distance calculations which becomes very slow as we try to optimize. By implementing a spatial hash grid, we can reduce this to near O(n) complexity for collision detection, enabling faster local optimization and better overall performance.

This approach will particularly benefit the local optimization phase where we're repeatedly checking validity of circle configurations.

def is_valid_solution(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> bool:
    """Check if all circles are within bounds and non-overlapping using spatial indexing for efficiency."""
    n = len(circles)

    # Check bounds
    x_coords = circles[:, 0]
    y_coords = circles[:, 1]
    radii = circles[:, 2]

    if np.any(x_coords - radii < 0) or np.any(x_coords + radii > rect_width) or \
       np.any(y_coords - radii < 0) or np.any(y_coords + radii > rect_height):
        return False

    # Use spatial indexing to detect overlaps efficiently
    # Create spatial hash grid
    cell_size = np.min(radii) * 2  # Cell size based on smallest radius
    if cell_size <= 0:
        cell_size = 0.01  # Fallback to minimum size

    # Grid dimensions
    grid_width = int(np.ceil(rect_width / cell_size)) + 1
    grid_height = int(np.ceil(rect_height / cell_size)) + 1

    # Initialize grid
    grid = {}

    # Place circles into grid cells
    for i in range(n):
        x, y, r = circles[i]
        # Get grid coordinates for this circle
        min_cell_x = int(max(0, (x - r) / cell_size))
        max_cell_x = int(min(grid_width - 1, (x + r) / cell_size))
        min_cell_y = int(max(0, (y - r) / cell_size))
        max_cell_y = int(min(grid_height - 1, (y + r) / cell_size))

        # Add circle to all relevant grid cells
        for gx in range(min_cell_x, max_cell_x + 1):
            for gy in range(min_cell_y, max_cell_y + 1):
                if (gx, gy) not in grid:
                    grid[(gx, gy)] = []
                grid[(gx, gy)].append(i)

    # Check for overlaps within each grid cell and adjacent cells
    for i in range(n):
        x1, y1, r1 = circles[i]
        # Get grid cell coordinates
        cell_x = int(x1 / cell_size)
        cell_y = int(y1 / cell_size)

        # Check nearby cells (including current cell)
        for dx in [-1, 0, 1]:
            for dy in [-1, 0, 1]:
                nx, ny = cell_x + dx, cell_y + dy
                if (nx, ny) in grid:
                    for j in grid[(nx, ny)]:
                        # Skip self-comparison
                        if i >= j:
                            continue
                        x2, y2, r2 = circles[j]
                        # Check if circles overlap
                        distance_squared = (x1 - x2)**2 + (y1 - y2)**2
                        radii_sum = r1 + r2
                        if distance_squared < radii_sum * radii_sum:
                            return False

    return True
