import numpy as np


def hexagonal_pack_21():
    """Create initial configuration using hexagonal packing pattern"""
    # Hexagonal packing parameters
    n = 21
    # Rectangle dimensions (perimeter = 4, so width + height = 2)
    width = 1.0
    height = 1.0

    # For hexagonal packing, calculate appropriate grid size
    # We'll use approximately sqrt(n) rows and columns
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Ensure we don't create too many circles
    total_cells = rows * cols
    if total_cells > n:
        # Adjust to fit exactly n circles
        while rows * cols > n:
            rows -= 1

    # Calculate spacing to fill the rectangle reasonably
    if rows > 0 and cols > 0:
        cell_width = width / cols
        cell_height = height / rows
    else:
        cell_width = width
        cell_height = height

    # Use smaller spacing to allow for radius adjustment
    spacing_x = max(0.05, cell_width * 0.9)
    spacing_y = max(0.05, cell_height * 0.9)

    max_radius = min(spacing_x, spacing_y) / 2.0

    # Create circles in hexagonal pattern
    circles = []
    for i in range(rows):
        for j in range(cols):
            if len(circles) >= n:
                break

            # Position in grid
            x = (j + 0.5) * spacing_x
            y = (i + 0.5) * spacing_y

            # Apply offset for hexagonal arrangement (odd rows offset)
            if i % 2 == 1:
                x += spacing_x / 2

            # Ensure within bounds (with margin for radius)
            x = np.clip(x, max_radius, width - max_radius)
            y = np.clip(y, max_radius, height - max_radius)

            circles.append([x, y, max_radius])

        if len(circles) >= n:
            break

    # Pad if we have fewer than n circles
    while len(circles) < n:
        circles.append([width/2, height/2, max_radius])

    return np.array(circles[:n])


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Create initial hexagonal arrangement
    circles = hexagonal_pack_21()

    # Apply simple correction to handle overlaps
    # This doesn't guarantee global optimality but provides a much better starting point
    n = len(circles)
    if n > 1:
        # Simple pairwise adjustment to prevent overlaps
        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]

                # Calculate distance
                dx = x2 - x1
                dy = y2 - y1
                dist = np.sqrt(dx*dx + dy*dy)

                # If overlapping, adjust radii
                if dist < (r1 + r2) and dist > 0:
                    # Reduce radii proportionally to maintain some space
                    reduction = (r1 + r2 - dist) / 2.0
                    new_r1 = max(0.001, r1 - reduction)
                    new_r2 = max(0.001, r2 - reduction)
                    circles[i, 2] = new_r1
                    circles[j, 2] = new_r2

    return circles
