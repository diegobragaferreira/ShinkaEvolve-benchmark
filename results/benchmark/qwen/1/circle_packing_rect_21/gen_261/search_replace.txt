<DESCRIPTION>
This improvement introduces Voronoi-based criticality calculation for mutation operators, which will help identify the most constrained regions more accurately. Instead of relying solely on geometric distance calculations, we'll compute Voronoi diagrams to determine actual geometric constraints, then use these to guide mutation steps more intelligently. This addresses the limitation of using only pairwise distances and allows for more precise control over mutation intensity in different regions of the packing.

    @staticmethod
    def compute_voronoi_criticality(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
        """Compute criticality based on Voronoi diagram - how much room each circle has"""
        n = len(circles)
        if n <= 1:
            return np.ones(n) * 0.5

        # Get positions
        positions = circles[:, :2]
        radii = circles[:, 2]

        criticality_scores = np.zeros(n)

        try:
            # Generate Voronoi diagram
            vor = Voronoi(positions)
        except:
            # Fallback if Voronoi fails - return uniform criticality
            return np.ones(n) * 0.5

        # For each circle, calculate minimum distance to Voronoi edges (actual geometric constraint)
        for i in range(n):
            min_dist_to_edge = float('inf')

            # Get Voronoi region for this point
            region_idx = vor.point_region[i]
            if region_idx != -1 and region_idx < len(vor.regions):
                region = vor.regions[region_idx]
                if len(region) > 0 and -1 not in region:
                    # Check distance to all Voronoi vertices in this region
                    for vertex_idx in region:
                        if vertex_idx < len(vor.vertices):
                            vertex = vor.vertices[vertex_idx]
                            dist = np.sqrt((positions[i,0] - vertex[0])**2 + (positions[i,1] - vertex[1])**2)
                            min_dist_to_edge = min(min_dist_to_edge, dist)

            # If we couldn't compute Voronoi edge distance, fall back to distance to nearest neighbor
            if min_dist_to_edge == float('inf'):
                # Get distances to all other circles
                distances = np.sqrt(np.sum((positions - positions[i])**2, axis=1))
                # Exclude self-distance
                distances[i] = float('inf')
                # Minimum distance to any other circle center
                min_distance = np.min(distances)
                if min_distance > 0:
                    min_dist_to_edge = min_distance

            # Also consider boundary constraints more accurately
            x, y = positions[i]
            r = radii[i]
            min_boundary_dist = min(x, y, rect_width - x, rect_height - y)

            # Criticality is inversely proportional to distance to nearest constraint
            # Smaller distances = more constrained = higher criticality
            # But also factor in boundary proximity
            combined_constraint = min_dist_to_edge + (0.1 * min_boundary_dist)

            if combined_constraint > 0:
                criticality_scores[i] = 1.0 / (combined_constraint + 0.001)
            else:
                criticality_scores[i] = 1000  # Very constrained

        # Normalize and ensure minimum value
        if np.max(criticality_scores) > 0:
            # Normalize to [0,1] where 0 = most constrained, 1 = least constrained
            criticality_scores = 1.0 / (1.0 + criticality_scores)
        else:
            criticality_scores = np.ones(n)

        # Handle edge cases
        criticality_scores[np.isnan(criticality_scores)] = 0.1
        criticality_scores[criticality_scores <= 0] = 0.1

        return criticality_scores
