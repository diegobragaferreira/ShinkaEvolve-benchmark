def compute_local_criticality(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """Compute enhanced criticality scores based on local constraints and boundary distances."""
    n = len(circles)
    criticality_scores = np.zeros(n)

    # Compute pairwise distances between all circles
    coords = circles[:, :2]
    radii = circles[:, 2]

    # Vectorized distance computation
    x_diff = coords[:, np.newaxis] - coords[np.newaxis, :]
    y_diff = coords[:, np.newaxis] - coords[np.newaxis, :]
    dists = np.sqrt(x_diff**2 + y_diff**2)

    # Set diagonal to infinity to avoid self-comparison
    np.fill_diagonal(dists, np.inf)

    # Find minimum distances to other circles for each circle
    min_neighbor_distances = np.min(dists, axis=1)

    # Compute boundary constraint scores
    boundary_distances = np.array([
        coords[:, 0],  # Distance to left boundary
        rect_width - coords[:, 0],  # Distance to right boundary
        coords[:, 1],  # Distance to bottom boundary
        rect_height - coords[:, 1]   # Distance to top boundary
    ])
    min_boundary_distances = np.min(boundary_distances, axis=0)

    # Combine constraints into criticality scores
    for i in range(n):
        # Neighbor-based criticality (inverse of minimum distance)
        neighbor_crit = 1.0 / (min_neighbor_distances[i] + 0.001) if min_neighbor_distances[i] < np.inf else 1000

        # Boundary-based criticality (more critical near boundaries)
        boundary_crit = 0
        if min_boundary_distances[i] < 0.05:  # Near boundary threshold
            boundary_crit = 1000 * (0.05 - min_boundary_distances[i])

        # Combined criticality score
        criticality_scores[i] = neighbor_crit + boundary_crit

    # Normalize to [0,1] range
    if np.max(criticality_scores) > 0:
        criticality_scores = criticality_scores / np.max(criticality_scores)

    # Ensure minimum value
    criticality_scores = np.maximum(criticality_scores, 0.01)

    return criticality_scores
