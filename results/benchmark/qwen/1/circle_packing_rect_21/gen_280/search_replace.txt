    def mutate_position(self, circles: np.ndarray, idx: int) -> np.ndarray:
        """Mutate position with adaptive boundary-aware perturbation."""
        new_circles = circles.copy()
        old_x, old_y = new_circles[idx, 0], new_circles[idx, 1]
        old_r = new_circles[idx, 2]

        # Calculate distance to each boundary
        dist_to_left = old_x - old_r
        dist_to_right = self.rect_width - old_x - old_r
        dist_to_bottom = old_y - old_r
        dist_to_top = self.rect_height - old_y - old_r

        # Minimum distance to any boundary
        min_boundary_dist = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)

        # Adaptive step size based on boundary proximity
        # Circles near boundaries get smaller steps to avoid constraint violations
        max_delta = min(0.05, max(0.005, min_boundary_dist * 0.3))

        # Use a more sophisticated sampling that respects boundary constraints
        # If close to boundary, sample more carefully to avoid going out of bounds
        if min_boundary_dist < 0.02:
            # For very tight boundary situations, use a more careful approach
            delta_x = np.random.uniform(-max_delta, max_delta)
            delta_y = np.random.uniform(-max_delta, max_delta)
        else:
            # Normal case
            delta_x = np.random.uniform(-max_delta, max_delta)
            delta_y = np.random.uniform(-max_delta, max_delta)

        new_x = old_x + delta_x
        new_y = old_y + delta_y

        # Ensure within bounds with margin (slightly more generous than before)
        margin = 0.005
        new_x = np.clip(new_x, old_r + margin, self.rect_width - old_r - margin)
        new_y = np.clip(new_y, old_r + margin, self.rect_height - old_r - margin)

        new_circles[idx, 1] = new_y
