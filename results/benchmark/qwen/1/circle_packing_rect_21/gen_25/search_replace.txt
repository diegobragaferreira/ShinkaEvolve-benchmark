        cell_size = min(width, height) * 0.3
        if cell_size <= 0:
            cell_size = 0.1

        # Create a hexagonal grid pattern
        for i in range(rows):
            for j in range(cols):
                if len(circles) >= 21:
                    break

                # Hexagonal offset for alternate rows
                x_offset = (j * cell_size * 1.732) + (i % 2) * (cell_size * 0.866)
