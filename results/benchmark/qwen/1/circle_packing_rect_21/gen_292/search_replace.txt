This improvement focuses on enhancing the mutation operator by implementing Voronoi-based adaptive mutation with criticality weighting. The key insight is that circles in high-constraint regions (small Voronoi cells) should receive smaller mutation steps to maintain stability, while circles in low-constraint regions can benefit from larger mutations for exploration. This builds upon the existing Voronoi criticality computation but refines the mutation strategy to be more targeted and effective.

The change modifies the `mutate` function to use a more sophisticated adaptive approach where mutation step sizes are inversely proportional to criticality scores, with additional boundary-aware corrections to ensure better constraint satisfaction.

def mutate(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """Improved mutation operator with Voronoi-based adaptive mutation and boundary awareness."""
    mutated = circles.copy()

    # Compute criticality scores
    criticality_scores = compute_voronoi_criticality(circles, rect_width, rect_height)

    for i in range(21):
        if np.random.random() < 0.3:  # Increased mutation rate to 0.3 for better exploration
            # Choose mutation type with bias towards position (70%)
            mutation_type = np.random.choice(['position', 'radius'], p=[0.7, 0.3])

            # Get criticality score for this circle (higher means more constrained)
            crit_score = criticality_scores[i]

            if mutation_type == 'position':
                # Mutate position with adaptive step size based on criticality
                # Criticality 0 = least critical, 1 = most critical
                # Use inverse relationship: more critical -> smaller steps
                step_size = 0.02 * (1.0 - crit_score) + 0.005  # Range: 0.005 to 0.02

                # Add noise to step size for better exploration
                step_size *= (1.0 + np.random.normal(0, 0.1))
                step_size = max(0.001, step_size)  # Ensure minimum step size

                mutated[i, 0] += np.random.normal(0, step_size)
                mutated[i, 1] += np.random.normal(0, step_size)

                # Boundary awareness: enforce bounds more strictly for critical circles
                x, y, r = mutated[i]
                if x - r < 0:
                    mutated[i, 0] = r
                elif x + r > rect_width:
                    mutated[i, 0] = rect_width - r

                if y - r < 0:
                    mutated[i, 1] = r
                elif y + r > rect_height:
                    mutated[i, 1] = rect_height - r

            else:
                # Mutate radius with adaptive step size based on criticality
                # For critical circles, make smaller changes; for less critical, larger changes
                max_radius_change = 0.01 * (1.0 - crit_score) + 0.001  # Range: 0.001 to 0.01

                # Add log-normal distribution for more natural radius changes
                log_std = 0.1 * (1.0 - crit_score) + 0.05  # Less variation for critical circles
                scale_factor = np.exp(np.random.normal(0, log_std))

                                        rect_width - mutated[i, 0], rect_height - mutated[i, 1])
                mutated[i, 2] = min(mutated[i, 2], max_possible_radius * 0.95)

    return mutated
