    def spherical_to_cartesian(self, r, theta, phi):
        """Convert spherical coordinates to cartesian coordinates for radial optimization."""
        x = r * np.sin(theta) * np.cos(phi)
        y = r * np.sin(theta) * np.sin(phi)
        z = r * np.cos(theta)
        return x, y, z

    def cartesian_to_spherical(self, x, y, z):
        """Convert cartesian coordinates to spherical coordinates."""
        r = np.sqrt(x*x + y*y + z*z)
        theta = np.arccos(z / (r + 1e-10))
        phi = np.arctan2(y, x)
        return r, theta, phi

    def _create_hexagonal_solution(self) -> np.ndarray:
        """Create hexagonal lattice initial solution with spherical optimization."""
        circles = np.zeros((21, 3))
        
        # Hexagonal grid parameters
        rows = int(np.ceil(np.sqrt(21)))
        cols = int(np.ceil(21 / rows))
        
        # Adjust for rectangular container
        if self.rect_width >= self.rect_height:
            grid_width = self.rect_width * 0.9
            grid_height = self.rect_height * 0.9
        else:
            grid_width = self.rect_width * 0.9
            grid_height = self.rect_height * 0.9
        
        # Calculate spacing
        cell_width = grid_width / cols
        cell_height = grid_height / rows
        min_cell_dim = min(cell_width, cell_height)
        hex_radius = min_cell_dim * 0.4
        
        # Arrange in hexagonal pattern
        placed = 0
        for row in range(rows):
            if placed >= 21:
                break
            for col in range(cols):
                if placed >= 21:
                    break
                    
                # Offset every other row for hexagonal pattern
                offset = (row % 2) * (cell_width / 2)
                x = offset + col * cell_width + cell_width / 2
                y = row * cell_height + cell_height / 2
                
                # Ensure within bounds
                x = np.clip(x, hex_radius, self.rect_width - hex_radius)
                y = np.clip(y, hex_radius, self.rect_height - hex_radius)
                
                # Adjust radius to prevent boundary issues
                max_radius = min(x, y, self.rect_width - x, self.rect_height - y)
                r = min(hex_radius, max_radius * 0.8)
                
                circles[placed] = [x, y, r]
                placed += 1
        
        # Fill remaining positions with small random circles
        for i in range(placed, 21):
            x = np.random.uniform(hex_radius, self.rect_width - hex_radius)
            y = np.random.uniform(hex_radius, self.rect_height - hex_radius)
            r = np.random.uniform(0.005, hex_radius * 0.5)
            circles[i] = [x, y, r]
            
        return circles

    def _create_voronoi_solution(self) -> np.ndarray:

        corner_positions = [

            (0.8 * self.rect_width, 0.8 * self.rect_height),
            (self.rect_width/2, self.rect_height/2)

        placed = 0
        for x, y in corner_positions:
            if placed >= 21:
                break
            r = min(x, y, self.rect_width - x, self.rect_height - y) * 0.15
            circles[placed] = [x, y, r]
            placed += 1
        
        # Fill remaining positions with greedy approach
        max_attempts = 10000
        for attempt in range(max_attempts):
            if placed >= 21:

            y = np.random.uniform(0.05 * self.rect_height, 0.95 * self.rect_height)

            min_dist = float('inf')
            for i in range(placed):
                existing_x, existing_y, existing_r = circles[i]
                distance = np.sqrt((x - existing_x)**2 + (y - existing_y)**2)
                min_dist = min(min_dist, distance)
            
            if min_dist > 0.1 * min(self.rect_width, self.rect_height):
                r = np.random.uniform(0.01, min(x, y, self.rect_width - x, self.rect_height - y) * 0.2)

        return circles

    def _create_random_spherical_solution(self) -> np.ndarray:

    def compute_voronoi_density(self, circles: np.ndarray) -> np.ndarray:
        """Compute density score based on Voronoi diagram for constraint prioritization."""
        if len(circles) < 3:
            return np.ones(len(circles)) * 0.5

        try:
            # Get circle centers
            points = circles[:, :2]

            # Shift points to avoid boundary issues  
            shifted_points = points.copy()
            shifted_points[:, 0] = np.clip(shifted_points[:, 0], 0.01, self.rect_width - 0.01)
            shifted_points[:, 1] = np.clip(shifted_points[:, 1], 0.01, self.rect_height - 0.01)

            # Compute Voronoi diagram
            vor = Voronoi(shifted_points)

            # Compute area of Voronoi cells
            areas = []
            for i in range(len(shifted_points)):
                # Get vertices of Voronoi cell for point i
                region = vor.regions[vor.point_region[i]]
                if -1 in region:
                    # Infinite region, skip
                    areas.append(1000000)  # Large area for infinite regions
                else:
                    # Compute polygon area
                    vertices = [vor.vertices[j] for j in region if j >= 0]
                    if len(vertices) < 3:
                        areas.append(1000000)
                    else:
                        # Simplified area calculation using cross product
                        vertices_array = np.array(vertices)
                        x = vertices_array[:, 0]
                        y = vertices_array[:, 1]
                        area = 0.5 * np.abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))
                        areas.append(area)

            # Convert to density score (higher area = lower density = less critical)
            areas = np.array(areas)
            # Normalize to [0,1] where 1 = most dense (most critical)
            normalized_areas = (areas.max() - areas) / (areas.max() - areas.min() + 1e-8)
            return normalized_areas  # Higher density = closer to 1

        except:
            # Fallback to uniform density if Voronoi fails
            return np.ones(len(circles)) * 0.5

    def evaluate_fitness(self, circles: np.ndarray) -> float:
        """Evaluate fitness with adaptive penalty system."""
        # Check constraints
        if not self._validate_constraints(circles):
            # Apply penalty based on number and severity of violations
            violations = self._count_violations(circles)
            penalty = self.penalty_coefficient * violations
            return np.sum(circles[:, 2]) - penalty
            
        return np.sum(circles[:, 2])

    def _validate_constraints(self, circles: np.ndarray) -> bool:
        """Check if all circles are within bounds and don't overlap."""
        # Check bounds
        for i in range(len(circles)):
            x, y, r = circles[i]
            if x - r < 0 or x + r > self.rect_width or y - r < 0 or y + r > self.rect_height:
                return False
        
        # Check overlaps
        n = len(circles)
        if n <= 1:
            return True
            
        # Use spatial grid for efficient overlap detection
        grid_size = 0.2
        grid_width = int(np.ceil(self.rect_width / grid_size))
        grid_height = int(np.ceil(self.rect_height / grid_size))
        grid = {}

        # Place circles in grid cells
        for i in range(n):
            x, y, r = circles[i]
            grid_x = int(x / grid_size)
            grid_y = int(y / grid_size)
            if (grid_x, grid_y) not in grid:
                grid[(grid_x, grid_y)] = []
            grid[(grid_x, grid_y)].append(i)

        # Check for overlaps using grid-based approach
        for i in range(n):
            x1, y1, r1 = circles[i]
            grid_x = int(x1 / grid_size)
            grid_y = int(y1 / grid_size)

            # Check this cell and adjacent cells
            for dx in [-1, 0, 1]:
                for dy in [-1, 0, 1]:
                    neighbor_cell = (grid_x + dx, grid_y + dy)
                    if neighbor_cell in grid:
                        for j in grid[neighbor_cell]:
                            if i != j:
                                x2, y2, r2 = circles[j]
                                distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                                if distance < (r1 + r2):
                                    return False
        return True

    def _count_violations(self, circles: np.ndarray) -> int:
        """Count constraint violations for penalty calculation."""
        violations = 0
        
        # Check boundary violations
        for i in range(len(circles)):
            x, y, r = circles[i]
            if x - r < 0 or x + r > self.rect_width or y - r < 0 or y + r > self.rect_height:
                violations += 1
                
        # Check overlap violations
        n = len(circles)
        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if distance < (r1 + r2):
                    violations += 1
                    
        return violations

        # Compute density scores (higher = more critical regions)

            if np.random.random() < mutation_prob:
                # Choose mutation type with preference for position (better for packing)
                mutation_type = np.random.choice(['position', 'radius'], p=[0.7, 0.3])

                density_factor = 1.0 + 0.5 * density_scores[i]  # More aggressive in dense regions
                
                if mutation_type == 'position':
                    # Position mutation with density-adapted step size
                    step_size = 0.02 * density_factor + np.random.random() * 0.01
                    
                    mutated[i, 0] += np.random.normal(0, step_size)
                    mutated[i, 1] += np.random.normal(0, step_size)
                    
                    # Boundary correction
                    mutated[i, 0] = np.clip(mutated[i, 0], mutated[i, 2], self.rect_width - mutated[i, 2])
                    mutated[i, 1] = np.clip(mutated[i, 1], mutated[i, 2], self.rect_height - mutated[i, 2])

                    mutated[i, 2] *= scale_factor
                    
                    # Ensure positive and bounded radius
                    mutated[i, 2] = max(0.001, mutated[i, 2])
                    max_radius = min(mutated[i, 0], mutated[i, 1], 
                                   self.rect_width - mutated[i, 0], self.rect_height - mutated[i, 1])
                    mutated[i, 2] = min(mutated[i, 2], max_radius * 0.95)
        
        return mutated

    def crossover_individuals(self, parent1: np.ndarray, parent2: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
        """Enhanced crossover with geometric consideration."""
        child1 = parent1.copy()
        child2 = parent2.copy()
        
        # Blend based on cluster analysis and geometric properties
        # First compute centroid of each parent
        centroid1 = np.mean(parent1[:, :2], axis=0)
        centroid2 = np.mean(parent2[:, :2], axis=0)
        
        # Determine which parent is more central
        dist1 = np.linalg.norm(centroid1 - np.array([self.rect_width/2, self.rect_height/2]))
        dist2 = np.linalg.norm(centroid2 - np.array([self.rect_width/2, self.rect_height/2]))
        
        # Use parent with centroid closer to center for position blending
        if dist1 < dist2:
            # Blend positions from parent1, radii from parent2
            alpha_pos = np.random.random() * 0.7 + 0.15  # Blend factor for position [0.15, 0.85]
            alpha_rad = np.random.random() * 0.6 + 0.2   # Blend factor for radius [0.2, 0.8]
        else:
            # Blend positions from parent2, radii from parent1
            alpha_pos = np.random.random() * 0.7 + 0.15
            alpha_rad = np.random.random() * 0.6 + 0.2
            
        # Position blending
        for i in range(21):
            if np.random.random() < 0.5:
                # Mix positions from both parents
                child1[i, :2] = parent1[i, :2] * alpha_pos + parent2[i, :2] * (1 - alpha_pos)
                child2[i, :2] = parent1[i, :2] * (1 - alpha_pos) + parent2[i, :2] * alpha_pos
                
                # Ensure positions remain within bounds
                child1[i, 0] = np.clip(child1[i, 0], child1[i, 2], self.rect_width - child1[i, 2])
                child1[i, 1] = np.clip(child1[i, 1], child1[i, 2], self.rect_height - child1[i, 2])
                child2[i, 0] = np.clip(child2[i, 0], child2[i, 2], self.rect_width - child2[i, 2])
                child2[i, 1] = np.clip(child2[i, 1], child2[i, 2], self.rect_height - child2[i, 2])
                
                # Radius blending
                child1[i, 2] = parent1[i, 2] * alpha_rad + parent2[i, 2] * (1 - alpha_rad)
                child2[i, 2] = parent1[i, 2] * (1 - alpha_rad) + parent2[i, 2] * alpha_rad
                
                # Ensure positive radii
                child1[i, 2] = max(0.001, child1[i, 2])
                child2[i, 2] = max(0.001, child2[i, 2])
        
        return child1, child2

    def repair_individual(self, circles: np.ndarray) -> np.ndarray:
        """Advanced repair mechanism focusing on constraint violation resolution."""
        repaired = circles.copy()
        
        # Step 1: Fix radii to be positive
        repaired[:, 2] = np.maximum(repaired[:, 2], 0.001)
        
        # Step 2: Fix boundary violations
        for i in range(len(repaired)):
            x, y, r = repaired[i]
            # Push back if out of bounds
            if x - r < 0:
                x = r
            elif x + r > self.rect_width:
                x = self.rect_width - r
                
            if y - r < 0:
                y = r
            elif y + r > self.rect_height:
                y = self.rect_height - r
                
            repaired[i] = [x, y, r]
            
        # Step 3: Resolve overlaps using iterative approach
        max_iterations = 100
        for iteration in range(max_iterations):
            conflicts = []
            n = len(repaired)
            
            # Spatial grid for efficient conflict detection
            grid_size = 0.1
            grid_width = int(np.ceil(self.rect_width / grid_size))
            grid_height = int(np.ceil(self.rect_height / grid_size))
            grid = {}
            
            # Place circles in grid cells
            for i in range(n):
                x, y, r = repaired[i]
                grid_x = int(x / grid_size)
                grid_y = int(y / grid_size)
                if (grid_x, grid_y) not in grid:
                    grid[(grid_x, grid_y)] = []
                grid[(grid_x, grid_y)].append(i)
            
            # Find conflicts
            for i in range(n):
                x1, y1, r1 = repaired[i]
                grid_x = int(x1 / grid_size)
                grid_y = int(y1 / grid_size)
                
                # Check this cell and adjacent cells
                for dx in [-1, 0, 1]:
                    for dy in [-1, 0, 1]:
                        neighbor_cell = (grid_x + dx, grid_y + dy)
                        if neighbor_cell in grid:
                            for j in grid[neighbor_cell]:
                                if i != j:
                                    x2, y2, r2 = repaired[j]
                                    distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                                    if distance < (r1 + r2):
                                        conflicts.append((i, j))
            
            if not conflicts:
                break
                
            # Resolve conflicts by moving circles apart
            for i, j in conflicts[:]:  # Copy to avoid modification during iteration
                x1, y1, r1 = repaired[i]
                x2, y2, r2 = repaired[j]
                
                dx = x2 - x1
                dy = y2 - y1
                distance = np.sqrt(dx*dx + dy*dy)
                
                if distance > 0:
                    # Move circles away from each other
                    overlap = (r1 + r2) - distance
                    move_distance = overlap / 2
                    
                    dx_norm = dx / distance
                    dy_norm = dy / distance
                    
                    # Apply movement with bias toward center
                    center_x, center_y = self.rect_width/2, self.rect_height/2
                    bias1 = np.sqrt((x1 - center_x)**2 + (y1 - center_y)**2) / (self.rect_width + self.rect_height)
                    bias2 = np.sqrt((x2 - center_x)**2 + (y2 - center_y)**2) / (self.rect_width + self.rect_height)
                    
                    move1 = move_distance * (1 - bias1) / (bias1 + bias2 + 1e-8)
                    move2 = move_distance * (1 - bias2) / (bias1 + bias2 + 1e-8)
                    
                    repaired[i, 0] -= dx_norm * move1 * 0.5
                    repaired[i, 1] -= dy_norm * move1 * 0.5
                    repaired[j, 0] += dx_norm * move2 * 0.5
                    repaired[j, 1] += dy_norm * move2 * 0.5
                    
                    # Keep within bounds
                    repaired[i, 0] = np.clip(repaired[i, 0], r1, self.rect_width - r1)
                    repaired[i, 1] = np.clip(repaired[i, 1], r1, self.rect_height - r1)
                    repaired[j, 0] = np.clip(repaired[j, 0], r2, self.rect_width - r2)
                    repaired[j, 1] = np.clip(repaired[j, 1], r2, self.rect_height - r2)
        
        return repaired

    def optimize(self) -> np.ndarray:
        """Main optimization loop using multi-stage spherical evolution."""
        # Initialize population
        population = self.initialize_population()
        best_overall_solution = None
        best_overall_fitness = -np.inf
        stagnation_count = 0
        previous_best = -np.inf
        
        # Two-phase optimization: Global coarse + local fine tuning
        for generation in range(self.generations):
            # Phase 1: Global optimization with adaptive parameters
            fitness_scores = []
            for individual in population:
                fitness = self.evaluate_fitness(individual)
                fitness_scores.append(fitness)
            
            # Sort by fitness (descending)
            sorted_indices = np.argsort(fitness_scores)[::-1]
            population = [population[i] for i in sorted_indices]
            fitness_scores = [fitness_scores[i] for i in sorted_indices]
            
            # Update best overall solution
            current_best_fitness = max(fitness_scores)
            if current_best_fitness > best_overall_fitness:
                best_overall_fitness = current_best_fitness
                best_overall_solution = population[0].copy()
            
            # Keep elite
            elite = population[:self.elite_size]
            
            # Generate new population (Phase 2: Fine-tuning)
            new_population = elite[:]
            
            # Create offspring using tournament selection and crossover
            while len(new_population) < self.population_size:
                # Tournament selection
                parent1_idx = sorted_indices[np.random.choice(min(self.tournament_size, len(sorted_indices)))]
                parent2_idx = sorted_indices[np.random.choice(min(self.tournament_size, len(sorted_indices)))]
                
                parent1 = population[parent1_idx].copy()
                parent2 = population[parent2_idx].copy()
                
                # Crossover
                child1, child2 = self.crossover_individuals(parent1, parent2)
                
                # Mutate
                child1 = self.mutate_individual(child1)
                child2 = self.mutate_individual(child2)
                
                # Repair
                child1 = self.repair_individual(child1)
                child2 = self.repair_individual(child2)
                
                new_population.extend([child1, child2])
            
            population = new_population[:self.population_size]
            
            # Convergence detection
            current_best = max(fitness_scores)
            if abs(current_best - previous_best) < 1e-6:
                stagnation_count += 1
            else:
                stagnation_count = 0
            previous_best = current_best
            
            # Early stopping if stagnated too long
            if stagnation_count > self.max_stagnation:
                print(f"Early stopping at generation {generation} due to convergence")
                break
            
            # Print progress
            if generation % 30 == 0:
                print(f"Generation {generation}: Best fitness = {current_best:.6f}")

        # Final validation and return best solution
        if best_overall_solution is not None:
            final_fitness = self.evaluate_fitness(best_overall_solution)
            if final_fitness == -np.inf:
                # Revert to a valid solution if needed
                return self._create_best_initial_solution()
            return best_overall_solution
        else:
            return self._create_best_initial_solution()

    def _create_best_initial_solution(self) -> np.ndarray:
        """Create the best initial solution by trying multiple strategies."""
        strategies = [
            self._create_hexagonal_solution,
            self._create_voronoi_solution,
            self._create_strategic_solution
        ]
        
        best_solution = None
        best_fitness = -np.inf
        
        for strategy in strategies:
            solution = strategy()
            fitness = self.evaluate_fitness(solution)
            if fitness > best_fitness:
                best_fitness = fitness
                best_solution = solution.copy()
        
        return best_solution if best_solution is not None else self._create_hexagonal_solution()

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """

    return optimizer.optimize()
