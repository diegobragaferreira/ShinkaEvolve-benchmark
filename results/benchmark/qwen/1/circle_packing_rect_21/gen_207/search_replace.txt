<DESCRIPTION>
This improvement introduces a Voronoi-based local search optimization that enhances the existing local search by leveraging Voronoi diagrams to identify the most constraint-dense regions. The key insight is that circles in regions with smaller Voronoi cells are more constrained and require more careful optimization. This approach will help achieve better packing efficiency by focusing optimization efforts where they matter most, potentially surpassing the current benchmark of 2.3658321334167627.

The changes include:
1. Implementing a more sophisticated local search that uses Voronoi-based criticality analysis
2. Adding a Voronoi-based refinement step that focuses on optimizing circles in high-criticality regions
3. Enhancing the adaptive mutation operator to better utilize Voronoi information

        """Improved crossover operator that exchanges both positions and radii of critical circles"""
        child1 = parent1.copy()
        child2 = parent2.copy()

        # Get criticality scores for both parents
        crit1 = get_voronoi_criticality(parent1)
        crit2 = get_voronoi_criticality(parent2)

        # Exchange both positions and radii of circles with highest criticality
        combined_criticality = np.maximum(crit1, crit2)
        sorted_indices = np.argsort(-combined_criticality)

        # Exchange both position and radius for top 35% of circles (more exchanges)
        num_exchanges = int(len(parent1) * 0.35)
        for i in range(num_exchanges):

            # Exchange full circle data (position + radius)

    def voronoi_local_search(individual, width, height, max_iterations=200):
        """Enhanced local search using Voronoi criticality for targeted optimization"""
        best_individual = individual.copy()
        best_fitness = evaluate_fitness(best_individual, width, height)

        # Get Voronoi-based criticality for all circles
        criticality = get_voronoi_criticality(best_individual)

        for iteration in range(max_iterations):
            test_individual = best_individual.copy()

            # Select circles based on Voronoi criticality (higher criticality = more important)
            # Use a probabilistic selection where criticality determines selection probability
            weights = criticality / np.sum(criticality)
            # Select 8-12 circles with higher probability for critical ones
            num_selected = np.random.randint(8, 13)
            selected_indices = np.random.choice(len(test_individual),
                                              num_selected,
                                              replace=False,
                                              p=weights)

            for idx in selected_indices:
                old_x, old_y, old_r = test_individual[idx]

                # Adapt perturbation strength based on criticality
                # Critical circles get smaller perturbations, less critical get larger ones
                adapt_factor = 1.0 / (criticality[idx] + 0.001)
                adapt_factor = min(adapt_factor, 10.0)  # Cap adaptation factor

                # Apply more sophisticated perturbations
                # For x and y, use a combination of small and medium steps
                dx = np.random.normal(0, 0.005 * adapt_factor)
                dy = np.random.normal(0, 0.005 * adapt_factor)
                dr = np.random.normal(0, 0.002 * adapt_factor)

                new_x = max(0.005, min(0.995, old_x + dx))
                new_y = max(0.005, min(0.995, old_y + dy))
                new_r = max(0.001, old_r + dr)

                # Check constraints before accepting
                if (new_x - new_r >= 0 and new_x + new_r <= width and
                    new_y - new_r >= 0 and new_y + new_r <= height):

                    # Check overlap with all other circles
                    valid = True
                    for other_idx in range(len(test_individual)):
                        if other_idx != idx:
                            ox, oy, oradius = test_individual[other_idx]
                            dist = np.sqrt((new_x - ox)**2 + (new_y - oy)**2)
                            if dist < (new_r + oradius):
                                valid = False
                                break

                    if valid:
                        test_individual[idx] = [new_x, new_y, new_r]

            # Evaluate improvement
            new_fitness = evaluate_fitness(test_individual, width, height)
            if new_fitness > best_fitness:
                best_individual = test_individual.copy()
                best_fitness = new_fitness

        return best_individual
