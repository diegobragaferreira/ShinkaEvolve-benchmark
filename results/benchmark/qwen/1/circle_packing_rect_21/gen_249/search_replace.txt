<DESCRIPTION>
The current mutation operator uses a constraint density calculation that combines neighbor distances, boundary distances, and neighbor counts. However, this approach doesn't fully capture the geometric constraints that actually matter for circle packing optimization. I'll implement a more sophisticated constraint density metric that specifically measures how much each circle's position and radius are "constrained" by its immediate neighbors, which will guide more intelligent mutation decisions.

This enhancement focuses on creating a better understanding of actual geometric constraints rather than simple heuristics, allowing for more targeted and effective mutations that respect the real limitations imposed by neighboring circles.

        """Compute constraint density scores for each circle based on multiple factors."""
        n = len(circles)
        if n <= 1:
            return np.ones(n) * 0.5

        # Get positions and radii
        positions = circles[:, :2]
        radii = circles[:, 2]

        # 1. Compute pairwise distances for neighbor density
        distances = cdist(positions, positions)

        # Mask diagonal (self-distances) and make them very large
        np.fill_diagonal(distances, np.inf)

        # For each circle, find minimum distance to neighbors
        min_distances = np.min(distances, axis=1)

        # 2. Compute boundary distances (how close each circle is to boundaries)
        boundary_distances = np.minimum(
            positions[:, 0],  # Distance to left boundary
            np.minimum(
                rect_width - positions[:, 0],  # Distance to right boundary
                np.minimum(
                    positions[:, 1],  # Distance to bottom boundary
                    rect_height - positions[:, 1]  # Distance to top boundary
                )
            )
        )

        # 3. Calculate local density (number of neighbors within a certain radius)
        # This gives us a measure of crowding
        neighborhood_radius = np.mean(radii) * 3  # Use average radius * 3 as reference
        neighbor_counts = np.zeros(n)

        for i in range(n):
            # Count neighbors within neighborhood radius
            neighbor_count = np.sum(distances[i] < neighborhood_radius)
            neighbor_counts[i] = neighbor_count

        # 4. Combine all metrics into a composite constraint density score
        # Normalize all metrics to [0,1] range
        if np.max(min_distances) > 0:
            normalized_min_dist = min_distances / np.max(min_distances)
        else:
            normalized_min_dist = np.zeros(n)

        if np.max(boundary_distances) > 0:
            normalized_boundary_dist = boundary_distances / np.max(boundary_distances)
        else:
            normalized_boundary_dist = np.zeros(n)

        normalized_neighbor_count = neighbor_counts / max(1, np.max(neighbor_counts))

        # Combine metrics with different weights:
        # - Close neighbors = more constrained (higher density)
        # - Close to boundaries = more constrained (higher density)
        # - High neighbor count = more constrained (higher density)
        constraint_density = (
            0.4 * (1.0 - normalized_min_dist) +  # 40% neighbor proximity
            0.3 * (1.0 - normalized_boundary_dist) +  # 30% boundary proximity
            0.3 * normalized_neighbor_count  # 30% local density
        )

        # Normalize final constraint density to [0,1] where 1 = most constrained
        if np.max(constraint_density) > 0:
            constraint_density = constraint_density / np.max(constraint_density)

        return constraint_density

    @staticmethod
    def compute_constraint_density(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
        """Compute constraint density scores for each circle based on actual geometric constraints."""
        n = len(circles)
        if n <= 1:
            return np.ones(n) * 0.5

        # Get positions and radii
        positions = circles[:, :2]
        radii = circles[:, 2]

        # Compute pairwise distances for neighbor analysis
        distances = cdist(positions, positions)

        # Mask diagonal (self-distances) and make them very large
        np.fill_diagonal(distances, np.inf)

        # Compute actual geometric constraint for each circle
        # This measures how much each circle is constrained by its neighbors
        constraint_scores = np.zeros(n)

        for i in range(n):
            # Find all neighbors that could potentially constrain this circle
            # A neighbor constrains if it's close enough to affect this circle's position/radius
            neighbor_distances = distances[i]

            # Only consider neighbors that are within a reasonable range (2x average radius)
            avg_radius = np.mean(radii)
            constraint_radius = avg_radius * 3.0

            # Find neighbors that could cause overlap constraints
            nearby_neighbors = np.where(neighbor_distances < constraint_radius)[0]

            # Initialize constraint score
            total_constraint = 0.0

            # Analyze how each neighbor constrains this circle
            for j in nearby_neighbors:
                if i != j:
                    # Distance between circle centers
                    dist_ij = neighbor_distances[j]
                    r_i = radii[i]
                    r_j = radii[j]

                    # How much this neighbor constrains our ability to move or resize
                    # If neighbor is too close to force a change in either circle
                    overlap_needed = (r_i + r_j) - dist_ij

                    if overlap_needed > 0:
                        # Overlap exists - increase constraint level
                        # Normalize by the amount of overlap
                        constraint_score = overlap_needed / (r_i + r_j + 1e-8)
                        total_constraint += constraint_score
                    else:
                        # No direct overlap, but proximity still matters
                        # Closer neighbors create more geometric constraints
                        proximity_factor = 1.0 / (1.0 + dist_ij / (constraint_radius + 1e-8))
                        constraint_score = proximity_factor * 0.3  # Base constraint from proximity
                        total_constraint += constraint_score

            # Add boundary constraint factor (how close to boundary)
            x, y = positions[i]
            r = radii[i]
            boundary_constraints = 0.0

            # Distance to each boundary
            dist_to_left = x - r
            dist_to_right = rect_width - x - r
            dist_to_bottom = y - r
            dist_to_top = rect_height - y - r

            # The closer to boundary, the more constrained
            min_boundary_dist = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)

            if min_boundary_dist < r * 0.3:  # Significant boundary constraint
                boundary_constraints = (r * 0.3 - min_boundary_dist) / (r * 0.3 + 1e-8)

            total_constraint += boundary_constraints * 0.5  # Scale boundary constraint

            constraint_scores[i] = total_constraint

        # Normalize constraint scores to [0,1] where higher = more constrained
        if np.max(constraint_scores) > 0:
            constraint_scores = constraint_scores / np.max(constraint_scores)

        # Inverse the constraint score so that lower constraint scores = more freedom for mutation
        # This makes it easier to move circles in less constrained regions
        constraint_density = 1.0 - constraint_scores

        # Clamp to [0,1] range
        constraint_density = np.clip(constraint_density, 0, 1)

        return constraint_density
