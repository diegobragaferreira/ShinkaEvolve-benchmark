<DESCRIPTION>
This enhancement focuses on improving the mutation operator by implementing a more sophisticated Voronoi-based adaptive mutation strategy. The key insight is to use Voronoi cell areas to determine mutation importance - circles in regions with smaller Voronoi cells (higher constraint density) should receive smaller, more precise mutations, while circles in regions with larger Voronoi cells (lower constraint density) can receive larger mutations. This approach helps avoid getting stuck in local optima while maintaining exploration capacity in unconstrained regions.

The specific changes include:
1. Implementing a refined Voronoi criticality computation that properly handles boundary conditions
2. Creating an adaptive mutation operator that scales mutation parameters based on Voronoi cell density
3. Adding proper boundary awareness and constraint validation in mutation operations
4. Improving the overall convergence behavior through better balance between exploration and exploitation

    @staticmethod
    def compute_voronoi_criticality(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
        """Compute criticality scores for each circle based on Voronoi diagram."""
        if len(circles) < 3:
            return np.ones(len(circles)) * 0.5

        try:
            # Get circle centers
            points = circles[:, :2]

            # Shift points to avoid boundary issues
            shifted_points = points.copy()
            shifted_points[:, 0] = np.clip(shifted_points[:, 0], 0.01, rect_width - 0.01)
            shifted_points[:, 1] = np.clip(shifted_points[:, 1], 0.01, rect_height - 0.01)

            # Compute Voronoi diagram
            vor = Voronoi(shifted_points)

            # Compute area of Voronoi cells
            areas = []
            for i in range(len(shifted_points)):
                # Get vertices of Voronoi cell for point i
                region = vor.regions[vor.point_region[i]]
                if -1 in region:
                    # Infinite region, skip
                    areas.append(1000000)  # Large area for infinite regions
                else:
                    # Compute polygon area
                    vertices = [vor.vertices[j] for j in region if j >= 0]
                    if len(vertices) < 3:
                        areas.append(1000000)
                    else:
                        # Simplified area calculation using cross product
                        vertices_array = np.array(vertices)
                        x = vertices_array[:, 0]
                        y = vertices_array[:, 1]
                        area = 0.5 * np.abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))
                        areas.append(area)

            # Convert to criticality score (lower area = higher criticality)
            areas = np.array(areas)
            # Normalize to [0,1] where 0 = most critical
            normalized_areas = (areas - areas.min()) / (areas.max() - areas.min() + 1e-8)
            return 1.0 - normalized_areas  # Higher criticality = closer to 1

        except:
            # Fallback to uniform criticality if Voronoi fails
            return np.ones(len(circles)) * 0.5

class MutationOperator:
    """Handles mutation operations with adaptive parameters."""

    @staticmethod
    def compute_voronoi_criticality(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
        """Compute criticality scores for each circle based on Voronoi diagram."""
        if len(circles) < 3:
            return np.ones(len(circles)) * 0.5

        try:
            # Get circle centers
            points = circles[:, :2]

            # Add boundary points to ensure proper Voronoi computation
            boundary_points = [
                [0, 0], [rect_width, 0], [0, rect_height], [rect_width, rect_height],
                [rect_width/2, 0], [rect_width/2, rect_height],
                [0, rect_height/2], [rect_width, rect_height/2]
            ]
            all_points = np.vstack([points, boundary_points])

            # Shift points to avoid boundary issues
            shifted_points = points.copy()
            shifted_points[:, 0] = np.clip(shifted_points[:, 0], 0.01, rect_width - 0.01)
            shifted_points[:, 1] = np.clip(shifted_points[:, 1], 0.01, rect_height - 0.01)

            # Compute Voronoi diagram with boundary points
            vor = Voronoi(all_points)

            # Compute area of Voronoi cells for original points
            areas = []
            for i in range(len(points)):
                # Get vertices of Voronoi cell for point i (excluding boundary points)
                region_idx = np.where(vor.point_region == i)[0][0] if i in vor.point_region else -1

                if region_idx != -1 and region_idx < len(vor.regions):
                    region = vor.regions[region_idx]
                    if -1 not in region and len(region) >= 3:
                        # Compute area of polygon
                        vertices = np.array([vor.vertices[j] for j in region])
                        # Simple polygon area calculation
                        if len(vertices) >= 3:
                            # Shoelace formula
                            x = vertices[:, 0]
                            y = vertices[:, 1]
                            area = 0.5 * np.abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))
                            areas.append(area)
                        else:
                            areas.append(1.0)
                    else:
                        areas.append(1.0)
                else:
                    areas.append(1.0)

            # Convert to criticality score (lower area = higher criticality)
            areas = np.array(areas)
            # Normalize to [0,1] where 0 = most critical
            normalized_areas = (areas - areas.min()) / (areas.max() - areas.min() + 1e-8)
            return 1.0 - normalized_areas  # Higher criticality = closer to 1

        except Exception as e:
            # Fallback to uniform criticality if Voronoi fails
            return np.ones(len(circles)) * 0.5
