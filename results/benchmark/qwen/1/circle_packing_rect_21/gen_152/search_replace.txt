<DESCRIPTION>
This change implements Voronoi-based constraint weighting for adaptive mutation, which is a key improvement suggested in the recommendations. The idea is to compute Voronoi diagrams for each configuration and weight radius mutations inversely to local Voronoi cell area, where smaller cells indicate higher constraint density requiring finer adjustments. This builds upon the existing Voronoi constraint computation but enhances the mutation strategy to be more sophisticated based on geometric density.

The approach:
1. Enhances the `compute_voronoi_constraints` function to provide more accurate constraint density information
2. Modifies `mutate_circles_adaptive` to use Voronoi cell areas for more precise mutation weights
3. Adds a more sophisticated spatial indexing method for improved collision detection
4. Implements better initialization strategies to start with more promising configurations

    # For each circle, count how many others are close enough to potentially collide
    for i in range(n):

def compute_voronoi_cell_areas(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:

        return np.ones(n) * 1.0  # Default to uniform weights

    centers = circles[:, :2]

    try:

        vor = Voronoi(centers)

        # Compute areas for each Voronoi cell
        cell_areas = np.zeros(n)

        for i in range(n):
            # Get vertices of the Voronoi cell for circle i
            # Find the ridges belonging to this cell
            ridge_indices = [j for j, ridge in enumerate(vor.ridge_vertices) if i in ridge and -1 not in ridge]

                            vertices.append(vor.vertices[v2])

def mutate_circles_adaptive(circles: np.ndarray,
                          constraint_densities: np.ndarray,
                          rect_width: float = 1.0,
                          rect_height: float = 1.0,
                          max_radius: float = 0.5) -> np.ndarray:
    """
    Mutate circle positions and radii with adaptive weights based on constraint density.
    """
    mutated = circles.copy()
    n = len(mutated)

    # Mutation parameters adapted based on constraint density
    for i in range(n):
        x, y, r = mutated[i]

        # Higher constraint density = more careful mutation
        density_weight = 1.0 + constraint_densities[i] * 2.0  # 1.0 to 3.0 range

        # Position mutation strength (smaller for high constraint areas)
        pos_strength = 0.02 / density_weight
        rad_strength = 0.01 / density_weight

        # Mutate position
        x += np.random.normal(0, pos_strength)
        y += np.random.normal(0, pos_strength)

        # Ensure position stays within bounds
        x = np.clip(x, r, rect_width - r)
        y = np.clip(y, r, rect_height - r)

        # Mutate radius
        r += np.random.normal(0, rad_strength)
        # Ensure radius remains positive and reasonable
        r = np.clip(r, 0.001, max_radius * 0.9)

        mutated[i] = [x, y, r]

    return mutated

        # Constraint density is inverse of cell area - smaller areas = higher density
        density_weight = 1.0 + constraint_densities[i] * 3.0  # Scale from 1.0 to 4.0

        pos_strength = 0.02 / density_weight

        y += np.random.normal(0, pos_strength)
