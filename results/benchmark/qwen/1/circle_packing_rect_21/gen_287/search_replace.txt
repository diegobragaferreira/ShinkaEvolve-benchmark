        return np.zeros(n)

        vor = Voronoi(centers)
    except:
        return np.ones(n) * 0.5

    for i in range(n):
        center_i = centers[i]
        nearby_count = 0
        max_radius = np.max(circles[:, 2])
        threshold = 3 * max_radius

        for j in range(n):
            if i != j:
                dist = np.linalg.norm(center_i - centers[j])

def compute_voronoi_criticality(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Compute criticality scores for each circle based on Voronoi diagram cell areas.
    Lower area = higher criticality (more constrained).
    """
    n = len(circles)
    if n < 3:
        return np.ones(n) * 0.5

    try:
        # Get circle centers
        points = circles[:, :2]

        # Shift points to avoid boundary issues
        shifted_points = points.copy()
        shifted_points[:, 0] = np.clip(shifted_points[:, 0], 0.01, rect_width - 0.01)
        shifted_points[:, 1] = np.clip(shifted_points[:, 1], 0.01, rect_height - 0.01)

        # Compute Voronoi diagram
        vor = Voronoi(shifted_points)

        # Compute area of Voronoi cells
        areas = []
        for i in range(len(shifted_points)):
            # Get vertices of Voronoi cell for point i
            region = vor.regions[vor.point_region[i]]
            if -1 in region:
                # Infinite region, use large area
                areas.append(1000000)
            else:
                # Compute polygon area
                vertices = [vor.vertices[j] for j in region if j >= 0]
                if len(vertices) < 3:
                    areas.append(1000000)
                else:
                    # Use shoelace formula for polygon area
                    vertices_array = np.array(vertices)
                    x = vertices_array[:, 0]
                    y = vertices_array[:, 1]
                    area = 0.5 * np.abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))
                    areas.append(area)

        # Convert to criticality score (lower area = higher criticality)
        areas = np.array(areas)
        # Normalize to [0,1] where 0 = most critical (smallest area)
        normalized_areas = (areas - areas.min()) / (areas.max() - areas.min() + 1e-8)
        criticality_scores = 1.0 - normalized_areas  # Higher criticality = closer to 1

        return criticality_scores

    except:
        # Fallback to uniform criticality if Voronoi fails
        return np.ones(len(circles)) * 0.5

def mutate_circles_adaptive(circles: np.ndarray,
                          criticality_scores: np.ndarray,
                          rect_width: float = 1.0,
                          rect_height: float = 1.0,
                          max_radius: float = 0.5) -> np.ndarray:
    """
    Mutate circle positions and radii with adaptive weights based on Voronoi-based criticality.
    High criticality (small Voronoi cells) = more conservative mutations.
    """
    mutated = circles.copy()
    n = len(mutated)

    for i in range(n):
        x, y, r = mutated[i]

        # Criticality-based mutation factors
        # High criticality (0-1 scale) -> smaller mutations
        criticality = criticality_scores[i]
        # Scale mutation strength: criticality 1 (high) = 0.1x base, criticality 0 (low) = 1x base
        mutation_scale = 0.1 + 0.9 * (1 - criticality)

        pos_strength = 0.02 * mutation_scale
        rad_strength = 0.01 * mutation_scale

        # Mutate position
        x += np.random.normal(0, pos_strength)
        y += np.random.normal(0, pos_strength)

        # Ensure position stays within bounds
        x = np.clip(x, r, rect_width - r)
        y = np.clip(y, r, rect_height - r)

        # Mutate radius
        r += np.random.normal(0, rad_strength)
        # Ensure radius remains positive and reasonable
        r = np.clip(r, 0.001, max_radius * 0.9)

        mutated[i] = [x, y, r]

    return mutated
