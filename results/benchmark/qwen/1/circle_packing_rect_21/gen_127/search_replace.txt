def calculate_voronoi_constraint_density(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """Calculate constraint density for each circle based on Voronoi diagram."""
    try:
        from scipy.spatial import Voronoi
        n = len(circles)
        if n < 2:
            return np.ones(n)

        points = circles[:, :2]
        vor = Voronoi(points)

        # Calculate Voronoi cell areas for each circle
        constraint_densities = np.zeros(n)

        for i in range(n):
            # Get Voronoi region for circle i
            region = []
            # Find region vertices associated with this point
            for j, point in enumerate(vor.points):
                if np.allclose(point, points[i], atol=1e-10):
                    # This is the correct point index
                    region_idx = j
                    break
            else:
                # Point not found, fallback to simple density calculation
                constraint_densities[i] = 1.0
                continue

            # We need to find the actual Voronoi cell for this point
            # For now, let's compute a simpler constraint density based on neighbors
            distances = np.sqrt(np.sum((points - points[i])**2, axis=1))
            distances[i] = np.inf  # Remove self-distance
            nearest_distances = np.sort(distances)[:min(5, len(distances))]  # Top 5 nearest

            # Constraint density is inversely proportional to average distance to neighbors
            if len(nearest_distances) > 0:
                avg_dist = np.mean(nearest_distances)
                # Higher density = closer neighbors, more constrained
                constraint_densities[i] = 1.0 / max(avg_dist, 1e-6)
            else:
                constraint_densities[i] = 1.0

        return constraint_densities

    except Exception:
        # Fallback to simple distance-based density calculation
        n = len(circles)
        constraint_densities = np.ones(n)
        if n <= 1:
            return constraint_densities

        positions = circles[:, :2]
        distances = cdist(positions, positions)

        # For each circle, count how many other circles are within a certain distance
        # This creates a sense of local crowding
        for i in range(n):
            # Count neighbors within a threshold distance (based on radius)
            neighbor_count = 0
            radius_i = circles[i, 2]
            for j in range(n):
                if i != j and distances[i, j] < radius_i * 3:  # Threshold
                    neighbor_count += 1
            # Normalize by number of circles
            constraint_densities[i] = neighbor_count / max(n - 1, 1)

        return constraint_densities

def mutate(circles: np.ndarray, mutation_rate: float = 0.3, constraint_densities: np.ndarray = None) -> np.ndarray:
    """Improved mutation operator with adaptive parameters based on constraint density."""
    mutated = circles.copy()

    # If constraint densities not provided, calculate them
    if constraint_densities is None:
        constraint_densities = calculate_voronoi_constraint_density(circles)

    for i in range(21):
        if np.random.random() < mutation_rate:
            # Choose mutation type with bias towards position
            mutation_type = np.random.choice(['position', 'radius'], p=[0.7, 0.3])

            # Get density for this circle (higher density = more constrained)
            density = constraint_densities[i]

            if mutation_type == 'position':
                # Mutate position (x, y) with adaptive step size based on density
                # More constrained areas get smaller steps
                base_step = 0.05 + np.random.random() * 0.05
                # Reduce step size for denser regions
                step_size = base_step * (1.0 / (1.0 + density * 0.5))
                mutated[i, 0] += np.random.normal(0, step_size)
                mutated[i, 1] += np.random.normal(0, step_size)
            else:
                # Mutate radius with adaptive scaling based on density
                # In dense regions, be more conservative
                base_scale = 0.2
                scale_factor = np.exp(np.random.normal(0, base_scale / (1.0 + density * 0.3)))
                mutated[i, 2] *= scale_factor
                mutated[i, 2] = max(0.001, mutated[i, 2])

    return mutated
