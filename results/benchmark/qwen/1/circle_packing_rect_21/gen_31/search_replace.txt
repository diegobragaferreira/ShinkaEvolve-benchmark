import numpy as np


def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Set up rectangle dimensions (perimeter = 4 means width + height = 2)
    rect_width = 1.0
    rect_height = 1.0

    # Use a hexagonal packing approach for good initial distribution
    # For 21 circles, we can use a rough 5x5 grid with offset rows
    n = 21

    # Initialize circles with some reasonable starting values
    circles = np.zeros((n, 3))

    # Hexagonal packing parameters
    sqrt3 = np.sqrt(3)

    # Calculate initial spacing
    max_radius = 0.1  # Initial guess
    spacing_x = 2 * max_radius
    spacing_y = sqrt3 * max_radius

    # Generate hexagonal grid
    row = 0
    col = 0
    idx = 0

    # Place circles in hexagonal pattern
    while idx < n:
        x = (col + (row % 2) * 0.5) * spacing_x + max_radius
        y = row * spacing_y + max_radius

        # Ensure within rectangle bounds
        if x <= rect_width - max_radius and y <= rect_height - max_radius:
            circles[idx] = [x, y, max_radius]
            idx += 1

        col += 1
        if col > 5:  # Limit columns to prevent infinite loop
            col = 0
            row += 1
            if row > 5:  # Limit rows
                break

    # Ensure we have exactly 21 circles even if our grid didn't fill perfectly
    while idx < n:
        circles[idx] = circles[idx-1][:]
        idx += 1

    # Simple local optimization to improve sum of radii
    # Try to increase radii while respecting constraints
    for _ in range(50):  # Limited iterations for speed
        improved = False
        for i in range(n):
            x, y, r = circles[i]
            new_r = min(r + 0.005, 0.2)  # Small increment, capped

            # Check if we can increase this radius
            valid = True
            for j in range(n):
                if i != j:
                    x2, y2, r2 = circles[j]
                    dist = np.sqrt((x - x2)**2 + (y - y2)**2)
                    if dist < new_r + r2:
                        valid = False
                        break

            # Also check boundary
            if valid and x - new_r >= 0 and x + new_r <= rect_width and y - new_r >= 0 and y + new_r <= rect_height:
                circles[i] = [x, y, new_r]
                improved = True

        if not improved:
            break

    return circles
