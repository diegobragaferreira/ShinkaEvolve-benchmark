def build_grid_index(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0,
                    cell_size_factor: float = 1.0) -> Tuple[np.ndarray, dict]:
    """
    Build a grid-based spatial index for efficient collision detection.
    """
    n = len(circles)
    if n == 0:
        return np.array([]), {}

    avg_radius = np.mean(circles[:, 2]) if n > 0 else 0.1
    cell_size = avg_radius * cell_size_factor

    if cell_size <= 0:
        cell_size = 0.1

    cols = max(1, int(rect_width / cell_size) + 1)
    rows = max(1, int(rect_height / cell_size) + 1)

    grid_cells = np.full((rows, cols), -1, dtype=np.int32)
    cell_dict = {}

    for i in range(n):
        x, y, r = circles[i]
        col = int(x / cell_size)
        row = int(y / cell_size)

        col = max(0, min(col, cols - 1))
        row = max(0, min(row, rows - 1))

        key = (row, col)
        if key not in cell_dict:
            cell_dict[key] = []
        cell_dict[key].append(i)

    return grid_cells, cell_dict

def grid_collision_check(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> bool:
    """
    Fast collision check using grid-based spatial indexing.
    """
    n = len(circles)

    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            return False

    if n <= 1:
        return True

    avg_radius = np.mean(circles[:, 2]) if n > 0 else 0.1
    cell_size = avg_radius * 1.5

    cols = max(1, int(rect_width / cell_size) + 1)
    rows = max(1, int(rect_height / cell_size) + 1)

    cell_dict = {}

    for i in range(n):
        x, y, r = circles[i]
        col = int(x / cell_size)
        row = int(y / cell_size)

        col = max(0, min(col, cols - 1))
        row = max(0, min(row, rows - 1))

        key = (row, col)
        if key not in cell_dict:
            cell_dict[key] = []
        cell_dict[key].append(i)

    for i in range(n):
        x1, y1, r1 = circles[i]

        col = int(x1 / cell_size)
        row = int(y1 / cell_size)
        col = max(0, min(col, cols - 1))
        row = max(0, min(row, rows - 1))

        for dr in [-1, 0, 1]:
            for dc in [-1, 0, 1]:
                neighbor_row = row + dr
                neighbor_col = col + dc

                if 0 <= neighbor_row < rows and 0 <= neighbor_col < cols:
                    key = (neighbor_row, neighbor_col)
                    if key in cell_dict:
                        for j in cell_dict[key]:
                            if i != j:
                                x2, y2, r2 = circles[j]

                                dx = x1 - x2
                                dy = y1 - y2
                                distance_sq = dx*dx + dy*dy
                                min_distance_sq = (r1 + r2) * (r1 + r2)

                                if distance_sq < min_distance_sq:
                                    return False

    return True

def is_valid_solution(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> bool:
    """
    Fast validity check using hybrid collision detection.
    """
    n = len(circles)

    # Check boundary constraints first
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            return False

    if n <= 1:
        return True

    # Use hybrid approach: grid for coarse check, KDTree for precise neighbor search
    try:
        # First perform grid-based collision check as quick filter
        if not grid_collision_check(circles, rect_width, rect_height):
            return False

        # Then use KDTree for final verification
        coords = circles[:, :2]
        radii = circles[:, 2]
        tree = cKDTree(coords)

        max_radius = np.max(radii)
        for i in range(n):
            neighbors = tree.query_ball_point(coords[i], 2 * max_radius)
            for j in neighbors:
                if i != j:
                    distance = np.linalg.norm(coords[i] - coords[j])
                    min_distance = radii[i] + radii[j]
                    if distance < min_distance:
                        return False
        return True
    except:
        # Fallback to brute force if spatial indexing fails
        for i in range(n):
            for j in range(i+1, n):
                distance = np.linalg.norm(coords[i] - coords[j])
                min_distance = radii[i] + radii[j]
                if distance < min_distance:
                    return False
        return True

def compute_voronoi_constraints(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Compute constraint density for each circle based on Voronoi cell areas.
    More sophisticated than simple neighbor counting - uses actual Voronoi cell areas.
    """
    n = len(circles)
    if n < 2:
        return np.zeros(n)

    centers = circles[:, :2]

    try:
        vor = Voronoi(centers)
    except:
        # Fallback to simpler constraint calculation if Voronoi fails
        return np.ones(n) * 0.5

    constraint_density = np.zeros(n)

    # For each circle, compute the area of its Voronoi cell
    # Smaller Voronoi cells = more constrained = higher criticality
    for i in range(n):
        # Find vertices of Voronoi cell for circle i
        # Voronoi regions are identified by the indices of generators
        # We'll compute the area using the polygon formed by the Voronoi vertices

        # Simple approximation: use minimum distance to neighbors as proxy for constraint
        center_i = centers[i]
        min_distance_to_neighbor = float('inf')

        for j in range(n):
            if i != j:
                dist = np.linalg.norm(center_i - centers[j])
                if dist < min_distance_to_neighbor:
                    min_distance_to_neighbor = dist

        # Convert distance to constraint score (smaller distance = more constrained)
        # Using inverse relationship to represent constraint density
        if min_distance_to_neighbor > 0:
            constraint_density[i] = 1.0 / (min_distance_to_neighbor + 0.001)
        else:
            constraint_density[i] = 1000  # Very constrained

    # Normalize the constraint scores
    if np.max(constraint_density) > 0:
        constraint_density = constraint_density / np.max(constraint_density)

    # Ensure minimum constraint value to prevent division by zero in later calculations
    constraint_density = np.maximum(constraint_density, 0.01)

    return constraint_density

def compute_voronoi_areas(circles: np.ndarray) -> np.ndarray:
    """
    Compute approximate Voronoi cell areas for each circle.
    """
    n = len(circles)
    if n < 2:
        return np.ones(n) * 1.0
    
    try:
        centers = circles[:, :2]
        vor = Voronoi(centers)
        
        areas = np.zeros(n)
        
        # Compute area of Voronoi cell for each generator
        for i in range(n):
            # Find Voronoi vertices associated with generator i
            # This is a simplified approach - in practice Voronoi cells can be complex
            # For now, we'll use a proxy: inverse square of minimum distance to neighbors
            center_i = centers[i]
            min_distance = float('inf')
            
            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(center_i - centers[j])
                    if dist < min_distance:
                        min_distance = dist
                        
            if min_distance > 0:
                areas[i] = 1.0 / (min_distance * min_distance + 0.0001)
            else:
                areas[i] = 1000.0
                
        # Normalize areas
        if np.max(areas) > 0:
            areas = areas / np.max(areas)
            
        return areas
    except:
        # Fallback to uniform areas
        return np.ones(n)

def initialize_multiple_patterns(n_circles: int, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Initialize circle positions using multiple patterns and select best.
    """
    patterns = []
    
    # Pattern 1: Hexagonal lattice
    try:
        hex_pattern = initialize_hexagonal_lattice(n_circles, rect_width, rect_height)
        patterns.append(hex_pattern)
    except:
        pass
    
    # Pattern 2: Grid pattern
    try:
        grid_pattern = initialize_grid_pattern(n_circles, rect_width, rect_height)
        patterns.append(grid_pattern)
    except:
        pass
    
    # Pattern 3: Spiral pattern
    try:
        spiral_pattern = initialize_spiral_pattern(n_circles, rect_width, rect_height)
        patterns.append(spiral_pattern)
    except:
        pass
    
    # Pattern 4: Random with overlap resolution
    try:
        random_pattern = initialize_random_constrained(n_circles, rect_width, rect_height)
        patterns.append(random_pattern)
    except:
        pass
    
    if not patterns:
        # Fallback to simple hexagonal
        return initialize_hexagonal_lattice(n_circles, rect_width, rect_height)
    
    # Evaluate patterns
    best_pattern = patterns[0]
    best_fitness = float('-inf')
    
    for pattern in patterns:
        fitness, _ = calculate_fitness(pattern, rect_width, rect_height)
        if fitness > best_fitness:
            best_fitness = fitness
            best_pattern = pattern
    
    return best_pattern

def initialize_hexagonal_lattice(n_circles: int, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Initialize circle positions using a hexagonal lattice pattern.
    """
    total_area = rect_width * rect_height
    circle_area = total_area / n_circles * 0.9
    estimated_radius = np.sqrt(circle_area / np.pi)

    side_length = 2 * estimated_radius
    cols = max(1, int(rect_width / side_length) + 1)
    rows = max(1, int(rect_height / (side_length * np.sqrt(3) / 2)) + 1)

    points = []
    for i in range(rows):
        for j in range(cols):
            x = (j + (i % 2) * 0.5) * side_length
            y = i * side_length * np.sqrt(3) / 2

            if x >= estimated_radius and x <= rect_width - estimated_radius and \
               y >= estimated_radius and y <= rect_height - estimated_radius:
                points.append([x, y])

    while len(points) < n_circles:
        x = random.uniform(estimated_radius, rect_width - estimated_radius)
        y = random.uniform(estimated_radius, rect_height - estimated_radius)
        points.append([x, y])

    points = points[:n_circles]

    circles = np.zeros((n_circles, 3))
    for i, (x, y) in enumerate(points):
        circles[i] = [x, y, estimated_radius * 0.8]

    return circles

def initialize_grid_pattern(n_circles: int, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Initialize circle positions using a grid pattern.
    """
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    
    # Adjust dimensions to ensure even spacing
    width_available = rect_width * 0.9
    height_available = rect_height * 0.9
    
    cell_width = width_available / cols
    cell_height = height_available / rows
    
    # Use minimum of both to make circles consistent
    max_radius = min(cell_width, cell_height) * 0.4
    
    circles = np.zeros((n_circles, 3))
    idx = 0
    
    for i in range(rows):
        for j in range(cols):
            if idx >= n_circles:
                break
            x = rect_width * 0.05 + j * cell_width + cell_width / 2
            y = rect_height * 0.05 + i * cell_height + cell_height / 2
            circles[idx] = [x, y, max_radius]
            idx += 1
    
    return circles

def initialize_spiral_pattern(n_circles: int, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Initialize circle positions using a spiral pattern.
    """
    circles = np.zeros((n_circles, 3))
    center_x, center_y = rect_width / 2, rect_height / 2
    max_radius = min(rect_width, rect_height) * 0.4
    
    # Spiral parameters
    angle_step = 2 * np.pi / 3.0
    radius_step = 0.05
    
    for i in range(n_circles):
        angle = i * angle_step
        radius = i * radius_step
        
        # Limit radius to stay within bounds
        radius = min(radius, max_radius * 0.8)
        
        x = center_x + radius * np.cos(angle)
        y = center_y + radius * np.sin(angle)
        
        # Clip to bounds
        x = np.clip(x, max_radius, rect_width - max_radius)
        y = np.clip(y, max_radius, rect_height - max_radius)
        
        circles[i] = [x, y, max_radius * 0.6]
    
    return circles

def initialize_random_constrained(n_circles: int, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
    """
    Initialize circle positions using random placement with overlap resolution.
    """
    circles = np.zeros((n_circles, 3))
    max_radius = min(rect_width, rect_height) * 0.1
    
    for i in range(n_circles):
        attempts = 0
        placed = False
        
        while not placed and attempts < 1000:
            # Random position
            x = random.uniform(max_radius, rect_width - max_radius)
            y = random.uniform(max_radius, rect_height - max_radius)
            r = random.uniform(max_radius * 0.3, max_radius)
            
            # Check if it conflicts with existing circles
            valid = True
            for j in range(i):
                existing_x, existing_y, existing_r = circles[j]
                dist = np.sqrt((x - existing_x)**2 + (y - existing_y)**2)
                if dist < (r + existing_r):
                    valid = False
                    break
                    
            if valid:
                circles[i] = [x, y, r]
                placed = True
            else:
                attempts += 1
        
        if not placed:
            # Fallback to simple random placement
            circles[i] = [
                random.uniform(max_radius, rect_width - max_radius),
                random.uniform(max_radius, rect_height - max_radius),
                random.uniform(max_radius * 0.3, max_radius)
            ]
    
    return circles

def calculate_fitness(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> Tuple[float, float]:
    """
    Calculate fitness of circle configuration with penalty for constraint violations.
    """
    n = len(circles)

    penalty = 0.0
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > rect_width or y - r < 0 or y + r > rect_height:
            overlap = 0.0
            if x - r < 0:
                overlap += abs(x - r) * 100
            if x + r > rect_width:
                overlap += abs(x + r - rect_width) * 100
            if y - r < 0:
                overlap += abs(y - r) * 100
            if y + r > rect_height:
                overlap += abs(y + r - rect_height) * 100
            penalty += overlap

    overlap_penalty = 0.0
    if n > 1:
        coords = circles[:, :2]
        radii = circles[:, 2]

        try:
            tree = cKDTree(coords)
            max_radius = np.max(radii)

            for i in range(n):
                neighbors = tree.query_ball_point(coords[i], 2 * max_radius)
                for j in neighbors:
                    if i != j:
                        distance = np.linalg.norm(coords[i] - coords[j])
                        min_distance = radii[i] + radii[j]

                        if distance < min_distance:
                            overlap_amount = min_distance - distance
                            overlap_penalty += overlap_amount * 1000
        except:
            for i in range(n):
                for j in range(i+1, n):
                    distance = np.linalg.norm(coords[i] - coords[j])
                    min_distance = radii[i] + radii[j]

                    if distance < min_distance:
                        overlap_amount = min_distance - distance
                        overlap_penalty += overlap_amount * 1000

    total_radius = np.sum(circles[:, 2])
    fitness = total_radius - penalty - overlap_penalty

    return fitness, overlap_penalty

def mutate_circles_adaptive(circles: np.ndarray,
                          constraint_densities: np.ndarray,
                          voronoi_areas: np.ndarray,
                          rect_width: float = 1.0,
                          rect_height: float = 1.0,
                          max_radius: float = 0.5) -> np.ndarray:
    """
    Mutate circle positions and radii with adaptive weights based on constraint density and Voronoi areas.
    """
    mutated = circles.copy()
    n = len(mutated)

    # Mutation parameters
    base_pos_strength = 0.015
    base_rad_strength = 0.008
    
    # Combine constraint density and Voronoi areas for weighted mutation
    # Circles in dense/constrained areas should have smaller mutations
    # Circles in sparse areas can have larger mutations
    combined_weights = constraint_densities * 0.6 + (1.0 - voronoi_areas) * 0.4
    
    for i in range(n):
        x, y, r = mutated[i]

        # Use combined weight for adaptive mutation
        weight = combined_weights[i]
        # Inverse relationship - higher constraint density = lower mutation strength
        pos_strength = base_pos_strength / (weight * 0.5 + 0.5)
        rad_strength = base_rad_strength / (weight * 0.5 + 0.5)
        
        # Cap maximum strength
        pos_strength = min(pos_strength, 0.05)
        rad_strength = min(rad_strength, 0.03)

        # Mutate position
        x += np.random.normal(0, pos_strength)
        y += np.random.normal(0, pos_strength)

        # Ensure position stays within bounds
        x = np.clip(x, r + 0.001, rect_width - r - 0.001)
        y = np.clip(y, r + 0.001, rect_height - r - 0.001)

        # Mutate radius
        r += np.random.normal(0, rad_strength)
        # Ensure radius remains positive and reasonable
        r = np.clip(r, 0.001, max_radius * 0.95)

        mutated[i] = [x, y, r]

    return mutated

def crossover_circles(parent1: np.ndarray, parent2: np.ndarray,
                     crossover_rate: float = 0.8) -> np.ndarray:
    """
    Perform uniform crossover between two circle configurations with constraint awareness.
    """
    if random.random() > crossover_rate:
        return parent1.copy()

    offspring = parent1.copy()
    n = len(parent1)

    # Hybrid crossover - mix good parts from both parents
    # Select crossover points based on constraint density (focus on better constrained areas)
    constraint1 = compute_voronoi_constraints(parent1)
    constraint2 = compute_voronoi_constraints(parent2)
    
    # Cross over based on average constraint density
    avg_constraint = (constraint1 + constraint2) / 2.0
    crossover_threshold = np.percentile(avg_constraint, 30)  # Top 30% get more crossover

    for i in range(n):
        # If constraint is high (dense area), do crossover more frequently
        if avg_constraint[i] >= crossover_threshold:
            if random.random() < 0.7:  # 70% chance for high constraint areas
                offspring[i] = parent2[i].copy()
        else:
            if random.random() < 0.3:  # 30% chance for low constraint areas
                offspring[i] = parent2[i].copy()

    return offspring

def refine_solution_fast(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0,
                        iterations: int = 150) -> np.ndarray:
    """
    Fast local refinement using gradient-like approach and constraint-aware moves.
    """
    refined = circles.copy()
    n = len(refined)

    if n <= 1:
        return refined

    # Precompute constraint information
    constraint_densities = compute_voronoi_constraints(refined, rect_width, rect_height)
    voronoi_areas = compute_voronoi_areas(refined)

    # Annealing schedule
    T_start = 1.0
    T_end = 0.01
    alpha = 0.95
    
    T = T_start

    for iter_num in range(iterations):
        # Work on one circle at a time
        for i in range(n):
            # Save current state
            old_x, old_y, old_r = refined[i]

            # Adaptive mutation based on constraint density and Voronoi areas
            weight = constraint_densities[i] * 0.6 + (1.0 - voronoi_areas[i]) * 0.4
            pos_strength = 0.01 / (weight * 0.5 + 0.5)
            rad_strength = 0.005 / (weight * 0.5 + 0.5)
            
            # Cap maximum strength
            pos_strength = min(pos_strength, 0.05)
            rad_strength = min(rad_strength, 0.03)

            # Try small random moves
            new_x = old_x + np.random.normal(0, pos_strength)
            new_y = old_y + np.random.normal(0, pos_strength)
            new_r = old_r + np.random.normal(0, rad_strength)

            # Clip to bounds
            new_x = np.clip(new_x, new_r + 0.001, rect_width - new_r - 0.001)
            new_y = np.clip(new_y, new_r + 0.001, rect_height - new_r - 0.001)
            new_r = np.clip(new_r, 0.001, rect_width / 2)

            # Test if this change improves fitness
            test_config = refined.copy()
            test_config[i] = [new_x, new_y, new_r]

            # Quick constraint check before full fitness evaluation
            if not is_valid_solution(test_config, rect_width, rect_height):
                continue

            # Check if this move improves fitness (with simulated annealing acceptance criteria)
            current_fitness, _ = calculate_fitness(refined, rect_width, rect_height)
            test_fitness, _ = calculate_fitness(test_config, rect_width, rect_height)

            if test_fitness > current_fitness:
                refined = test_config
            elif random.random() < np.exp((test_fitness - current_fitness) / T):
                # Accept worse solution with probability based on temperature
                refined = test_config

        # Cool down temperature
        if iter_num > 0 and iter_num % 50 == 0:
            T = max(T_end, T * alpha)

        # Occasionally do larger moves to escape local optima
        if random.random() < 0.03 and iter_num > iterations // 3:
            # Do bigger perturbation
            i = random.randint(0, n-1)
            old_x, old_y, old_r = refined[i]
            
            # Use larger mutation for escaping local optima
            large_pos_strength = pos_strength * 2.0
            large_rad_strength = rad_strength * 2.0
            
            new_x = old_x + np.random.normal(0, large_pos_strength)
            new_y = old_y + np.random.normal(0, large_pos_strength)
            new_r = old_r + np.random.normal(0, large_rad_strength)

            # Clip to bounds
            new_x = np.clip(new_x, new_r + 0.001, rect_width - new_r - 0.001)
            new_y = np.clip(new_y, new_r + 0.001, rect_height - new_r - 0.001)
            new_r = np.clip(new_r, 0.001, rect_width / 2)

            if is_valid_solution(test_config, rect_width, rect_height):
                test_config[i] = [new_x, new_y, new_r]
                test_fitness, _ = calculate_fitness(test_config, rect_width, rect_height)
                if test_fitness > current_fitness:
                    refined = test_config

    return refined

def refine_boundary_circles(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0,
                           iterations: int = 75) -> np.ndarray:
    """
    Specialized refinement for circles near boundaries to better utilize container space.
    """
    refined = circles.copy()
    n = len(refined)

    if n <= 1:
        return refined

    # Define boundary thresholds (circles within this distance from edge are considered boundary circles)
    boundary_threshold = min(rect_width, rect_height) * 0.1

    for iter_num in range(iterations):
        # Process circles near boundaries first (they're more constrained)
        for i in range(n):
            x, y, r = refined[i]

            # Check if circle is near any boundary
            near_boundary = (
                x - r <= boundary_threshold or
                x + r >= rect_width - boundary_threshold or
                y - r <= boundary_threshold or
                y + r >= rect_height - boundary_threshold
            )

            if not near_boundary:
                continue  # Skip non-boundary circles for this refinement step

            # For boundary circles, use more aggressive but careful updates
            old_x, old_y, old_r = x, y, r

            # Use larger mutation for boundary circles to push them into better positions
            pos_strength = 0.04  # Larger than normal
            rad_strength = 0.015  # Larger than normal

            # Try moves that might help utilize boundary space better
            new_x = old_x + np.random.normal(0, pos_strength)
            new_y = old_y + np.random.normal(0, pos_strength)
            new_r = old_r + np.random.normal(0, rad_strength)

            # Ensure boundary constraints are maintained
            new_x = np.clip(new_x, new_r + 0.001, rect_width - new_r - 0.001)
            new_y = np.clip(new_y, new_r + 0.001, rect_height - new_r - 0.001)
            new_r = np.clip(new_r, 0.001, rect_width / 2)

            # Test if this change improves fitness
            test_config = refined.copy()
            test_config[i] = [new_x, new_y, new_r]

            # Quick constraint check before full fitness evaluation
            if not is_valid_solution(test_config, rect_width, rect_height):
                continue

            # Check if this move improves fitness
            current_fitness, _ = calculate_fitness(refined, rect_width, rect_height)
            test_fitness, _ = calculate_fitness(test_config, rect_width, rect_height)

            if test_fitness > current_fitness:
                refined = test_config

    return refined

def optimize_with_voronoi_evolution(n_circles: int = 21,
                                  rect_width: float = 1.0,
                                  rect_height: float = 1.0,
                                  population_size: int = 150,
                                  generations: int = 120) -> np.ndarray:
    """
    Optimize circle packing using Voronoi-enhanced evolutionary algorithm.
    """
    # Phase 1: Multi-pattern initialization to get diverse starting points
    print("Initializing population with multiple patterns...")
    population = []
    for _ in range(population_size):
        circles = initialize_multiple_patterns(n_circles, rect_width, rect_height)
        # Add small random noise to initial positions
        for i in range(n_circles):
            circles[i][0] += random.uniform(-0.02, 0.02)
            circles[i][1] += random.uniform(-0.02, 0.02)
            circles[i][0] = np.clip(circles[i][0], circles[i][2], rect_width - circles[i][2])
            circles[i][1] = np.clip(circles[i][1], circles[i][2], rect_height - circles[i][2])
        population.append(circles)

    # Evolutionary loop with adaptive parameters
    best_fitness_history = []
    stagnation_counter = 0

    # Phase 2: Main evolutionary optimization
    for gen in range(generations):
        # Evaluate fitness of population
        fitness_scores = []
        for circles in population:
            fitness, _ = calculate_fitness(circles, rect_width, rect_height)
            fitness_scores.append(fitness)

        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]  # Descending order
        population = [population[i] for i in sorted_indices]
        fitness_scores.sort(reverse=True)

        best_fitness_history.append(fitness_scores[0])

        # Print progress
        if gen % 20 == 0:
            print(f"Generation {gen}, Best fitness: {fitness_scores[0]:.6f}")

        # Check for stagnation
        if len(best_fitness_history) >= 5:
            recent_improvement = best_fitness_history[-1] - best_fitness_history[-5]
            if recent_improvement < 1e-6:
                stagnation_counter += 1
            else:
                stagnation_counter = 0
                
            if stagnation_counter > 5 and gen > 30:
                # Restart with new diverse initialization
                print(f"Stagnation detected at generation {gen}, restarting...")
                population = []
                for _ in range(population_size):
                    circles = initialize_multiple_patterns(n_circles, rect_width, rect_height)
                    # Add small random noise to initial positions
                    for i in range(n_circles):
                        circles[i][0] += random.uniform(-0.02, 0.02)
                        circles[i][1] += random.uniform(-0.02, 0.02)
                        circles[i][0] = np.clip(circles[i][0], circles[i][2], rect_width - circles[i][2])
                        circles[i][1] = np.clip(circles[i][1], circles[i][2], rect_height - circles[i][2])
                    population.append(circles)
                stagnation_counter = 0
        else:
            stagnation_counter = 0

        # Create new generation
        new_population = [population[0]]  # Elitism - keep best individual

        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 5
            tournament_indices = random.sample(range(population_size), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]

            # Select parent
            parent1 = population[winner_index]

            # Select second parent
            tournament_indices.remove(winner_index)
            tournament_fitness.remove(max(tournament_fitness))
            winner_index2 = tournament_indices[np.argmax(tournament_fitness)]
            parent2 = population[winner_index2]

            # Crossover
            offspring = crossover_circles(parent1, parent2)

            # Compute constraint densities and mutate adaptively
            constraint_densities = compute_voronoi_constraints(offspring, rect_width, rect_height)
            voronoi_areas = compute_voronoi_areas(offspring)
            offspring = mutate_circles_adaptive(offspring, constraint_densities, voronoi_areas,
                                              rect_width, rect_height,
                                              max_radius=min(rect_width, rect_height) / 2)

            new_population.append(offspring)

        population = new_population[:population_size]

    # Phase 3: Final refinement
    print("Performing final refinement...")
    # Return best solution from the population
    best_index = np.argmax([calculate_fitness(ind, rect_width, rect_height)[0] for ind in population])
    best_solution = population[best_index]
    
    # Apply multi-stage refinement
    refined_solution = refine_solution_fast(best_solution, rect_width, rect_height, 200)
    boundary_refined = refine_boundary_circles(refined_solution, rect_width, rect_height, 100)

    return boundary_refined

def circle_packing21() -> np.ndarray:
    """
    Places 21 non-overlapping circles inside a rectangle of perimeter 4 in order to maximize the sum of their radii.

    Returns:
        circles: np.array of shape (21,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    # Rectangle dimensions: perimeter = 4 => width + height = 2
    # Optimize rectangle aspect ratio for better packing
    rect_width = 1.2
    rect_height = 0.8

    # Run Voronoi-enhanced optimization
    try:
        best_solution = optimize_with_voronoi_evolution(
            n_circles=21,

            rect_height=rect_height,

    except Exception as e:
