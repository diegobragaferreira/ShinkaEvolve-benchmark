        n_circles = 21

        cols = 5

        # Calculate spacing to fit within rectangle
        cell_size = min(width, height) * 0.25
        if cell_size <= 0:
            cell_size = 0.1

        # Create a hexagonal grid pattern
        for i in range(rows):
            for j in range(cols):
                if len(circles) >= n_circles:
                    break

                # Hexagonal offset for alternate rows
                x_offset = (j * cell_size * 1.732) + (i % 2) * (cell_size * 0.866)
                y_offset = i * cell_size * 0.75

                # Check if position is within bounds

                    y = y_offset + cell_size * 0.5

                        # Use smaller radius to allow for optimization

                        y = np.clip(y, r, height - r)
