<DESCRIPTION>
The current Voronoi-based criticality computation in the mutation operator is computationally expensive and sometimes fails due to numerical instability. I'll improve this by creating a more efficient and robust criticality calculation that uses a hybrid approach combining Voronoi analysis for the core geometric constraints with simpler distance-based heuristics as fallbacks. This will reduce computational overhead while maintaining the beneficial properties of Voronoi-based constraint awareness.

The key improvement is to implement a faster Voronoi-based criticality calculation that:
1. Uses a more stable Voronoi computation method
2. Falls back gracefully to simpler distance-based criticality when Voronoi fails
3. Is more efficient by reducing unnecessary calculations
4. Maintains the beneficial property of prioritizing mutations in constrained regions

    @staticmethod
    def compute_voronoi_criticality(circles: np.ndarray, rect_width: float = 1.0, rect_height: float = 1.0) -> np.ndarray:
        """Compute criticality scores for each circle based on Voronoi diagram and geometric constraints."""
        if len(circles) < 2:
            return np.ones(len(circles)) * 0.5

        try:
            # Get circle centers
            points = circles[:, :2]

            # Use a more numerically stable approach for Voronoi computation
            # Clip points to avoid extreme boundary conditions
            clipped_points = points.copy()
            clip_margin = 0.01
            clipped_points[:, 0] = np.clip(clipped_points[:, 0], clip_margin, rect_width - clip_margin)
            clipped_points[:, 1] = np.clip(clipped_points[:, 1], clip_margin, rect_height - clip_margin)

            # For performance, we'll compute simplified criticality metrics
            # Instead of full Voronoi cell areas, we'll use:
            # 1. Local density (based on number of close neighbors)
            # 2. Boundary proximity
            # 3. Minimum distance to neighbors

            # Compute pairwise distances
            distances = cdist(clipped_points, clipped_points)

            # Mask diagonal (self-distances) and set to large value
            np.fill_diagonal(distances, np.inf)

            # Compute minimum distances to neighbors
            min_distances = np.min(distances, axis=1)

            # Compute local density (number of neighbors within a reasonable radius)
            # Use average radius as reference for neighborhood radius
            avg_radius = np.mean(circles[:, 2]) if len(circles) > 0 else 0.1
            neighborhood_radius = avg_radius * 2.5

            # Count neighbors in neighborhood
            neighbor_counts = np.sum(distances < neighborhood_radius, axis=1)

            # Normalize all metrics to [0,1]
            if np.max(min_distances) > 0:
                normalized_min_dist = min_distances / np.max(min_distances)
            else:
                normalized_min_dist = np.zeros_like(min_distances)

            if np.max(neighbor_counts) > 0:
                normalized_neighbor_count = neighbor_counts / np.max(neighbor_counts)
            else:
                normalized_neighbor_count = np.zeros_like(neighbor_counts)

            # Compute boundary distance metrics
            boundary_distances = np.minimum(
                clipped_points[:, 0],  # Distance to left boundary
                np.minimum(
                    rect_width - clipped_points[:, 0],  # Distance to right boundary
                    np.minimum(
                        clipped_points[:, 1],  # Distance to bottom boundary
                        rect_height - clipped_points[:, 1]  # Distance to top boundary
                    )
                )
            )

            if np.max(boundary_distances) > 0:
                normalized_boundary = boundary_distances / np.max(boundary_distances)
            else:
                normalized_boundary = np.zeros_like(boundary_distances)

            # Combine criticality factors with appropriate weights:
            # - 40%: Inverse of minimum distance to neighbors (closer = more constrained)
            # - 30%: Neighbor count (more neighbors = more constrained)
            # - 30%: Inverse of boundary distance (closer to boundary = more constrained)
            criticality = (
                0.4 * (1.0 - normalized_min_dist) +
                0.3 * normalized_neighbor_count +
                0.3 * (1.0 - normalized_boundary)
            )

            # Ensure values are in [0, 1] range
            criticality = np.clip(criticality, 0, 1)

            # Invert so that lower criticality means more freedom for mutation
            # This way, circles with criticality ~0 are free to mutate more
            return 1.0 - criticality

        except Exception as e:
            # Fallback to distance-based criticality if Voronoi fails
            # Use simpler but robust approach
            if len(circles) > 1:
                positions = circles[:, :2]
                distances = cdist(positions, positions)
                np.fill_diagonal(distances, np.inf)
                min_distances = np.min(distances, axis=1)

                # Normalize distances (smaller = more critical)
                if np.max(min_distances) > 0:
                    normalized_distances = (min_distances - min_distances.min()) / (min_distances.max() - min_distances.min() + 1e-8)
                else:
                    normalized_distances = np.zeros_like(min_distances)

                # Higher criticality = more constrained
                criticality = 1.0 - normalized_distances
                return np.clip(criticality, 0, 1)
            else:
                return np.ones(len(circles)) * 0.5
