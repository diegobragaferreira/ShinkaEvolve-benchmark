    def initialize_with_voronoi():
        """Initialize circle positions using Voronoi-based approach with enhanced boundary sampling"""
        # Generate initial points using a grid-like pattern with some randomness
        n_points = 128  # Increase points for better Voronoi coverage
        points = []

        # Create a systematic point distribution with boundary emphasis
        for i in range(12):  # More grid points
            for j in range(12):
                x = 0.05 + i * (0.9 / 11) + np.random.uniform(-0.01, 0.01)
                y = 0.05 + j * (0.9 / 11) + np.random.uniform(-0.01, 0.01)
                if 0 <= x <= 1 and 0 <= y <= 1:
                    points.append([x, y])

        # Add boundary points to encourage large circles near edges
        boundary_points = []
        for _ in range(32):  # Add extra boundary points
            side = np.random.randint(0, 4)
            if side == 0:  # Top
                boundary_points.append([np.random.random(), 1.0])
            elif side == 1:  # Bottom
                boundary_points.append([np.random.random(), 0.0])
            elif side == 2:  # Left
                boundary_points.append([0.0, np.random.random()])
            else:  # Right
                boundary_points.append([1.0, np.random.random()])

        points.extend(boundary_points)
        points = np.array(points)

        try:
