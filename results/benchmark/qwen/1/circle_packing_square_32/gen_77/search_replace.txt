    def initialize_strategic_positions(self) -> List[Tuple[float, float]]:
        """Generate strategic positions for initial circle placement using enhanced Voronoi-inspired approach."""
        # Start with corner and edge positions (these are most likely to accommodate large circles)
        positions = [
            (0.1, 0.1), (0.1, 0.9), (0.9, 0.1), (0.9, 0.9),  # corners
            (0.5, 0.1), (0.5, 0.9), (0.1, 0.5), (0.9, 0.5),  # edges
            (0.5, 0.5),  # center
        ]

        # Add more strategic positions in a grid pattern
        grid_positions = []
        spacing = 0.2  # Spacing between grid points

        # Generate a grid of points
        for i in range(1, 5):  # 4x4 internal grid
            for j in range(1, 5):
                x = i * spacing
                y = j * spacing
                grid_positions.append((x, y))

        # Add boundary points with emphasis on areas where large circles can fit
        boundary_points = []
        for i in range(1, 10):
            # Top and bottom edges
            boundary_points.extend([(i*0.1, 0.05), (i*0.1, 0.95)])
            # Left and right edges
            boundary_points.extend([(0.05, i*0.1), (0.95, i*0.1)])

        # Combine and weight positions for better boundary coverage
        all_positions = positions + grid_positions + boundary_points

        # Weight positions closer to boundaries more heavily since they're more promising for large circles
        weighted_positions = []
        for pos in all_positions:
            x, y = pos
            # Calculate distance to nearest boundary
            dist_to_boundary = min(x, 1-x, y, 1-y)
            # Higher weight for positions closer to boundaries (more potential for large circles)
            weight = 1.0 + 4.0 * (1.0 - dist_to_boundary)  # Up to 5x weighting for boundary positions
            weighted_positions.append((pos[0], pos[1], weight))

        # Sort by weight (descending) and keep top positions
        weighted_positions.sort(key=lambda p: p[2], reverse=True)
        return [(p[0], p[1]) for p in weighted_positions[:len(all_positions)]]  # Return just coordinates
