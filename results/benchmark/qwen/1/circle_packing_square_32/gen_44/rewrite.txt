import numpy as np
from scipy.spatial import Voronoi
from scipy.spatial.distance import cdist
import random
from copy import deepcopy

def circle_packing32() -> np.ndarray:
    """
    Places 32 non-overlapping circles in the unit square in order to maximize the sum of radii.
    Uses Voronoi-guided evolutionary optimization for improved performance.
    
    Returns:
        circles: np.array of shape (32,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    np.random.seed(42)
    random.seed(42)
    
    def is_valid_configuration(circles):
        """Check if all circles fit in square and don't overlap"""
        n = len(circles)
        # Check containment constraints
        for i in range(n):
            x, y, r = circles[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        
        # Check overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                distance = np.sqrt((x1-x2)**2 + (y1-y2)**2)
                if distance < r1 + r2:
                    return False
        return True
    
    def compute_sum_radii(circles):
        """Compute sum of all radii"""
        return sum(circle[2] for circle in circles)
    
    def generate_voronoi_initialization(n_points=32):
        """Generate initial configuration using Voronoi diagram approach"""
        # Start with random points inside square
        points = np.random.rand(n_points, 2)
        
        # Find Voronoi vertices and filter valid ones
        try:
            vor = Voronoi(points)
            # Get rid of infinite vertices
            finite_regions = []
            for region in vor.point_region:
                if region != -1:  # Skip infinite regions
                    finite_regions.append(region)
            
            # Generate candidate positions from Voronoi vertices
            candidates = []
            for vertex_idx in range(len(vor.vertices)):
                x, y = vor.vertices[vertex_idx]
                # Only keep vertices that are inside the unit square
                if 0 <= x <= 1 and 0 <= y <= 1:
                    candidates.append([x, y])
            
            # If we have enough valid candidates, sample them
            if len(candidates) >= n_points:
                selected = random.sample(candidates, n_points)
                return np.array(selected)
            else:
                # Fall back to random points if we don't have enough Voronoi vertices
                return np.random.rand(n_points, 2)
                
        except Exception:
            # Fallback to pure random initialization
            return np.random.rand(n_points, 2)
    
    def optimize_radius_config(circles, max_iter=1000):
        """Optimize radii for fixed positions using gradient-like approach"""
        n = len(circles)
        current = deepcopy(circles)
        
        # Simple gradient ascent for radius optimization
        for _ in range(max_iter):
            improved = False
            
            # Try to increase each radius independently
            for i in range(n):
                old_r = current[i][2]
                new_r = old_r + 0.001
                
                # Check if we can increase this radius
                valid = True
                for j in range(n):
                    if i != j:
                        x1, y1, r1 = current[i]
                        x2, y2, r2 = current[j]
                        distance = np.sqrt((x1-x2)**2 + (y1-y2)**2)
                        if distance < r1 + r2 + 0.001:  # Allow small margin
                            valid = False
                            break
                
                # Check containment
                x, y, r = current[i]
                if r > x or r > (1-x) or r > y or r > (1-y):
                    valid = False
                
                if valid and new_r < min(x, 1-x, y, 1-y):
                    # Check if new configuration is valid
                    temp_config = deepcopy(current)
                    temp_config[i][2] = new_r
                    
                    # Check if the new configuration is still valid
                    if is_valid_configuration(temp_config):
                        current[i][2] = new_r
                        improved = True
                        
            if not improved:
                break
                
        return current
    
    def generate_offspring(parent1, parent2, mutation_rate=0.1):
        """Create offspring through crossover and mutation"""
        n = len(parent1)
        child = deepcopy(parent1)
        
        # Crossover: mix positions from both parents
        for i in range(n):
            if random.random() < 0.5:
                child[i][0] = parent2[i][0]
                child[i][1] = parent2[i][1]
        
        # Mutation: slightly perturb positions
        for i in range(n):
            if random.random() < mutation_rate:
                child[i][0] += np.random.normal(0, 0.01)
                child[i][1] += np.random.normal(0, 0.01)
                # Keep within bounds
                child[i][0] = max(0, min(1, child[i][0]))
                child[i][1] = max(0, min(1, child[i][1]))
        
        # Ensure valid radii
        for i in range(n):
            # Set initial radius based on available space
            x, y = child[i][0], child[i][1]
            min_dist = min(x, 1-x, y, 1-y)
            child[i][2] = min_dist * 0.4  # Conservative initial value
            
        return child
    
    # Main evolutionary algorithm
    population_size = 50
    generations = 100
    elite_size = 5
    
    # Initialize population with Voronoi-based configurations
    population = []
    for _ in range(population_size):
        # Get initial Voronoi-based positions
        positions = generate_voronoi_initialization(32)
        
        # Create circles with initial radii (conservative estimates)
        circles = []
        for i in range(32):
            x, y = positions[i]
            # Initial conservative radius estimate
            min_dist = min(x, 1-x, y, 1-y)
            r = min_dist * 0.4
            circles.append([x, y, r])
        
        # Optimize the radii for this configuration
        optimized = optimize_radius_config(circles)
        population.append(optimized)
    
    # Evolution loop
    for gen in range(generations):
        # Evaluate fitness
        fitness_scores = []
        for individual in population:
            if is_valid_configuration(individual):
                fitness = compute_sum_radii(individual)
            else:
                fitness = 0  # Invalid configurations get zero fitness
            fitness_scores.append(fitness)
        
        # Sort by fitness descending
        sorted_indices = sorted(range(len(fitness_scores)), 
                               key=lambda i: fitness_scores[i], reverse=True)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Print best fitness
        if gen % 20 == 0:
            print(f"Generation {gen}: Best fitness = {fitness_scores[0]:.6f}")
        
        # Elitism: keep top individuals
        new_population = population[:elite_size]
        
        # Create offspring
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 5
            tournament_indices = random.sample(range(elite_size), tournament_size)
            winner_idx = max(tournament_indices, key=lambda i: fitness_scores[i])
            parent1 = population[winner_idx]
            
            tournament_indices = random.sample(range(elite_size), tournament_size)
            winner_idx = max(tournament_indices, key=lambda i: fitness_scores[i])
            parent2 = population[winner_idx]
            
            # Create offspring
            child = generate_offspring(parent1, parent2)
            
            # Optimize radii of offspring
            child_optimized = optimize_radius_config(child)
            
            # Add to new population
            new_population.append(child_optimized)
        
        population = new_population
    
    # Return best solution
    best_solution = population[0]
    # Final optimization step
    final_optimized = optimize_radius_config(best_solution)
    
    # Convert to required format (32,3) array
    result = np.zeros((32, 3))
    for i, circle in enumerate(final_optimized):
        result[i] = circle
    
    # Ensure it's still valid
    if not is_valid_configuration(result):
        # If somehow invalid, create a fallback
        fallback = np.random.rand(32, 3)
        for i in range(32):
            fallback[i, 0] = fallback[i, 0] * 0.8 + 0.1  # Keep away from edges
            fallback[i, 1] = fallback[i, 1] * 0.8 + 0.1
            fallback[i, 2] = 0.02  # Small radius
        return fallback
    
    return result
