This change will help maintain consistent behavior under edge cases and improve convergence reliability.

    def calculate_min_max_ratio(self, points: np.ndarray) -> float:
        """Calculate the ratio of minimum to maximum distance between all point pairs."""
        if len(points) < 2:
            return 0.0

        # Use squareform for more numerically stable distance matrix calculation
        distance_matrix = squareform(pdist(points))

        # Set diagonal to infinity to exclude self-distances
        np.fill_diagonal(distance_matrix, np.inf)

        # Handle edge cases
        if np.all(distance_matrix == np.inf):
            return 0.0

        d_min = np.min(distance_matrix)
        d_max = np.max(distance_matrix)

        # Avoid division by zero
        if d_max <= 0:
            return 0.0

        return d_min / d_max
