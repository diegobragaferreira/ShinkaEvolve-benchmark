import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import differential_evolution, minimize
import time
import random

class PointDispersionOptimizer:
    def __init__(self, n_points=16, dimension=2, bounds=(0, 1)):
        self.n_points = n_points
        self.dimension = dimension
        self.bounds = bounds
        self.best_points = None
        self.best_ratio = 0
        
    def calculate_distance_matrix(self, points):
        """Calculate pairwise distance matrix"""
        return squareform(pdist(points))
    
    def fitness_function(self, points_flat):
        """Calculate fitness as min/max distance ratio"""
        # Reshape flat array back to points
        points = points_flat.reshape(-1, self.dimension)
        
        # Ensure points are within bounds
        points = np.clip(points, self.bounds[0], self.bounds[1])
        
        # Calculate distance matrix
        try:
            dist_matrix = self.calculate_distance_matrix(points)
            
            # Set diagonal to large value to avoid zero distances
            np.fill_diagonal(dist_matrix, np.inf)
            
            # Calculate min and max distances
            min_dist = np.min(dist_matrix)
            max_dist = np.max(dist_matrix)
            
            # Avoid division by zero
            if max_dist == 0:
                return 0
                
            # Return ratio (we want to maximize this)
            ratio = min_dist / max_dist
            
            # Store best solution
            if ratio > self.best_ratio:
                self.best_ratio = ratio
                self.best_points = points.copy()
                
            return ratio
            
        except Exception:
            return 0
    
    def initialize_population(self, pop_size=50):
        """Initialize population with diverse point configurations"""
        population = []
        for _ in range(pop_size):
            # Mix of different initialization strategies
            strategy = random.choice(['random', 'grid', 'spiral'])
            if strategy == 'random':
                points = np.random.uniform(self.bounds[0], self.bounds[1], 
                                         (self.n_points, self.dimension))
            elif strategy == 'grid':
                # Create grid-like initialization
                side = int(np.ceil(np.sqrt(self.n_points)))
                x = np.linspace(self.bounds[0], self.bounds[1], side)
                y = np.linspace(self.bounds[0], self.bounds[1], side)
                grid_x, grid_y = np.meshgrid(x, y)
                points = np.column_stack([grid_x.ravel(), grid_y.ravel()])[:self.n_points]
                # Add small random perturbations
                points += np.random.normal(0, 0.01, points.shape)
            else:  # spiral
                angles = np.linspace(0, 4*np.pi, self.n_points)
                radii = np.linspace(0.1, 0.4, self.n_points)
                x = radii * np.cos(angles) + 0.5
                y = radii * np.sin(angles) + 0.5
                points = np.column_stack([x, y])
            
            # Ensure all points are within bounds
            points = np.clip(points, self.bounds[0], self.bounds[1])
            population.append(points.flatten())
        
        return population
    
    def crossover(self, parent1, parent2):
        """Perform crossover between two parents"""
        crossover_point = len(parent1) // 2
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(self, individual, mutation_rate=0.1):
        """Apply mutation to an individual"""
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.01)
        return mutated
    
    def optimize_with_de(self, max_iter=200):
        """Optimize using differential evolution"""
        # Flatten bounds for scipy
        flat_bounds = [(self.bounds[0], self.bounds[1]) for _ in range(self.n_points * self.dimension)]
        
        # Use differential evolution
        result = differential_evolution(
            self.fitness_function,
            flat_bounds,
            maxiter=max_iter,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42,
            disp=False
        )
        
        return result.x
    
    def optimize_with_local_refinement(self, initial_points):
        """Refine solution using local optimization"""
        # Define bounds for each variable
        bounds = [(self.bounds[0], self.bounds[1]) for _ in range(self.n_points * self.dimension)]
        
        # Use L-BFGS-B for local refinement
        result = minimize(
            self.fitness_function,
            initial_points,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-8},
            callback=self._optimization_callback
        )
        
        return result.x
    
    def _optimization_callback(self, x):
        """Callback function for optimization progress"""
        pass

def min_max_dist_dim2_16() -> np.ndarray:
    """
    Creates 16 points in 2 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (16,2) containing the (x,y) coordinates of the 16 points.

    """
    
    # Setup optimizer
    optimizer = PointDispersionOptimizer(n_points=16, dimension=2)
    
    # Set seed for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    # Start timer
    start_time = time.time()
    
    # Method 1: Try differential evolution
    try:
        # Run differential evolution
        final_solution = optimizer.optimize_with_de(max_iter=150)
        
        # Local refinement
        refined_solution = optimizer.optimize_with_local_refinement(final_solution)
        
        # Reshape result
        points = refined_solution.reshape(-1, 2)
        
    except Exception as e:
        # Fallback to random initialization if optimization fails
        print(f"Optimization failed: {e}")
        points = np.random.uniform(0, 1, (16, 2))
    
    # Ensure points are within bounds
    points = np.clip(points, 0, 1)
    
    return points
