                    tol=1e-6
                )

                if result.success:
                    optimized_points = result.x.reshape(-1, self.dimensions)
                    optimized_points = np.clip(optimized_points, 0, 1)
                    ratio = self._calculate_min_max_ratio(optimized_points)

                    if ratio > best_ratio:
                        best_ratio = ratio
                        best_points = optimized_points.copy()

            except Exception:
                continue

        return best_points, best_ratio

    def optimize(self) -> np.ndarray:
        """Main optimization process with enhanced strategies."""
        if self._time_remaining() < 10:
            # Fallback to simple approach
            return np.random.rand(self.n_points, self.dimensions)

        # Step 1: Generate initial population
        initial_configs = self._generate_initial_population()

        # Step 2: Evaluate initial configurations
        evaluated_configs = self._evaluate_initial_configurations(initial_configs)

        # Step 3: Select best initial configuration
        best_initial_config = evaluated_configs[0][0]
        best_ratio = evaluated_configs[0][1]

        # Step 4: Apply global search to the best initial
        if self._time_remaining() > 25:
            global_points, global_ratio = self._global_search(best_initial_config, max_iter=150)

            if global_ratio > best_ratio:
                best_ratio = global_ratio
                best_initial_config = global_points

        # Step 5: Apply adaptive local refinement
        if self._time_remaining() > 20:
            # Adaptive refinement with multiple strategies
            adaptive_points, adaptive_ratio = self._adaptive_local_refinement(best_initial_config, max_iter=200)
            if adaptive_ratio > best_ratio:
                best_ratio = adaptive_ratio
                best_initial_config = adaptive_points

        # Step 6: Additional refinement with different approach
        if self._time_remaining() > 15:
            # Try another approach with more iterations
            final_points, final_ratio = self._adaptive_local_refinement(best_initial_config, max_iter=250)
            if final_ratio > best_ratio:
                best_ratio = final_ratio
                best_initial_config = final_points

        # Step 7: Final refinement with potentially more iterations
        if self._time_remaining() > 10:
            # More thorough refinement if time allows
            final_points, final_ratio = self._adaptive_local_refinement(best_initial_config, max_iter=300)
            if final_ratio > best_ratio:
                best_ratio = final_ratio
                best_initial_config = final_points

        return best_initial_config
