        # 6. Fibonacci spiral pattern for enhanced distribution
        fibonacci_points = []
        golden_ratio = (1 + np.sqrt(5)) / 2
        for i in range(16):
            angle = i * 2 * np.pi / golden_ratio  # Golden angle increment
            radius = 0.4 * np.sqrt(i / 15.0) if i > 0 else 0.0
            x = 0.5 + radius * np.cos(angle)
            y = 0.5 + radius * np.sin(angle)
            fibonacci_points.append([x, y])
        configs.append(np.array(fibonacci_points).clip(0, 1))
