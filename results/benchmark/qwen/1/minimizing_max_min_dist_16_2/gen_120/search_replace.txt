            # Track objective values for early stopping
            return -fitness_function(x)  # Negative because minimize function

        # Add callback for tracking progress
        objective_values = []
        def callback(xk):

            objective_values.append(obj_val)

            if len(objective_values) > 10:

                if recent_improvement < 1e-12:
                    return True  # Stop optimization
            return False

                'gtol': 1e-12,
                'maxcor': 20  # Number of corrections to store
            },
            callback=callback if max_iter > 50 else None
        )
