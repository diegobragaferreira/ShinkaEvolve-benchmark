The modification ensures consistent behavior under edge cases and helps avoid issues related to sparse or ill-conditioned distance matrices, particularly useful for configurations approaching symmetry or collinearity.

    def calculate_min_max_ratio(points):
        """Calculate the ratio of minimum to maximum distance"""
        if len(points) < 2:
            return 0.0
        # Use squareform for better numerical stability
        distances = squareform(pdist(points))
        # Mask diagonal elements (distance to self is 0)
        np.fill_diagonal(distances, np.inf)
        # Get min and max distances
        d_min = np.min(distances)
        d_max = np.max(distances)
        # Avoid division by zero or invalid ratios
        if d_max <= 0 or np.isinf(d_min):
            return 0.0
        return d_min / d_max
