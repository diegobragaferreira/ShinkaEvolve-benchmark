    def calculate_min_max_ratio(points):
        """Calculate the ratio of minimum to maximum distance with improved numerical stability"""
        # Apply boundary padding to ensure points are safely away from edges
        eps = 1e-8
        padded_points = np.clip(points, eps, 1-eps)
        distances = pdist(padded_points)
        if len(distances) == 0 or np.max(distances) <= 0:
            return 0.0
        d_min = np.min(distances)
        d_max = np.max(distances)
        if d_max <= 0:
            return 0.0
        return d_min / d_max
