    def calculate_min_max_ratio(self, points: np.ndarray) -> float:
        """Calculate the ratio of minimum to maximum distance between all point pairs."""
        if len(points) < 2:
            return 0.0

        # Use squareform for better numerical stability
        distances = squareform(pdist(points))

        # Handle edge cases
        # Find off-diagonal elements (non-zero distances)
        non_zero_distances = distances[distances > 0]

        if len(non_zero_distances) == 0:
            return 0.0

        d_min = np.min(non_zero_distances)
        d_max = np.max(non_zero_distances)

        # Avoid division by zero
        if d_max <= 0:
            return 0.0

        return d_min / d_max
