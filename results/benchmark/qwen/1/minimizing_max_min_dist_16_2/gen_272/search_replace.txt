        # 7. Thomson problem-inspired initialization (electrostatic equilibrium)
        # Generate points that approximately solve the Thomson problem for 16 electrons on a sphere
        # Projected to 2D using stereographic projection
        thomson_points = []
        # Use known approximate solutions for 16 points on sphere
        # These are based on the fact that 16 points arranged to minimize energy
        # give good uniformity
        np.random.seed(42)
        for i in range(16):
            # Generate points using Fibonacci-based distribution on sphere
            # Then project to 2D
            phi = np.arccos(-1 + (2 * i) / 15.0)  # Polar angle
            theta = np.sqrt(16 * 15) * phi  # Azimuthal angle

            # Convert to Cartesian coordinates on unit sphere
            x_sphere = np.sin(phi) * np.cos(theta)
            y_sphere = np.sin(phi) * np.sin(theta)
            z_sphere = np.cos(phi)

            # Stereographic projection from south pole to plane
            # This maps the sphere to the plane in a way that preserves angles
            x_proj = x_sphere / (1 - z_sphere)
            y_proj = y_sphere / (1 - z_sphere)

            # Scale and shift to [0.1, 0.9] x [0.1, 0.9] to avoid boundaries
            x_2d = 0.4 + 0.4 * x_proj / np.sqrt(2)  # Scale appropriately
            y_2d = 0.4 + 0.4 * y_proj / np.sqrt(2)

            # Clip to ensure within bounds
            thomson_points.append([np.clip(x_2d, 0.1, 0.9), np.clip(y_2d, 0.1, 0.9)])

        configs.append(np.array(thomson_points).clip(0, 1))
