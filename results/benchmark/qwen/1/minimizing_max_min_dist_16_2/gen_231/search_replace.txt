<DESCRIPTION>
The current implementation uses fixed noise levels for perturbing initial points during multi-start optimization. This approach may not be optimal for all scenarios - sometimes too much noise can disrupt good configurations while too little may get stuck in local optima. I'll implement adaptive noise scaling that adjusts based on the quality of previous optimizations, allowing for more intelligent exploration of the search space.

    def run_multi_start_optimization(self):
        """Run optimization with multiple restarts and strategies."""
        # Generate initial strategies
        strategies = self.generate_initial_strategies()

        # Find the best initial configuration
        best_initial, initial_ratio = self.evaluate_all_strategies(strategies)

        # Initialize best results
        self.best_points = best_initial.copy()
        self.best_ratio = initial_ratio

        # Track optimization history for adaptive noise
        optimization_history = []

        # Multi-start optimization with different initial variations
        for restart in range(5):
            # Generate new variation of the initial points with adaptive noise
            np.random.seed(restart + 1000)
            perturbed_points = best_initial.copy()

            # Adaptive noise scaling based on performance history
            if len(optimization_history) > 0:
                # Calculate average improvement from previous runs
                avg_improvement = np.mean([improvement for _, improvement in optimization_history[-3:]])
                # Start with base noise but scale based on performance
                noise_scale = max(0.005, 0.03 - avg_improvement * 0.5)
            else:
                noise_scale = 0.03

            noise_level = noise_scale + restart * 0.005  # Gradually increasing noise
            perturbed_points += np.random.normal(0, noise_level, best_initial.shape)
            perturbed_points = np.clip(perturbed_points, 0, 1)

            # Optimize this variant
            optimized_points = self.robust_optimization(perturbed_points, max_evaluations=500)
            optimized_ratio = PointConfiguration(optimized_points).compute_min_max_ratio()

            # Record this optimization attempt
            improvement = optimized_ratio - self.best_ratio
            optimization_history.append((restart, improvement))

            if optimized_ratio > self.best_ratio:
                self.best_ratio = optimized_ratio
                self.best_points = optimized_points.copy()

        # Final optimization on the best configuration found
        final_points = self.robust_optimization(self.best_points, max_evaluations=300)
        final_ratio = PointConfiguration(final_points).compute_min_max_ratio()

        # One final refinement attempt with adaptive noise
        np.random.seed(9999)
        # Use smaller adaptive noise for final refinement
        final_noise = max(0.001, 0.005 * (1.0 - min(len(optimization_history), 5) * 0.1))
        last_attempt = final_points + np.random.normal(0, final_noise, final_points.shape)
        last_attempt = np.clip(last_attempt, 0, 1)
        refined_final = self.robust_optimization(last_attempt, max_evaluations=200)
        refined_ratio = PointConfiguration(refined_final).compute_min_max_ratio()

        if refined_ratio > final_ratio:
            return refined_final
        else:
            return final_points
