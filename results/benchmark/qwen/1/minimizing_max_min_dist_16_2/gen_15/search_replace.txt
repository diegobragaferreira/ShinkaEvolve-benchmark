import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist


def min_max_dist_dim2_16() -> np.ndarray:
    """
    Creates 16 points in 2 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (16,2) containing the (x,y) coordinates of the 16 points.
    """

    def objective(x):

        points = x.reshape(-1, 2)

        # Compute pairwise distances
        distances = pdist(points)

        # Compute min and max distances
        d_min = np.min(distances)
        d_max = np.max(distances)

        # Return negative ratio to maximize (since we're minimizing the negative)
        if d_max == 0:
            return -1.0

    # Arrange points in a roughly hexagonal pattern within the unit square
    np.random.seed(42)

    # Using a modified hexagonal grid pattern
    n_points = 16
    points = np.zeros((n_points, 2))

    # Create a hexagonal-like arrangement
    rows = 4
    cols = 4

    # Hexagonal grid with slight randomness to avoid perfect symmetry
    row_spacing = 1.0 / (rows - 1) if rows > 1 else 1.0
    col_spacing = 1.0 / (cols - 1) if cols > 1 else 1.0

        for j in range(cols):

                break

            points[i * cols + j] = [x + np.random.normal(0, 0.01), y + np.random.normal(0, 0.01)]

    # Ensure points stay within bounds
    points = np.clip(points, 0, 1)

    # Flatten for optimization
    x0 = points.flatten()

    # Define bounds for each coordinate (0 to 1 for both x and y)
    bounds = [(0, 1) for _ in range(32)]  # 16 points * 2 coordinates each

    # First stage: Use differential evolution for global optimization
    de_result = differential_evolution(
        objective,
        bounds,
        seed=42,
        maxiter=100,  # Reduced iterations for faster global search
        popsize=15,   # Standard population size
        tol=1e-6,     # Reasonable tolerance for global search
        recombination=0.7,
        mutation=(0.5, 1.0),
        disp=False
    )

    # Second stage: Local refinement with L-BFGS-B using the DE result as starting point
    # This helps achieve better convergence to a locally optimal solution
    lbfgs_result = minimize(
        objective,
        de_result.x,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12},
        callback=None
    )

    # Return the refined points from the local optimization
    points = lbfgs_result.x.reshape(-1, 2)

    return points
