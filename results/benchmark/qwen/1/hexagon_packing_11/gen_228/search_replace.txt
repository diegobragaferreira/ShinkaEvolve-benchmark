def local_refinement(individual, max_iter=50, tolerance=1e-6):
    """Apply local refinement to improve individual quality"""
    # More sophisticated local search with adaptive steps and better convergence
    best_individual = individual.copy()
    best_fitness = evaluate_solution(best_individual)

    # Track improvement history for adaptive behavior
    improvement_history = []
    stagnation_count = 0

    for iteration in range(max_iter):
        improved = False
        # Store candidates to test in batch to avoid excessive function calls
        candidate_changes = []

        # Try multiple perturbations per hexagon
        for i in range(len(best_individual)):
            for j in range(3):  # x, y, angle
                original_value = best_individual[i][j]
                # Try several step sizes for better exploration
                step_sizes = [0.01, 0.02, 0.05]
                for step_size in step_sizes:
                    # Try positive and negative steps
                    for step in [-step_size, step_size]:
                        test_individual = best_individual.copy()
                        if j < 2:  # x or y
                            test_individual[i][j] = original_value + step
                        else:  # angle
                            test_individual[i][j] = (original_value + step) % 360
                        candidate_changes.append((test_individual, i, j, step))

        # Evaluate all candidates
        best_candidate = None
        best_candidate_fitness = best_fitness

        for test_individual, i, j, step in candidate_changes:
            fitness = evaluate_solution(test_individual)
            if fitness > best_candidate_fitness:
                best_candidate_fitness = fitness
                best_candidate = test_individual

        # Accept the best candidate if it improves fitness
        if best_candidate is not None and best_candidate_fitness > best_fitness:
            best_individual = best_candidate
            best_fitness = best_candidate_fitness
            improved = True
            improvement_history.append(best_fitness)
            stagnation_count = 0
        else:
            stagnation_count += 1
            improvement_history.append(best_fitness)

        # Early termination conditions
        if not improved:
            if stagnation_count > 15:  # No improvement in last 15 iterations
                break
            # Check if we're approaching a local optimum
            if len(improvement_history) >= 5:
                recent_improvements = improvement_history[-5:]
                if max(recent_improvements) - min(recent_improvements) < tolerance:
                    break

    return best_individual
