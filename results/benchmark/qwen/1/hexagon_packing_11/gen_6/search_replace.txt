def hexagon_vertices(center_x, center_y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given center, angle, and side length."""
    angle_rad = np.deg2rad(angle_deg)
    # Vertices of a regular hexagon with side length 1 centered at origin
    base_vertices = np.array([
        [1.0, 0.0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1.0, 0.0],
        [-0.5, -np.sqrt(3)/2],
        [0.5, -np.sqrt(3)/2]
    ])

    # Rotate and translate
    cos_a = np.cos(angle_rad)
    sin_a = np.sin(angle_rad)
    rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
    rotated_vertices = base_vertices @ rotation_matrix.T
    return rotated_vertices + np.array([center_x, center_y])

def check_hexagon_containment(hex_vertices, outer_center_x, outer_center_y, outer_radius):
    """Check if all vertices of a hexagon are inside the outer hexagon."""
    outer_vertices = hexagon_vertices(outer_center_x, outer_center_y, 0, outer_radius)

    # Check if each vertex of the inner hexagon is inside the outer hexagon
    for vertex in hex_vertices:
        x, y = vertex
        # Check if point is inside hexagon using cross product method
        # For a regular hexagon centered at origin with radius r, point is inside if
        # distance to center is <= r and it's on the correct side of all edges
        dist_to_center = np.sqrt((x - outer_center_x)**2 + (y - outer_center_y)**2)
        if dist_to_center > outer_radius:
            return False

    return True

def hexagon_distance(v1, v2):
    """Calculate minimum distance between two hexagons."""
    # Simplified distance calculation - return distance between centroids
    # In practice, would compute actual minimum distance between polygons
    return np.linalg.norm(np.mean(v1, axis=0) - np.mean(v2, axis=0))

def check_overlap(hex1_vertices, hex2_vertices):
    """Check if two hexagons overlap using distance-based approximation."""
    # If distance between centers is less than sum of radii, they might overlap
    center1 = np.mean(hex1_vertices, axis=0)
    center2 = np.mean(hex2_vertices, axis=0)
    distance = np.linalg.norm(center1 - center2)

    # For unit hexagons, the minimum distance between centers for non-overlapping is 2
    return distance < 1.99  # Allow small tolerance
