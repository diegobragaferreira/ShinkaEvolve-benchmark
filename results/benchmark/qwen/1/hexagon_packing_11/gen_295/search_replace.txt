    def create_initial_configuration(self):
        """Create a better initial configuration using Voronoi-inspired arrangement"""
        # Import required modules here to avoid circular imports
        from scipy.spatial import Voronoi
        import random

        # Generate Voronoi-based initial configuration
        np.random.seed(42)

        # Generate random points that will serve as Voronoi seeds
        n_seeds = 15  # More seeds than hexagons for better distribution
        points = np.random.uniform(-10, 10, size=(n_seeds, 2))

        try:
            # Compute Voronoi diagram
            vor = Voronoi(points)

            # Find finite Voronoi cells and their centroids
            centroids = []
            for i, region in enumerate(vor.regions):
                if len(region) > 0 and -1 not in region:
                    # Get vertices of this region
                    vertices = [vor.vertices[j] for j in region if j >= 0]
                    if vertices:
                        # Calculate centroid
                        centroid = np.mean(vertices, axis=0)
                        # Only keep centroids within reasonable bounds
                        if -15 <= centroid[0] <= 15 and -15 <= centroid[1] <= 15:
                            centroids.append(centroid)

            # If we have enough centroids, use them; otherwise generate random points
            if len(centroids) >= 11:
                selected_centroids = centroids[:11]
            else:
                # Fall back to random points if Voronoi didn't work well
                selected_centroids = np.random.uniform(-10, 10, size=(11, 2))
        except:
            # If Voronoi computation fails, fall back to deterministic arrangement
            selected_centroids = np.random.uniform(-10, 10, size=(11, 2))

        # Convert to list of positions
        base_positions = [list(centroid) for centroid in selected_centroids]

        # Add some randomness to avoid symmetric solutions
        for i in range(len(base_positions)):
            base_positions[i][0] += np.random.uniform(-0.5, 0.5)
            base_positions[i][1] += np.random.uniform(-0.5, 0.5)

        # Add random rotations to break symmetry
        initial_angles = [np.random.uniform(0, 360) for _ in range(11)]
        return base_positions, initial_angles
