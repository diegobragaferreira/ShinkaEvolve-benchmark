import numpy as np
from scipy.spatial.distance import cdist
from shapely.geometry import Polygon, Point
from shapely.ops import unary_union
import time
from functools import lru_cache

# Constants
UNIT_HEX_RADIUS = 1.0  # Side length of unit hexagon
HEX_VERTICES = 6
HEX_ANGLE = 2 * np.pi / HEX_VERTICES

def generate_hexagon_vertices(center, radius, rotation=0):
    """Generate vertices of a regular hexagon."""
    angles = np.linspace(0, 2*np.pi, HEX_VERTICES + 1) + rotation
    x = center[0] + radius * np.cos(angles)
    y = center[1] + radius * np.sin(angles)
    return list(zip(x, y))

def get_hexagon_polygon(center, radius, rotation=0):
    """Create a Shapely Polygon for a hexagon."""
    vertices = generate_hexagon_vertices(center, radius, rotation)
    return Polygon(vertices)

def check_containment(hexagon_poly, outer_hex_poly):
    """Check if hexagon is fully contained in outer hexagon."""
    return outer_hex_poly.contains(hexagon_poly)

def check_overlap(hex1_poly, hex2_poly):
    """Check if two hexagons overlap."""
    return hex1_poly.intersects(hex2_poly)

@lru_cache(maxsize=1000)
def calculate_outer_hexagon_size(inner_hex_data):
    """
    Calculate minimum outer hexagon size needed to contain all inner hexagons.
    Returns (outer_radius, outer_center_x, outer_center_y).
    """
    # Convert inner hex data to polygons
    hexagons = []
    for i, (cx, cy, rot) in enumerate(inner_hex_data):
        hex_poly = get_hexagon_polygon((cx, cy), UNIT_HEX_RADIUS, rot)
        hexagons.append(hex_poly)

    # Find bounding box of all hexagon centers
    centers = np.array([[h[0], h[1]] for h in inner_hex_data])
    min_x, max_x = centers[:, 0].min(), centers[:, 0].max()
    min_y, max_y = centers[:, 1].min(), centers[:, 1].max()

    # Calculate approximate outer hexagon size
    # We need to ensure all vertices of inner hexagons are within outer hexagon
    outer_radius = 0

    # Check vertex containment for worst-case scenario
    for i, (cx, cy, rot) in enumerate(inner_hex_data):
        vertices = generate_hexagon_vertices((cx, cy), UNIT_HEX_RADIUS, rot)
        for vx, vy in vertices:
            dist = np.sqrt((vx - 0)**2 + (vy - 0)**2)  # Assuming outer hexagon centered at origin
            outer_radius = max(outer_radius, dist)

    # Add some margin for safety
    outer_radius *= 1.1

    return outer_radius

def evaluate_fitness(individual):
    """
    Evaluate fitness of a solution (inverse of outer hexagon size).
    individual: array of shape (11, 3) where each row is (x, y, angle)
    """
    # Convert to tuple for hashing
    individual_tuple = tuple(tuple(row) for row in individual)

    # Calculate outer hexagon size
    outer_radius = calculate_outer_hexagon_size(individual_tuple)

    # Return inverse of outer radius (higher is better)
    return 1.0 / outer_radius if outer_radius > 0 else 0

def create_individual():
    """Create a new random individual."""
    # Start with center hexagon
    individual = [[0.0, 0.0, 0.0]]

    # Add remaining hexagons with some geometric reasoning
    # Try to place them around the center in a hexagonal pattern
    for i in range(1, 11):
        # Place in hexagonal pattern around center
        angle = (i - 1) * (2 * np.pi / 6)  # 60-degree steps
        distance = 2.0  # Start with reasonable distance

        # Add some randomness
        dx = distance * np.cos(angle) + np.random.uniform(-0.5, 0.5)
        dy = distance * np.sin(angle) + np.random.uniform(-0.5, 0.5)

        # Random orientation
        rotation = np.random.uniform(0, 2 * np.pi)

        individual.append([dx, dy, rotation])

    return np.array(individual)

def create_initial_population(pop_size):
    """Create initial population."""
    return [create_individual() for _ in range(pop_size)]

def mutate_individual(individual, mutation_rate=0.1):
    """Mutate an individual."""
    mutated = individual.copy()

    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            # Mutate position slightly
            mutated[i][0] += np.random.normal(0, 0.2)
            mutated[i][1] += np.random.normal(0, 0.2)

            # Mutate rotation
            mutated[i][2] += np.random.normal(0, 0.2)
            mutated[i][2] = mutated[i][2] % (2 * np.pi)

    return mutated

def crossover_individuals(parent1, parent2):
    """Crossover two individuals."""
    child1 = parent1.copy()
    child2 = parent2.copy()

    # Single-point crossover on positions
    crossover_point = np.random.randint(1, len(parent1))

    # Swap segments
    child1[:crossover_point] = parent2[:crossover_point]
    child2[:crossover_point] = parent1[:crossover_point]

    return child1, child2

def optimize_hexagon_packing():
    """Main optimization function."""
    # Parameters
    pop_size = 50
    generations = 100
    elite_size = 5

    # Create initial population
    population = create_initial_population(pop_size)

    best_fitness = 0
    best_individual = None

    # Optimization loop
    for gen in range(generations):
        # Evaluate fitness for all individuals
        fitness_scores = [(ind, evaluate_fitness(ind)) for ind in population]

        # Sort by fitness
        fitness_scores.sort(key=lambda x: x[1], reverse=True)

        # Track best
        current_best_fitness = fitness_scores[0][1]
        if current_best_fitness > best_fitness:
            best_fitness = current_best_fitness
            best_individual = fitness_scores[0][0].copy()

        # Print progress
        if gen % 20 == 0:
            print(f"Generation {gen}: Best fitness = {best_fitness:.6f}")

        # Create new population
        new_population = []

        # Elitism - keep best individuals
        for i in range(elite_size):
            new_population.append(fitness_scores[i][0].copy())

        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness_scores[i][1] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]

            parent1 = population[winner_idx]
            parent2 = population[np.random.choice(len(population))]

            child1, child2 = crossover_individuals(parent1, parent2)

            child1 = mutate_individual(child1)
            child2 = mutate_individual(child2)

            new_population.extend([child1, child2])

        # Trim to exact population size
        population = new_population[:pop_size]

    # Final refinement using local search
    refined_individual = best_individual.copy()
    original_fitness = evaluate_fitness(best_individual)

    # Try small local perturbations
    for _ in range(100):
        candidate = refined_individual.copy()
        # Small random perturbations
        for i in range(len(candidate)):
            if np.random.random() < 0.3:
                candidate[i][0] += np.random.normal(0, 0.05)
                candidate[i][1] += np.random.normal(0, 0.05)
                candidate[i][2] += np.random.normal(0, 0.05)
                candidate[i][2] = candidate[i][2] % (2 * np.pi)

        candidate_fitness = evaluate_fitness(candidate)
        if candidate_fitness > original_fitness:
            refined_individual = candidate
            original_fitness = candidate_fitness

    return refined_individual, original_fitness
