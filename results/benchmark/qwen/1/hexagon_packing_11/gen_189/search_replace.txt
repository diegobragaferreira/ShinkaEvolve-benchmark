def adaptive_mutation_schedule(iteration, maxiter):
    """Adaptive mutation rate that decreases over time for better convergence"""
    # Start with higher mutation rate for exploration
    # Gradually decrease to favor exploitation
    return 0.8 - (0.7 * iteration / maxiter)

def local_optimization_refinement(initial_params, bounds):
    """Apply local optimization to refine the solution"""
    from scipy.optimize import minimize

    def objective(params):
        return evaluate_solution(params)

    # Use L-BFGS-B for local refinement
    result = minimize(
        objective,
        initial_params,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 50, 'ftol': 1e-6},
        jac=False
    )

    return result.x if result.success else initial_params
