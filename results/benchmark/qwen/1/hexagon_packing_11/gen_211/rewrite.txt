# Set seeds for reproducibility
np.random.seed(42)

def generate_hexagon_vertices(center_x, center_y, angle_deg, side_length=1):
    """Generate vertices of a regular hexagon given center, angle, and side length"""
    angle_rad = np.radians(angle_deg)
    # Vertices of a regular hexagon with side length 1, centered at origin
    base_vertices = np.array([
        [1, 0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1, 0],
        [-0.5, -np.sqrt(3)/2],
        [0.5, -np.sqrt(3)/2]
    ])

    # Rotate and translate
    cos_a, sin_a = np.cos(angle_rad), np.sin(angle_rad)
    rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
    rotated_vertices = base_vertices @ rotation_matrix.T
    translated_vertices = rotated_vertices + np.array([center_x, center_y])

    return translated_vertices

def check_containment(hexagon_poly, outer_hex_poly):
    """Check if hexagon is fully contained within outer hexagon"""
    return outer_hex_poly.contains(hexagon_poly)

def check_overlap(hex1_poly, hex2_poly):
    """Check if two hexagons overlap"""
    return hex1_poly.intersects(hex2_poly)

def calculate_outer_hex_side_length(inner_hex_data):
    """Calculate minimum side length of outer hexagon needed to contain all inner hexagons"""
    # Get all vertices of all inner hexagons
    all_vertices = []
    for i in range(len(inner_hex_data)):
        center_x, center_y, angle = inner_hex_data[i]
        vertices = generate_hexagon_vertices(center_x, center_y, angle)
        all_vertices.extend(vertices)

    if len(all_vertices) == 0:
        return 10.0
        
    all_vertices = np.array(all_vertices)

    # Find bounding box
    min_x, max_x = np.min(all_vertices[:, 0]), np.max(all_vertices[:, 0])
    min_y, max_y = np.min(all_vertices[:, 1]), np.max(all_vertices[:, 1])

    # Calculate approximate side length 
    width = max_x - min_x
    height = max_y - min_y
    side_len_width = width / 2.0
    side_len_height = height / (np.sqrt(3))
    estimated_side_length = max(side_len_width, side_len_height) * 1.1  # Add small buffer

    return estimated_side_length

def create_hexagon_polygons(inner_hex_data):
    """Create shapely polygons for all inner hexagons"""
    hex_polygons = []
    for i in range(len(inner_hex_data)):
        center_x, center_y, angle = inner_hex_data[i]
        vertices = generate_hexagon_vertices(center_x, center_y, angle)
        hex_polygons.append(Polygon(vertices))
    return hex_polygons

def evaluate_fitness(individual_flat):
    """
    Evaluate fitness of solution - maximize 1/outer_hex_side_length.
    individual_flat: flattened array of shape (33,) containing [x1,y1,a1,x2,y2,a2,...,x11,y11,a11]
    Returns negative value because we want to maximize 1/R (minimize R)
    """
    try:
        # Reshape individual into hexagon data
        inner_hex_data = individual_flat.reshape(-1, 3)
        
        # Create polygons for all inner hexagons
        hex_polygons = create_hexagon_polygons(inner_hex_data)
        
        # Calculate outer hexagon side length
        outer_side_length = calculate_outer_hex_side_length(inner_hex_data)
        
        # Create outer hexagon polygon
        outer_vertices = generate_hexagon_vertices(0, 0, 0, outer_side_length)
        outer_polygon = Polygon(outer_vertices)
        
        # Check containment for all hexagons
        for poly in hex_polygons:
            if not check_containment(poly, outer_polygon):
                return 1e6  # Penalty for containment failure
        
        # Check overlaps between all pairs of hexagons
        for i in range(len(hex_polygons)):
            for j in range(i+1, len(hex_polygons)):
                if check_overlap(hex_polygons[i], hex_polygons[j]):
                    return 1e6  # Penalty for overlap
        
        # Return negative inverse of outer radius (since we're minimizing)
        return -1.0 / outer_side_length if outer_side_length > 0 else 1e6
        
    except Exception as e:
        return 1e6  # Penalty for any error

def optimize_single_pattern(initial_pattern):
    """Optimize a single pattern using L-BFGS with constraint handling"""
    # Flatten initial pattern
    initial_flat = initial_pattern.flatten()
    
    # Define bounds for optimization (positions: -10 to 10, angles: 0 to 360)
    bounds = []
    for _ in range(11):
        bounds.extend([(-10, 10), (-10, 10), (0, 360)])
    
    # Use L-BFGS optimization for fine-tuning
    try:
        result = minimize(
            evaluate_fitness,
            initial_flat,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6},
            callback=None
        )
        
        if result.success:
            optimized_result = result.x.reshape(-1, 3)
            return optimized_result
    except:
        pass
    
    return initial_pattern

def hexagon_packing_11():
    """
    Constructs a packing of 11 disjoint unit regular hexagons inside a larger regular hexagon, maximizing 1/outer_hex_side_length.
    Returns
        inner_hex_data: np.ndarray of shape (11,3), where each row is of the form (x, y, angle_degrees) containing the (x,y) coordinates and angle_degree of the respective inner hexagon.
        outer_hex_data: np.ndarray of shape (3,) of form (x,y,angle_degree) containing the (x,y) coordinates and angle_degree of the outer hexagon.
        outer_hex_side_length: float representing the side length of the outer hexagon.
    """
    start_time = time.time()
    max_time = 175  # Leave some margin for cleanup
    
    best_fitness = -1e6
    best_solution = None
    
    # Generate initial patterns
    initial_patterns = create_pattern_initialization()
    
    # Optimize each pattern
    for pattern in initial_patterns:
        if time.time() - start_time > max_time:
            break
            
        # Optimize this pattern
        optimized_pattern = optimize_single_pattern(pattern)
        
        # Evaluate fitness of optimized pattern
        fitness = evaluate_fitness(optimized_pattern.flatten())
        
        if fitness < best_fitness:
            best_fitness = fitness
            best_solution = optimized_pattern.copy()
    
    # If no good solution was found, fallback to a predefined good configuration
    if best_solution is None:
        best_solution = np.array([
            [0, 0, 0],
            [-2.5, 0, 0],
            [2.5, 0, 0],
            [-1.25, 2.17, 0],
            [1.25, 2.17, 0],
            [-1.25, -2.17, 0],
            [1.25, -2.17, 0],
            [-3.75, 2.17, 0],
            [3.75, 2.17, 0],
            [-3.75, -2.17, 0],
            [3.75, -2.17, 0],
        ])
    
    # Final evaluation to get precise outer side length
    final_fitness = evaluate_fitness(best_solution.flatten())
    outer_side_length = -1.0 / final_fitness if final_fitness != 0 else 10.0
    
    if outer_side_length > 100:
        outer_side_length = 10.0
    
    # Center the outer hexagon at origin
    outer_hex_data = np.array([0, 0, 0])

    return best_solution, outer_hex_data, outer_side_length
