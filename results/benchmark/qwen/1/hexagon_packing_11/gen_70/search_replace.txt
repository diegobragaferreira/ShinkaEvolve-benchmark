<DESCRIPTION>
The current implementation uses brute force pairwise collision detection which becomes inefficient as the search space grows. I'll implement a spatial indexing approach using a grid-based partitioning system to pre-filter potential collision pairs before doing expensive polygon intersection tests. This will significantly reduce the computational complexity of overlap checking, especially important for the 11-hexagon problem where many pairwise checks are performed.

This approach will:
1. Add a spatial hash grid to quickly identify nearby hexagons
2. Reduce overlap checking from O(n²) to approximately O(n) on average
3. Maintain the same accuracy while improving performance
4. Allow for more aggressive optimization strategies due to faster evaluations
</DESCRIPTION>

def build_spatial_grid(hex_polygons, cell_size=2.0):
    """Build a spatial grid for efficient collision detection"""
    grid = defaultdict(list)

    for i, poly in enumerate(hex_polygons):
        # Get bounding box of polygon
        bounds = poly.bounds
        min_x, min_y, max_x, max_y = bounds

        # Calculate grid cell indices
        min_cell_x = int(min_x // cell_size)
        max_cell_x = int(max_x // cell_size)
        min_cell_y = int(min_y // cell_size)
        max_cell_y = int(max_y // cell_size)

        # Add to all relevant cells
        for x in range(min_cell_x, max_cell_x + 1):
            for y in range(min_cell_y, max_cell_y + 1):
                grid[(x, y)].append(i)

    return grid

def get_potential_collisions(grid, hex_polygons, cell_size=2.0):
    """Get potential collision pairs using spatial grid"""
    collisions = set()

    for i, poly in enumerate(hex_polygons):
        bounds = poly.bounds
        min_x, min_y, max_x, max_y = bounds

        # Calculate grid cell indices
        min_cell_x = int(min_x // cell_size)
        max_cell_x = int(max_x // cell_size)
        min_cell_y = int(min_y // cell_size)
        max_cell_y = int(max_y // cell_size)

        # Check all cells that might contain neighbors
        for x in range(min_cell_x, max_cell_x + 1):
            for y in range(min_cell_y, max_cell_y + 1):
                for j in grid[(x, y)]:
                    if i < j:  # Avoid duplicate pairs and self-checks
                        collisions.add((i, j))

    return collisions
