<DESCRIPTION>
Adding spatial indexing to improve collision detection efficiency. The current implementation uses O(n²) pairwise intersection checks which becomes very slow as the number of hexagons increases. By implementing a simple grid-based spatial index, we can significantly reduce the number of expensive intersection tests needed. This will allow for more effective optimization and potentially better solutions within the time constraints.

def hexagon_intersects(hex1_vertices, hex2_vertices):
    """Check if two hexagons intersect using Shapely."""
    poly1 = Polygon(hex1_vertices)
    poly2 = Polygon(hex2_vertices)
    return poly1.intersects(poly2)

def build_spatial_index(hex_data, cell_size=5.0):
    """Build a simple grid-based spatial index for fast neighbor lookup."""
    index = defaultdict(list)
    for i, hex_info in enumerate(hex_data):
        center_x, center_y = hex_info[0], hex_info[1]
        grid_x = int(center_x // cell_size)
        grid_y = int(center_y // cell_size)
        index[(grid_x, grid_y)].append(i)
    return index

def get_potential_collisions(hex_data, spatial_index, cell_size=5.0):
    """Get pairs of hexagons that might collide based on spatial indexing."""
    collisions = set()
    for i, hex_info in enumerate(hex_data):
        center_x, center_y = hex_info[0], hex_info[1]
        grid_x = int(center_x // cell_size)
        grid_y = int(center_y // cell_size)

        # Check neighboring cells for potential collisions
        for dx in [-1, 0, 1]:
            for dy in [-1, 0, 1]:
                key = (grid_x + dx, grid_y + dy)
                if key in spatial_index:
                    for j in spatial_index[key]:
                        if i < j:  # Avoid duplicate pairs and self-comparison
                            collisions.add((i, j))
    return collisions
