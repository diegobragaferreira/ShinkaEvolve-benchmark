<DESCRIPTION>
This improvement modifies the differential evolution approach to use adaptive parameters that dynamically adjust based on optimization progress. The key insight is that early generations benefit from high exploration (high mutation rate, high crossover rate) while later generations benefit from high exploitation (low mutation rate, high crossover rate). This adaptive scheduling helps avoid premature convergence and improves overall optimization performance.

The changes include:
1. Implementing adaptive DE parameters with dynamic mutation and crossover rates
2. Adding convergence tracking to detect when to adjust parameters
3. Improving the initial population generation to be more diverse
4. Enhancing the local refinement process with better cooling schedule

These changes aim to push the solution closer to the benchmark of 0.2544 by enabling better exploration in early stages and more focused exploitation in later stages.

    # Phase 1: Generate geometrically informed initial solution

    initial_angles = [0.0] * 11

    # Phase 2: Use a hybrid approach with adaptive differential evolution
    bounds = []
    # Position bounds
    for _ in range(22):
        bounds.append((-8.0, 8.0))  # Tighter bounds for faster convergence
    # Angle bounds
    for _ in range(11):
        bounds.append((0.0, 360.0))

    try:

        def adaptive_de_strategy(func, bounds, maxiter=50, popsize=15, seed=42):
            # Initialize parameters with high exploration values
            mutation_rate = 0.9
            crossover_rate = 0.9
            patience_counter = 0
            last_best_fitness = float('inf')
            improvement_threshold = 1e-5

            # Initialize population
            np.random.seed(seed)
            population = []
            for i in range(popsize):
                individual = []
                for lower, upper in bounds:
                    individual.append(np.random.uniform(lower, upper))
                population.append(individual)

            history = []

            for generation in range(maxiter):
                # Evaluate fitness
                fitnesses = [func(ind) for ind in population]
                best_fitness = min(fitnesses)
                history.append(best_fitness)

                # Check for improvement
                if len(history) > 1 and abs(history[-2] - best_fitness) < improvement_threshold:
                    patience_counter += 1
                else:
                    patience_counter = 0

                # Adaptive parameter adjustment based on progress
                if patience_counter > 5 and generation > 10:
                    # Decrease mutation rate and increase crossover rate to exploit
                    mutation_rate = max(0.2, mutation_rate * 0.95)
                    crossover_rate = min(0.9, crossover_rate + 0.05)
                elif generation > 20 and len(history) > 10:
                    # Further tune if no significant improvement in recent history
                    if abs(history[-10] - history[-1]) < 1e-6:
                        mutation_rate = max(0.1, mutation_rate * 0.9)
                        crossover_rate = min(0.95, crossover_rate + 0.02)

                # Select parents
                tournament_size = 3
                selected = []
                for _ in range(popsize):
                    tournament = [np.random.choice(population) for _ in range(tournament_size)]
                    tournament_fitness = [func(ind) for ind in tournament]
                    winner = tournament[np.argmin(tournament_fitness)]
                    selected.append(winner)

                new_population = []
                for i in range(popsize):

                    a, b, c = np.random.choice(selected, 3, replace=False)
                    mutant = []
                    for j in range(len(bounds)):
                        if np.random.rand() < mutation_rate:
                            mutant.append(a[j] + 0.5 * (b[j] - c[j]))
                        else:
                            mutant.append(selected[i][j])

                    # Boundary handling
                    for j, (lower, upper) in enumerate(bounds):
                        mutant[j] = max(lower, min(upper, mutant[j]))

                    # Crossover
                    trial = []
                    for j in range(len(bounds)):
                        if np.random.rand() < crossover_rate:
                            trial.append(mutant[j])
                        else:
                            trial.append(selected[i][j])

                    new_population.append(trial)

                population = new_population

            # Return best individual
            fitnesses = [func(ind) for ind in population]
            best_idx = np.argmin(fitnesses)
            return population[best_idx]

        temperature = 5.0

        max_iterations = 500

        for iteration in range(max_iterations):
            if temperature < min_temperature:

            for _ in range(3):

                        best_positions[hex_idx] = old_pos
                        best_angles[hex_idx] = old_angle

                # Check if we need to cool down
                if iteration % 20 == 0:
