def create_spatial_index(hex_vertices_list):
    """Create a spatial index for fast collision detection using optimized grid structure"""
    # Create bounding boxes for all hexagons
    boxes = []
    for i, vertices in enumerate(hex_vertices_list):
        box = get_bounding_box(vertices)
        boxes.append((box, i))

    # Optimize grid cell size based on hexagon properties
    # Hexagon diameter is 2, so we use a cell size slightly larger than hexagon diameter
    cell_size = 2.5  # Slightly larger than hexagon diameter to minimize overlap checks
    spatial_index = defaultdict(list)

    for box, idx in boxes:
        # Grid cell coordinates - use floor division for proper cell assignment
        min_x, min_y, max_x, max_y = box
        cell_min_x = int(min_x // cell_size)
        cell_min_y = int(min_y // cell_size)
        cell_max_x = int(max_x // cell_size)
        cell_max_y = int(max_y // cell_size)

        # Add to all relevant grid cells
        for cx in range(cell_min_x, cell_max_x + 1):
            for cy in range(cell_min_y, cell_max_y + 1):
                spatial_index[(cx, cy)].append(idx)

    return spatial_index
