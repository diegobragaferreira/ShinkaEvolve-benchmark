2. Creating hexagon configurations based on these Voronoi seeds with strategic spacing
3. Combining the Voronoi-based configurations with existing initialization methods to increase diversity

def create_initial_population(pop_size, n_inner_hex):
    """Create initial population with diverse and strategically placed arrangements"""
    population = []

    # Generate multiple good starting solutions
    for _ in range(pop_size):
        # Create a variety of initial configurations
        config_type = random.randint(0, 3)

        if config_type == 0:
            # Honeycomb-like arrangement (most promising)
            individual = []
            # Center hexagon
            individual.append([0.0, 0.0, random.uniform(0, 360)])

            # First ring around center (6 hexagons)
            for i in range(6):
                angle = i * 60
                center_x = 2.0 * np.cos(np.radians(angle))
                center_y = 2.0 * np.sin(np.radians(angle))
                individual.append([center_x, center_y, random.uniform(0, 360)])

            # Second ring (12 hexagons) - fill remaining spots
            for i in range(5):  # Just 5 more to make 11 total
                angle = i * 30  # Different angles for second ring
                center_x = 3.5 * np.cos(np.radians(angle))
                center_y = 3.5 * np.sin(np.radians(angle))
                individual.append([center_x, center_y, random.uniform(0, 360)])

            # Pad if necessary
            while len(individual) < n_inner_hex:
                individual.append([random.uniform(-4, 4), random.uniform(-4, 4), random.uniform(0, 360)])

        elif config_type == 1:
            # Spiral arrangement
            individual = []
            # Center hexagon
            individual.append([0.0, 0.0, random.uniform(0, 360)])

            # Spiral pattern outward
            radius_step = 1.5
            angle_step = 30
            current_radius = radius_step
            current_angle = 0

                center_x = current_radius * np.cos(np.radians(current_angle))
                center_y = current_radius * np.sin(np.radians(current_angle))
                individual.append([center_x, center_y, random.uniform(0, 360)])
                current_angle += angle_step
                if current_angle >= 360:
                    current_angle = 0
                    current_radius += radius_step

        elif config_type == 2:
            # Diamond pattern
            individual = []
            # Center hexagon
            individual.append([0.0, 0.0, random.uniform(0, 360)])

            # Positions arranged in diamond pattern
            positions = [

                (-1.25, 1.25), (1.25, 1.25), (-1.25, -1.25), (1.25, -1.25),
                (-3.75, 0), (3.75, 0)
            ]

            # Pad if necessary
            while len(individual) < n_inner_hex:
                individual.append([random.uniform(-4, 4), random.uniform(-4, 4), random.uniform(0, 360)])

            # Random placement with more structured positioning

            # Center hexagon
            individual.append([0.0, 0.0, random.uniform(0, 360)])

            # Place hexagons in strategic clusters
            for i in range(1, n_inner_hex):
                # Cluster around specific regions
                if i <= 3:  # Near center

def generate_voronoi_initialization(n_inner_hex, attempts=5):
    """
    Generate initial configuration based on Voronoi-like distribution to ensure good spatial spread
    """
    # Use a more sophisticated approach: generate pseudo-random points with minimal distance constraints
    # This mimics a Voronoi diagram by ensuring minimal proximity between points

    points = []
    max_attempts = 1000

    for attempt in range(max_attempts):
        # Generate a random point in a reasonable area
        angle = random.uniform(0, 2 * np.pi)
        radius = random.uniform(0, 4.0)  # Within reasonable bounds
        x = radius * np.cos(angle)
        y = radius * np.sin(angle)

        # Check minimum distance to existing points
        valid = True
        for px, py in points:
            dist = np.sqrt((x - px)**2 + (y - py)**2)
            if dist < 1.5:  # Minimum distance constraint
                valid = False
                break

        if valid:
            points.append((x, y))
            if len(points) >= n_inner_hex:
                break

    # If we couldn't generate enough points, fill with random ones
    while len(points) < n_inner_hex:
        points.append((random.uniform(-4, 4), random.uniform(-4, 4)))

    # Create individual from points
    individual = []
    for i, (x, y) in enumerate(points[:n_inner_hex]):
        individual.append([x, y, random.uniform(0, 360)])

    return individual
