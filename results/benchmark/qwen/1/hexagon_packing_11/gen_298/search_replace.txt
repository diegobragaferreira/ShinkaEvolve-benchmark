def build_hierarchical_spatial_grid(hexagons, coarse_grid_size=2.8, fine_grid_size=1.4):
    """Build hierarchical spatial grid with adaptive cell sizing based on density"""
    # First, build coarse grid for distant regions
    coarse_grid = defaultdict(list)
    fine_grid = defaultdict(list)

    # Calculate average spacing to determine when to use fine grid
    if len(hexagons) > 1:
        positions = np.array([hex_poly.centroid.coords[0] for hex_poly in hexagons])
        # Compute pairwise distances to estimate density
        distances = cdist(positions, positions)
        np.fill_diagonal(distances, np.inf)
        min_distances = np.min(distances, axis=1)
        avg_min_distance = np.mean(min_distances)

        # Use fine grid when hexagons are relatively close together
        use_fine_grid = avg_min_distance < 2.0  # Threshold for dense regions
    else:
        use_fine_grid = False

    for i, hex_poly in enumerate(hexagons):
        bbox = hex_poly.bounds
        min_x, min_y, max_x, max_y = bbox

        # Always use coarse grid for broad spatial partitioning
        for x in range(int(min_x/coarse_grid_size), int(max_x/coarse_grid_size)+1):
            for y in range(int(min_y/coarse_grid_size), int(max_y/coarse_grid_size)+1):
                coarse_grid[(x,y)].append(i)

        # Use fine grid in dense regions
        if use_fine_grid:
            for x in range(int(min_x/fine_grid_size), int(max_x/fine_grid_size)+1):
                for y in range(int(min_y/fine_grid_size), int(max_y/fine_grid_size)+1):
                    fine_grid[(x,y)].append(i)

    return coarse_grid, fine_grid, use_fine_grid
