def hexagon_vertices(center, size, angle_degrees):
    """Generate vertices of a regular hexagon given center, size, and rotation."""
    angle_rad = np.radians(angle_degrees)
    # Vertices of unit hexagon centered at origin
    unit_vertices = np.array([
        [1, 0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1, 0],
        [-0.5, -np.sqrt(3)/2],
        [0.5, -np.sqrt(3)/2]
    ])
    # Scale and rotate
    vertices = unit_vertices * size
    cos_a, sin_a = np.cos(angle_rad), np.sin(angle_rad)
    rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
    vertices = vertices @ rotation_matrix.T + center
    return vertices

def calculate_outer_hexagon_size(inner_hex_data):
    """Calculate the minimum outer hexagon size needed to contain all inner hexagons."""
    # Get vertices of all inner hexagons
    max_dist_from_center = 0
    for i in range(len(inner_hex_data)):

        angle = inner_hex_data[i, 2]

        for vertex in vertices:
            dist = np.linalg.norm(vertex)
            max_dist_from_center = max(max_dist_from_center, dist)

    # Account for some padding to ensure full containment
    # The outer hexagon needs to be large enough so that its vertices are beyond
    # all inner hexagon vertices. The radius of outer hexagon should be
    # approximately max_dist_from_center + epsilon
    return max_dist_from_center + 0.1
