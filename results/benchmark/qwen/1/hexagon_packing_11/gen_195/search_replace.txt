def simulated_annealing_refinement(initial_individual, max_iterations=500):
    """
    Perform local refinement using simulated annealing on a given solution
    """
    current_individual = initial_individual.copy()
    current_fitness = evaluate_fitness_adaptive(current_individual)

    # Simulated annealing parameters
    temperature = 1.0
    cooling_rate = 0.98
    min_temperature = 0.001

    for iteration in range(max_iterations):
        # Perturb one random hexagon
        hex_index = np.random.randint(0, 11)
        param_index = hex_index * 3

        # Create a neighbor solution by perturbing position and angle
        neighbor_individual = current_individual.copy()

        # Small random perturbations
        neighbor_individual[param_index + 0] += np.random.normal(0, 0.05)  # x position
        neighbor_individual[param_index + 1] += np.random.normal(0, 0.05)  # y position
        neighbor_individual[param_index + 2] += np.random.normal(0, 5)     # angle (degrees)

        # Ensure angle stays in [0, 360]
        neighbor_individual[param_index + 2] = neighbor_individual[param_index + 2] % 360

        # Evaluate neighbor
        neighbor_fitness = evaluate_fitness_adaptive(neighbor_individual)

        # Accept or reject the neighbor
        if neighbor_fitness > current_fitness:
            # Always accept better solutions
            current_individual = neighbor_individual
            current_fitness = neighbor_fitness
        else:
            # Accept worse solutions with probability based on temperature
            delta = neighbor_fitness - current_fitness
            acceptance_probability = np.exp(delta / temperature)
            if np.random.rand() < acceptance_probability:
                current_individual = neighbor_individual
                current_fitness = neighbor_fitness

        # Cool down temperature
        temperature = max(min_temperature, temperature * cooling_rate)

    return current_individual, current_fitness
