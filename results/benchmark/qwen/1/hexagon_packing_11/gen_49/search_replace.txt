<DESCRIPTION>
Implement adaptive mutation rate scheduling in differential evolution to improve convergence. The idea is to start with a higher mutation rate (0.8) for early generations to encourage exploration, then gradually decrease it to 0.1 in later generations to promote convergence towards optimal solutions. This builds upon the proven effectiveness of adaptive control mechanisms shown in successful optimization programs.

The key insight is that early exploration needs higher diversity, while later exploitation benefits from fine-tuning with lower mutation rates.

def optimize_hexagon_packing():
    """Main optimization function with multi-start approach"""
    # Generate multiple initial populations
    initial_populations = generate_initial_population(5)

    best_result = None
    best_score = float('inf')

    # Run optimization from multiple starting points
    for i, initial_solution in enumerate(initial_populations):
        try:
            # Set bounds for optimization
            bounds = []
            # Position bounds
            for _ in range(22):
                bounds.append((-10.0, 10.0))  # X and Y coordinates
            # Angle bounds
            for _ in range(11):
                bounds.append((0.0, 360.0))   # Rotation angles

            # Use adaptive DE parameters with custom implementation
            maxiter = 100
            popsize = 15

            def adaptive_differential_evolution(func, bounds, maxiter, popsize, seed=None):
                """Custom differential evolution with adaptive mutation rate"""
                # Initialize random number generator
                rng = np.random.default_rng(seed)

                # Generate initial population
                population = []
                for _ in range(popsize):
                    individual = []
                    for (min_val, max_val) in bounds:
                        individual.append(rng.uniform(min_val, max_val))
                    population.append(individual)

                # Track best solution
                best_individual = None
                best_fitness = float('inf')

                for generation in range(maxiter):
                    # Adaptive mutation rate: start high, decrease over time
                    mutation_rate = 0.8 - (0.7 * generation / maxiter)  # Decreases from 0.8 to 0.1

                    # Evaluate current population
                    fitnesses = []
                    for individual in population:
                        fitness = func(np.array(individual))
                        fitnesses.append(fitness)
                        if fitness < best_fitness:
                            best_fitness = fitness
                            best_individual = individual[:]

                    # Create new population
                    new_population = []
                    for j in range(popsize):
                        # Select three different individuals
                        candidates = list(range(popsize))
                        candidates.remove(j)
                        selected = rng.choice(candidates, 3, replace=False)

                        # Differential evolution mutation
                        mutant = []
                        for k in range(len(bounds)):
                            if rng.random() < mutation_rate or generation == 0:
                                # Mutation using differential evolution
                                mutant.append(population[selected[0]][k] +
                                            0.8 * (population[selected[1]][k] - population[selected[2]][k]))
                            else:
                                mutant.append(population[j][k])

                            # Ensure bounds are respected
                            min_val, max_val = bounds[k]
                            mutant[k] = np.clip(mutant[k], min_val, max_val)

                        # Crossover
                        trial = []
                        for k in range(len(bounds)):
                            if rng.random() < 0.9 or k == 0:  # Keep some randomness
                                trial.append(mutant[k])
                            else:
                                trial.append(population[j][k])

                        new_population.append(trial)

                    population = new_population

                return type('Result', (), {
                    'x': np.array(best_individual),
                    'fun': best_fitness
                })()

            # Run custom adaptive differential evolution
            result = adaptive_differential_evolution(
                lambda sol: evaluate_solution(sol, use_spatial_index=(i < 3)),  # Use spatial index for first 3 starts
                bounds,
                maxiter=maxiter,
                popsize=popsize,
                seed=42+i  # Different seed for each start
            )

            # Evaluate final result
            final_score = evaluate_solution(result.x, use_spatial_index=False)

            if final_score < best_score:
                best_score = final_score
                best_result = result

        except Exception as e:
            print(f"Start {i} failed: {e}")
            continue

        # Fallback to simple solution
        raise RuntimeError("All optimization attempts failed")

    final_angles = best_result.x[22:]

    # Refine the solution with local optimization
