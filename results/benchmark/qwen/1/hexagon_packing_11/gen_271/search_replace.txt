    def check_overlap_fast(self, hex1_vertices: np.ndarray, hex2_vertices: np.ndarray) -> bool:
        """Fast overlap check using polygon intersection with early termination"""
        try:
            poly1 = Polygon(hex1_vertices)
            poly2 = Polygon(hex2_vertices)

            # Early termination: if bounding boxes don't intersect, no overlap possible
            if not poly1.bounds or not poly2.bounds:
                return False

            # Check if bounding boxes intersect first (very fast)
            bbox1 = poly1.bounds
            bbox2 = poly2.bounds
            if (bbox1[2] < bbox2[0] or bbox1[0] > bbox2[2] or
                bbox1[3] < bbox2[1] or bbox1[1] > bbox2[3]):
                return False

            # Check if polygons actually intersect
            return poly1.intersects(poly2)
        except:
            return False

    def check_hexagon_overlap_direct(self, center1: np.ndarray, angle1: float,
                                   center2: np.ndarray, angle2: float) -> bool:
        """Direct overlap check using distance and geometric properties of hexagons"""
