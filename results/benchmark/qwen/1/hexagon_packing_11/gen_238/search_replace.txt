def compute_voronoi_hex_centers(n_points=11, bounds=(-5, 5)):
    """Generate initial hexagon centers using Voronoi diagram."""
    # Generate random points
    np.random.seed(42)
    points = np.random.uniform(bounds[0], bounds[1], size=(n_points, 2))

    # Compute Voronoi diagram
    vor = Voronoi(points)

    # Take the centroids of the finite Voronoi cells
    centroids = []
    for region in vor.regions:
        if len(region) > 0 and -1 not in region:
            points_in_region = [vor.vertices[i] for i in region if i >= 0]
            if points_in_region:
                centroid = np.mean(points_in_region, axis=0)
                # Only keep centroids within bounds
                if bounds[0] <= centroid[0] <= bounds[1] and bounds[0] <= centroid[1] <= bounds[1]:
                    centroids.append(centroid)

    # If we don't have enough centroids, add some random ones
    while len(centroids) < n_points:
        centroids.append(np.random.uniform(bounds[0], bounds[1], size=2))

    # Take first n_points
    return np.array(centroids[:n_points])

def compute_structured_hex_centers(n_points=11, bounds=(-5, 5)):
    """Generate initial hexagon centers using structured hexagonal arrangement."""
    # Create a hexagonal lattice arrangement that naturally fits 11 hexagons
    # Start with central hexagon
    centers = [[0.0, 0.0]]

    # Add surrounding hexagons in concentric rings
    # First ring (6 hexagons)
    ring1_radius = 2.0  # Adjusted spacing for unit hexagons
    for i in range(6):
        angle = i * 60  # 60 degree increments
        x = ring1_radius * math.cos(math.radians(angle))
        y = ring1_radius * math.sin(math.radians(angle))
        centers.append([x, y])

    # Second ring (if needed)
    if len(centers) < n_points:
        ring2_radius = 3.5
        additional_count = min(n_points - len(centers), 6)
        for i in range(additional_count):
            angle = i * 60  # 60 degree increments
            x = ring2_radius * math.cos(math.radians(angle))
            y = ring2_radius * math.sin(math.radians(angle))
            centers.append([x, y])

    # Fill remaining positions if needed
    while len(centers) < n_points:
        # Random positions within bounds but with preference for central area
