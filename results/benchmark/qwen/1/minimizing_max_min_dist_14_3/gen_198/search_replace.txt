    def generate_initial_points(n=14):
        """Generate multiple types of initial configurations"""
        configs = []

        # 1. Icosahedron-based points
        configs.append(('icosahedron', icosahedron_points(n)))

        # 2. Fibonacci spiral on sphere
        points_fib = np.zeros((n, 3))
        golden_angle = np.pi * (3 - np.sqrt(5))
        for i in range(n):
            y = 1 - (i / float(n - 1)) * 2  # y goes from 1 to -1
            radius = np.sqrt(1 - y * y)
            theta = golden_angle * i
            x = np.cos(theta) * radius
            z = np.sin(theta) * radius
            points_fib[i] = [x, y, z]
        configs.append(('fibonacci', points_fib))

        # 3. Sobol sequence points (added for better space filling)
        try:
            from sobol_seq import i4_sobol_generate
            sobol_points = i4_sobol_generate(3, n)
            points_sobol = np.zeros((n, 3))
            for i in range(n):
                # Map Sobol points to sphere using spherical coordinates
                u = sobol_points[i, 0]  # Uniform in [0,1]
                v = sobol_points[i, 1]  # Uniform in [0,1]

                theta = 2 * np.pi * u  # azimuthal angle
                phi = np.arccos(2 * v - 1)  # polar angle

                x = np.sin(phi) * np.cos(theta)
                y = np.sin(phi) * np.sin(theta)
                z = np.cos(phi)

                points_sobol[i] = [x, y, z]
            configs.append(('sobol', points_sobol))
        except ImportError:
            # Fallback to random if sobol not available
            np.random.seed(42)
            points_random = np.random.rand(n, 3) * 2 - 1  # [-1, 1]
            norms = np.linalg.norm(points_random, axis=1)
            points_random = points_random / np.maximum(norms[:, np.newaxis], 1e-12) * 0.9
            configs.append(('random', points_random))

        # 4. Random points with normalization
        np.random.seed(42)
        points_random = np.random.rand(n, 3) * 2 - 1  # [-1, 1]
        norms = np.linalg.norm(points_random, axis=1)
        points_random = points_random / np.maximum(norms[:, np.newaxis], 1e-12) * 0.9
        configs.append(('random', points_random))

        return configs
