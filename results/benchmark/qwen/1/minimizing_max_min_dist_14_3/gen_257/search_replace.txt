    def sobol_initialization(self) -> np.ndarray:
        """Initialize points using Sobol sequence for better space-filling properties."""
        sampler = qmc.Sobol(d=self.d, seed=42)
        points = sampler.random(n=self.n_points)
        # Scale to unit sphere
        points = points * 2 - 1  # Map to [-1, 1]^3
        norms = np.linalg.norm(points, axis=1, keepdims=True)
        # Avoid division by zero
        norms = np.where(norms == 0, 1.0, norms)
        points = points / norms
        return points

    def adaptive_constraint_tightening(self, iteration: int, max_iterations: int) -> Tuple[float, float]:
        """Dynamically adjust constraint bounds during optimization iterations."""
        # Start with relaxed bounds and tighten over time
        start_bound = 2.0
        end_bound = 1.0
        alpha = iteration / max_iterations if max_iterations > 0 else 0
        current_bound = start_bound + (end_bound - start_bound) * alpha
        return (-current_bound, current_bound)

    def hybrid_refinement(self, initial_points: np.ndarray, max_iter: int = 1000) -> Tuple[np.ndarray, float]:
        """Apply hybrid refinement using both SLSQP and L-BFGS-B optimizers."""
        # First stage: SLSQP for robust global convergence
        points_slsqp, ratio_slsqp = self.optimize_single_start(
            initial_points, max_iter=max_iter, method='SLSQP', coarse=False
        )

        # Second stage: L-BFGS-B with tighter tolerances for local improvement
        points_lbfgsb, ratio_lbfgsb = self.optimize_single_start(
            points_slsqp, max_iter=max_iter, method='L-BFGS-B', coarse=False
        )

        # Return the better of the two refinements
        if ratio_lbfgsb > ratio_slsqp:
            return points_lbfgsb, ratio_lbfgsb
        else:
            return points_slsqp, ratio_slsqp
