    def _generate_sobol_points(self) -> np.ndarray:
        """Generate points using Sobol sequence for better space-filling properties."""
        # Create Sobol sequence sampler
        sampler = qmc.Sobol(d=3, seed=42)

        # Generate more points than needed to ensure good distribution
        n_samples = max(self.n * 2, 32)
        points = sampler.random(n=n_samples)

        # Scale to [-1, 1]^3 cube
        points = points * 2 - 1

        # Project to unit sphere
        norms = np.linalg.norm(points, axis=1, keepdims=True)
        # Avoid division by zero
        norms = np.where(norms == 0, 1.0, norms)
        points = points / norms
