def sobol_like_distribution(n):

    # Use Fibonacci base with additional low-discrepancy perturbations
    points = []
    phi = np.pi * (3 - np.sqrt(5))

    # Generate Fibonacci points but with Sobol-like spacing adjustments
    for i in range(n):
        # Standard Fibonacci approach
        y = 1 - (i / float(n - 1)) * 2
        radius = np.sqrt(1 - y * y)
        theta = phi * i

        x = np.cos(theta) * radius
        z = np.sin(theta) * radius

        # Apply small perturbations to improve distribution (Sobol-like)
        # Use quasi-random sequence for perturbations
        pert_x = 0.02 * np.sin(i * 1.618) * np.cos(i * 0.785)
        pert_y = 0.02 * np.cos(i * 1.618) * np.sin(i * 0.785)
        pert_z = 0.02 * np.sin(i * 0.785) * np.cos(i * 1.618)

        x += pert_x
        y += pert_y
        z += pert_z

        # Normalize to unit sphere
        norm = np.sqrt(x*x + y*y + z*z)
        if norm > 0:
            x /= norm
            y /= norm
            z /= norm

def optimize_points(initial_points, max_iter=1000, method='L-BFGS-B'):
    """Optimize point positions to maximize min/max distance ratio"""

    def objective(x):
        # Reshape flat array back to points
        points = x.reshape(-1, 3)
        # We want to maximize the ratio, so we minimize its negative
        ratio = min_max_dist_ratio(points)
        if ratio <= 0:
            return 1e10  # Penalize invalid configurations
        return -ratio

    # Flatten initial points for optimization
    x0 = initial_points.flatten()

    # Define bounds (points should stay within [-1,1] for sphere)
    bounds = [(-1, 1) for _ in range(len(x0))]

    # Use L-BFGS-B for optimization with bounds
    result = minimize(
        objective,
        x0,
        method=method,
        bounds=bounds,
        options={'maxiter': max_iter},
        tol=1e-8
    )

    # Return optimized points
    if result.success:
        return result.x.reshape(-1, 3)
    else:
        # If optimization fails, return original points
        return initial_points


def optimize_with_multi_start(initial_points, num_starts=5, max_iter=1000):
    """Run optimization from multiple starting points to find better solution"""
    best_ratio = -np.inf
    best_points = initial_points.copy()

    for i in range(num_starts):
        # Create perturbed version of initial points to ensure diversity
        if i == 0:
            # First iteration uses original points
            current_points = initial_points.copy()
        else:
            # Subsequent iterations use slightly perturbed versions
            np.random.seed(42 + i)  # Different seed for each start
            perturbed_points = initial_points + np.random.normal(0, 0.02, initial_points.shape)
            # Normalize to unit sphere

            norms = np.where(norms == 0, 1, norms)
            current_points = perturbed_points / norms

        # Optimize from this starting point
        optimized_points = optimize_points(current_points, max_iter)

        # Calculate ratio for this optimization run
        ratio = min_max_dist_ratio(optimized_points)

        # Keep track of best solution
        if ratio > best_ratio:
            best_ratio = ratio
            best_points = optimized_points.copy()

    return best_points
