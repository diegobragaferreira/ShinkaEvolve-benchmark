import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, cdist
from scipy.stats import qmc
import time
from typing import Tuple, Optional
import random

class SphereEvolutionOptimizer:
    """A novel evolutionary optimizer for point distribution in 3D space using geometric insights."""

    def __init__(self, n_points: int = 14, d: int = 3):
        self.n_points = n_points
        self.d = d
        self.best_ratio = -np.inf
        self.best_points = None
        self.max_evaluations = 10000  # Limit evaluations for performance control

    def fibonacci_spiral_sphere(self) -> np.ndarray:
        """Generate points on a sphere using Fibonacci spiral method."""
        points = []
        phi = np.pi * (3 - np.sqrt(5))  # golden angle

        for i in range(self.n_points):
            y = 1 - (i / float(self.n_points - 1)) * 2  # y goes from 1 to -1
            radius = np.sqrt(1 - y * y)  # radius at y

            theta = phi * i  # golden angle increment

            x = np.cos(theta) * radius
            z = np.sin(theta) * radius

            points.append([x, y, z])

        return np.array(points)

    def initialize_sphere_packing(self) -> np.ndarray:
        """Initialize points using a novel sphere packing approach with iterative refinement."""
        # Start with Fibonacci spiral
        base_points = self.fibonacci_spiral_sphere()
        
        # Apply controlled perturbation
        np.random.seed(42)
        perturbation = np.random.normal(0, 0.03, (self.n_points, self.d))
        points = base_points + perturbation
        
        # Spring model relaxation for better distribution
        for iteration in range(50):
            # Calculate all pairwise distances
            distances = cdist(points, points)
            np.fill_diagonal(distances, np.inf)
            
            # Calculate forces between points (repulsion proportional to inverse distance)
            forces = np.zeros_like(points)
            for i in range(self.n_points):
                for j in range(i+1, self.n_points):
                    if distances[i,j] > 0:
                        # Force vector from i to j
                        diff = points[j] - points[i]
                        dist = np.linalg.norm(diff)
                        if dist > 1e-10:  # Avoid division by zero
                            force_magnitude = 1.0 / (dist * dist * dist)
                            forces[i] += force_magnitude * diff / dist
                            forces[j] -= force_magnitude * diff / dist
            
            # Apply forces with damping
            points += 0.01 * forces
            
            # Project back to sphere surface
            norms = np.linalg.norm(points, axis=1, keepdims=True)
            norms = np.where(norms == 0, 1.0, norms)
            points = points / norms
        
        return points

    def sobol_initialization(self) -> np.ndarray:
        """Initialize points using Sobol sequence for better space-filling properties."""
        sampler = qmc.Sobol(d=self.d, seed=42)
        points = sampler.random(n=self.n_points)
        # Scale to unit sphere
        points = points * 2 - 1  # Map to [-1, 1]^3
        norms = np.linalg.norm(points, axis=1, keepdims=True)
        # Avoid division by zero
        norms = np.where(norms == 0, 1.0, norms)
        points = points / norms
        return points

    def adaptive_initialization(self, method: str = 'sphere_packing') -> np.ndarray:
        """Create diverse initial point configurations with improved strategies."""
        if method == 'sphere_packing':
            points = self.initialize_sphere_packing()
        elif method == 'sobol':
            points = self.sobol_initialization()
        else:
            # Random initialization with better spread
            np.random.seed(42)
            points = np.random.rand(self.n_points, self.d) * 2 - 1
            # Normalize to unit sphere
            norms = np.linalg.norm(points, axis=1, keepdims=True)
            norms = np.where(norms == 0, 1.0, norms)
            points = points / norms

        return points

    def min_max_dist_ratio(self, points: np.ndarray) -> float:
        """Calculate the ratio of minimum to maximum distance with numerical stability."""
        if len(points) < 2:
            return 0.0

        try:
            distances = pdist(points)
            if len(distances) == 0:
                return 0.0

            # Early termination: if we have very small distances already,
            # we don't need to compute the full ratio
            min_dist = np.min(distances)
            max_dist = np.max(distances)

            # Avoid division by zero or near-zero values
            if max_dist <= 1e-12:
                return 0.0
            
            # Add small epsilon to prevent numerical issues
            epsilon = 1e-15
            return min_dist / (max_dist + epsilon)
        except Exception:
            return 0.0

    def distance_variance_penalty(self, points: np.ndarray) -> float:
        """Calculate variance penalty to promote uniform distribution."""
        if len(points) < 2:
            return 0.0
            
        distances = pdist(points)
        if len(distances) == 0:
            return 0.0
            
        return np.var(distances)

    def multi_objective_fitness(self, points: np.ndarray) -> float:
        """Multi-objective fitness combining min/max ratio and uniformity."""
        ratio = self.min_max_dist_ratio(points)
        variance_penalty = self.distance_variance_penalty(points)
        # Weighted combination favoring ratio but penalizing variance
        return ratio - 0.1 * variance_penalty

    def setup_constraint(self) -> dict:
        """Setup constraint dictionary for optimization."""
        def constraint_sphere(x_flat):
            points_reshaped = x_flat.reshape(self.n_points, self.d)
            norms = np.linalg.norm(points_reshaped, axis=1)
            return norms - 1.0
        
        return {'type': 'eq', 'fun': constraint_sphere}

    def normalize_points(self, points: np.ndarray) -> np.ndarray:
        """Normalize points to unit sphere ensuring numerical stability."""
        norms = np.linalg.norm(points, axis=1, keepdims=True)
        # Avoid division by zero with epsilon
        epsilon = 1e-15
        norms = np.where(norms < epsilon, 1.0, norms)
        return points / norms

    def optimize_with_evolutionary_approach(self, initial_points: np.ndarray) -> Tuple[np.ndarray, float]:
        """Use a novel evolutionary approach for optimization."""
        # Initialize population
        population_size = 20
        population = []
        fitness_scores = []
        
        # Create initial population
        for i in range(population_size):
            if i == 0:
                # First individual is the input
                individual = initial_points.copy()
            else:
                # Create diversified individuals
                np.random.seed(i + 42)
                jitter = np.random.normal(0, 0.02, initial_points.shape)
                individual = initial_points + jitter
                # Normalize to sphere
                individual = self.normalize_points(individual)
            
            population.append(individual)
            fitness_scores.append(self.multi_objective_fitness(individual))
        
        # Evolutionary loop
        max_generations = 100
        elite_count = 4
        
        for generation in range(max_generations):
            # Sort by fitness
            sorted_indices = np.argsort(fitness_scores)[::-1]
            sorted_population = [population[i] for i in sorted_indices]
            sorted_fitness = [fitness_scores[i] for i in sorted_indices]
            
            # Update best solution
            if sorted_fitness[0] > self.best_ratio:
                self.best_ratio = sorted_fitness[0]
                self.best_points = sorted_population[0].copy()
            
            # Create new population
            new_population = []
            new_fitness = []
            
            # Elitism: keep best individuals
            for i in range(elite_count):
                new_population.append(sorted_population[i].copy())
                new_fitness.append(sorted_fitness[i])
            
            # Generate offspring through crossover and mutation
            while len(new_population) < population_size:
                # Tournament selection
                parent1_idx = self.tournament_selection(sorted_fitness, tournament_size=3)
                parent2_idx = self.tournament_selection(sorted_fitness, tournament_size=3)
                
                # Crossover
                child = self.uniform_crossover(sorted_population[parent1_idx], 
                                               sorted_population[parent2_idx])
                
                # Mutation
                child = self.gaussian_mutation(child, mutation_rate=0.1, sigma=0.01)
                
                # Normalize
                child = self.normalize_points(child)
                
                new_population.append(child)
                new_fitness.append(self.multi_objective_fitness(child))
            
            population = new_population
            fitness_scores = new_fitness
            
            # Adaptive parameters based on convergence
            if generation > 20:
                # Increase mutation rate if stagnation detected
                if abs(sorted_fitness[0] - sorted_fitness[-1]) < 1e-6:
                    for i in range(len(new_population)):
                        new_population[i] = self.gaussian_mutation(new_population[i], 
                                                                  mutation_rate=0.15, 
                                                                  sigma=0.02)
                        new_population[i] = self.normalize_points(new_population[i])
                        new_fitness[i] = self.multi_objective_fitness(new_population[i])

        # Final optimization of best solution using local method
        best_individual = population[0]
        try:
            x0 = best_individual.flatten()
            
            def objective(x_flat):
                points_reshaped = x_flat.reshape(self.n_points, self.d)
                normalized_points = self.normalize_points(points_reshaped)
                return -self.min_max_dist_ratio(normalized_points)

            constraints = self.setup_constraint()
            bounds = [(-2, 2) for _ in range(self.n_points * self.d)]

            result = minimize(
                objective,
                x0,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12},
                tol=1e-12
            )
            
            if result.success:
                optimized_points = result.x.reshape(self.n_points, self.d)
                optimized_points = self.normalize_points(optimized_points)
                final_ratio = self.min_max_dist_ratio(optimized_points)
                
                if final_ratio > self.best_ratio:
                    self.best_ratio = final_ratio
                    self.best_points = optimized_points.copy()
                    
        except Exception:
            pass
            
        return self.best_points if self.best_points is not None else initial_points, self.best_ratio

    def tournament_selection(self, fitness_scores, tournament_size=3):
        """Select an individual using tournament selection."""
        tournament_indices = np.random.choice(len(fitness_scores), 
                                            size=tournament_size, 
                                            replace=False)
        tournament_fitness = [fitness_scores[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmax(tournament_fitness)]
        return winner_index

    def uniform_crossover(self, parent1, parent2):
        """Uniform crossover operator for 3D points."""
        mask = np.random.rand(*parent1.shape) > 0.5
        child = np.where(mask, parent1, parent2)
        return child

    def gaussian_mutation(self, points, mutation_rate=0.1, sigma=0.01):
        """Gaussian mutation operator."""
        mask = np.random.rand(*points.shape) < mutation_rate
        noise = np.random.normal(0, sigma, points.shape)
        mutated = points + mask * noise
        return mutated

    def multi_start_evolutionary_optimization(self) -> np.ndarray:
        """Perform multi-start optimization using evolutionary approach."""
        # Try multiple initialization strategies
        init_methods = ['sphere_packing', 'sobol', 'random']
        num_starts = 15  # Increase number of starts for better exploration

        for method_idx, method in enumerate(init_methods):
            for start_idx in range(num_starts // len(init_methods)):
                try:
                    # Different seeding for better exploration
                    seed_val = 42 + method_idx * 100 + start_idx * 10
                    np.random.seed(seed_val)
                    
                    if method == 'sphere_packing':
                        points = self.adaptive_initialization('sphere_packing')
                    elif method == 'sobol':
                        points = self.adaptive_initialization('sobol')
                    else:
                        points = self.adaptive_initialization('random')
                    
                    # Apply evolutionary optimization
                    optimized_points, current_ratio = self.optimize_with_evolutionary_approach(points)
                    
                    if current_ratio > self.best_ratio:
                        self.best_ratio = current_ratio
                        self.best_points = optimized_points.copy()
                        
                except Exception:
                    continue

        # Return the best solution found, fallback to sphere packing if needed
        if self.best_points is not None:
            return self.best_points
        else:
            return self.adaptive_initialization('sphere_packing')

def min_max_dist_dim3_14() -> np.ndarray:
    """
    Creates 14 points in 3 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (14,3) containing the (x,y,z) coordinates of the 14 points.
    """
    optimizer = SphereEvolutionOptimizer(n_points=14, d=3)
    return optimizer.multi_start_evolutionary_optimization()
