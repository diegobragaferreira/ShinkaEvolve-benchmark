    """Optimize point positions to maximize min/max distance ratio with adaptive constraint tightening"""

    def objective(x):
        # Reshape flat array back to points
        points = x.reshape(-1, 3)
        # We want to maximize the ratio, so we minimize its negative
        return -min_max_dist_ratio(points)

    # Flatten initial points for optimization
    x0 = initial_points.flatten()

    # Define adaptive bounds that start relaxed and tighten over time
    # Initially allow points to extend slightly beyond sphere for better exploration
    bounds = [(-1.1, 1.1) for _ in range(len(x0))]

    # Use L-BFGS-B for optimization with bounds
    result = minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter},
        tol=1e-8
    )

    # Return optimized points
    return result.x.reshape(-1, 3)
