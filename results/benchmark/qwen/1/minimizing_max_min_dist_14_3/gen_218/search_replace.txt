    def _sobol_initialization(self) -> np.ndarray:
        """Generate points using Sobol sequence for better space-filling properties."""
        # Generate Sobol sequence in 3D
        sobol_points = i4_sobol_generate(self.dimension, self.num_points, skip=1)
        # Scale to [-1, 1] range
        sobol_points = sobol_points * 2 - 1

        # Project to unit sphere
        norms = np.linalg.norm(sobol_points, axis=1, keepdims=True)
        sphere_points = sobol_points / np.maximum(norms, 1e-12)

        return sphere_points

    def _random_initialization(self) -> np.ndarray:
        """Generate random points on unit sphere."""
        np.random.seed(42)
        points = np.random.randn(self.num_points, self.dimension)
        norms = np.linalg.norm(points, axis=1, keepdims=True)
        return points / np.maximum(norms, 1e-12)
