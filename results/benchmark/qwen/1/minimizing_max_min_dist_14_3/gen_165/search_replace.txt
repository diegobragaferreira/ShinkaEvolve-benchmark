    def enhanced_sobol_fibonacci_initialization(n):
        """
        Enhanced initialization using Sobol sequence combined with Fibonacci-inspired distribution.
        This provides superior space-filling properties while maintaining mathematical structure.
        """
        # Use Sobol sequence for well-distributed initial points
        from scipy.stats import qmc
        sampler = qmc.Sobol(d=3, seed=42)
        sobol_points = sampler.random(n=n)

        # Transform Sobol points to spherical coordinates with Fibonacci-like distribution
        points = []
        for i in range(n):
            # Map Sobol uniform points to sphere using Fibonacci-inspired distribution
            # This preserves good angular distribution properties of Fibonacci
            theta = math.acos(1 - 2 * (i / (n - 1)))  # Standard Fibonacci theta
            phi = (i * 2.414213562) % (2 * math.pi)   # Golden ratio multiple for phi

            # Add slight perturbation from Sobol sequence for better distribution
            theta_pert = theta * (0.95 + 0.1 * sobol_points[i, 0])  # Perturb theta slightly
            phi_pert = phi * (0.98 + 0.04 * sobol_points[i, 1])     # Perturb phi slightly

            x = math.sin(theta_pert) * math.cos(phi_pert)
            y = math.sin(theta_pert) * math.sin(phi_pert)
            z = math.cos(theta_pert)
            points.append([x, y, z])

        np.random.seed(42)
        # Use stronger but controlled perturbations to escape local minima

        norms = np.linalg.norm(points, axis=1)
