import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.stats import qmc
import math

def min_max_dist_dim3_14() -> np.ndarray:
    """
    Creates 14 points in 3 dimensions in order to maximize the ratio of minimum to maximum distance.

    Returns
        points: np.ndarray of shape (14,3) containing the (x,y,z) coordinates of the 14 points.
    """
    
    def distance_ratio(points_flat):
        """Calculate the ratio of minimum to maximum distance"""
        points = points_flat.reshape(-1, 3)
        distances = squareform(pdist(points))
        # Set diagonal to large value so it doesn't affect min/max
        np.fill_diagonal(distances, np.inf)
        min_dist = np.min(distances)
        max_dist = np.max(distances)
        if max_dist == 0:
            return 0
        return min_dist / max_dist
    
    def objective_function(points_flat):
        """Minimize negative of distance ratio with penalty for numerical instability"""
        points = points_flat.reshape(-1, 3)
        distances = squareform(pdist(points))
        np.fill_diagonal(distances, np.inf)
        min_dist = np.min(distances)
        max_dist = np.max(distances)
        if max_dist == 0:
            return 0
        ratio = min_dist / max_dist
        
        # Add penalty for very small ratios to avoid numerical issues
        if ratio < 1e-10:
            return -1e-10  # Return a small negative value
            
        return -ratio
    
    def sobol_sphere_initialization(n):
        """
        Generate initial points using Sobol sequence projected onto sphere.
        This provides superior space filling compared to standard methods.
        """
        # Generate Sobol sequence in [0,1]^3
        sampler = qmc.Sobol(d=3, seed=42)
        points = sampler.random(n=n)
        
        # Convert to unit sphere using spherical coordinates
        # Map [0,1] to [0,pi] for theta and [0,2pi] for phi
        theta = points[:, 0] * np.pi  # 0 to pi
        phi = points[:, 1] * 2 * np.pi  # 0 to 2pi
        r = points[:, 2]  # Radius component
        
        # Convert to Cartesian coordinates
        x = r * np.sin(theta) * np.cos(phi)
        y = r * np.sin(theta) * np.sin(phi)
        z = r * np.cos(theta)
        
        # Normalize to unit sphere
        points_cartesian = np.column_stack([x, y, z])
        norms = np.linalg.norm(points_cartesian, axis=1)
        normalized_points = points_cartesian / norms[:, np.newaxis]
        
        # Add slight jitter to prevent symmetrical issues
        np.random.seed(42)
        noise = np.random.normal(0, 0.01, normalized_points.shape)
        normalized_points += noise
        
        # Renormalize
        norms = np.linalg.norm(normalized_points, axis=1)
        normalized_points = normalized_points / norms[:, np.newaxis]
        
        return normalized_points
    
    def adaptive_optimization(x0, maxiter=1000):
        """
        Adaptive optimization with dynamic parameter adjustment
        """
        # Initialize parameters
        current_x = x0.copy()
        current_value = objective_function(current_x)
        prev_value = current_value
        history = [current_value]
        
        # Constraint function
        def constraint_func(x):
            points = x.reshape(-1, 3)
            norms = np.linalg.norm(points, axis=1)
            return norms - 1.0  # Should be near 0 for unit sphere
        
        constraints = []
        for i in range(14):
            constraints.append({'type': 'eq', 'fun': lambda x, i=i: constraint_func(x)[i]})
        
        bounds = [(-1.1, 1.1)] * len(x0)
        
        # Main optimization loop with adaptive parameters
        for iteration in range(maxiter):
            # Adjust parameters based on convergence history
            if len(history) >= 5:
                recent_change = abs(history[-1] - history[-5])
                if recent_change < 1e-6:  # Slow convergence - tighten tolerances
                    ftol = 1e-10
                    gtol = 1e-10
                    max_iter_current = 50
                else:
                    ftol = 1e-6
                    gtol = 1e-6
                    max_iter_current = 100
            else:
                ftol = 1e-6
                gtol = 1e-6
                max_iter_current = 100
            
            # Optimization step
            try:
                result = minimize(
                    objective_function,
                    current_x,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': max_iter_current, 'ftol': ftol, 'gtol': gtol},
                    tol=ftol
                )
                
                if result.success:
                    current_x = result.x
                    current_value = objective_function(current_x)
                    history.append(current_value)
                    
                    # Early stopping if improvement is minimal
                    if len(history) >= 2 and abs(current_value - prev_value) < 1e-8:
                        break
                        
                    prev_value = current_value
                else:
                    # If optimization fails, use previous good solution
                    break
                    
            except Exception:
                break
                
        return current_x
    
    best_ratio = -np.inf
    best_points = None
    
    # Multi-start optimization with enhanced initialization
    for restart in range(15):  # Increase restarts for better exploration
        np.random.seed(42 + restart)
        
        # Use Sobol-based initialization
        initial_points = sobol_sphere_initialization(14)
        
        # Add slight random perturbation to break symmetry
        perturbation = np.random.normal(0, 0.005, initial_points.shape)
        initial_points += perturbation
        
        # Ensure normalization
        norms = np.linalg.norm(initial_points, axis=1)
        initial_points = initial_points / norms[:, np.newaxis]
        
        x0 = initial_points.flatten()
        
        # Optimize with adaptive approach
        try:
            optimized_points = adaptive_optimization(x0, maxiter=800)
            
            # Calculate final ratio
            ratio = distance_ratio(optimized_points)
            
            if ratio > best_ratio:
                best_ratio = ratio
                best_points = optimized_points.copy()
                
        except Exception as e:
            continue
    
    # If no good solution found, fallback to Sobol initialization
    if best_points is None:
        initial_points = sobol_sphere_initialization(14)
        best_points = initial_points.flatten()
    
    # Convert back to 14x3 array
    final_points = best_points.reshape(14, 3)
    
    return final_points
