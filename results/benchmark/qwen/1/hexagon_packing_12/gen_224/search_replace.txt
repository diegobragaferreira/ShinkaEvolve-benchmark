        """Quick check using bounding circle: if center is too far from outer center, hexagon can't fit."""
        dist = np.sqrt((center[0] - self.outer_center[0])**2 + (center[1] - self.outer_center[1])**2)

        return dist + 1.0 <= outer_side_length
