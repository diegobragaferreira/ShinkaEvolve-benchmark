import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import math
from shapely.geometry import Polygon, Point
from shapely.ops import unary_union


def hexagon_vertices(center_x, center_y, side_length=1, angle_degrees=0):
    """Generate vertices of a regular hexagon."""
    angle_rad = math.radians(angle_degrees)
    vertices = []
    for i in range(6):
        angle = angle_rad + i * math.pi / 3
        x = center_x + side_length * math.cos(angle)
        y = center_y + side_length * math.sin(angle)
        vertices.append((x, y))
    return np.array(vertices)


def check_hexagon_containment(hex_center_x, hex_center_y, side_length, outer_hex_side_length, outer_hex_angle=0):
    """Check if a hexagon is fully contained within outer hexagon."""
    inner_vertices = hexagon_vertices(hex_center_x, hex_center_y, side_length)
    outer_vertices = hexagon_vertices(0, 0, outer_hex_side_length, outer_hex_angle)
    outer_polygon = Polygon(outer_vertices)

    # Check all vertices are inside
    for vx, vy in inner_vertices:
        if not outer_polygon.contains(Point(vx, vy)):
            return False
    return True


def check_hexagon_overlap(hex1_center_x, hex1_center_y, hex1_angle, hex2_center_x, hex2_center_y, hex2_angle):
    """Check if two hexagons overlap."""
    hex1_vertices = hexagon_vertices(hex1_center_x, hex1_center_y, 1, hex1_angle)
    hex2_vertices = hexagon_vertices(hex2_center_x, hex2_center_y, 1, hex2_angle)

    hex1_polygon = Polygon(hex1_vertices)
    hex2_polygon = Polygon(hex2_vertices)

    return hex1_polygon.intersects(hex2_polygon)


def compute_outer_hex_side_length(inner_hex_data):
    """Compute the minimum side length of outer hexagon required to contain all inner hexagons."""
    # This is a simplified approach - in practice we'd want to do this more carefully
    max_dist = 0
    for i in range(len(inner_hex_data)):
        cx, cy, _ = inner_hex_data[i]
        # Distance from center to corner of unit hexagon
        dist_to_corner = 1  # for unit hexagon
        max_dist = max(max_dist, abs(cx) + dist_to_corner, abs(cy) + dist_to_corner)

    # Add some buffer to ensure complete containment
    return max_dist * 1.1


def objective_function(params):
    """Minimize negative inverse outer hex side length (maximize 1/R)."""
    # Unpack parameters: 12 hexagons * 3 params each = 36 parameters
    # But we'll use a more structured approach
    inner_hex_params = params.reshape(12, 3)  # 12 hexagons, 3 params each

    # Compute outer hexagon side length needed
    outer_side_length = compute_outer_hex_side_length(inner_hex_params)

    # Return negative inverse (since we want to maximize 1/R, we minimize -1/R)
    return -1.0 / outer_side_length


def constraint_containment(params):
    """Constraint ensuring all inner hexagons are contained within outer hexagon."""
    inner_hex_params = params.reshape(12, 3)
    # For now we'll use a simplified constraint - we'll return a positive value if any hex is outside
    # This will be used as a penalty in optimization

    outer_side_length = compute_outer_hex_side_length(inner_hex_params)

    # Simple containment test: check if hexagons are reasonably within bounds
    penalty = 0
    for i in range(len(inner_hex_params)):
        cx, cy, _ = inner_hex_params[i]
        distance_from_center = math.sqrt(cx*cx + cy*cy)
        # If distance exceeds outer hex side length, apply penalty
        if distance_from_center > outer_side_length:
            penalty += (distance_from_center - outer_side_length) * 1000

    return penalty


def constraint_nonoverlap(params):
    """Constraint ensuring no two inner hexagons overlap."""
    inner_hex_params = params.reshape(12, 3)
    penalty = 0

    # Check all pairs of hexagons
    for i in range(len(inner_hex_params)):
        for j in range(i+1, len(inner_hex_params)):
            cx1, cy1, angle1 = inner_hex_params[i]
            cx2, cy2, angle2 = inner_hex_params[j]

            if check_hexagon_overlap(cx1, cy1, angle1, cx2, cy2, angle2):
                # Apply strong penalty for overlaps
                penalty += 1000000

    return penalty
