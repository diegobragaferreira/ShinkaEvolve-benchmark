import numpy as np

from numba import njit

# Constants for hexagon geometry
HEX_RADIUS = 1.0  # unit hexagon radius
HEX_APOGEE = HEX_RADIUS * np.sqrt(3) / 2  # distance from center to side
HEX_SIDE = HEX_RADIUS  # side length for unit hexagon

@njit
def hexagon_vertices(x, y, angle_deg):
    """Generate vertices of a unit regular hexagon centered at (x,y) with given rotation"""
    angle_rad = np.radians(angle_deg)
    # Vertices of unit hexagon centered at origin
    base_vertices = np.array([
        [1.0, 0.0],
        [0.5, np.sqrt(3)/2],
        [-0.5, np.sqrt(3)/2],
        [-1.0, 0.0],
        [-0.5, -np.sqrt(3)/2],
        [0.5, -np.sqrt(3)/2]
    ])

    # Rotate and translate
    cos_a = np.cos(angle_rad)
    sin_a = np.sin(angle_rad)
    rot_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])

    rotated_vertices = base_vertices @ rot_matrix.T
    return rotated_vertices + np.array([x, y])

@njit
def point_in_hexagon(point_x, point_y, hex_x, hex_y, angle_deg):
    """Check if a point is inside a hexagon"""
    vertices = hexagon_vertices(hex_x, hex_y, angle_deg)
    poly = Polygon(vertices)
    return poly.contains(Point(point_x, point_y))

def check_overlap_pair(hex1_params, hex2_params):
    """Check if two hexagons overlap using fast approximation"""
    x1, y1, angle1 = hex1_params
    x2, y2, angle2 = hex2_params

    # Fast distance check first
    dx = x1 - x2
    dy = y1 - y2
    dist_sq = dx*dx + dy*dy

    # If centers are too far apart, they definitely don't overlap
    if dist_sq > (2 + 2*HEX_APOGEE)**2:
        return False

    # Generate vertices for both hexagons
    verts1 = hexagon_vertices(x1, y1, angle1)
    verts2 = hexagon_vertices(x2, y2, angle2)

    poly1 = Polygon(verts1)
    poly2 = Polygon(verts2)

    # Check if polygons intersect
    return poly1.intersects(poly2)

def check_containment(hex_params, outer_hex_center_x, outer_hex_center_y, outer_hex_radius):
    """Check if all vertices of a hexagon are contained within the outer hexagon"""
    x, y, angle = hex_params
    vertices = hexagon_vertices(x, y, angle)

    # Outer hexagon centered at origin with given radius
    outer_verts = hexagon_vertices(outer_hex_center_x, outer_hex_center_y, 0)
    outer_poly = Polygon(outer_verts)

    # Check that all vertices are inside the outer hexagon
    for vx, vy in vertices:
        if not outer_poly.contains(Point(vx, vy)):
            return False

    return True

def compute_outer_hex_radius(inner_hex_data, outer_hex_center=(0, 0)):
    """Compute minimum radius needed for outer hexagon to contain all inner hexagons"""
    max_dist = 0.0
    outer_x, outer_y = outer_hex_center

    for i in range(len(inner_hex_data)):
        x, y, angle = inner_hex_data[i]
        vertices = hexagon_vertices(x, y, angle)

        # Find maximum distance from outer center to any vertex
        for vx, vy in vertices:
            dist_sq = (vx - outer_x)**2 + (vy - outer_y)**2
            max_dist = max(max_dist, dist_sq)

    return np.sqrt(max_dist) + 0.1  # Add small buffer

def total_volume_constraint(inner_hex_data, outer_hex_radius):
    """Constraint function for optimization"""
    # Check overlaps between all pairs
    n = len(inner_hex_data)
    for i in range(n):
        for j in range(i+1, n):
            if check_overlap_pair(inner_hex_data[i], inner_hex_data[j]):
                return False  # Violates non-overlap constraint

    # Check containment
    for i in range(n):
        if not check_containment(inner_hex_data[i], 0, 0, outer_hex_radius):
            return False  # Violates containment constraint

    return True

def objective_function(params):
    """Minimize the negative of 1/outer_hex_radius (equivalent to maximizing 1/outer_hex_radius)"""
    # Extract parameters
    inner_params = params[:36].reshape((12, 3))  # 12 hexagons, 3 params each (x, y, angle)

    # Compute effective outer hexagon radius
    outer_radius = compute_outer_hex_radius(inner_params)

    # Return negative inverse radius to convert to minimization problem
    return -1.0 / (outer_radius + 1e-10)  # Small epsilon to avoid division by zero

def evaluate_solution(inner_hex_data, outer_hex_center=(0, 0)):
    """Evaluate the complete solution for metrics"""
    # Ensure we don't violate constraints
    n = len(inner_hex_data)

    # Check all pairwise overlaps
    for i in range(n):
        for j in range(i+1, n):
            if check_overlap_pair(inner_hex_data[i], inner_hex_data[j]):
                return 0.0  # Invalid solution

    # Check containment
    outer_radius = compute_outer_hex_radius(inner_hex_data, outer_hex_center)

    # Check if all vertices are contained
    for i in range(n):
        if not check_containment(inner_hex_data[i], outer_hex_center[0], outer_hex_center[1], outer_radius):
            return 0.0  # Invalid solution

    inv_outer_radius = 1.0 / (outer_radius + 1e-10)
    return inv_outer_radius
