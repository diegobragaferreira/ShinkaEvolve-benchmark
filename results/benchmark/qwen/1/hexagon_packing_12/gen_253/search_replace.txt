def create_hexagon_vertices(center, side_length, rotation_degrees):
    """Create vertices of a regular hexagon."""
    angle_rad = np.radians(rotation_degrees)
    angle_step = 2 * np.pi / 6
    vertices = []
    for i in range(6):
        angle = angle_step * i + angle_rad
        x = center[0] + side_length * np.cos(angle)
        y = center[1] + side_length * np.sin(angle)
        vertices.append((x, y))
    return np.array(vertices)

def check_overlap_pair(hex1_vertices, hex2_vertices):
    """Check if two hexagons overlap using fast geometric checks."""
    # Use a simple distance-based overlap check for speed
    # Calculate centroids
    hex1_centroid_x = 0.0
    hex1_centroid_y = 0.0
    hex2_centroid_x = 0.0
    hex2_centroid_y = 0.0

    for i in range(6):
        hex1_centroid_x += hex1_vertices[i, 0]
        hex1_centroid_y += hex1_vertices[i, 1]
        hex2_centroid_x += hex2_vertices[i, 0]
        hex2_centroid_y += hex2_vertices[i, 1]

    hex1_centroid_x /= 6
    hex1_centroid_y /= 6
    hex2_centroid_x /= 6
    hex2_centroid_y /= 6

    # Calculate distance between centroids
    dist_sq = (hex1_centroid_x - hex2_centroid_x)**2 + (hex1_centroid_y - hex2_centroid_y)**2

    # If centroids are too far apart, they don't overlap
    # Circumradius of unit hexagon is 1, so max distance between centers for overlap is ~2
    if dist_sq > 4.0:  # 2^2
        return False

    # For close hexagons, do a more detailed check using vertex distances
    # Minimum distance between any pair of vertices
    min_dist = np.inf
    for i in range(6):
        for j in range(6):
            dist_sq = (hex1_vertices[i, 0] - hex2_vertices[j, 0])**2 + (hex1_vertices[i, 1] - hex2_vertices[j, 1])**2
            if dist_sq < min_dist:
                min_dist = dist_sq

    # If minimum vertex distance is small enough, they overlap
    return min_dist < 1e-6  # Threshold for overlap
