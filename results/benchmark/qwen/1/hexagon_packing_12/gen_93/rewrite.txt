def create_hexagon_vertices(center_x, center_y, side_length=1, rotation_deg=0):
    """Create vertices of a regular hexagon."""
    rotation_rad = np.radians(rotation_deg)
    angles = np.linspace(0, 2*np.pi, 7) + rotation_rad
    vertices = np.column_stack([
        center_x + side_length * np.cos(angles),
        center_y + side_length * np.sin(angles)
    ])
    return vertices

def check_overlap(hex1_vertices, hex2_vertices):
    """Check if two hexagons overlap using Shapely."""
    poly1 = Polygon(hex1_vertices)
    poly2 = Polygon(hex2_vertices)
    return poly1.intersects(poly2)

def evaluate_solution(params, outer_side_length=None):
    """Evaluate a solution and return a fitness value."""
    # Reshape params into (12, 3) array: [x, y, angle] for each hexagon
    hex_params = params.reshape(-1, 3)
    
    # Get positions and rotations
    centers = hex_params[:, :2]
    rotations = hex_params[:, 2]
    
    # Create hexagon vertices
    hex_vertices_list = []
    for i, (center, rotation) in enumerate(zip(centers, rotations)):
        vertices = create_hexagon_vertices(center[0], center[1], 1, rotation)
        hex_vertices_list.append(vertices)
    
    # Check for overlaps
    penalty = 0
    for i in range(len(hex_vertices_list)):
        for j in range(i+1, len(hex_vertices_list)):
            if check_overlap(hex_vertices_list[i], hex_vertices_list[j]):
                penalty += 1000
    
    # If overlaps exist, return poor fitness
    if penalty > 0:
        return penalty + 1e6
    
    # Check containment
    if outer_side_length is not None:
        outer_vertices = create_hexagon_vertices(0, 0, outer_side_length, 0)
        for vertices in hex_vertices_list:
            if not check_containment(vertices, outer_vertices):
                penalty += 1000
    
    if penalty > 0:
        return penalty + 1e6
    
    # If we get here, solution is feasible
    # Minimize outer hexagon side length (maximize 1/outer_side_length)
    if outer_side_length is None:
        # Calculate minimum bounding box
        min_x, max_x = np.inf, -np.inf
        min_y, max_y = np.inf, -np.inf
        
        for vertices in hex_vertices_list:
            min_x = min(min_x, np.min(vertices[:, 0]))
            max_x = max(max_x, np.max(vertices[:, 0]))
            min_y = min(min_y, np.min(vertices[:, 1]))
            max_y = max(max_y, np.max(vertices[:, 1]))
        
        # Estimate required outer hexagon size
        width = max_x - min_x
        height = max_y - min_y
        # For hexagon, estimate side length from bounding box
        outer_side_length = max(width, height) / (2 * np.sqrt(3)) * 2
        
    # Return negative because we want to maximize 1/outer_side_length
    return -1.0 / outer_side_length

def get_symmetric_initial_positions(n):
    """Generate symmetric initial positions for hexagons."""
    positions = []
    
    # Center hexagon
    positions.append([0, 0, 0])
    
    # Surrounding hexagons in two rings
    # First ring: 6 hexagons around the center
    for i in range(6):
        angle = i * np.pi / 3
        x = np.cos(angle) * 2
        y = np.sin(angle) * 2
        positions.append([x, y, 0])
    
    # Second ring: 6 more hexagons
    for i in range(6):
        angle = i * np.pi / 3 + np.pi / 6
        x = np.cos(angle) * 3.5
        y = np.sin(angle) * 3.5  
        positions.append([x, y, 0])
    
    return np.array(positions)
