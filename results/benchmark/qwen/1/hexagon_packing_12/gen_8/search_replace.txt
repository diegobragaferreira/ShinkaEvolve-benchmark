import numpy as np
from math import sqrt, cos, sin, pi


def hexagon_vertices(center_x, center_y, side_length=1, rotation=0):
    """Generate vertices of a regular hexagon"""
    vertices = []
    for i in range(6):
        angle = rotation + i * pi / 3
        x = center_x + side_length * cos(angle)
        y = center_y + side_length * sin(angle)
        vertices.append((x, y))
    return np.array(vertices)


def distance_point_to_hexagon(point, hex_center, hex_side, hex_rotation):
    """Calculate minimum distance from point to hexagon boundary"""
    px, py = point
    hex_vertices = hexagon_vertices(hex_center[0], hex_center[1], hex_side, hex_rotation)

    # Find the closest vertex
    min_dist = float('inf')
    for vx, vy in hex_vertices:
        dist = sqrt((px - vx)**2 + (py - vy)**2)
        min_dist = min(min_dist, dist)

    return min_dist


def check_hexagon_containment(inner_hex_center, inner_hex_side, outer_hex_center, outer_hex_side, outer_hex_rotation):
    """Check if inner hexagon is fully contained in outer hexagon"""
    inner_vertices = hexagon_vertices(inner_hex_center[0], inner_hex_center[1], inner_hex_side, 0)

    # Check if all vertices are inside outer hexagon
    for vx, vy in inner_vertices:
        # Simple containment check - distance to center vs outer hex radius
        dx = vx - outer_hex_center[0]
        dy = vy - outer_hex_center[1]
        distance_from_center = sqrt(dx**2 + dy**2)

        # For a regular hexagon with side length s, the distance from center to vertex is s
        # We need to check that this distance is less than or equal to outer_hex_side
        if distance_from_center > outer_hex_side:
            return False

    return True


def calculate_outer_hex_side_length(inner_hex_data, outer_hex_center=[0, 0]):
    """Calculate minimal outer hexagon side length needed to contain all inner hexagons"""
    max_distance = 0
    inner_hex_side = 1

    for i in range(len(inner_hex_data)):
        center_x, center_y, _ = inner_hex_data[i]
        # Distance from outer center to inner hex center
        distance = sqrt((center_x - outer_hex_center[0])**2 + (center_y - outer_hex_center[1])**2)
        # Add hexagon radius (distance from center to vertex)
        max_distance = max(max_distance, distance + inner_hex_side)

    return max_distance
