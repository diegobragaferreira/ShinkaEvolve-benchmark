import numpy as np

def create_hexagon_vertices(center, side_length, angle_degrees):
    """Create vertices of a regular hexagon given center, side length, and rotation."""
    angle_rad = np.deg2rad(angle_degrees)
    angles = np.linspace(0, 2*np.pi, 7) + angle_rad
    vertices = np.column_stack([
        center[0] + side_length * np.cos(angles),
        center[1] + side_length * np.sin(angles)
    ])
    return vertices


def check_containment(hexagon_vertices, outer_hex_center, outer_hex_side_length):
    """Check if all vertices of a hexagon are inside the outer hexagon."""
    outer_vertices = create_hexagon_vertices(outer_hex_center, outer_hex_side_length, 0)

    # Create polygon from outer hexagon vertices
    outer_polygon = [(v[0], v[1]) for v in outer_vertices[:-1]]

    # Check if all inner hexagon vertices are inside outer hexagon
    for vertex in hexagon_vertices:
        x, y = vertex
        # Point-in-polygon test using winding number or ray casting
        # For simplicity, we'll check distance to center and use approximate containment
        dist_to_center = np.sqrt((x - outer_hex_center[0])**2 + (y - outer_hex_center[1])**2)
        # Maximum distance from center for a hexagon of side length L is L * sqrt(3)
        max_dist = outer_hex_side_length * np.sqrt(3)
        if dist_to_center > max_dist:
            return False
    return True


def compute_hexagon_distance(v1, v2):
    """Compute minimum distance between two hexagons given their vertices."""
    # Simplified approach: distance between centers minus radii
    # For unit hexagons, radius is 1
    center1 = np.mean(v1, axis=0)
    center2 = np.mean(v2, axis=0)
    dist_centers = np.linalg.norm(center1 - center2)
    min_dist = dist_centers - 2  # subtract twice the radius (2 units for unit hexagons)
    return min_dist


def objective_function(x):
    """Objective function to maximize 1/R (minimize R)."""
    # x contains: [R, centers_x1, centers_y1, ..., centers_x12, centers_y12]
    R = x[0]
    # Extract centers
    centers = x[1:].reshape(-1, 2)

    # Calculate penalty for violations
    penalty = 0

    # Check containment: all inner hexagons must be inside outer hexagon
    outer_center = np.array([0, 0])
    for i, center in enumerate(centers):
        # Distance from center of inner hexagon to center of outer hexagon
        dist = np.linalg.norm(center - outer_center)
        # For unit hexagon, maximum distance from center to vertex is sqrt(3) ~ 1.732
        # So for containment: dist + sqrt(3) <= R
        if dist + np.sqrt(3) > R:
            penalty += (dist + np.sqrt(3) - R)**2

    # Check non-overlap: minimum distance between hexagons
    for i in range(len(centers)):
        for j in range(i+1, len(centers)):
            # Get vertices of both hexagons
            h1_vertices = create_hexagon_vertices(centers[i], 1, 0)
            h2_vertices = create_hexagon_vertices(centers[j], 1, 0)

            # Compute distances
            dist = compute_hexagon_distance(h1_vertices, h2_vertices)
            if dist < 0:  # Overlap exists
                penalty += dist**2

    # Return negative of 1/R plus penalties (since we want to minimize)
    return -1/R + penalty
