<DESCRIPTION>
This change implements a more efficient bounding volume hierarchy (BVH) approach for overlap detection, which should significantly improve performance over the current cKDTree-based method. The BVH approach will provide faster overlap checking by first testing bounding boxes before doing expensive polygon intersection tests. This addresses the computational bottleneck identified in the overlap detection phase.

The key improvements:
1. Replace cKDTree-based spatial indexing with BVH approach for overlap detection
2. Implement proper bounding box construction for hexagons
3. Add efficient bounding box intersection tests
4. Use the BVH for early rejection of non-colliding pairs

    @staticmethod
    def _build_bvh_for_hexagons(positions: np.ndarray, angles: np.ndarray) -> List[Tuple[Tuple[float, float], Tuple[float, float], int]]:
        """Build a simple bounding volume hierarchy for hexagons using axis-aligned bounding boxes."""
        # For each hexagon, compute its AABB (axis-aligned bounding box)
        boxes = []
        for i, (pos, angle) in enumerate(zip(positions, angles)):
            vertices = HexagonPacker._hexagon_vertices(pos[0], pos[1], np.deg2rad(angle))
            min_x = vertices[:, 0].min()
            max_x = vertices[:, 0].max()
            min_y = vertices[:, 1].min()
            max_y = vertices[:, 1].max()
            boxes.append(((min_x, min_y), (max_x, max_y), i))
        return boxes

    @staticmethod
    def _bvh_intersect_boxes(box1: Tuple[Tuple[float, float], Tuple[float, float], int],
                           box2: Tuple[Tuple[float, float], Tuple[float, float], int]) -> bool:
        """Check if two bounding boxes intersect."""
        (min_x1, min_y1), (max_x1, max_y1) = box1[0], box1[1]
        (min_x2, min_y2), (max_x2, max_y2) = box2[0], box2[1]
        return not (max_x1 < min_x2 or max_x2 < min_x1 or max_y1 < min_y2 or max_y2 < min_y1)

    @staticmethod
    def _fast_check_overlaps_bvh(positions: np.ndarray, angles: np.ndarray) -> bool:
        """Fast overlap checking using BVH approach."""
        # Build bounding boxes for all hexagons
        boxes = HexagonPacker._build_bvh_for_hexagons(positions, angles)

        # Use the BVH approach for overlap detection
        n_hexagons = len(positions)
        # Check all pairs using bounding boxes as first filter
        for i in range(n_hexagons):
            for j in range(i+1, n_hexagons):
                # Early rejection using bounding boxes
                if HexagonPacker._bvh_intersect_boxes(boxes[i], boxes[j]):
                    # If bounding boxes overlap, do detailed polygon intersection test
                    hex1_vertices = HexagonPacker._hexagon_vertices(positions[i][0], positions[i][1],
                                                                  np.deg2rad(angles[i]))
                    hex2_vertices = HexagonPacker._hexagon_vertices(positions[j][0], positions[j][1],
                                                                  np.deg2rad(angles[j]))

                    if HexagonPacker._fast_check_overlap_pair_fast(hex1_vertices, hex2_vertices):
                        return True  # Found overlap

        return False  # No overlaps found
