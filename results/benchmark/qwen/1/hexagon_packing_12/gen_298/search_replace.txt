class HexagonGeometry:
    """Handles hexagon creation and geometric operations."""

    @staticmethod
    def create_vertices(center: Tuple[float, float], side_length: float, rotation_degrees: float) -> np.ndarray:
        """Create vertices of a regular hexagon."""
        angle_rad = np.radians(rotation_degrees)
        angle_step = 2 * np.pi / 6
        vertices = []
        for i in range(6):
            angle = angle_step * i + angle_rad
            x = center[0] + side_length * np.cos(angle)
            y = center[1] + side_length * np.sin(angle)
            vertices.append((x, y))
        return np.array(vertices)

    @staticmethod
    def get_circumradius(side_length: float) -> float:
        """Get the circumradius of a regular hexagon."""
        return side_length

class HexagonGeometry:
    """Handles hexagon creation and geometric operations."""

    @staticmethod
    @jit(nopython=True)
    def create_vertices_numba(center_x: float, center_y: float, side_length: float, rotation_degrees: float) -> np.ndarray:
        """Create vertices of a regular hexagon using Numba for speed."""
        angle_rad = np.radians(rotation_degrees)
        angle_step = 2 * np.pi / 6
        vertices = np.empty((6, 2))
        for i in range(6):
            angle = angle_step * i + angle_rad
            x = center_x + side_length * np.cos(angle)
            y = center_y + side_length * np.sin(angle)
            vertices[i] = [x, y]
        return vertices

    @staticmethod
    def create_vertices(center: Tuple[float, float], side_length: float, rotation_degrees: float) -> np.ndarray:
        """Create vertices of a regular hexagon."""
        return HexagonGeometry.create_vertices_numba(center[0], center[1], side_length, rotation_degrees)

    @staticmethod
    @jit(nopython=True)
    def get_circumradius_numba(side_length: float) -> float:
        """Get the circumradius of a regular hexagon using Numba."""
        return side_length

    def get_circumradius(side_length: float) -> float:
        """Get the circumradius of a regular hexagon."""

class ConstraintChecker:
    """Handles constraint validation for hexagon packing."""

    def __init__(self, outer_center: Tuple[float, float] = (0, 0)):
        self.outer_center = outer_center

    def check_containment(self, hex_vertices: np.ndarray, outer_side_length: float) -> bool:
        """Check if all vertices of a hexagon are inside the outer hexagon."""
        outer_vertices = HexagonGeometry.create_vertices(self.outer_center, outer_side_length, 0)
        outer_polygon = Polygon(outer_vertices)

        for vertex in hex_vertices:
            point = Point(vertex)
            if not outer_polygon.contains(point):
                return False
        return True

    def check_overlap_pair(self, hex1_vertices: np.ndarray, hex2_vertices: np.ndarray) -> bool:
        """Check if two hexagons overlap using Shapely."""
        hex1_polygon = Polygon(hex1_vertices)
        hex2_polygon = Polygon(hex2_vertices)
        return hex1_polygon.intersects(hex2_polygon)

    def get_outer_hex_side_from_config(self, inner_hex_data: np.ndarray, center: Tuple[float, float] = (0, 0)) -> float:
        """Compute the minimum required outer hexagon side length from current configuration."""
        if len(inner_hex_data) == 0:
            return 100.0

        max_dist = 0.0
        circumradius = HexagonGeometry.get_circumradius(1.0)

        for i in range(len(inner_hex_data)):

            dist = np.sqrt((cx - center[0])**2 + (cy - center[1])**2)
            dist_to_edge = dist + circumradius
            max_dist = max(max_dist, dist_to_edge)

        return max_dist * 2.0  # Diameter gives us the side length for a hexagon

class ConstraintChecker:
    """Handles constraint validation for hexagon packing."""

    def __init__(self, outer_center: Tuple[float, float] = (0, 0)):
        self.outer_center = outer_center

    @staticmethod
    @jit(nopython=True)
    def check_containment_numba(hex_vertices: np.ndarray, outer_side_length: float, outer_center_x: float, outer_center_y: float) -> bool:
        """Fast containment check using Numba."""
        # This is a simplified check that avoids creating Polygon objects for speed
        # We'll check if all vertices are within a circle of appropriate radius
        outer_radius = outer_side_length
        for i in range(len(hex_vertices)):
            vx, vy = hex_vertices[i]
            dist_sq = (vx - outer_center_x)**2 + (vy - outer_center_y)**2
            if dist_sq > outer_radius**2:
                return False
        return True

    def check_containment(self, hex_vertices: np.ndarray, outer_side_length: float) -> bool:
        """Check if all vertices of a hexagon are inside the outer hexagon."""
        # Using the fast numba version
        return ConstraintChecker.check_containment_numba(
            hex_vertices, outer_side_length, self.outer_center[0], self.outer_center[1]
        )

    def check_overlap_pair(self, hex1_vertices: np.ndarray, hex2_vertices: np.ndarray) -> bool:
        """Check if two hexagons overlap using Shapely."""
        hex1_polygon = Polygon(hex1_vertices)
        hex2_polygon = Polygon(hex2_vertices)
        return hex1_polygon.intersects(hex2_polygon)

    @staticmethod
    @jit(nopython=True)
    def get_outer_hex_side_from_config_numba(inner_hex_data: np.ndarray, center_x: float, center_y: float) -> float:
        """Fast computation of outer hexagon side length using Numba."""
        if len(inner_hex_data) == 0:

        max_dist = 0.0
        circumradius = 1.0  # For unit hexagons

        for i in range(len(inner_hex_data)):

        return max_dist * 2.0  # Diameter gives us the side length for a hexagon

    def get_outer_hex_side_from_config(self, inner_hex_data: np.ndarray, center: Tuple[float, float] = (0, 0)) -> float:
        """Compute the minimum required outer hexagon side length from current configuration."""
        return ConstraintChecker.get_outer_hex_side_from_config_numba(
            inner_hex_data, center[0], center[1]
        )
