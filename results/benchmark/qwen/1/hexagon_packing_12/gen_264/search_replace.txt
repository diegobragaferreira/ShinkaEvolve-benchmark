@jit(nopython=True)
def compute_hexagon_bounding_box_jit(x, y, angle_deg):
    """Compute axis-aligned bounding box for a hexagon using Numba."""
    vertices = hexagon_vertices(x, y, angle_deg)
    min_x = vertices[0, 0]
    max_x = vertices[0, 0]
    min_y = vertices[0, 1]
    max_y = vertices[0, 1]

    for i in range(1, 6):
        vx, vy = vertices[i]
        if vx < min_x: min_x = vx
        if vx > max_x: max_x = vx
        if vy < min_y: min_y = vy
        if vy > max_y: max_y = vy

    return min_x, max_x, min_y, max_y
