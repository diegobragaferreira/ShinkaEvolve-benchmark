@jit(nopython=True)
def create_unit_hexagon_vertices(center_x, center_y, rotation):
    """Create vertices of a unit regular hexagon with Numba JIT optimization."""
    angle_rad = rotation * np.pi / 180
    angle_step = 2 * np.pi / 6
    vertices = np.empty((6, 2))
    for i in range(6):
        angle = angle_step * i + angle_rad
        x = center_x + 1.0 * np.cos(angle)
        y = center_y + 1.0 * np.sin(angle)
        vertices[i] = (x, y)
    return vertices

@jit(nopython=True)
def point_in_hexagon_fast(point_x, point_y, hex_center_x, hex_center_y, rotation):
    """Fast check if a point is inside a regular hexagon using distance to edges."""
    # For a unit regular hexagon, distance from center to edge = sqrt(3)/2
    # Transform point to hexagon's coordinate system
    cos_rot = np.cos(rotation * np.pi / 180)
    sin_rot = np.sin(rotation * np.pi / 180)
    dx = point_x - hex_center_x
    dy = point_y - hex_center_y
    rot_x = dx * cos_rot + dy * sin_rot
    rot_y = -dx * sin_rot + dy * cos_rot

    # Distance from center to edge in x and y directions for unit hexagon
    edge_distance_x = np.sqrt(3) / 2
    edge_distance_y = 0.5

    # Check if point is within bounds
    return abs(rot_x) <= edge_distance_x and abs(rot_y) <= edge_distance_y and \
           abs(rot_x) + abs(rot_y) <= np.sqrt(3)

def create_unit_hexagon(center=(0,0), rotation=0):
    """Create a unit regular hexagon with given center and rotation."""
    vertices = create_unit_hexagon_vertices(center[0], center[1], rotation)
    return Polygon(vertices)
