class BVHNode:
    """Simple bounding volume hierarchy node for spatial acceleration"""
    def __init__(self, bounds=None, objects=None, left=None, right=None):
        self.bounds = bounds  # [min_x, min_y, max_x, max_y]
        self.objects = objects or []
        self.left = left
        self.right = right

@njit
def compute_bounding_box(vertices):
    """Compute axis-aligned bounding box for hexagon vertices"""
    min_x = vertices[0, 0]
    max_x = vertices[0, 0]
    min_y = vertices[0, 1]
    max_y = vertices[0, 1]

    for i in range(1, len(vertices)):
        min_x = min(min_x, vertices[i, 0])
        max_x = max(max_x, vertices[i, 0])
        min_y = min(min_y, vertices[i, 1])
        max_y = max(max_y, vertices[i, 1])

    return np.array([min_x, min_y, max_x, max_y])

def build_bvh(objects, max_objects_per_node=4):
    """Build a simple BVH for objects with bounding boxes"""
    if not objects:
        return None

    # Compute bounding boxes for all objects
    bounds_list = []
    for obj in objects:
        # Assuming object has a method to get its bounding box
        bounds_list.append(obj['bounds'])

    bounds_array = np.array(bounds_list)

    # Create root node
    root_bounds = np.array([
        np.min(bounds_array[:, 0]),
        np.min(bounds_array[:, 1]),
        np.max(bounds_array[:, 2]),
        np.max(bounds_array[:, 3])
    ])

    return build_bvh_recursive(objects, bounds_array, root_bounds, max_objects_per_node)

def build_bvh_recursive(objects, bounds_array, root_bounds, max_objects_per_node):
    """Recursively build BVH"""
    if len(objects) <= max_objects_per_node:
        return BVHNode(bounds=root_bounds, objects=objects)

    # Find split axis (longest dimension)
    extent_x = root_bounds[2] - root_bounds[0]
    extent_y = root_bounds[3] - root_bounds[1]
    split_axis = 0 if extent_x > extent_y else 1

    # Sort by median along split axis
    if split_axis == 0:
        # Split along x-axis
        sorted_indices = np.argsort(bounds_array[:, 0] + bounds_array[:, 2])  # Sort by midpoint
    else:
        # Split along y-axis
        sorted_indices = np.argsort(bounds_array[:, 1] + bounds_array[:, 3])  # Sort by midpoint

    mid = len(objects) // 2
    left_indices = sorted_indices[:mid]
    right_indices = sorted_indices[mid:]

    # Build left and right subtrees
    left_objects = [objects[i] for i in left_indices]
    right_objects = [objects[i] for i in right_indices]

    # Compute bounds for children
    left_bounds = np.array([
        np.min(bounds_array[left_indices, 0]),
        np.min(bounds_array[left_indices, 1]),
        np.max(bounds_array[left_indices, 2]),
        np.max(bounds_array[left_indices, 3])
    ])

    right_bounds = np.array([
        np.min(bounds_array[right_indices, 0]),
        np.min(bounds_array[right_indices, 1]),
        np.max(bounds_array[right_indices, 2]),
        np.max(bounds_array[right_indices, 3])
    ])

    left_child = build_bvh_recursive(left_objects, bounds_array[left_indices], left_bounds, max_objects_per_node)
    right_child = build_bvh_recursive(right_objects, bounds_array[right_indices], right_bounds, max_objects_per_node)

    return BVHNode(bounds=root_bounds, left=left_child, right=right_child)

def check_overlap_bvh_recursive(node1, node2, objects1, objects2):
    """Recursively check overlap between two BVH nodes"""
    if not node1 or not node2:
        return False

    # Quick bounds check
    if not intersect_bounds(node1.bounds, node2.bounds):
        return False

    # If both nodes are leaves, check pairs
    if node1.objects and node2.objects:
        for obj1 in node1.objects:
            for obj2 in node2.objects:
                # Use the fast overlap check for actual overlap
                if check_overlap_bvh_numba(obj1['vertices'], obj2['vertices']):
                    return True
        return False

    # If one is leaf and other is internal, traverse appropriately
    if node1.objects and not node2.objects:
        return check_overlap_bvh_recursive(node1, node2.left, objects1, objects2) or \
               check_overlap_bvh_recursive(node1, node2.right, objects1, objects2)
    elif not node1.objects and node2.objects:
        return check_overlap_bvh_recursive(node1.left, node2, objects1, objects2) or \
               check_overlap_bvh_recursive(node1.right, node2, objects1, objects2)
    else:
        # Both internal, recurse
        return (check_overlap_bvh_recursive(node1.left, node2.left, objects1, objects2) or
                check_overlap_bvh_recursive(node1.left, node2.right, objects1, objects2) or
                check_overlap_bvh_recursive(node1.right, node2.left, objects1, objects2) or
                check_overlap_bvh_recursive(node1.right, node2.right, objects1, objects2))
