import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
UNIT_HEX_RADIUS = 1.0  # radius of unit hexagon circumcircle
UNIT_HEX_WIDTH = 2.0  # width of unit hexagon
UNIT_HEX_HEIGHT = np.sqrt(3.0)  # height of unit hexagon


def get_hexagon_vertices(center_x, center_y, angle_deg, radius=1.0):
    """Get vertices of a regular hexagon given center, angle, and radius."""
    angle_rad = np.radians(angle_deg)
    angles = np.linspace(0, 2*np.pi, 7) + angle_rad
    x = center_x + radius * np.cos(angles)
    y = center_y + radius * np.sin(angles)
    return np.column_stack((x, y))


def check_containment(hex_vertices, outer_center_x, outer_center_y, outer_radius):
    """Check if all vertices of a hexagon are inside the outer hexagon."""
    outer_vertices = get_hexagon_vertices(outer_center_x, outer_center_y, 0, outer_radius)

    # Check if each vertex of inner hexagon is inside outer hexagon
    for vx, vy in hex_vertices:
        # Point-in-polygon test using winding number or ray casting
        # For hexagon, we can use the fact that it's convex
        # We'll use the cross product method for point-in-polygon

        # Convert to shapely polygon for robust containment testing
        try:
            from shapely.geometry import Point, Polygon
            outer_polygon = Polygon(outer_vertices)
            inner_point = Point(vx, vy)
            if not outer_polygon.contains(inner_point):
                return False
        except ImportError:
            # Fallback: manual containment check for hexagon
            # This is a simplified version - assumes hexagon is regular and centered
            dx = vx - outer_center_x
            dy = vy - outer_center_y
            distance = np.sqrt(dx*dx + dy*dy)
            if distance >= outer_radius:
                return False
    return True


def check_overlap(hex1_vertices, hex2_vertices):
    """Check if two hexagons overlap using separating axis theorem."""
    try:
        from shapely.geometry import Polygon
        poly1 = Polygon(hex1_vertices)
        poly2 = Polygon(hex2_vertices)
        return poly1.intersects(poly2)
    except ImportError:
        # Simplified fallback: check if any vertices of one hexagon are inside the other
        # This is a rough approximation but works for basic overlap detection
        for vx, vy in hex1_vertices:
            # Check if this vertex is inside hex2
            inside = point_in_hexagon(vx, vy, hex2_vertices)
            if inside:
                return True
        for vx, vy in hex2_vertices:
            # Check if this vertex is inside hex1
            inside = point_in_hexagon(vx, vy, hex1_vertices)
            if inside:
                return True
        return False


def point_in_hexagon(px, py, hex_vertices):
    """Check if a point is inside a hexagon using ray casting."""
    # Simple point-in-polygon test (assuming convex polygon)
    n = len(hex_vertices)
    inside = False

    p1x, p1y = hex_vertices[0]
    for i in range(1, n+1):
        p2x, p2y = hex_vertices[i % n]
        if py > min(p1y, p2y):
            if py <= max(p1y, p2y):
                if px <= max(p1x, p2x):
                    if p1y != p2y:
                        xinters = (py - p1y) * (p2x - p1x) / (p2y - p1y) + p1x
                    if p1x == p2x or px <= xinters:
                        inside = not inside
        p1x, p1y = p2x, p2y

    return inside


def calculate_outer_hex_radius(inner_hex_data, outer_center_x=0, outer_center_y=0):
    """Estimate the minimal outer hexagon radius needed to contain all inner hexagons."""
    max_dist = 0
    for i in range(len(inner_hex_data)):
        cx, cy, angle = inner_hex_data[i]
        vertices = get_hexagon_vertices(cx, cy, angle)

        # Calculate distance from center to each vertex
        for vx, vy in vertices:
            dist = np.sqrt((vx - outer_center_x)**2 + (vy - outer_center_y)**2)
            max_dist = max(max_dist, dist)

    return max_dist + UNIT_HEX_RADIUS  # Add buffer for hexagon size


def objective_function(x):
    """Objective function to minimize (negative of 1/outer_radius)."""
    # Decode parameters: 12 hexagons with (x,y,angle) each, plus outer hexagon center
    # x contains [cx1, cy1, angle1, ..., cx12, cy12, angle12, outer_cx, outer_cy, outer_radius]
    n = 12
    inner_params = x[:3*n].reshape((n, 3))
    outer_center_x = x[3*n]
    outer_center_y = x[3*n+1]
    outer_radius = x[3*n+2]

    # Check feasibility
    if outer_radius < UNIT_HEX_RADIUS:
        return 1e6  # penalty for invalid radius

    # Check containment
    for i in range(n):
        cx, cy, angle = inner_params[i]
        vertices = get_hexagon_vertices(cx, cy, angle)
        if not check_containment(vertices, outer_center_x, outer_center_y, outer_radius):
            return 1e6  # penalty for containment violation

    # Check overlaps
    for i in range(n):
        for j in range(i+1, n):
            cx1, cy1, angle1 = inner_params[i]
            cx2, cy2, angle2 = inner_params[j]
            vertices1 = get_hexagon_vertices(cx1, cy1, angle1)
            vertices2 = get_hexagon_vertices(cx2, cy2, angle2)
            if check_overlap(vertices1, vertices2):
                return 1e6  # penalty for overlap

    # Return negative of 1/outer_radius (to maximize 1/outer_radius)
    return -1.0 / outer_radius
