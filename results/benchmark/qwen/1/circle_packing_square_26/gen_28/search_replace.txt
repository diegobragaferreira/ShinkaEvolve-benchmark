def generate_voronoi_initial_config(n):
    """Generate initial circle configuration based on adaptive Voronoi-like principles"""
    circles = np.zeros((n, 3))

    # Distribute points more evenly using a modified grid
    # We'll use a honeycomb-like approach for better distribution
    sqrt_n = int(math.ceil(math.sqrt(n)))

    # Create approximately square grid
    rows = math.ceil(n / sqrt_n)
    cols = math.ceil(n / rows)

    # Grid spacing (with margin)
    spacing_x = 0.95 / (cols + 1)
    spacing_y = 0.95 / (rows + 1)

    # Generate initial points
    count = 0
    for i in range(rows):
        for j in range(cols):
            if count >= n:
                break
            # Add some randomness to avoid perfect grid pattern
            x = (j + 1) * spacing_x + random.uniform(-spacing_x/4, spacing_x/4)
            y = (i + 1) * spacing_y + random.uniform(-spacing_y/4, spacing_y/4)
            # Initial radius based on spacing and some randomness
            r = min(spacing_x, spacing_y) * random.uniform(0.3, 0.5)
            circles[count] = [x, y, r]
            count += 1

    # Adjust for containment and initial overlap prevention
    for i in range(n):
        x, y, r = circles[i]
        # Bound radii to stay within square
        circles[i] = [x, y, min(r, x, 1-x, y, 1-y)]

    return circles

def poisson_disk_sampling(radius, width=1.0, height=1.0, k=30):
    """
    Generate points using Poisson disk sampling for uniform distribution.
    Returns array of (x, y) coordinates.
    """
    # Grid for efficient neighbor lookup
    cell_size = radius / np.sqrt(2)
    grid_width = int(width / cell_size) + 1
    grid_height = int(height / cell_size) + 1
    grid = [[None for _ in range(grid_width)] for _ in range(grid_height)]

    # Active list for points to consider
    active_list = []
    points = []

    # Add first point randomly
    x = random.uniform(0, width)
    y = random.uniform(0, height)
    points.append((x, y))
    active_list.append((x, y))

    # Add to grid
    grid_x, grid_y = int(x / cell_size), int(y / cell_size)
    grid[grid_y][grid_x] = (x, y)

    while active_list:
        # Pick random point from active list
        idx = random.randint(0, len(active_list) - 1)
        point = active_list[idx]

        # Try up to k times to place a new point
        placed = False
        for _ in range(k):
            angle = random.uniform(0, 2 * math.pi)
            r = random.uniform(radius, 2 * radius)
            new_x = point[0] + r * math.cos(angle)
            new_y = point[1] + r * math.sin(angle)

            # Check if point is within bounds
            if new_x < 0 or new_x >= width or new_y < 0 or new_y >= height:
                continue

            # Check if point is too close to existing points
            grid_x = int(new_x / cell_size)
            grid_y = int(new_y / cell_size)

            valid = True
            # Check nearby cells in grid
            for i in range(max(0, grid_y - 2), min(grid_height, grid_y + 3)):
                for j in range(max(0, grid_x - 2), min(grid_width, grid_x + 3)):
                    if grid[i][j] is not None:
                        px, py = grid[i][j]
                        dist_sq = (new_x - px)**2 + (new_y - py)**2
                        if dist_sq < radius**2:
                            valid = False
                            break
                if not valid:
                    break

            if valid:
                points.append((new_x, new_y))
                active_list.append((new_x, new_y))

                # Add to grid
                grid[grid_y][grid_x] = (new_x, new_y)
                placed = True
                break

        if not placed:
            active_list.pop(idx)

    return np.array(points)

def generate_voronoi_initial_config(n):
    """Generate initial circle configuration using Poisson disk sampling for better distribution"""
    circles = np.zeros((n, 3))

    # Generate Poisson disk points
    # Use a reasonable minimum radius to ensure good distribution
    min_radius = 0.1
    points = poisson_disk_sampling(min_radius, width=0.95, height=0.95, k=30)

    # If we don't have enough points, add more using a grid fallback
    if len(points) < n:
        # Use regular grid for remaining points
        sqrt_n = int(math.ceil(math.sqrt(n)))
        rows = math.ceil(n / sqrt_n)
        cols = math.ceil(n / rows)
        spacing_x = 0.95 / (cols + 1)
        spacing_y = 0.95 / (rows + 1)

        for i in range(rows):
            for j in range(cols):
                if len(points) >= n:
                    break
                x = (j + 1) * spacing_x + random.uniform(-spacing_x/4, spacing_x/4)
                y = (i + 1) * spacing_y + random.uniform(-spacing_y/4, spacing_y/4)
                points = np.vstack([points, [x, y]])

    # Trim to exactly n points
    points = points[:n]

    # Assign initial radii based on density and spacing considerations
    for i, (x, y) in enumerate(points):
        # Initial radius based on proximity to neighbors
        min_dist = float('inf')
        for j, (other_x, other_y) in enumerate(points):
            if i != j:
                dist = np.sqrt((x - other_x)**2 + (y - other_y)**2)
                min_dist = min(min_dist, dist)

        # Set initial radius to be about 15% of minimum neighbor distance
        # or a reasonable fraction if no neighbors exist
        initial_r = min(0.15, min_dist * 0.15) if min_dist < float('inf') else 0.05
        # Ensure it's within bounds
        initial_r = min(initial_r, x, 1-x, y, 1-y)
        circles[i] = [x, y, max(0.001, initial_r)]

    return circles
