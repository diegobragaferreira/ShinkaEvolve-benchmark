    Apply constraint-aware local search to improve the solution.
    This attempts to iteratively improve the configuration by:
    1. Resolving overlap constraints through position adjustments
    2. Maximizing radii while maintaining constraints

    improved_circles = circles.copy()

    n = len(improved_circles)

    for iteration in range(max_iterations):
        # Track if any improvement was made
        improved = False

        for i in range(n):

            max_radius = min(x, 1-x, y, 1-y)

            # Find neighboring circles to check constraints
            neighbors = []
            for j in range(n):
                if i != j:
                    x2, y2, r2 = improved_circles[j]
                    dist_sq = (x - x2)**2 + (y - y2)**2
                    min_dist_sq = (r + r2)**2
                    neighbors.append((j, dist_sq, min_dist_sq))

            min_dist_to_neighbor = float('inf')

    This attempts to iteratively improve the configuration by:
    1. Resolving overlap constraints through position adjustments
    2. Maximizing radii while maintaining constraints

    improved_circles = circles.copy()

    n = len(improved_circles)

    for iteration in range(max_iterations):
        # Track if any improvement was made
        improved = False

        # Sort circles by how constrained they are (smaller radii or more neighbors)
        # This prioritizes fixing the most problematic circles first
        circle_constraints = []
        for i in range(n):
            x, y, r = improved_circles[i]

            # Count overlapping neighbors
            overlap_count = 0
            max_radius = min(x, 1-x, y, 1-y)
            neighbors = []

            for j in range(n):
                if i != j:
                    x2, y2, r2 = improved_circles[j]
                    dist_sq = (x - x2)**2 + (y - y2)**2
                    min_dist_sq = (r + r2)**2
                    if dist_sq < min_dist_sq:
                        overlap_count += 1
                        neighbors.append((j, dist_sq, min_dist_sq))

            # Circle constraint score: higher means more constrained
            constraint_score = overlap_count + (0.5 if r < 0.05 else 0)
            circle_constraints.append((constraint_score, i))

        # Sort by constraint score (descending - most constrained first)
        circle_constraints.sort(reverse=True)

        # Process circles in order of constraint
