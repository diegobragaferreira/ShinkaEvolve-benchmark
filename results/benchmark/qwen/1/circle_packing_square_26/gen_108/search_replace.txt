    def _generate_voronoi_density_initial_config(self):
        """Generate initial circle configuration using golden spiral with density-aware radius assignment."""
        circles = np.zeros((self.N_CIRCLES, 3))

        # Golden spiral distribution
        golden_angle = np.pi * (3 - np.sqrt(5))

        # Create a coarse density grid to understand spatial distribution
        grid_size = 8
        density_grid = np.zeros((grid_size, grid_size))

        # Sample points for density estimation
        sample_points = []
        sample_count = min(100, self.N_CIRCLES * 2)  # Sample more points for better density estimate

        # Use a subset of spiral points for density estimation
        for i in range(sample_count):
            r = np.sqrt(i / (sample_count - 1)) if sample_count > 1 else 0
            theta = i * golden_angle
            x = 0.4 * r * np.cos(theta) + 0.5
            y = 0.4 * r * np.sin(theta) + 0.5
            if 0.01 <= x <= 0.99 and 0.01 <= y <= 0.99:
                sample_points.append([x, y])

        # Estimate density in grid cells
        for point in sample_points:
            x, y = point
            grid_x = int(x * grid_size)
            grid_y = int(y * grid_size)
            grid_x = min(grid_x, grid_size - 1)
            grid_y = min(grid_y, grid_size - 1)
            density_grid[grid_y, grid_x] += 1

        # Normalize density grid
        max_density = np.max(density_grid)
        if max_density > 0:
            density_grid = density_grid / max_density

        for i in range(self.N_CIRCLES):
            # Improved spiral layout with better distribution
            r = np.sqrt(i / (self.N_CIRCLES - 1)) if self.N_CIRCLES > 1 else 0
            theta = i * golden_angle

            # Convert to Cartesian coordinates in [0.1, 0.9] range to keep margin
            x = 0.4 * r * np.cos(theta) + 0.5
            y = 0.4 * r * np.sin(theta) + 0.5

            # Apply small random perturbations for diversity
            x += random.uniform(-0.015, 0.015)
            y += random.uniform(-0.015, 0.015)

            # Clamp to valid range
            x = max(0.01, min(0.99, x))
            y = max(0.01, min(0.99, y))

            # Calculate density-based initial radius
            grid_x = int(x * grid_size)
            grid_y = int(y * grid_size)
            grid_x = min(grid_x, grid_size - 1)
            grid_y = min(grid_y, grid_size - 1)

            # Density factor: lower density = higher radius potential
            density_factor = 1.0 / (density_grid[grid_y, grid_x] + 0.1)  # Add small value to avoid division by zero

            # Set initial radius based on distance to borders, availability and density
            min_dist_to_edge = min(x, 1-x, y, 1-y)
            # Base radius calculation with some randomness and density consideration
            base_radius = min(min_dist_to_edge, 0.15) * random.uniform(0.5, 0.8) * density_factor
            r = max(0.005, base_radius)

            circles[i] = [x, y, r]

        return circles
