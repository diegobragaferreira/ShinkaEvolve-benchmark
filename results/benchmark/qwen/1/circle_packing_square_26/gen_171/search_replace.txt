    def _generate_voronoi_initial_config(self):
        """Generate initial circle configuration using Voronoi-inspired distribution."""
        circles = np.zeros((self.N_CIRCLES, 3))

        # Create points using a Voronoi-like approach
        # Generate points that are roughly evenly distributed
        n_points = max(30, self.N_CIRCLES * 2)  # More points for better Voronoi coverage
        points = np.random.rand(n_points, 2)

        # Select subset that provides good spread
        # Using a simplified approach: select points that are reasonably far apart
        selected_points = [points[0]]  # Always include first point
        for point in points[1:]:
            # Check distance to all selected points
            min_dist = min(np.sqrt(np.sum((point - selected_point)**2))
                          for selected_point in selected_points)
            # Only add if sufficiently far from existing points
            if min_dist > 0.15 or len(selected_points) < 10:  # Allow some close points initially
                selected_points.append(point)
                if len(selected_points) >= self.N_CIRCLES:
                    break

        # Ensure we have enough points
        while len(selected_points) < self.N_CIRCLES:
            # Add random points in valid regions
            selected_points.append(np.random.rand(2))

        selected_points = np.array(selected_points[:self.N_CIRCLES])

        # Now create circles with positions from these points
        for i, (x, y) in enumerate(selected_points):
            # Scale to interior of unit square to allow for margins
            x = 0.1 + 0.8 * x
            y = 0.1 + 0.8 * y

            # Set initial radius based on distance to borders and available space
            min_dist_to_edge = min(x, 1-x, y, 1-y)
            # Base radius calculation with some randomness
            base_radius = min(min_dist_to_edge, 0.15) * random.uniform(0.7, 0.95)
            r = max(0.005, base_radius)

            circles[i] = [x, y, r]

        return circles
