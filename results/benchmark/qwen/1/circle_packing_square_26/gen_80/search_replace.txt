        # Create a more structured grid-based initialization with some randomness
        # This creates better initial spatial distribution for circle packing

        grid_size = int(np.ceil(np.sqrt(self.N_CIRCLES)))
        if grid_size * grid_size < self.N_CIRCLES:
            grid_size += 1

        spacing_x = 0.9 / (grid_size + 1)
        spacing_y = 0.9 / (grid_size + 1)

        # Generate points on a grid with slight jitter
        count = 0
        for i in range(grid_size):
            for j in range(grid_size):
                if count >= self.N_CIRCLES:
