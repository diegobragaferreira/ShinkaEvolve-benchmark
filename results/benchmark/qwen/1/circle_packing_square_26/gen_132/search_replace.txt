I'll implement a hierarchical clustering approach that first groups the unit square into regions and then places circles in these clusters using a more strategic approach. This should provide better spatial distribution and potentially lead to higher sum of radii.

The key insight is to first identify promising regions using a coarse-grained clustering approach, then place circles strategically within these regions to avoid poor initial positioning that leads to suboptimal convergence.

import numpy as np
from scipy.spatial import Voronoi
from scipy.spatial.distance import cdist
import time

def _initialize_voronoi_placement(n_circles: int, seed: int = 42) -> np.ndarray:
    """Initialize circle positions using a refined Voronoi-based approach."""
    np.random.seed(seed)

    # Create a denser grid for better distribution
    grid_size = max(5, int(np.ceil(np.sqrt(n_circles)) * 1.5))
    x_coords = np.linspace(0.05, 0.95, grid_size)
    y_coords = np.linspace(0.05, 0.95, grid_size)
    grid_points = np.array([[x, y] for x in x_coords for y in y_coords])

    # Take the first n_circles points (or generate more if needed)
    if len(grid_points) >= n_circles:
        initial_points = grid_points[:n_circles]
    else:
        # If we don't have enough grid points, add random points
        extra_points = n_circles - len(grid_points)
        random_points = np.random.uniform(0.05, 0.95, (extra_points, 2))
        initial_points = np.vstack([grid_points, random_points])

    # Create Voronoi diagram and use centroids as circle centers
    try:
        vor = Voronoi(initial_points)
        # Use the original points as circle centers (more reliable than vertices)
        centroids = vor.points
        # Filter out points that are too close to edges or invalid
        valid_centroids = []
        for point in centroids:
            if 0.05 <= point[0] <= 0.95 and 0.05 <= point[1] <= 0.95:
                valid_centroids.append(point)

        if len(valid_centroids) < n_circles:
            # Fill in missing points randomly
            remaining = n_circles - len(valid_centroids)
            random_points = np.random.uniform(0.05, 0.95, (remaining, 2))
            final_points = np.vstack([valid_centroids, random_points])
        else:
            final_points = np.array(valid_centroids[:n_circles])

    except Exception:
        # Fallback to random initialization if Voronoi fails
        final_points = np.random.uniform(0.05, 0.95, (n_circles, 2))

    return final_points

import numpy as np
from scipy.spatial import Voronoi
from scipy.spatial.distance import cdist
from sklearn.cluster import AgglomerativeClustering
import time

def _initialize_hierarchical_placement(n_circles: int, seed: int = 42) -> np.ndarray:
    """Initialize circle positions using hierarchical clustering for better spatial distribution."""
    np.random.seed(seed)

    # First, create a coarse grid to identify promising regions
    coarse_grid_size = max(3, int(np.ceil(np.sqrt(n_circles))))
    x_coords = np.linspace(0.05, 0.95, coarse_grid_size)
    y_coords = np.linspace(0.05, 0.95, coarse_grid_size)
    coarse_points = np.array([[x, y] for x in x_coords for y in y_coords])

    # Use hierarchical clustering to group regions
    clustering = AgglomerativeClustering(n_clusters=min(n_circles, len(coarse_points)),
                                        linkage='ward')
    cluster_labels = clustering.fit_predict(coarse_points)

    # Select representative points from each cluster
    selected_points = []
    for cluster_id in range(max(cluster_labels) + 1):
        cluster_mask = cluster_labels == cluster_id
        if np.any(cluster_mask):
            # Get the centroid of the cluster
            cluster_points = coarse_points[cluster_mask]
            centroid = np.mean(cluster_points, axis=0)
            selected_points.append(centroid)

    # If we don't have enough points, fill with random ones
    while len(selected_points) < n_circles:
        random_point = np.random.uniform(0.05, 0.95, (2,))
        selected_points.append(random_point)

    # Trim to exactly n_circles
    final_points = np.array(selected_points[:n_circles])

    # Add some randomness to prevent perfect grid patterns that might not generalize well
    noise_level = 0.05
    final_points += np.random.uniform(-noise_level, noise_level, final_points.shape)

    # Ensure all points are within bounds
    final_points[:, 0] = np.clip(final_points[:, 0], 0.05, 0.95)
    final_points[:, 1] = np.clip(final_points[:, 1], 0.05, 0.95)

    return final_points
