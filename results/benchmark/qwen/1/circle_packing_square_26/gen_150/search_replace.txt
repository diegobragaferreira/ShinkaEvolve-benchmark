def _validate_circle_placement(circles: np.ndarray) -> bool:
    """Validate that circles are within bounds and don't overlap."""
    n = len(circles)

    # Check containment constraints
    for i in range(n):
        x, y, r = circles[i]
        if r <= 0 or x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return False

    # Check non-overlap constraints using KDTree for efficiency
    points = circles[:, :2]
    tree = cKDTree(points)

    # Find all pairs within distance 2*r (minimum separation needed to avoid overlap)
    pairs = tree.query_pairs(2 * min(circles[:, 2]), output_type='ndarray')

    for i, j in pairs:
        x1, y1, r1 = circles[i]
        x2, y2, r2 = circles[j]
        distance_sq = (x1 - x2)**2 + (y1 - y2)**2
        min_distance_sq = (r1 + r2)**2
        if distance_sq < min_distance_sq:
            return False

    return True

def _calculate_diversity(population: list) -> float:
    """Calculate population diversity based on Euclidean distances between individuals."""
    if len(population) < 2:
        return 0.0

    # Flatten all individuals for distance calculation
    flat_pop = np.array([ind for ind in population]).reshape(len(population), -1)

    # Calculate pairwise distances
    distances = []
    for i in range(len(flat_pop)):
        for j in range(i+1, len(flat_pop)):
            dist = np.linalg.norm(flat_pop[i] - flat_pop[j])
            distances.append(dist)

    if distances:
        return np.std(distances)
    return 0.0

def _speciate_population(population: list, n_species: int = 5) -> list:
    """Group similar individuals into species based on geometric similarity."""
    if len(population) < n_species:
        return [population]

    # Convert to numpy array for easier processing
    flat_pop = np.array([ind for ind in population]).reshape(len(population), -1)

    # Calculate pairwise distances between individuals
    n = len(flat_pop)
    distance_matrix = np.zeros((n, n))

    for i in range(n):
        for j in range(n):
            if i != j:
                distance_matrix[i, j] = np.linalg.norm(flat_pop[i] - flat_pop[j])
            else:
                distance_matrix[i, j] = float('inf')

    # Simple clustering based on minimum distance to other individuals
    species = [[] for _ in range(n_species)]

    # Assign each individual to a species (simple round-robin assignment for now)
    for i, individual in enumerate(population):
        species[i % n_species].append(individual)

    return species

def _adaptive_mutation_rate(population: list, generation: int) -> float:
    """Calculate adaptive mutation rate based on population diversity and generation."""
    diversity = _calculate_diversity(population)
    max_gen = 100  # Estimated max generations

    # Base mutation rate that decreases over time
    base_rate = 0.15 - (generation / max_gen) * 0.10  # From 0.15 to 0.05
    base_rate = max(base_rate, 0.05)

    # Increase mutation rate when diversity is low (to escape local optima)
    diversity_factor = max(0.8, 1.0 - diversity / 10.0)

    return base_rate * diversity_factor
