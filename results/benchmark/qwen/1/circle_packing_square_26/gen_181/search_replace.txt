        circles = np.zeros((self.N_CIRCLES, 3))

        # Create more sophisticated initial point distribution using a modified Poisson disk sampling approach
        # but start with a grid and add jitter to avoid regular patterns
        points = []

        spacing = 1.0 / (grid_size + 1)

        for i in range(grid_size):
            for j in range(grid_size):
                if len(points) >= self.N_CIRCLES:
                    break
                x = (j + 1) * spacing + random.uniform(-spacing/4, spacing/4)
                y = (i + 1) * spacing + random.uniform(-spacing/4, spacing/4)
                # Only add if within the valid region

                    points.append([x, y])

        # Ensure we have enough points

        # Move points to maximize minimum distances to each other
