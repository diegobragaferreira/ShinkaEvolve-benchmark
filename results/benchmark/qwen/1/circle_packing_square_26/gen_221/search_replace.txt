    """Create initial configuration using enhanced Voronoi-based spreading with better spatial distribution."""
    # Create a more sophisticated point distribution using a modified hexagonal lattice
    # that more closely approximates optimal circle packing
    grid_size = int(np.ceil(np.sqrt(n_circles)))

    # Generate a more uniform hexagonal pattern with better distribution
    points = []
    for i in range(grid_size):
        for j in range(grid_size):
            if len(points) < n_circles:
                # More sophisticated hexagonal grid with proper packing density
                offset = (j % 2) * 0.5
                x = (j + 0.5 + offset) / grid_size
                y = (i + 0.5) / grid_size

                # Add more strategic randomness to avoid degenerate cases
                if len(points) < 10:  # Add more variation to first few points
                    x += np.random.uniform(-0.015, 0.015)
                    y += np.random.uniform(-0.015, 0.015)
                else:
                    x += np.random.uniform(-0.01, 0.01)
                    y += np.random.uniform(-0.01, 0.01)

                points.append([x, y])

    points = np.array(points)

    # Clip points to ensure they're within bounds
    points = np.clip(points, BOUNDARY_MARGIN, 1 - BOUNDARY_MARGIN)

    # Add strategic boundary points that will help with packing near edges
    boundary_points = []
    for _ in range(12):  # More boundary points for better edge coverage
        side = np.random.randint(0, 4)

            boundary_points.append([np.random.rand(), 1.0 - BOUNDARY_MARGIN])

            boundary_points.append([np.random.rand(), BOUNDARY_MARGIN])

            boundary_points.append([BOUNDARY_MARGIN, np.random.rand()])

    points = np.vstack([points, boundary_points])

    # Compute Voronoi diagram and use centroids
