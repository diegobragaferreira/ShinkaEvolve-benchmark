import numpy as np
from scipy.spatial import cKDTree
from scipy.spatial.distance import cdist
import random
from collections import defaultdict
import time

# Global constants
POPULATION_SIZE = 100
GENERATIONS = 200
TOURNAMENT_SIZE = 5
MUTATION_RATE = 0.1
CROSSOVER_RATE = 0.8
LOCAL_SEARCH_ITERATIONS = 10
SEED = 42

random.seed(SEED)
np.random.seed(SEED)

def create_voronoi_initialization(n):
    """Create initial configuration using Voronoi-like distribution"""
    # Generate points in a grid pattern
    points = []
    grid_size = int(np.ceil(np.sqrt(n)))
    
    # Create a roughly uniform distribution of points
    for i in range(grid_size):
        for j in range(grid_size):
            x = (i + 0.5) / grid_size
            y = (j + 0.5) / grid_size
            if len(points) < n:
                points.append([x, y])
    
    # Add some randomness to avoid perfect grid
    for point in points:
        point[0] += (np.random.random() - 0.5) * 0.1
        point[1] += (np.random.random() - 0.5) * 0.1
    
    # Ensure points are within bounds
    points = np.clip(points, 0, 1)
    
    # Initialize circles with small radii
    circles = np.zeros((n, 3))
    for i, (x, y) in enumerate(points):
        circles[i] = [x, y, 0.01]
    
    return circles

def is_valid_configuration(circles):
    """Check if configuration satisfies all constraints"""
    n = len(circles)
    
    # Check containment
    for i in range(n):
        x, y, r = circles[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return False
    
    # Check overlaps using KDTree for efficiency
    positions = circles[:, :2]
    tree = cKDTree(positions)
    
    # Check pairwise distances
    for i in range(n):
        x1, y1, r1 = circles[i]
        # Find neighbors within distance 2*r1 (potential overlaps)
        neighbors = tree.query_ball_point([x1, y1], 2*r1)
        for j in neighbors:
            if i != j:
                x2, y2, r2 = circles[j]
                distance = np.sqrt((x1-x2)**2 + (y1-y2)**2)
                if distance < r1 + r2:
                    return False
    
    return True

def calculate_sum_radii(circles):
    """Calculate the sum of all radii"""
    return np.sum(circles[:, 2])

def constraint_aware_local_search(circles, max_iterations=10):
    """Perform local search to improve configuration while respecting constraints"""
    best_circles = circles.copy()
    best_sum = calculate_sum_radii(best_circles)
    
    for _ in range(max_iterations):
        # Try to increase radii while maintaining constraints
        improved = False
        
        # Try increasing radii one by one
        for i in range(len(circles)):
            x, y, r = best_circles[i]
            
            # Calculate maximum possible radius without violating constraints
            max_radius = min(x, 1-x, y, 1-y)
            
            # Find minimum distance to other circles
            min_distance = float('inf')
            for j in range(len(circles)):
                if i != j:
                    x2, y2, r2 = best_circles[j]
                    distance = np.sqrt((x-x2)**2 + (y-y2)**2)
                    min_distance = min(min_distance, distance)
            
            # Maximum allowable radius considering overlaps
            if min_distance != float('inf'):
                max_radius = min(max_radius, min_distance - 0.001)
            
            # Try to increase radius up to maximum
            new_r = min(r + 0.001, max_radius)
            if new_r > r + 1e-6:
                # Update temporarily and check validity
                temp_circles = best_circles.copy()
                temp_circles[i, 2] = new_r
                
                # Check if this change maintains validity
                if is_valid_configuration(temp_circles):
                    best_circles[i, 2] = new_r
                    improved = True
        
        if not improved:
            break
            
        current_sum = calculate_sum_radii(best_circles)
        if current_sum > best_sum:
            best_sum = current_sum
        else:
            break
    
    return best_circles

def crossover(parent1, parent2):
    """Perform crossover operation"""
    if np.random.random() > CROSSOVER_RATE:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    # Simple uniform crossover
    child1 = parent1.copy()
    child2 = parent2.copy()
    
    for i in range(n):
        if np.random.random() > 0.5:
            child1[i] = parent2[i].copy()
            child2[i] = parent1[i].copy()
    
    return child1, child2

def mutate(circles, mutation_rate=0.1):
    """Mutate the configuration"""
    mutated = circles.copy()
    n = len(mutated)
    
    for i in range(n):
        if np.random.random() < mutation_rate:
            # Mutate position and radius
            mutated[i, 0] = max(0, min(1, mutated[i, 0] + np.random.normal(0, 0.01)))
            mutated[i, 1] = max(0, min(1, mutated[i, 1] + np.random.normal(0, 0.01)))
            mutated[i, 2] = max(0.001, mutated[i, 2] + np.random.normal(0, 0.005))
    
    return mutated

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select individual using tournament selection"""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmax([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def circle_packing26() -> np.ndarray:
    """
    Places 26 non-overlapping circles in the unit square in order to maximize the sum of radii.

    Returns:
        circles: np.array of shape (26,3), where the i-th row (x,y,r) stores the (x,y) coordinates of the i-th circle of radius r.
    """
    n = 26
    population = []
    
    # Create initial population with Voronoi initialization
    for _ in range(POPULATION_SIZE):
        init_config = create_voronoi_initialization(n)
        # Add some randomness to initial configurations
        for i in range(n):
            init_config[i, 0] += (np.random.random() - 0.5) * 0.05
            init_config[i, 1] += (np.random.random() - 0.5) * 0.05
        population.append(init_config)
    
    best_fitness = -float('inf')
    best_individual = None
    
    # Evolutionary loop
    for generation in range(GENERATIONS):
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            # Validate and repair if needed
            if is_valid_configuration(individual):
                fitness = calculate_sum_radii(individual)
                fitnesses.append(fitness)
            else:
                # If invalid, repair it
                repaired = constraint_aware_local_search(individual, LOCAL_SEARCH_ITERATIONS)
                if is_valid_configuration(repaired):
                    fitness = calculate_sum_radii(repaired)
                    fitnesses.append(fitness)
                    # Replace in population with repaired version
                    idx = population.index(individual)
                    population[idx] = repaired
                else:
                    # Penalize invalid configurations heavily
                    fitnesses.append(-1000)
        
        # Track best solution
        current_best_idx = np.argmax(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        
        if current_best_fitness > best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
        
        # Check if we've beaten the benchmark
        if best_fitness > 2.6358627564136983:
            break
        
        # Create new population
        new_population = []
        
        # Elitism: keep top 10%
        elite_count = POPULATION_SIZE // 10
        sorted_indices = np.argsort(fitnesses)[::-1][:elite_count]
        for i in sorted_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < POPULATION_SIZE:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, MUTATION_RATE)
            child2 = mutate(child2, MUTATION_RATE)
            
            # Local search refinement
            child1 = constraint_aware_local_search(child1, LOCAL_SEARCH_ITERATIONS)
            child2 = constraint_aware_local_search(child2, LOCAL_SEARCH_ITERATIONS)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:POPULATION_SIZE]
        
        # Adaptive mutation rate based on population diversity
        if generation > 50:
            diversity = np.std([calculate_sum_radii(ind) for ind in population])
            if diversity < 0.01:  # Low diversity, increase mutation
                MUTATION_RATE = 0.15
            else:
                MUTATION_RATE = 0.1
    
    # Final refinement
    if best_individual is not None:
        final_refined = constraint_aware_local_search(best_individual, LOCAL_SEARCH_ITERATIONS * 2)
        if is_valid_configuration(final_refined):
            return final_refined
    
    # Return the best configuration found
    return best_individual if best_individual is not None else create_voronoi_initialization(n)
