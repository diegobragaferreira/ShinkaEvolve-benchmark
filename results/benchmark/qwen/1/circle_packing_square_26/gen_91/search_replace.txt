2. Placing circles with maximum possible radius in Voronoi cells

def create_voronoi_initialization(n_circles, n_candidates=1000):
    """Create initial configuration using Voronoi-based approach"""
    # Generate random candidate points
    candidates = np.random.rand(n_candidates, 2)

    # Add boundary points to ensure coverage
    boundary_points = np.array([
        [0, 0], [0, 1], [1, 0], [1, 1],
        [0.5, 0], [0.5, 1], [0, 0.5], [1, 0.5]
    ])
    candidates = np.vstack([candidates, boundary_points])

    # Compute Voronoi diagram
    vor = Voronoi(candidates)

    # Get Voronoi vertices and regions
    regions = []
    for i in range(len(vor.points)):
        region = []
        for j, vertex_idx in enumerate(vor.point_region[i]):
            if vertex_idx != -1:
                region.append(vor.vertices[vertex_idx])
        regions.append(region)

    # Select candidate points that are likely to be good for circle placement
    selected_points = []
    for i, (point, region) in enumerate(zip(vor.points, regions)):
        if len(region) > 0:
            # Calculate the area of the Voronoi cell
            area = calculate_polygon_area(np.array(region))
            if area > 0.0001:  # Only consider reasonably sized cells
                selected_points.append(point)

    # If we don't have enough points, fill with additional random ones
    if len(selected_points) < n_circles:
        additional_points = np.random.rand(n_circles - len(selected_points), 2)
        selected_points.extend(additional_points)

    selected_points = selected_points[:n_circles]

    # Place circles with maximum possible radius at each selected point
    circles = []
    for i, (x, y) in enumerate(selected_points):
        # Maximum radius at this point without violating boundaries
        max_radius = min(x, 1-x, y, 1-y)
        if max_radius > 0:
            circles.append([x, y, max_radius])
        else:
            # Fallback to small random circle if boundary constrained
            circles.append([x, y, 0.01])

    return circles

def calculate_polygon_area(vertices):
    """Calculate area of polygon using shoelace formula"""
    if len(vertices) < 3:
        return 0
    x = vertices[:, 0]
    y = vertices[:, 1]
    return 0.5 * abs(sum(x[i] * y[i+1] - x[i+1] * y[i] for i in range(len(x)-1)) +
                     x[-1] * y[0] - x[0] * y[-1])
