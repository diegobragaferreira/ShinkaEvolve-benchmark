    circles = np.zeros((n, 3))

    # Create a more sophisticated initial distribution using a modified hexagonal grid
    # that better reflects optimal packing densities

    # Determine grid dimensions based on packing efficiency
    phi = (1 + math.sqrt(5)) / 2  # Golden ratio for better packing
    rows = int(math.ceil(math.sqrt(n / 0.866)))  # Hexagonal packing density ~0.866
    cols = int(math.ceil(n / rows))

    # Adjust for a slightly denser packing arrangement
    if rows * cols < n:
        rows += 1
        cols = int(math.ceil(n / rows))

    # Calculate spacing with consideration for edge effects
    spacing_x = 0.98 / cols
    spacing_y = 0.98 / rows

    # Use golden ratio spacing for better distribution
    golden_spacing_x = spacing_x * phi / 2
    golden_spacing_y = spacing_y * phi / 2

    # Generate points in a hexagonal pattern with slight perturbations
    count = 0
    for i in range(rows):
        for j in range(cols):
            if count >= n:
                break

            # Create hexagonal offset pattern
            offset = (i % 2) * (golden_spacing_x / 2)
            x = (j * golden_spacing_x + offset) + 0.01
            y = (i * golden_spacing_y) + 0.01

            # Add more randomness for better exploration
            x += random.uniform(-spacing_x/8, spacing_x/8)
            y += random.uniform(-spacing_y/8, spacing_y/8)

            # Ensure points are within bounds
            x = max(0.01, min(0.99, x))
            y = max(0.01, min(0.99, y))

            # Initial radius estimation based on local density
            # Larger radius in regions with fewer nearby points
            r = min(spacing_x, spacing_y) * random.uniform(0.3, 0.6)

            # Ensure the radius fits within bounds
            r = min(r, x, 1-x, y, 1-y)

            # Ensure minimum radius for stability
            r = max(0.005, r)

            circles[count] = [x, y, r]
            count += 1

    # If we didn't place enough circles, fill remaining positions
    while count < n:

        r = max(0.005, min(0.5, r))
