    def _generate_voronoi_initial_config(self):
        """Generate initial circle configuration based on improved Voronoi-like distribution."""
        circles = np.zeros((self.N_CIRCLES, 3))

        # Use a more sophisticated approach inspired by Voronoi diagrams
        # Generate points using a spiral pattern in the unit square
        golden_angle = np.pi * (3 - np.sqrt(5))

        for i in range(self.N_CIRCLES):
            # Spiral pattern with some randomness for better distribution
            r = np.sqrt(i / (self.N_CIRCLES - 1)) if self.N_CIRCLES > 1 else 0
            theta = i * golden_angle

            # Convert to Cartesian coordinates in [0.1, 0.9] range to keep margin
            x = 0.8 * r * np.cos(theta) + 0.5
            y = 0.8 * r * np.sin(theta) + 0.5

            # Apply small random perturbations for better exploration
            x += random.uniform(-0.02, 0.02)
            y += random.uniform(-0.02, 0.02)

            # Clamp to valid range
            x = max(0.01, min(0.99, x))
            y = max(0.01, min(0.99, y))
