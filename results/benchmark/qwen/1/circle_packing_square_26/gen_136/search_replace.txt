    # Constraint-aware crossover that maintains validity
    def constraint_aware_crossover(ind1, ind2):
        # Perform standard crossover
        child1, child2 = tools.cxTwoPoint(ind1, ind2)

        # Convert to circles arrays for validation
        circles1 = np.array(child1).reshape(-1, 3)
        circles2 = np.array(child2).reshape(-1, 3)

        # Repair children if necessary
        if not _validate_circle_placement(circles1):
            # Simple repair: keep parent 1's structure but adjust positions/radii to be valid
            circles1 = _repair_circles(circles1)
            child1 = circles1.flatten().tolist()

        if not _validate_circle_placement(circles2):
            # Simple repair: keep parent 2's structure but adjust positions/radii to be valid
            circles2 = _repair_circles(circles2)
            child2 = circles2.flatten().tolist()

        return child1, child2

    # Repair function to fix invalid circle configurations
    def _repair_circles(circles):
        n = len(circles)
        repaired = circles.copy()

        # Repair each circle individually to satisfy constraints
        for i in range(n):
            x, y, r = repaired[i]

            # Fix containment constraints
            r = min(r, x, 1-x, y, 1-y)
            r = max(r, 0.005)  # Minimum radius

            # Ensure at least minimum radius
            if r < 0.005:
                r = 0.005

            # Adjust position to maintain containment
            x = np.clip(x, r, 1-r)
            y = np.clip(y, r, 1-r)

            repaired[i] = [x, y, r]

        # Resolve overlaps through iterative adjustment
        for _ in range(100):  # Limited iterations to prevent infinite loop
            any_changed = False

            # Check all pairs of circles for overlap
            for i in range(n):
                for j in range(i+1, n):
                    x1, y1, r1 = repaired[i]
                    x2, y2, r2 = repaired[j]

                    distance_sq = (x1 - x2)**2 + (y1 - y2)**2
                    min_distance_sq = (r1 + r2)**2

                    if distance_sq < min_distance_sq:
                        # Circles overlap, resolve by moving them apart
                        dx = x2 - x1
                        dy = y2 - y1
                        distance = np.sqrt(distance_sq)

                        if distance > 1e-10:  # Avoid division by zero
                            # Move circles apart along the vector connecting their centers
                            move_distance = (min_distance_sq**0.5 - distance) / 2.0

                            # Normalize direction vector
                            nx = dx / distance
                            ny = dy / distance

                            # Apply movement (split equally between both circles)
                            move_x = nx * move_distance
                            move_y = ny * move_distance

                            # Adjust positions
                            repaired[i][0] -= move_x * 0.5
                            repaired[i][1] -= move_y * 0.5
                            repaired[j][0] += move_x * 0.5
                            repaired[j][1] += move_y * 0.5

                            # Keep within bounds
                            repaired[i][0] = np.clip(repaired[i][0], repaired[i][2], 1-repaired[i][2])
                            repaired[i][1] = np.clip(repaired[i][1], repaired[i][2], 1-repaired[i][2])
                            repaired[j][0] = np.clip(repaired[j][0], repaired[j][2], 1-repaired[j][2])
                            repaired[j][1] = np.clip(repaired[j][1], repaired[j][2], 1-repaired[j][2])

                            any_changed = True

            if not any_changed:
                break

        return repaired
