            # This creates a more even distribution than pure random placement

            spacing_x = 1.0 / (grid_size + 1)
            spacing_y = 1.0 / (grid_size + 1)

            # Generate seed points with some jitter for diversity
            seed_points = []
            for i in range(grid_size):
                for j in range(grid_size):
                    if len(seed_points) >= N_CIRCLES:
                        break
                    x = (i + 1) * spacing_x + np.random.uniform(-spacing_x/6, spacing_x/6)
                    y = (j + 1) * spacing_y + np.random.uniform(-spacing_y/6, spacing_y/6)
                    seed_points.append([x, y])

            seed_points = np.array(seed_points[:N_CIRCLES])
