def _initialize_voronoi_placement(n_circles: int, seed: int = 42) -> np.ndarray:
    """Initialize circle positions using a improved Voronoi-based approach with better distribution."""
    np.random.seed(seed)

    # Create a more refined and strategic grid for better initial distribution
    # Base grid size calculation to ensure good coverage
    grid_size = max(5, int(np.ceil(np.sqrt(n_circles)) + 2))

    # Create a structured grid with jittering to avoid regular patterns
    x_coords = np.linspace(0.05, 0.95, grid_size)
    y_coords = np.linspace(0.05, 0.95, grid_size)

    # Add jitter to grid points to break regularity
    jitter_magnitude = 0.02
    grid_points = []
    for i, x in enumerate(x_coords):
        for j, y in enumerate(y_coords):
            # Apply jitter based on position to get more uniform distribution
            jitter_x = np.random.uniform(-jitter_magnitude, jitter_magnitude)
            jitter_y = np.random.uniform(-jitter_magnitude, jitter_magnitude)
            grid_points.append([x + jitter_x, y + jitter_y])

    grid_points = np.array(grid_points)

    # If we have enough points, take the first n_circles
    if len(grid_points) >= n_circles:
        # Select points that are furthest apart for better distribution
        selected_indices = [0]  # Always include first point
        remaining_points = grid_points[1:]

        # Greedily select points that are farthest from already selected points
        for _ in range(n_circles - 1):
            if len(remaining_points) == 0:
                break
            max_dist = -1
            max_idx = 0

            # Find point that maximizes minimum distance to selected points
            for idx, point in enumerate(remaining_points):
                min_dist = float('inf')
                for selected_point in grid_points[selected_indices]:
                    dist = np.sqrt((point[0] - selected_point[0])**2 +
                                 (point[1] - selected_point[1])**2)
                    min_dist = min(min_dist, dist)

                if min_dist > max_dist:
                    max_dist = min_dist
                    max_idx = idx

            selected_indices.append(max_idx + 1)  # +1 because we skipped first point
            remaining_points = np.delete(remaining_points, max_idx, axis=0)

        initial_points = grid_points[selected_indices]
    else:
        # If we don't have enough points, first fill with grid points
        initial_points = grid_points
        # Then fill with random points to reach target count
        extra_points_needed = n_circles - len(initial_points)
        random_points = np.random.uniform(0.05, 0.95, (extra_points_needed, 2))
        initial_points = np.vstack([initial_points, random_points])

    try:
        vor = Voronoi(initial_points)

        centroids = vor.points

        valid_centroids = []
