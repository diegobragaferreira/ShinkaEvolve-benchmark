        # Generate diverse initial points with strategic sampling
        points = []

        for _ in range(30):

            if side == 0:  # Top
                points.append([np.random.rand(), 1.0])
            elif side == 1:  # Bottom
                points.append([np.random.rand(), 0.0])
            elif side == 2:  # Left
                points.append([0.0, np.random.rand()])
            else:  # Right
                points.append([1.0, np.random.rand()])

        # Add center points for better interior coverage
        for _ in range(10):
            points.append([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])

        # Add more systematic grid-like points
        grid_points = []
        for i in range(5):
            for j in range(5):

            circles = np.zeros((n_circles, 3))
