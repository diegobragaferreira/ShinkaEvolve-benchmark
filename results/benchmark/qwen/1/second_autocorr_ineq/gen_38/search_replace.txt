import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import convolve
import time

def compute_autoconvolution_norms(f_values):
    """Compute the norms needed for C2 calculation with proper numerical handling."""
    # Convert to numpy array
    f = np.array(f_values)

    # Compute autoconvolution using discrete convolution
    # This computes the full convolution, but we only need the valid part
    g_full = convolve(f, f, mode='full')
    g = g_full[len(f)-1:]  # Take the part corresponding to valid convolution

    # Compute L2 norm squared
    norm_2_sq = np.sum(g**2)

    # Compute L1 norm
    norm_1 = np.sum(np.abs(g))

    # Compute L-infinity norm
    norm_inf = np.max(np.abs(g))

    return norm_2_sq, norm_1, norm_inf, g

def compute_c2(f_values):
    """Compute C2 value for given function values."""
    try:
        norm_2_sq, norm_1, norm_inf, _ = compute_autoconvolution_norms(f_values)

        # Avoid division by zero or very small numbers
        if norm_1 <= 1e-12 or norm_inf <= 1e-12:
            return 0.0

        c2 = norm_2_sq / (norm_1 * norm_inf)
        return c2
    except Exception:
        return 0.0
