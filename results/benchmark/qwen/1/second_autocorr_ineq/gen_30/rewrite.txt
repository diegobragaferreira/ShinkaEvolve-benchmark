@jit(nopython=True)
def compute_autoconvolution_norms_numba(f_values):
    """Optimized computation of autoconvolution norms using numba"""
    n = len(f_values)
    
    # Initialize autoconvolution array
    g = np.zeros(2*n - 1)
    
    # Compute convolution manually for efficiency
    for i in range(n):
        for j in range(n):
            g[i + j] += f_values[i] * f_values[j]
    
    # Keep only center portion
    half_len = n - 1
    g_center = g[half_len:-half_len]
    
    # Compute norms
    norm_2_squared = np.sum(g_center**2)
    norm_1 = np.sum(np.abs(g_center))
    norm_inf = np.max(np.abs(g_center))
    
    return norm_2_squared, norm_1, norm_inf

def compute_autoconvolution_norms(f_values):
    """Compute the norms needed for C2 calculation with proper handling"""
    try:
        if not f_values:
            return 0.0, 0.0, 0.0, 0.0

        # Convert to numpy array for easier manipulation
        f = np.array(f_values, dtype=np.float64)

        # Ensure non-negative values
        f = np.maximum(f, 0.0)

        # Compute autoconvolution g = f * f
        g = np.convolve(f, f, mode='full')

        # Keep only the valid convolution part (middle)
        half_len = len(f) - 1
        g_valid = g[half_len:-half_len]

        # Compute norms
        norm_2_squared = np.sum(g_valid**2)
        norm_1 = np.sum(np.abs(g_valid))
        norm_inf = np.max(np.abs(g_valid))

        # Avoid division by zero
        if norm_1 == 0 or norm_inf == 0:
            return 0.0, 0.0, 0.0, 0.0

        # C2 = ||g||₂² / (||g||₁ · ||g||∞)
        c2 = norm_2_squared / (norm_1 * norm_inf)

        return c2, norm_2_squared, norm_1, norm_inf
    except Exception:
        return 0.0, 0.0, 0.0, 0.0

def evaluate_c2(individual):
    """Evaluate fitness of individual (step function) for maximizing C2"""
    try:
        # Ensure non-negative values
        individual = np.maximum(individual, 0.0)

        # Compute C2 value
        c2, _, _, _ = compute_autoconvolution_norms(individual)

        # Return negative because we want to maximize
        return -c2
    except Exception:
        # Return very poor fitness if error occurs
        return 1e10

def adaptive_sophisticated_initialization(n_steps):
    """
    Create a more sophisticated initial step function pattern
    that balances high energy concentration with structural diversity
    """
    # Create a wavelet-inspired pattern with alternating peaks and valleys
    f_values = []
    
    # Create base pattern with multiple frequency components
    base_freq = 2 * np.pi / n_steps
    
    # Generate pattern with multiple harmonics for better energy distribution
    for i in range(n_steps):
        # Base sine wave with varying amplitude
        t = i * base_freq
        amplitude = 0.5 + 0.5 * np.sin(2*t) + 0.2 * np.sin(5*t) + 0.1 * np.sin(10*t)
        
        # Add some random component for diversity
        noise = np.random.normal(0, 0.03)
        value = max(0, amplitude + noise)
        f_values.append(value)
    
    # Apply a smoothing window to reduce sharp transitions
    window_size = max(1, n_steps // 50)
    smoothed = []
    for i in range(len(f_values)):
        start = max(0, i - window_size)
        end = min(len(f_values), i + window_size)
        avg = np.mean(f_values[start:end])
        smoothed.append(avg)
    
    # Normalize to maintain reasonable magnitude
    total = sum(smoothed)
    if total > 0:
        smoothed = [x * n_steps / total for x in smoothed]
    
    return smoothed

def hybrid_local_refinement(initial_solution, max_iter=20):
    """
    Apply local optimization to refine the solution
    """
    # Use L-BFGS-B for local refinement
    try:
        bounds = [(0.0, 10.0) for _ in range(len(initial_solution))]
        
        # Apply bounds constraint and local minimization
        result = minimize(
            evaluate_c2,
            initial_solution,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter},
            tol=1e-6
        )
        
        if result.success:
            return result.x.tolist()
    except:
        pass
    return initial_solution

def multi_phase_optimization(n_steps):
    """
    Multi-phase optimization to balance exploration and exploitation
    """
    # Phase 1: Global exploration with diversified population
    phase1_populations = []
    for _ in range(5):
        init_sol = adaptive_sophisticated_initialization(n_steps)
        phase1_populations.append(init_sol)
    
    # Evaluate all initial solutions
    scores = []
    for sol in phase1_populations:
        try:
            score = evaluate_c2(sol)
            scores.append((score, sol))
        except:
            scores.append((1e10, sol))
    
    # Select best candidates
    best_candidates = sorted(scores, key=lambda x: x[0])[:3]
    best_solutions = [sol for _, sol in best_candidates]
    
    # Phase 2: Local refinement of best candidates
    refined_solutions = []
    for sol in best_solutions:
        refined = hybrid_local_refinement(sol, max_iter=15)
        refined_solutions.append(refined)
    
    # Final evaluation and selection
    final_scores = [(evaluate_c2(sol), sol) for sol in refined_solutions]
    best_final = min(final_scores, key=lambda x: x[0])
    
    return best_final[1]

def construct_function() -> list[float]:
    """Function to construct step-function with high C2 value using hybrid optimization"""
    # Set seeds for reproducibility
    random.seed(42)
    np.random.seed(42)
    
    start_time = time.time()
    
    try:
        # Use a larger number of steps for better resolution
        n_steps = 1000
        
        # Multi-phase optimization approach
        best_solution = multi_phase_optimization(n_steps)
        
        # Ensure non-negative values
        best_solution = [max(0, x) for x in best_solution]
        
        # Normalize to avoid extreme values that might cause numerical issues
        total = sum(best_solution)
        if total > 0:
            best_solution = [x / total * len(best_solution) for x in best_solution]
        
        # Final check to ensure we're within reasonable bounds
        best_solution = [min(10.0, max(0.0, x)) for x in best_solution]
        
        return best_solution

    except Exception as e:
        # Fallback to simple approach if evolution fails
        print(f"Fallback due to error: {e}")
        return adaptive_sophisticated_initialization(500)
