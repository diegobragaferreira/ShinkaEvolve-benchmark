@jit(nopython=True)
def compute_sparse_autoconvolution(f_vals):
    """
    Efficiently compute autoconvolution using sparse representation.
    Switches between direct O(n^2) and FFT-based O(n log n) computation
    depending on input size for optimal performance.
    """
    n = len(f_vals)

    # For small arrays, use direct computation for better numerical stability
    if n <= 100:
        # Initialize result array - will be length 2*n-1
        g = np.zeros(2*n - 1, dtype=np.float64)

        # Direct computation: g[i] = sum_{j+k=i} f[j] * f[k]
        for i in range(n):
            for j in range(n):
                g[i + j] += f_vals[i] * f_vals[j]

        return g
    else:
        # For larger arrays, use FFT-based approach for better performance
        # Zero-pad to avoid circular convolution effects
        padded_length = 2 * n - 1
        f_padded = np.zeros(padded_length)
        f_padded[:n] = f_vals

        # Use FFT-based convolution (equivalent to f*f)
        # This requires padding to avoid wraparound effects
        from scipy.fft import fft, ifft

        F = fft(f_padded)
        G_fft = F * F  # Point-wise multiplication in frequency domain
        g = ifft(G_fft).real[:padded_length]

        return g
