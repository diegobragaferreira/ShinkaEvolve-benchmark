2. Adding adaptive population sizing based on problem complexity

4. Adding a post-processing step that applies stochastic perturbations to escape local optima
5. Improving numerical stability handling in norm computations

def adaptive_evolutionary_optimization(f_vals: np.ndarray, max_evals: int = 1000) -> np.ndarray:
    """
    Perform enhanced adaptive evolutionary optimization with multi-stage approach
    """
    n_steps = len(f_vals)

    # Set bounds for each parameter
    bounds = [(0.0, 3.0) for _ in range(n_steps)]

    def objective(f_vals_array):
        return -compute_autoconvolution_norms(f_vals_array)[0]  # Negative because we minimize

    try:
        # Phase 1: Coarse global search with larger population
        popsize_phase1 = min(20, max(10, n_steps // 8))
        maxiter_phase1 = min(30, max_evals // popsize_phase1)

        result1 = differential_evolution(
            objective,
            bounds,
            seed=42,
            maxiter=maxiter_phase1,
            popsize=popsize_phase1,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.7,
            disp=False
        )

        optimized_values = result1.x if result1.success else f_vals

        # Phase 2: Fine-grained local search with smaller population
        popsize_phase2 = min(10, max(5, n_steps // 15))
        maxiter_phase2 = min(20, (max_evals - popsize_phase1 * maxiter_phase1) // popsize_phase2)

        if maxiter_phase2 > 0:
            result2 = differential_evolution(
                objective,
                bounds,
                seed=42,
                maxiter=maxiter_phase2,
                popsize=popsize_phase2,
                tol=1e-6,
                mutation=(0.7, 1.0),
                recombination=0.8,
                x0=optimized_values,
                disp=False
            )

            if result2.success:
                optimized_values = result2.x

        # Phase 3: Local refinement with L-BFGS-B
        refined_result = minimize(
            objective,
            optimized_values,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 25, 'ftol': 1e-8, 'gtol': 1e-8}
        )

        if refined_result.success:
            final_values = refined_result.x
        else:
            final_values = optimized_values

        # Phase 4: Stochastic perturbation to escape local optima
        # Add controlled noise to potentially escape suboptimal solutions
        noise_level = 0.02
        perturbed_values = final_values + np.random.normal(0, noise_level, len(final_values))
        perturbed_values = np.clip(perturbed_values, 0, None)

        # Refine the perturbed solution
        final_refinement = minimize(
            objective,
            perturbed_values,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 15, 'ftol': 1e-8, 'gtol': 1e-8}
        )

        if final_refinement.success:
            final_values = final_refinement.x

    except Exception:
        # Fallback to simple approach
        final_values = f_vals

    return np.clip(final_values, 0, None)
