    - Mix of high and low values to encourage diverse convolution properties
    - Smooth transitions to avoid numerical issues
    - Concentrated mass in center to promote good C2 behavior

    # Create base pattern with alternating regions
    f_vals = np.zeros(n)

    # Divide into segments
    segment_size = max(1, n // 12)

    # Create alternating high/low pattern with some smoothness
    for i in range(0, n, segment_size):

    """
    Create intelligent initial candidates using Fourier-based pattern generation:
    - Mix of fundamental and harmonic frequencies for rich spectral content
    - Smooth transitions to avoid numerical issues
    - Concentrated mass in center to promote good C2 behavior
    """
    # Create base pattern using Fourier synthesis
    x = np.linspace(-1, 1, n)
    f_vals = np.zeros(n)

    # Add multiple sinusoidal components to create complex structure
    # Fundamental frequency component
    f_vals += 0.5 * np.sin(2 * np.pi * x) + 0.3 * np.cos(4 * np.pi * x)

    # Harmonic components
    f_vals += 0.2 * np.sin(6 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)

    # Add a broader envelope for structure
    envelope = np.exp(-0.5 * (x / 0.4)**2)
    f_vals = f_vals * envelope + envelope * 0.5

    # Add some random variations to break symmetry
    f_vals += np.random.normal(0, 0.05, n)

    # Apply smoothing to ensure good transitions
    if n > 10:
        kernel_size = max(3, min(7, n // 15))
        if kernel_size > 1:
