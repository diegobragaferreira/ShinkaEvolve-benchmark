import numpy as np

class NormCalculator:
    """Computes autoconvolution norms with numerical stability optimizations."""
    
    @staticmethod
    def compute_autoconvolution_norms(f_values: np.ndarray) -> Tuple[float, float, float]:
        """
        Compute the three norms needed for C2 calculation:
        ||g||₂² (L2 norm squared), ||g||₁ (L1 norm), ||g||∞ (L-infinity norm)
        
        Uses improved trapezoidal integration for better accuracy
        """
        # Ensure non-negative values
        f = np.maximum(f_values, 0.0)
        
        # Compute autoconvolution g = f * f (discrete convolution)
        g = convolve(f, f, mode='full')
        
        # Extract the central portion that represents the main interval
        n = len(f)
        middle_idx = n - 1
        half_width = n
        
        # Take the central part of the convolution
        g_centered = g[middle_idx - half_width + 1 : middle_idx + half_width]
        
        # Compute the norms using improved trapezoidal integration
        g_squared = g_centered ** 2
        g_abs = np.abs(g_centered)
        
        # ||g||₂² - integrate using trapezoidal rule properly for piecewise linear function
        # Since g is piecewise linear (convolution of step functions),
        # we compute integral of g^2 using trapezoidal rule on g^2
        norm_g2_sq = 0.0
        if len(g_centered) >= 2:
            # Use trapezoidal rule: ∫g^2 ≈ h/2 * (g[0]^2 + 2*sum(g[i]^2) + g[n-1]^2)
            # where h is the step width (here assumed to be 1 for unit spacing)
            norm_g2_sq = g_squared[0] + g_squared[-1]
            norm_g2_sq += 2.0 * np.sum(g_squared[1:-1])
            norm_g2_sq *= 0.5  # h/2 where h=1
            
        # ||g||₁ - sum of absolute values
        norm_g1 = np.sum(g_abs)
        
        # ||g||∞ - maximum absolute value
        norm_ginf = np.max(g_abs)
        
        return norm_g2_sq, norm_g1, norm_ginf

    @staticmethod
    @jit
    def compute_autoconvolution_jax(f_values: jnp.ndarray) -> jnp.ndarray:
        """JAX version of autoconvolution computation"""
        # Using JAX's convolution operation
        g = jnp.convolve(f_values, f_values, mode='full')
        return g

    @staticmethod
    @jit
    def compute_norms_jax(g_vals: jnp.ndarray) -> Tuple[jnp.ndarray, jnp.ndarray, jnp.ndarray]:
        """JAX version of norm computation"""
        g_squared = g_vals ** 2
        g_abs = jnp.abs(g_vals)

        # ||g||₂² - sum of squares
        norm_g2_sq = jnp.sum(g_squared)

        # ||g||₁ - sum of absolute values
        norm_g1 = jnp.sum(g_abs)

        # ||g||∞ - maximum absolute value
        norm_ginf = jnp.max(g_abs)

        return norm_g2_sq, norm_g1, norm_ginf

class StepFunction:
    """Represents a step function with utility methods for optimization."""
    
    def __init__(self, values: List[float]):
        self.values = np.array(values, dtype=np.float64)
        self._validate()
        
    def _validate(self):
        """Validate that step function values are finite and non-negative."""
        if not np.all(np.isfinite(self.values)):
            raise ValueError("Step function contains infinite or NaN values")
        if np.any(self.values < 0):
            warnings.warn("Negative values found, clipping to zero")
            self.values = np.maximum(self.values, 0.0)
    
    def get_values(self) -> np.ndarray:
        """Get the current step function values."""
        return self.values.copy()
    
    def set_values(self, new_values: List[float]):
        """Set new step function values."""
        self.values = np.array(new_values, dtype=np.float64)
        self._validate()
    
    def normalize(self):
        """Normalize the step function to prevent extreme values."""
        max_val = np.max(self.values)
        if max_val > 0:
            self.values = self.values / max_val
    
    def smooth(self, kernel_size: int = None):
        """Apply Gaussian smoothing to the step function."""
        if len(self.values) < 3:
            return
            
        if kernel_size is None:
            kernel_size = min(21, len(self.values) // 10)
        if kernel_size % 2 == 0:
            kernel_size += 1
            
        sigma = kernel_size / 6.0
        x = np.arange(kernel_size) - kernel_size // 2
        gaussian_kernel = np.exp(-x**2 / (2 * sigma**2))
        gaussian_kernel /= np.sum(gaussian_kernel)
        
        # Apply convolution to smooth the function
        self.values = np.convolve(self.values, gaussian_kernel, mode='same')
        self.values = np.maximum(self.values, 0.0)

class EvolutionaryOptimizer:
    """Main optimization engine with adaptive strategies."""
    
    def __init__(self, max_time_seconds: float = 90.0, seed: int = 42):
        self.max_time_seconds = max_time_seconds
        self.seed = seed
        self.norm_calculator = NormCalculator()
        
    def _create_initial_population(self, n_steps: int, population_size: int = 10) -> List[np.ndarray]:
        """Generate diverse initial population based on mathematical insights."""
        population = []
        
        for _ in range(population_size):
            # Strategy selection with more sophisticated patterns
            strategy = np.random.choice([
                'uniform', 'alternating', 'gaussian', 'mixed', 'wavelet'
            ])
            
            if strategy == 'uniform':
                values = [0.5] * n_steps
            elif strategy == 'alternating':
                values = [np.random.uniform(0.7, 1.0) if i % 2 == 0 
                         else np.random.uniform(0.0, 0.3) for i in range(n_steps)]
            elif strategy == 'gaussian':
                x = np.linspace(-1, 1, n_steps)
                mu, sigma = 0, 0.3
                gauss = np.exp(-0.5 * ((x - mu) / sigma) ** 2) / (sigma * np.sqrt(2 * np.pi))
                scale_factor = 1.0 / np.max(gauss)
                values = (gauss * scale_factor).tolist()
            elif strategy == 'wavelet':
                # Create wavelet-inspired pattern
                x = np.linspace(-1, 1, n_steps)
                values = np.zeros(n_steps)
                # Multiple wavelet-like components
                for i in range(3):
                    mu = -0.5 + i * 0.3
                    sigma = 0.1 + i * 0.05
                    amplitude = 0.5 + 0.3 * np.random.random()
                    values += amplitude * np.exp(-((x - mu)**2) / (2 * sigma**2))
                values = values.tolist()
            else:  # mixed
                values = [np.random.uniform(0.7, 1.0) if i % 2 == 0 
                         else np.random.uniform(0.0, 0.3) for i in range(n_steps)]
                # Apply Gaussian smoothing
                if n_steps >= 5:
                    kernel_size = min(11, n_steps // 10)
                    if kernel_size % 2 == 0:
                        kernel_size += 1
                    sigma = kernel_size / 6.0
                    x = np.arange(kernel_size) - kernel_size // 2
                    gaussian_kernel = np.exp(-x**2 / (2 * sigma**2))
                    gaussian_kernel /= np.sum(gaussian_kernel)
                    values = np.convolve(values, gaussian_kernel, mode='same').tolist()
            
            # Add noise for exploration
            noise_level = 0.05
            values = [max(0, val + np.random.normal(0, noise_level)) for val in values]
            
            population.append(np.array(values))
            
        return population
    
    def _evaluate_fitness(self, individual: np.ndarray) -> float:
        """Evaluate fitness (negative C2) of an individual."""
        try:
            # Ensure non-negative values
            individual = np.maximum(individual, 0.0)
            
            # Compute norms
            norm_g2_sq, norm_g1, norm_ginf = self.norm_calculator.compute_autoconvolution_norms(individual)
            
            # Avoid division by zero
            if norm_g1 < 1e-15 or norm_ginf < 1e-15:
                return -0.0  # Return worst possible fitness
            
            # C2 = ||g||₂² / (||g||₁ · ||g||∞)
            c2 = norm_g2_sq / (norm_g1 * norm_ginf)
            
            # Return negative because we want to maximize C2
            return -c2
            
        except Exception:
            return -1e10  # Very poor fitness for invalid cases
    
    def _evaluate_fitness_jax(self, individual: np.ndarray) -> float:
        """JAX version of fitness evaluation for improved gradient estimation"""
        if not JAX_AVAILABLE:
            return self._evaluate_fitness(individual)

        try:
            # Convert to JAX arrays
            f_vals = jnp.array(np.maximum(individual, 0.0), dtype=jnp.float32)

            # Compute autoconvolution using JAX
            g_vals = self.norm_calculator.compute_autoconvolution_jax(f_vals)

            # Extract centered values
            n = len(individual)
            middle_idx = n - 1
            half_width = n
            g_centered = g_vals[middle_idx - half_width + 1 : middle_idx + half_width]

            # Compute norms using JAX
            norm_g2_sq, norm_g1, norm_ginf = self.norm_calculator.compute_norms_jax(g_centered)

            # Avoid division by zero
            if norm_g1 < 1e-15 or norm_ginf < 1e-15:
                return -0.0

            # C2 = ||g||₂² / (||g||₁ · ||g||∞)
            c2 = norm_g2_sq / (norm_g1 * norm_ginf)

            return float(-c2)

        except Exception:
            return -1e10

    def _adaptive_evolutionary_search(self, initial_individual: np.ndarray) -> np.ndarray:
        """Perform adaptive evolutionary search with multiple phases."""
        start_time = time.time()
        n_dimensions = len(initial_individual)
        
        # Phase 1: Differential Evolution (Global Search) - Enhanced settings
        bounds = [(0, 3) for _ in range(n_dimensions)]
        max_iter_phase1 = min(50, int(self.max_time_seconds * 0.6))
        popsize_phase1 = min(20, max(5, n_dimensions // 8))
        
        try:
            def objective(x):
                return self._evaluate_fitness(x)
                
            de_result = differential_evolution(
                objective,
                bounds,
                maxiter=max_iter_phase1,
                popsize=popsize_phase1,
                tol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=self.seed,
                disp=False,
                strategy='best1bin'  # Better strategy for this problem
            )
            
            best_solution = de_result.x if de_result.success else initial_individual
            
        except Exception:
            best_solution = initial_individual
            
        # Phase 2: Local Refinement (Fine-tuning) - Enhanced with hybrid approach
        try:
            # Use a hybrid approach: try both JAX and standard gradient methods
            if JAX_AVAILABLE:
                # Create gradient function using JAX
                @jit
                def grad_objective(x):
                    return grad(lambda x: self._evaluate_fitness_jax(np.array(x)))(x)
                
                def objective(x):
                    return self._evaluate_fitness_jax(x)
            else:
                def objective(x):
                    return self._evaluate_fitness(x)
                
            # Try L-BFGS-B first, then fall back to Nelder-Mead
            refined_result = minimize(
                objective,
                best_solution,
                method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': max(15, int(self.max_time_seconds * 0.25)),
                    'ftol': 1e-8,
                    'gtol': 1e-8
                }
            )
            
            # If L-BFGS-B fails, try a more robust method
            if not refined_result.success:
                refined_result = minimize(
                    objective,
                    best_solution,
                    method='Nelder-Mead',
                    options={
                        'maxiter': max(10, int(self.max_time_seconds * 0.1)),
                        'ftol': 1e-6,
                        'xtol': 1e-6
                    }
                )
                
            final_solution = refined_result.x if refined_result.success else best_solution
                
        except Exception:
            final_solution = best_solution
            
        # Post-processing - Enhanced with normalization and smoothing
        final_solution = np.maximum(final_solution, 0.0)
        
        # Normalize to maintain reasonable scale
        total = np.sum(final_solution)
        if total > 0:
            final_solution = final_solution / total * n_dimensions * 0.5
            
        # Apply light smoothing to make solution more stable
        if len(final_solution) > 10:
            try:
                smoother = StepFunction(final_solution.tolist())
                smoother.smooth(min(11, len(final_solution) // 10))
                final_solution = smoother.get_values()
            except:
                pass
                
        return final_solution
    
    def optimize(self, n_steps: int = None) -> np.ndarray:
        """
        Main optimization routine.
        
        Args:
            n_steps: Number of steps in the function (random if None)
            
        Returns:
            Optimized step function values
        """
        if n_steps is None:
            n_steps = np.random.randint(800, 4000)  # Broader range for better exploration
            
        # Set seeds for reproducibility
        np.random.seed(self.seed)
        
        # Generate initial population
        initial_population = self._create_initial_population(n_steps, 6)  # Reduced population size for speed
        
        # Select the best initial function
        best_initial = min(initial_population, key=self._evaluate_fitness)
        
        # Perform adaptive evolutionary search
        optimized_values = self._adaptive_evolutionary_search(best_initial)
        
        return optimized_values

def construct_function() -> List[float]:
    """
    Main entry point for constructing step-function with high C2 value.
    
    Returns:
        List of step function heights that maximize C2
    """
    # Create optimizer with time constraints
    optimizer = EvolutionaryOptimizer(max_time_seconds=90.0, seed=42)
    
    try:
        # Perform optimization
        optimized_function = optimizer.optimize()
    except Exception as e:
        # Fallback to simple initialization if optimization fails
        print(f"Optimization failed with error: {e}. Using fallback.")
        # Generate simple pattern
        n_steps = np.random.randint(800, 4000)  # Broader range
        f_values = [0.5] * n_steps
        optimized_function = np.array(f_values)
    
    # Ensure valid output format
    return optimized_function.tolist()
