def compute_c2_jax(f_vals):
    """JAX-based C2 computation for automatic differentiation"""
    # Convert to JAX array
    f = jnp.array(f_vals)

    # Compute autoconvolution
    n = len(f_vals)
    g_len = 2 * n - 1
    g = jnp.zeros(g_len)

    # Manual convolution (can be optimized further)
    for i in range(n):
        for j in range(n):
            idx = i + j
            if 0 <= idx < g_len:
                g = g.at[idx].add(f[i] * f[j])

    # Compute norms using JAX operations
    g_l1 = jnp.sum(jnp.abs(g))
    g_l2_sq = jnp.sum(g ** 2)
    g_max = jnp.max(jnp.abs(g))

    # Compute C2 with numerical stability
    epsilon = 1e-16
    c2 = jnp.where(
        (g_l1 > epsilon) & (g_max > epsilon),
        g_l2_sq / (g_l1 * g_max),
        0.0
    )

    return c2

def evaluate_c2_jax(params):
    """Wrapper for evaluating C2 using JAX for gradient computation"""
    return compute_c2_jax(params)
