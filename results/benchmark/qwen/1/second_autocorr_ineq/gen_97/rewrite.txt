import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import convolve
import numba
from typing import List

@numba.jit(nopython=True)
def compute_autoconvolution_norms(f_vals: np.ndarray, n_points: int = 15000) -> tuple:
    """
    Compute the norms for autoconvolution g = f*f using piecewise linear integration
    """
    # Create step function on [-1/4, 1/4] with given values
    step_width = 0.5 / len(f_vals)
    x = np.linspace(-0.25, 0.25, len(f_vals))

    # Create piecewise constant function
    f = np.zeros(n_points)
    x_grid = np.linspace(-0.25, 0.25, n_points)

    # Interpolate step function onto grid more accurately
    for i in range(len(f_vals)):
        start_x = -0.25 + i * step_width
        end_x = start_x + step_width
        start_idx = max(0, int((start_x + 0.25) / 0.5 * n_points))
        end_idx = min(n_points, int((end_x + 0.25) / 0.5 * n_points))
        if i == len(f_vals) - 1:
            end_idx = n_points
        f[start_idx:end_idx] = f_vals[i]

    # Compute autoconvolution g = f * f (discrete convolution)
    g = convolve(f, f[::-1], mode='full')
    g = g[len(g)//2:]  # Take positive half

    # Truncate to match x_grid size
    g = g[:n_points]

    # Compute norms using trapezoidal-like piecewise integration
    # For ||g||_2^2: integrate g^2 using improved trapezoidal rule
    g_squared = g * g
    # Using corrected trapezoidal rule: (y1 + y2) * h / 2
    # But since we're integrating g^2, we use a modified version
    # For accurate quadratic integration, we use Simpson's-like approach
    # Here we do weighted average of consecutive points for better precision
    h = x_grid[1] - x_grid[0]
    norm_g2_sq = 0.0
    for i in range(len(g_squared)-1):
        # Trapezoidal approximation for integral of g^2
        norm_g2_sq += (g_squared[i] + g_squared[i+1]) * h / 2

    # For ||g||_1
    norm_g1 = 0.0
    for i in range(len(g)):
        norm_g1 += np.abs(g[i]) * h

    # For ||g||_inf
    norm_ginf = 0.0
    for i in range(len(g)):
        if np.abs(g[i]) > norm_ginf:
            norm_ginf = np.abs(g[i])

    return norm_g2_sq, norm_g1, norm_ginf

@numba.jit(nopython=True)
def evaluate_c2(f_vals: np.ndarray) -> float:
    """Evaluate C2 = ||g||_2^2 / (||g||_1 * ||g||_inf)"""
    norm_g2_sq, norm_g1, norm_ginf = compute_autoconvolution_norms(f_vals)

    # Handle numerical issues
    if norm_g1 < 1e-15 or norm_ginf < 1e-15:
        return 0.0

    return norm_g2_sq / (norm_g1 * norm_ginf)

def generate_multiscale_initialization(n_steps: int) -> List[float]:
    """Generate multiple initial patterns at different scales for better exploration"""
    patterns = []
    
    # Pattern 1: Gaussian-like with sharp center
    x = np.linspace(0, 1, n_steps)
    pattern1 = 1.5 * np.exp(-((x - 0.5) * 5)**2) + 0.3
    patterns.append(pattern1.tolist())
    
    # Pattern 2: Alternating high-low regions
    pattern2 = []
    for i in range(n_steps):
        if i % 2 == 0:
            pattern2.append(np.random.uniform(0.7, 1.0))
        else:
            pattern2.append(np.random.uniform(0.1, 0.4))
    patterns.append(pattern2)
    
    # Pattern 3: Multi-peak pattern
    pattern3 = []
    for i in range(n_steps):
        x_pos = i / (n_steps - 1) if n_steps > 1 else 0.5
        val = 0.5 * np.exp(-((x_pos - 0.3) * 8)**2) + \
              1.0 * np.exp(-((x_pos - 0.5) * 6)**2) + \
              0.3 * np.exp(-((x_pos - 0.7) * 8)**2)
        pattern3.append(val)
    patterns.append(pattern3)
    
    # Normalize all patterns
    normalized_patterns = []
    for pattern in patterns:
        total = sum(pattern)
        if total > 0:
            normalized_patterns.append([x/total * 2.0 for x in pattern])
        else:
            normalized_patterns.append([1.0/n_steps] * n_steps)
    
    return normalized_patterns

def adaptive_evolutionary_optimization(initial_guess: List[float], max_iterations: int = 120) -> List[float]:
    """Perform adaptive evolutionary optimization with multiple refinement stages"""
    n_steps = len(initial_guess)
    
    # Stage 1: Coarse global search with larger population
    def objective(f_vals):
        return -evaluate_c2(np.array(f_vals))  # Negative because we minimize
    
    bounds = [(0.0, 3.0) for _ in range(n_steps)]
    
    # Use a larger population for better exploration
    popsize = max(20, n_steps // 50)  # Adaptive population size
    popsize = min(popsize, 25)  # Cap at reasonable size
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            seed=42,
            maxiter=max_iterations,
            popsize=popsize,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.7
        )
        
        if not result.success:
            raise Exception("Differential evolution failed")
            
        optimized_values = result.x
        
        # Stage 2: Local refinement with adaptive iterations
        refine_iter = max(30, n_steps // 50)  # Adaptive refinement iterations
        refine_iter = min(refine_iter, 80)   # Cap refinement iterations
        
        refined_result = minimize(
            objective,
            optimized_values,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': refine_iter, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        final_values = refined_result.x if refined_result.success else optimized_values
        
        # Stage 3: Stochastic perturbation to escape local optima
        # Add small random noise to prevent premature convergence
        noise_level = 0.01
        perturbed_values = final_values + np.random.normal(0, noise_level, len(final_values))
        perturbed_values = np.clip(perturbed_values, 0, None)
        
        # Final refinement with perturbed values
        final_refine_result = minimize(
            objective,
            perturbed_values,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if final_refine_result.success:
            final_values = final_refine_result.x
            
    except Exception as e:
        # Fallback to initial guess if optimization fails
        print(f"Optimization failed with error: {e}")
        final_values = initial_guess
    
    # Ensure non-negative values and normalize appropriately
    final_values = np.clip(final_values, 0, None)
    total = np.sum(final_values)
    if total > 0:
        final_values = final_values / total * 2.0
    
    return final_values.tolist()

def construct_function() -> List[float]:
    """Function to construct step-function with high C2 value using enhanced optimization."""
    # Use a larger number of steps for better resolution
    n_steps = 200  # Increased from 150 for better resolution
    
    # Generate multiple initial patterns to improve chance of finding global optimum
    initial_patterns = generate_multiscale_initialization(n_steps)
    
    # Evaluate all initial patterns and select the best one
    best_c2 = -np.inf
    best_pattern = None
    
    for pattern in initial_patterns:
        try:
            c2 = evaluate_c2(np.array(pattern))
            if c2 > best_c2:
                best_c2 = c2
                best_pattern = pattern
        except:
            continue
    
    # If none worked, use first pattern as fallback
    if best_pattern is None:
        best_pattern = initial_patterns[0]
    
    # Apply adaptive evolutionary optimization
    optimized_values = adaptive_evolutionary_optimization(best_pattern, max_iterations=120)
    
    # Convert to list of floats
    return [float(x) for x in optimized_values]
