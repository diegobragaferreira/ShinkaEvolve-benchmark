@jax_jit
def compute_autoconvolution_jax(f_vals_jax):
    """JAX version of autoconvolution for gradient computation"""
    # Convert to JAX array
    f_vals = f_vals_jax

    # For autoconvolution using JAX operations
    n = len(f_vals)
    g = jnp.zeros(2*n - 1)

    # Vectorized autoconvolution
    for i in range(n):
        g = g.at[i:i+n].add(f_vals[i] * f_vals)

    # Trim to center portion
    offset = (n - 1) // 2
    g_trimmed = g[offset:(2*n-1)-offset]

    return g_trimmed

def compute_c2_jax(g_vals):
    """JAX version of C2 computation"""
    if len(g_vals) == 0:
        return 0.0

def compute_autoconvolution_jax(f_vals_jax):
    """Vectorized JAX version of autoconvolution for gradient computation"""
    # Convert to JAX array
    f_vals = f_vals_jax

    # Vectorized autoconvolution using outer product - much faster than loop
    # This computes the full convolution matrix and extracts the relevant part
    f_ext = jnp.pad(f_vals, (0, len(f_vals) - 1), mode='constant')
    g_full = jnp.convolve(f_ext, f_vals, mode='full')

    # Keep only the central portion (autoconvolution)
    n = len(f_vals)
    offset = n - 1
    g_trimmed = g_full[offset:offset + n]

    return g_trimmed
