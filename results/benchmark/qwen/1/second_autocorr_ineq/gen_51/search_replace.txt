def compute_autoconvolution_numba(f_vals, n_points=1000):

    Compute autoconvolution g = f*f using trapezoidal integration approach.
    Returns g values at discrete points.
    """
    # Create convolution grid
    conv_len = 2 * len(f_vals) - 1
    g = np.zeros(conv_len)

    # Compute autoconvolution using discrete convolution
    for i in range(len(f_vals)):
        for j in range(len(f_vals)):
            idx = i + j
            if idx < conv_len:
                g[idx] += f_vals[i] * f_vals[j]

    # Downsample to target number of points for accuracy
    if conv_len > n_points:
        g = g[::conv_len // n_points]

    return g

@jit(nopython=True)
def compute_norms_numba(g_vals):
    """
    Computes the three required norms efficiently using proper integration
    """
    if len(g_vals) == 0:
        return 0.0, 0.0, 0.0

    # L1 norm (sum of absolute values)
    l1 = 0.0
    for i in range(len(g_vals)):
        l1 += abs(g_vals[i])

    # L-infinity norm (maximum absolute value)
    l_inf = 0.0
    for i in range(len(g_vals)):
        abs_val = abs(g_vals[i])
        if abs_val > l_inf:
            l_inf = abs_val

    # L2^2 norm using trapezoidal integration
    # For piecewise linear integration of g^2, we use:
    # integral g^2 dt ≈ h * (g[0]^2 + 2*sum(g[i]^2) + g[n-1]^2)/2
    # where h is the step size
    l2_sq = 0.0
    if len(g_vals) >= 2:
        # Trapezoidal rule for L2 norm squared
        # Using the form: h * (y0^2 + 2*y1^2 + 2*y2^2 + ... + yn-1^2)/2
        l2_sq = g_vals[0]*g_vals[0] + g_vals[-1]*g_vals[-1]
        for i in range(1, len(g_vals)-1):
            l2_sq += 2 * g_vals[i] * g_vals[i]

        # Assuming unit step width for simplicity in this discrete computation
        # In practice, we need to account for actual domain width
        # For a domain of [-1/2, 1/2], the width is 1, and we have len(g_vals) points
        # So step width h = 1/(len(g_vals)-1) if len > 1, else 1
        h = 1.0 / (len(g_vals) - 1) if len(g_vals) > 1 else 1.0
        l2_sq *= h / 2.0
    elif len(g_vals) == 1:
        l2_sq = g_vals[0] * g_vals[0]
    else:
        l2_sq = 0.0

    return l2_sq, l1, l_inf
