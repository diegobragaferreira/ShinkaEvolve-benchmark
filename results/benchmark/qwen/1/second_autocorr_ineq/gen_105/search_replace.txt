def compute_autoconvolution(f_vals):
    """
    Efficiently compute autoconvolution using scipy's convolution or FFT methods
    """
    f_vals = np.array(f_vals, dtype=np.float64)

    # For better performance with large arrays, use FFT-based convolution
    n = len(f_vals)
    # Pad to next power of 2 for FFT efficiency
    fft_size = 2**int(np.ceil(np.log2(2*n - 1)))

    # Use FFT-based convolution
    f_padded = np.pad(f_vals, (0, fft_size - n), 'constant')
    f_fft = np.fft.fft(f_padded)
    g_fft = f_fft * f_fft
    g = np.real(np.fft.ifft(g_fft))

    # Keep only the valid convolution part
    g_valid = g[n-1:2*n-1]

    return g_valid

def compute_norms(g_vals):
    """
    Compute L1, L2^2, and L-infinity norms efficiently
    """
    # L1 norm (sum of absolute values)
    l1_norm = np.sum(np.abs(g_vals))

    # L2^2 norm (sum of squares)
    l2_sq_norm = np.sum(g_vals**2)

    # L-infinity norm (maximum absolute value)
    linf_norm = np.max(np.abs(g_vals))

    return l1_norm, l2_sq_norm, linf_norm

def compute_c2(f_vals):
    """
    Compute C2 value using optimized functions
    """
    try:
        # Compute autoconvolution
        g_vals = compute_autoconvolution(f_vals)

        # Compute norms
        l1, l2_sq, linf = compute_norms(g_vals)

        # Avoid division by zero
        if l1 <= 1e-15 or linf <= 1e-15:
            return 0.0

        # Return C2 value
        return l2_sq / (l1 * linf)

        return 0.0
