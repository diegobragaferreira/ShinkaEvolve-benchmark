# Custom sparse convex optimization solver
class SparseConvexOptimizer:
    def __init__(self, max_iterations=1000, tolerance=1e-8):
        self.max_iterations = max_iterations
        self.tolerance = tolerance
    
    def solve_sparse_convex(self, initial_guess, target_structure=None):
        """
        Solve the sparse convex optimization problem for step function construction
        """
        n_steps = len(initial_guess)
        
        # Initialize with structured approach
        x0 = np.array(initial_guess)
        
        # Define the optimization problem
        def objective(x):
            # Compute the autoconvolution norms
            try:
                norm_g_2_squared, norm_g_1, norm_g_inf = self._compute_autoconvolution_norms(x)
                
                # Avoid division by zero
                if norm_g_1 <= 1e-15 or norm_g_inf <= 1e-15:
                    return float('inf')
                
                c2 = norm_g_2_squared / (norm_g_1 * norm_g_inf)
                return -c2  # Minimize negative C2 to maximize C2
            except Exception:
                return float('inf')
        
        def gradient(x):
            # Simple finite difference gradient approximation
            eps = 1e-6
            grad_vals = np.zeros_like(x)
            for i in range(len(x)):
                x_plus = x.copy()
                x_minus = x.copy()
                x_plus[i] += eps
                x_minus[i] -= eps
                grad_vals[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
            return grad_vals
        
        # Use L-BFGS-B optimizer with bounds
        bounds = [(0, 1.0) for _ in range(n_steps)]
        
        try:
            result = minimize(
                objective,
                x0,
                method='L-BFGS-B',
                bounds=bounds,
                jac=gradient,
                options={'maxiter': self.max_iterations, 'gtol': self.tolerance}
            )
            
            if result.success:
                optimized_x = np.maximum(result.x, 0)
                if np.sum(optimized_x) > 0:
                    optimized_x = optimized_x / np.sum(optimized_x)
                return optimized_x.tolist()
        except Exception:
            pass
            
        return x0.tolist()
    
    def _compute_autoconvolution_norms(self, f_values: np.ndarray) -> tuple:
        """
        Compute the norms ||g||₂², ||g||₁, and ||g||∞ for the autoconvolution g = f*f
        Using proper piecewise linear integration for ||g||₂²
        """
        f = np.array(f_values)
        
        # Compute autoconvolution g = f * f using discrete convolution
        g = np.convolve(f, f, mode='full')
        
        # Compute the norms
        # ||g||₂² using proper piecewise integration
        # For consecutive pairs (y1, y2) with unit spacing:
        # integral of y^2 ≈ (1/3)(y1^2 + y1*y2 + y2^2)
        norm_g_2_squared = 0.0
        for i in range(len(g) - 1):
            y1 = g[i]
            y2 = g[i + 1]
            norm_g_2_squared += (y1 * y1 + y1 * y2 + y2 * y2) / 3.0
        
        # ||g||₁ = sum(|g[i]|)
        norm_g_1 = np.sum(np.abs(g))
        
        # ||g||∞ = max(|g[i]|)
        norm_g_inf = np.max(np.abs(g))
        
        return norm_g_2_squared, norm_g_1, norm_g_inf

def compute_autoconvolution_norms(f_values: List[float]) -> tuple:
    """
    Compute the norms ||g||₂², ||g||₁, and ||g||∞ for the autoconvolution g = f*f
    Using proper piecewise linear integration for ||g||₂² as specified in requirements
    """
    f = np.array(f_values)
    
    # Compute autoconvolution g = f * f using discrete convolution
    g = np.convolve(f, f, mode='full')
    
    # Compute the norms
    # ||g||₂² = sum(g[i]²) using proper piecewise integration
    norm_g_2_squared = 0.0

    # For piecewise linear integration, we use trapezoidal-like approach:
    # for consecutive pairs of points (y1, y2) with unit spacing:
    # integral of y^2 ≈ (1/3)(y1^2 + y1*y2 + y2^2)
    for i in range(len(g) - 1):
        y1 = g[i]
        y2 = g[i + 1]
        norm_g_2_squared += (y1 * y1 + y1 * y2 + y2 * y2) / 3.0

    # ||g||₁ = sum(|g[i]|)
    norm_g_1 = np.sum(np.abs(g))

    # ||g||∞ = max(|g[i]|)
    norm_g_inf = np.max(np.abs(g))

    return norm_g_2_squared, norm_g_1, norm_g_inf

def evaluate_c2(f_values: List[float]) -> float:
    """
    Evaluate C₂ = ||g||₂² / (||g||₁ · ||g||∞) for given step function
    """
    try:
        norm_g_2_squared, norm_g_1, norm_g_inf = compute_autoconvolution_norms(f_values)

        # Avoid division by zero with stricter thresholds
        if norm_g_1 <= 1e-15 or norm_g_inf <= 1e-15:
            return 0.0

        c2 = norm_g_2_squared / (norm_g_1 * norm_g_inf)
        return c2
    except Exception:
        return 0.0

def generate_structured_initialization(n_steps: int) -> List[float]:
    """
    Generate structured initial configuration using mathematical insights
    Instead of random initialization, use a mathematically motivated approach
    """
    # Create a base function with strategic structure
    f = np.zeros(n_steps)
    
    # Apply mathematical intuition: create structure that balances flatness 
    # and energy concentration for optimal autoconvolution
    x = np.linspace(-1, 1, n_steps)
    
    # Use a combination of sinusoidal and step patterns for rich convolution properties
    base_pattern = 0.5 + 0.4 * np.sin(3 * np.pi * x) + 0.3 * np.cos(5 * np.pi * x)
    
    # Add step-like features at multiple scales
    step_positions = np.linspace(0.1, 0.9, 8)
    for pos in step_positions:
        step_width = max(1, n_steps // 20)
        start_idx = max(0, int(pos * n_steps - step_width // 2))
        end_idx = min(n_steps, int(pos * n_steps + step_width // 2))
        f[start_idx:end_idx] = np.maximum(f[start_idx:end_idx], 0.8)
    
    # Ensure the pattern is positive and add some random variation
    f = np.clip(base_pattern + f * 0.5, 0, None)
    
    # Add structured peaks to encourage complex convolution behavior
    n_peaks = 4
    for i in range(n_peaks):
        peak_pos = 0.1 + i * 0.2  # Evenly spaced peaks
        peak_width = max(1, n_steps // 15)
        start = max(0, int(peak_pos * n_steps - peak_width // 2))
        end = min(n_steps, int(peak_pos * n_steps + peak_width // 2))
        f[start:end] = np.maximum(f[start:end], 0.7 + 0.3 * np.random.random(end - start))
    
    # Normalize to ensure valid probability distribution
    if np.sum(f) > 0:
        f = f / np.sum(f)
        
    return f.tolist()

def generate_sparse_initialization(n_steps: int) -> List[float]:
    """
    Generate sparse initialization focused on key structural elements
    This exploits the fact that optimal solutions often have sparse structure
    """
    # Start with sparsely populated function
    f = np.zeros(n_steps)
    
    # Place high-value regions at strategic locations
    # These locations are chosen to maximize convolution interaction
    n_regions = 6
    for i in range(n_regions):
        # Position chosen to balance convolution effects
        pos = 0.1 + i * 0.15 + np.random.random() * 0.05
        width = max(1, n_steps // 12 + np.random.randint(-2, 3))
        start = max(0, int(pos * n_steps - width // 2))
        end = min(n_steps, int(pos * n_steps + width // 2))
        f[start:end] = np.maximum(f[start:end], 0.7 + 0.2 * np.random.random(end - start))
    
    # Add some smooth modulation
    x = np.linspace(-1, 1, n_steps)
    modulation = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    f = f * modulation
    
    # Ensure non-negativity
    f = np.clip(f, 0, None)
    
    # Normalize
    if np.sum(f) > 0:
        f = f / np.sum(f)
        
    return f.tolist()

def advanced_convex_optimization_approach() -> List[float]:
    """
    Advanced convex optimization approach focusing on mathematical properties
    """
    n_steps = 500
    
    # Generate multiple structured initializations
    initial_solutions = []
    
    # Strategy 1: Structured mathematical initialization
    init1 = generate_structured_initialization(n_steps)
    initial_solutions.append(init1)
    
    # Strategy 2: Sparse optimization initialization  
    init2 = generate_sparse_initialization(n_steps)
    initial_solutions.append(init2)
    
    # Strategy 3: Random but well-distributed initialization
    init3 = np.random.random(n_steps)
    init3 = init3 / np.sum(init3) if np.sum(init3) > 0 else init3
    initial_solutions.append(init3.tolist())
    
    # Evaluate all initial solutions
    best_c2 = -np.inf
    best_solution = None
    
    for i, sol in enumerate(initial_solutions):
        c2 = evaluate_c2(sol)
        if c2 > best_c2:
            best_c2 = c2
            best_solution = sol.copy()
    
    # Apply convex optimization with structured approach
    if best_solution is not None:
        try:
            optimizer = SparseConvexOptimizer(max_iterations=500)
            optimized_solution = optimizer.solve_sparse_convex(best_solution)
            optimized_c2 = evaluate_c2(optimized_solution)
            
            if optimized_c2 > best_c2:
                best_c2 = optimized_c2
                best_solution = optimized_solution
        except Exception:
            pass
    
    return best_solution if best_solution is not None else [1.0/n_steps] * n_steps

    """

    try:
        # Use advanced convex optimization approach
        final_solution = advanced_convex_optimization_approach()

        final_solution = np.array(final_solution)
        final_solution = np.clip(final_solution, 0, None)
        if np.sum(final_solution) > 0:

        return [1.0/500] * 500
