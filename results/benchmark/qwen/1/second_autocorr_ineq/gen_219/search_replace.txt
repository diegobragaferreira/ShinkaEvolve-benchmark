@numba.jit(nopython=True)
def compute_autoconvolution_norms(f_vals: np.ndarray, n_points: int = 10000) -> tuple:
    """
    Compute the norms for autoconvolution g = f*f using piecewise linear integration
    """
    # Create step function on [-1/4, 1/4] with given values
    step_width = 0.5 / len(f_vals)
    x = np.linspace(-0.25, 0.25, len(f_vals))

    # Create piecewise constant function
    f = np.zeros(n_points)
    x_grid = np.linspace(-0.25, 0.25, n_points)

    # Interpolate step function onto grid
    for i in range(len(f_vals)):
        start_idx = max(0, int((x[i] + 0.25) / step_width * n_points / 0.5))
        end_idx = min(n_points, int((x[i+1] if i+1 < len(x) else 0.25 + step_width) + 0.25) / step_width * n_points / 0.5)
        if i == len(f_vals) - 1:
            end_idx = n_points
        f[start_idx:end_idx] = f_vals[i]

    # Compute autoconvolution g = f * f (discrete convolution)
    g = convolve(f, f[::-1], mode='full')
    g = g[len(g)//2:]  # Take positive half

    # Truncate to match x_grid size
    g = g[:n_points]

    # Compute norms using trapezoidal-like piecewise integration
    # For ||g||_2^2: integrate g^2
    g_squared = g * g
    norm_g2_sq = np.sum((g_squared[:-1] + g_squared[1:]) * (x_grid[1] - x_grid[0]) / 2)

    # For ||g||_1
    norm_g1 = np.sum(np.abs(g)) * (x_grid[1] - x_grid[0])

    # For ||g||_inf
    norm_ginf = np.max(np.abs(g))

    return norm_g2_sq, norm_g1, norm_ginf

@numba.jit(nopython=True)
def compute_autoconvolution_norms(f_vals: np.ndarray, n_points: int = 10000) -> tuple:
    """
    Compute the norms for autoconvolution g = f*f using FFT-based convolution for efficiency
    """
    # Create step function on [-1/4, 1/4] with given values
    step_width = 0.5 / len(f_vals)
    x = np.linspace(-0.25, 0.25, len(f_vals))

    # Create piecewise constant function
    f = np.zeros(n_points)
    x_grid = np.linspace(-0.25, 0.25, n_points)

    # Interpolate step function onto grid using simpler approach
    # Just directly assign values to appropriate segments
    for i in range(len(f_vals)):
        # Calculate grid indices for this step
        start_pos = -0.25 + i * step_width
        end_pos = start_pos + step_width

        # Find corresponding grid indices
        start_idx = int((start_pos + 0.25) / step_width * n_points / 0.5)
        end_idx = int((end_pos + 0.25) / step_width * n_points / 0.5)

        # Clamp indices to valid range
        start_idx = max(0, min(start_idx, n_points))
        end_idx = max(0, min(end_idx, n_points))

        if start_idx < end_idx:
            f[start_idx:end_idx] = f_vals[i]

    # Compute autoconvolution using FFT for better performance
    # Pad to next power of 2 for FFT efficiency
    padded_len = 1
    while padded_len < 2 * n_points:
        padded_len *= 2

    # Zero pad f to the required length
    f_padded = np.zeros(padded_len)
    f_padded[:len(f)] = f

    # Use FFT to compute convolution
    f_fft = fft(f_padded)
    g_fft = f_fft * np.conj(f_fft)  # Autoconvolution in frequency domain
    g = ifft(g_fft).real

    # Take the first n_points elements (corresponding to the original domain)
    g = g[:n_points]

    # Compute norms using trapezoidal-like piecewise integration
    # For ||g||_2^2: integrate g^2
    g_squared = g * g
    norm_g2_sq = np.sum((g_squared[:-1] + g_squared[1:]) * (x_grid[1] - x_grid[0]) / 2)

    # For ||g||_1
    norm_g1 = np.sum(np.abs(g)) * (x_grid[1] - x_grid[0])

    # For ||g||_inf
    norm_ginf = np.max(np.abs(g))

    return norm_g2_sq, norm_g1, norm_ginf
