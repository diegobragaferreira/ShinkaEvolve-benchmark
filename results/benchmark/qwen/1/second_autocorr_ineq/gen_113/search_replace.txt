@jit(nopython=True)
def compute_autoconvolution_numba(f_vals, dx):
    """
    Efficiently compute autoconvolution using Numba JIT compilation
    """
    n = len(f_vals)
    # Using the mathematical property that f*f = sum_{i,j} f[i]*f[j] * delta_{i+j}
    # For step functions, we can precompute the result more efficiently

    # Autoconvolution result has length 2*n - 1
    g = np.zeros(2*n - 1)

    # More efficient computation: for each pair of indices, accumulate into correct position
    # This is a direct implementation of discrete convolution
    for i in range(n):
        for j in range(n):
            idx = i + j
            if 0 <= idx < len(g):
                g[idx] += f_vals[i] * f_vals[j]

    return g

def compute_autoconvolution_fft(f_vals):
    """
    Compute autoconvolution using FFT for better efficiency with large arrays
    """
    # Pad to power of 2 for FFT efficiency
    n = len(f_vals)
    padded_n = 2**(int(np.ceil(np.log2(2*n - 1))) + 1)

    # Zero-pad the input
    f_padded = np.pad(f_vals, (0, padded_n - n), mode='constant')

    # Compute FFT
    f_fft = np.fft.fft(f_padded)

    # Compute autoconvolution in frequency domain
    g_fft = f_fft.conj() * f_fft

    # Inverse FFT
    g = np.real(np.fft.ifft(g_fft))

    # Return only the valid portion (first 2*n-1 elements)
    return g[:2*n-1]

def compute_norms_numba(g_vals):
    """
    Compute L1, L2^2, and L-infinity norms efficiently
    """
    # Vectorized computation for better performance
    l1_norm = np.sum(np.abs(g_vals))
    l2_sq_norm = np.sum(g_vals * g_vals)
    linf_norm = np.max(np.abs(g_vals))

    return l1_norm, l2_sq_norm, linf_norm

def create_bell_shaped_pattern(n_steps):
    """Create a bell-shaped base pattern"""
    x = np.linspace(0, 1, n_steps)
    # Create Gaussian-like shape with emphasis on edges
    pattern = 1.0 + 0.8 * np.exp(-15 * (x - 0.5)**2) - 0.3 * np.exp(-5 * x**2) - 0.3 * np.exp(-5 * (1-x)**2)
    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

    """Create a balanced pattern with moderate variation"""
    # Create a pattern that maintains balance between high and low values
    pattern = np.ones(n_steps) * 0.5
    # Add some structured variation
    for i in range(0, n_steps, 10):
        if i < n_steps:

    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def create_sawtooth_pattern(n_steps):
    """Create a sawtooth-like pattern"""
    pattern = np.zeros(n_steps)
    cycle_length = max(10, n_steps // 10)
    for i in range(n_steps):
        cycle_pos = i % cycle_length
        pattern[i] = 0.5 + 0.5 * np.sin(2 * np.pi * cycle_pos / cycle_length)
    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def create_double_peak_pattern(n_steps):
    """Create a pattern with double peaks"""
    pattern = np.zeros(n_steps)
    left_peak = n_steps // 3
    right_peak = 2 * n_steps // 3
    peak_width = max(1, n_steps // 10)

    # Create two peaks
    for i in range(n_steps):
        if abs(i - left_peak) <= peak_width:
            pattern[i] = 1.0 + 0.5 * np.exp(-0.5 * ((i - left_peak) / peak_width)**2)
        elif abs(i - right_peak) <= peak_width:
            pattern[i] = 1.0 + 0.5 * np.exp(-0.5 * ((i - right_peak) / peak_width)**2)
        else:
            pattern[i] = 0.3

    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def evaluate_step_function(f_vals):
    """
    Evaluate a step function and return C2 value
    """
    try:
        # Ensure non-negative values
        f_vals = np.array([max(0.0, x) for x in f_vals])

        # Handle empty or invalid inputs
        if len(f_vals) == 0 or np.isnan(np.sum(f_vals)) or np.isinf(np.sum(f_vals)):
            return 0.0

        # If all values are zero, return 0
        if np.sum(f_vals) == 0:
            return 0.0

        # Use a fixed dx for consistent spacing
        dx = 0.5 / len(f_vals) if len(f_vals) > 0 else 0.001

        # Compute C2 value
        c2 = compute_c2_numba(f_vals, dx)

        # Check for valid results
        if np.isnan(c2) or np.isinf(c2) or c2 < 0:
            return 0.0

        return c2
    except Exception as e:
        # Debugging information
        # print(f"Evaluation error: {e}")
        return 0.0
