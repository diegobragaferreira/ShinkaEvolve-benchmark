def compute_autoconvolution_norms(f_values):
    """
    Compute the three norms needed for C2 calculation using piecewise linear integration.

    Args:
        f_values: list of step heights

    Returns:
        tuple: (||g||₂², ||g||₁, ||g||∞)
    """
    if not f_values:
        return 0.0, 0.0, 0.0

    # Convert to numpy array for easier manipulation
    f = np.array(f_values)
    n = len(f)

    # Define domain [-1/4, 1/4] with equal spacing
    domain_width = 0.5
    dx = domain_width / (n - 1) if n > 1 else 1.0

    # Compute autoconvolution g = f * f using discrete convolution
    # This represents the convolution integral with piecewise linear interpolation
    g = np.convolve(f, f, mode='full')

    # Adjust indices to map back to [-1/4, 1/4] domain
    # The convolution has 2*n-1 elements, centered around 0
    g_len = len(g)
    half_len = g_len // 2

    # Extract the central part that corresponds to domain [-1/4, 1/4]
    # We need to properly account for the convolution scaling
    g_centered = g[half_len:(half_len + n)]

    # Normalize by dx to get proper scaling for integration
    g_scaled = g_centered * dx

    # Compute norms
    # ||g||₂² using trapezoidal rule with piecewise linear approximation
    # For piecewise linear segments, the integral of g² is approximated:
    # ∫g² ≈ Σ (dx/3)(y_i² + y_i*y_{i+1} + y_{i+1}²)
    g_squared = g_scaled ** 2

    # For L2 norm squared using trapezoidal integration for piecewise linear segments
    # We use the formula adapted for piecewise linear functions
    if len(g_scaled) < 2:
        g_l2_squared = g_scaled[0]**2 if len(g_scaled) == 1 else 0.0
    else:
        # Using the trapezoidal rule approximation for piecewise linear functions
        # For segment from i to i+1 with values y_i, y_{i+1},
        # ∫g² ≈ (dx/3)(y_i² + y_i*y_{i+1} + y_{i+1}²)
        g_l2_squared = (dx/3) * (
            g_scaled[0]**2 +
            g_scaled[0]*g_scaled[1] +
            g_scaled[1]**2
        )
        for i in range(1, len(g_scaled)-1):
            g_l2_squared += (dx/3) * (
                g_scaled[i]**2 +
                g_scaled[i]*g_scaled[i+1] +
                g_scaled[i+1]**2
            )

    # ||g||₁ = sum of absolute values
    g_l1 = np.sum(np.abs(g_scaled))

    # ||g||∞ = maximum absolute value
    g_linf = np.max(np.abs(g_scaled)) if len(g_scaled) > 0 else 0.0

    return g_l2_squared, g_l1, g_linf

def evaluate_c2(f_values):
    """
    Evaluate C2 value for given step function.

    Args:
        f_values: list of step heights

    Returns:
        float: C2 value
    """
    try:
        g_l2_sq, g_l1, g_linf = compute_autoconvolution_norms(f_values)

        # Avoid division by zero
        if g_l1 <= 1e-15 or g_linf <= 1e-15:
            return 0.0

        c2 = g_l2_sq / (g_l1 * g_linf)
        return c2
    except Exception:
        return 0.0
