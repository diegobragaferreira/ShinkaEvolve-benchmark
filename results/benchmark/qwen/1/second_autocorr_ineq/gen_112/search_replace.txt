def adaptive_local_search(current_solution, current_c2, max_evaluations=100):
    """
    Enhanced local search with multiple strategies
    """
    best_solution = current_solution.copy()
    best_c2 = current_c2

    # Strategy 1: Simulated Annealing-like approach
    temperature = 1.0
    min_temperature = 0.01
    cooling_rate = 0.95

    for i in range(max_evaluations // 3):
        if temperature < min_temperature:
            break

        # Create neighbor by perturbing values
        trial = []
        for val in best_solution:
            # Perturb by up to 25%
            perturbation = np.random.uniform(0.75, 1.25)
            new_val = max(0.0, val * perturbation)
            trial.append(new_val)

        trial_c2 = evaluate_c2_fast(trial)
        if trial_c2 > best_c2:
            best_solution = trial
            best_c2 = trial_c2
        elif np.random.random() < np.exp((trial_c2 - best_c2) / temperature):
            # Accept worse solution with probability
            best_solution = trial
            best_c2 = trial_c2

        temperature *= cooling_rate

    # Strategy 2: Gradient-free local search with coordinate descent
    for i in range(max_evaluations // 3):
        # Try small perturbations to individual coordinates
        trial = best_solution.copy()
        idx = np.random.randint(len(trial))
        # Perturb selected coordinate
        trial[idx] = max(0.0, trial[idx] * np.random.uniform(0.9, 1.1))

        trial_c2 = evaluate_c2_fast(trial)
        if trial_c2 > best_c2:
            best_solution = trial
            best_c2 = trial_c2

    # Strategy 3: Multi-point neighborhood search
    for i in range(max_evaluations // 3):
        # Create trial by modifying several coordinates
        trial = best_solution.copy()
        num_changes = max(1, len(trial) // 10)  # Change about 10% of coordinates

            idx = np.random.randint(len(trial))

        trial_c2 = evaluate_c2_fast(trial)

            best_c2 = trial_c2

    return best_solution, best_c2
