import numpy as np
from scipy.optimize import differential_evolution
from numba import jit
import time

@jit(nopython=True)
def compute_autoconvolution_numba(f_vals, n_points=1000):
    """
    Compute autoconvolution g = f*f using trapezoidal integration approach.
    Returns g values at discrete points.
    """
    # Create convolution grid
    conv_len = 2 * len(f_vals) - 1
    g = np.zeros(conv_len)

    # Compute autoconvolution using discrete convolution
    for i in range(len(f_vals)):
        for j in range(len(f_vals)):
            idx = i + j
            if idx < conv_len:
                g[idx] += f_vals[i] * f_vals[j]

    # Downsample to target number of points for accuracy
    if conv_len > n_points:
        g = g[::conv_len // n_points]

    return g

@jit(nopython=True)
def compute_norms_numba(g_vals):
    """
    Computes the three required norms efficiently
    """
    # L2 norm squared
    l2_sq = np.sum(g_vals * g_vals)

    # L1 norm
    l1 = np.sum(np.abs(g_vals))

    # L-infinity norm
    l_inf = np.max(np.abs(g_vals))

    return l2_sq, l1, l_inf

def compute_c2(f_vals):
    """Compute C2 value for given step function values"""
    try:
        # Ensure non-negative values
        f_vals = np.maximum(f_vals, 0)

        # Compute autoconvolution
        g_vals = compute_autoconvolution_numba(f_vals, 5000)

        # Compute norms
        l2_sq, l1, l_inf = compute_norms_numba(g_vals)

        # Avoid division by zero
        if l1 <= 1e-12 or l_inf <= 1e-12:
            return 0.0

        # Compute C2
        c2 = l2_sq / (l1 * l_inf)
        return c2

    except Exception:
        return 0.0

def sophisticated_initialization(size):
    """Create a smart initial population for evolution"""
    # Create alternating high-low segments
    segments = size // 10
    if segments == 0:
        segments = 1

    f_vals = []
    for i in range(segments):
        segment_size = size // segments
        # Alternate between high and low values
        if i % 2 == 0:
            f_vals.extend([1.0] * segment_size)
        else:
            f_vals.extend([0.1] * segment_size)

    # Trim or extend to exact size
    if len(f_vals) > size:
        f_vals = f_vals[:size]
    elif len(f_vals) < size:
        f_vals.extend([0.1] * (size - len(f_vals)))

    # Add some noise to make it more diverse
    np.random.seed(42)
    noise = np.random.normal(0, 0.1, size)
    f_vals = np.array(f_vals) + noise
    f_vals = np.maximum(f_vals, 0)  # Ensure non-negative

    return f_vals.tolist()

def objective_function(x):
    """Objective function to minimize (negative C2)"""
    # Convert to numpy array
    f_vals = np.array(x)
    # Compute negative C2 (because we want to maximize C2)
    return -compute_c2(f_vals)

def evolutionary_optimization():
    """Optimize step function using differential evolution"""
    # Set up bounds (0 to 1 for step heights)
    size = 1000
    bounds = [(0, 1.0)] * size

    # Use sophisticated initialization
    x0 = sophisticated_initialization(size)

    # Run differential evolution
    result = differential_evolution(
        objective_function,
        bounds,
        maxiter=50,
        popsize=15,
        seed=42,
        disp=False,
        tol=1e-6
    )

    return result.x

def construct_function() -> list[float]:
    """Function to construct step-function with high C2 value."""
    start_time = time.time()

    try:
        # Use evolutionary optimization
        best_f_vals = evolutionary_optimization()

        # Check if it's a good enough solution
        c2_val = compute_c2(best_f_vals)
        print(f"Best C2 found: {c2_val}")

        # Return the optimized values
        return best_f_vals.tolist()

    except Exception as e:
        print(f"Error in optimization: {e}")
        # Fallback to random initialization
        return sophisticated_initialization(1000)
