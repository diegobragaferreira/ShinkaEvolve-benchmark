    Perform adaptive gradient-based optimization with momentum and learning rate scheduling
    """

    n_steps = len(f_start)

    # Use scipy's L-BFGS-B as it provides good control over constraints and is robust
    def objective(x):
        return -evaluate_c2(x.tolist())

    @jit
    def jax_objective(x):
        return -evaluate_c2_jax(x)

    # Use JAX automatic differentiation instead of finite differences
    jax_grad = grad(jax_objective)

    def gradient(x):
        # Use JAX's automatic differentiation for precise gradients
        grad_vals = jax_grad(x)
