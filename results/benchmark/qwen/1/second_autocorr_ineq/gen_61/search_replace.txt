def compute_autoconvolution_norms(f_values):
    """Compute the norms needed for C2 calculation with optimized computation"""
    try:
        if not f_values:
            return 0.0, 0.0, 0.0, 0.0

        # Convert to numpy array for easier manipulation
        f = np.array(f_values, dtype=np.float64)

        # Ensure non-negative values
        f = np.maximum(f, 0.0)

        # Compute autoconvolution g = f * f using optimized manual computation
        n = len(f)

        # Initialize autoconvolution array with correct size
        g = np.zeros(2*n - 1, dtype=np.float64)

        # Manual convolution computation - more efficient for our specific case
        # Each element g[k] = sum_{i+j=k} f[i] * f[j]
        # Using the fact that for fixed k, this is the convolution at position k
        for i in range(n):
            for j in range(n):
                g[i + j] += f[i] * f[j]

        half_len = n - 1
