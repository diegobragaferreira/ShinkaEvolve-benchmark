@jax_jit
def compute_autoconvolution_jax(f_vals):
    """JAX-based autoconvolution computation"""
    # Use JAX's convolution operation for better numerical stability
    f = jnp.array(f_vals)
    # Compute f * f with proper padding
    result = convolve(f, f, mode='full')
    # Trim to appropriate size (center portion)
    center = len(result) // 2
    trimmed = result[center - len(f) + 1:center + len(f) - 1]
    return trimmed

@jax_jit
def compute_norms_jax(g_vals):
    """JAX-based computation of norms"""
    g = jnp.array(g_vals)
    # L1 norm
    l1_norm = jnp.sum(jnp.abs(g))
    # L2^2 norm
    l2_sq_norm = jnp.sum(g * g)
    # L-infinity norm
    linf_norm = jnp.max(jnp.abs(g))
    return l1_norm, l2_sq_norm, linf_norm

@jax_jit
def compute_c2_jax(f_vals):
    """Compute C2 value using JAX for automatic differentiation"""
    g = compute_autoconvolution_jax(f_vals)
    l1, l2_sq, linf = compute_norms_jax(g)

    # Avoid division by zero
    safe_l1 = jnp.where(l1 <= 1e-15, 1e-15, l1)
    safe_linf = jnp.where(linf <= 1e-15, 1e-15, linf)

    return l2_sq / (safe_l1 * safe_linf)

# PyTorch-based autoconvolution and norm computation
def compute_autoconvolution_torch(f_vals):
    """PyTorch-based autoconvolution computation"""
    # Convert to PyTorch tensor
    f = torch.tensor(f_vals, dtype=torch.float64)
    # Compute f * f with proper padding using PyTorch's convolution
    f_padded = f.unsqueeze(0).unsqueeze(0)  # Add batch and channel dimensions
    # Use torch.nn.functional.conv1d for convolution
    kernel = f.unsqueeze(0).unsqueeze(0)  # Create kernel
    result = torch.nn.functional.conv1d(f_padded, kernel, padding=len(f)-1, groups=1)
    # Extract center portion
    center = result.shape[-1] // 2
    trimmed = result[0, 0, center - len(f) + 1:center + len(f) - 1]
    return trimmed

def compute_norms_torch(g_vals):

def adaptive_gradient_optimization(initial_solution):
    """
    Use gradient-based optimization with PyTorch automatic differentiation
    """
    # Convert to PyTorch tensor with gradient tracking
    f_torch = torch.tensor(initial_solution, dtype=torch.float64, requires_grad=True)

    # Set learning rate and iterations
    learning_rate = 0.01
    iterations = 20

    try:
        # Gradient descent loop
        for _ in range(iterations):
            # Compute C2 value (forward pass)
            c2 = compute_autoconvolution_norms_torch(f_torch.tolist())

            # Zero gradients
            if f_torch.grad is not None:
                f_torch.grad.zero_()

            # Compute gradient using PyTorch autograd
            # For this we need to compute the gradient manually since we're passing a list
            # Let's compute it more directly
            f_torch.requires_grad_(True)
            c2 = compute_autoconvolution_norms_torch(f_torch.tolist())
            c2.backward()

            # Update parameters (with clipping to prevent negative values)
            with torch.no_grad():
                f_torch -= learning_rate * f_torch.grad
                f_torch = torch.clamp(f_torch, min=0.0)

                # Normalize to maintain reasonable scale
                total = torch.sum(f_torch)
                if total > 0:
                    f_torch = f_torch / total * len(initial_solution)

        # Convert back to Python list
        result = f_torch.tolist()
        return result

    except Exception as e:
        print(f"Gradient optimization failed: {e}")
        # Fallback to original solution if gradient optimization fails
        return initial_solution
