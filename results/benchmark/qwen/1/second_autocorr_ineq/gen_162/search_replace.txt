@jax_jit
def compute_autoconvolution_jax(f_vals):
    """Compute autoconvolution using JAX for better gradient computation"""
    # Convert to JAX array
    f_array = jnp.array(f_vals)
    n = len(f_vals)

    # Convolution result has length 2*n-1
    g_len = 2 * n - 1

    # Manual convolution implementation for compatibility
    g = jnp.zeros(g_len)

    for i in range(n):
        for j in range(n):
            idx = i + j
            g = g.at[idx].add(f_array[i] * f_array[j])

    return g

@jax_jit
def compute_c2_jax(g_vals):
    """Compute C2 value using JAX with proper integration"""
    if len(g_vals) == 0:
        return 0.0

    # Convert to JAX arrays
    g_array = jnp.array(g_vals)

    # Compute norms using JAX operations
    g_l1 = jnp.sum(jnp.abs(g_array))
    g_max = jnp.max(jnp.abs(g_array))

    # Compute L2^2 norm using trapezoidal integration in JAX
    if len(g_vals) >= 2:
        # Trapezoidal rule for integral of g^2
        h = 1.0 / (len(g_vals) - 1)
        # For the first and last elements, use 1 weight each
        # For middle elements, use 2 weights each
        g_l2_sq = g_array[0]**2 + g_array[-1]**2
        g_l2_sq = g_l2_sq + 2 * jnp.sum(g_array[1:-1]**2)
        g_l2_sq = g_l2_sq * h / 2.0
    else:
        g_l2_sq = g_array[0]**2

    # Compute C2 with improved numerical stability
    epsilon = 1e-16
    c2 = jnp.where((g_l1 > epsilon) & (g_max > epsilon),
                   g_l2_sq / (g_l1 * g_max),
                   0.0)

    return c2

def jax_based_local_search(initial_params, max_iter=50):
    """Local search using JAX automatic differentiation for better gradients"""
    # Create a JAX-compatible objective function
    @jax_jit
    def objective_jax(params):
        # Ensure non-negative values using JAX operations
        f_vals = jnp.maximum(params, 0)
        # Compute autoconvolution using JAX
        g_vals = compute_autoconvolution_jax(f_vals)
        # Compute C2 using JAX
        c2 = compute_c2_jax(g_vals)
        return -c2  # Negative because we're minimizing

    # Compute gradient using JAX
    grad_fn = jax.grad(objective_jax)

    # Convert to JAX arrays for gradient computation
    params = jnp.array(initial_params, dtype=jnp.float32)

    # Simple gradient descent with learning rate
    learning_rate = 0.01

    for i in range(max_iter):
        try:
            # Compute gradient
            grads = grad_fn(params)

            # Update parameters
            params = params - learning_rate * grads

            # Ensure non-negativity (projection)
            params = jnp.maximum(params, 0)

        except Exception:
            break

    # Convert back to numpy array
    return np.array(params)
