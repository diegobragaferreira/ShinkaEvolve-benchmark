import numpy as np

# Global constants
N_BINS = 1000
DOMAIN = [-0.25, 0.25]
STEP_WIDTH = (DOMAIN[1] - DOMAIN[0]) / N_BINS

@jit(nopython=True)
def compute_autoconvolution_numba(f_vals):
    """Compute autoconvolution using fast Numba implementation"""
    n = len(f_vals)
    # Convolution result has length 2*n-1
    g_len = 2 * n - 1
    g = np.zeros(g_len)

    # Compute convolution manually for efficiency
    for i in range(n):
        for j in range(n):
            idx = i + j
            if 0 <= idx < g_len:
                g[idx] += f_vals[i] * f_vals[j]

    return g

@jit(nopython=True)
def compute_c2_numba(g_vals):
    """Compute C2 value using fast Numba implementation with proper integration"""
    if len(g_vals) == 0:
        return 0.0

    # Compute norms using trapezoidal integration for L2^2
    g_l2_sq = 0.0
    g_l1 = 0.0
    g_max = 0.0

    # For L1 norm (sum of absolute values)
    for i in range(len(g_vals)):
        g_l1 += abs(g_vals[i])

    # For infinity norm (max absolute value)
    for i in range(len(g_vals)):
        if abs(g_vals[i]) > g_max:
            g_max = abs(g_vals[i])

    # Compute L2^2 norm using trapezoidal integration
    if len(g_vals) >= 2:
        g_l2_sq = g_vals[0]*g_vals[0] + g_vals[-1]*g_vals[-1]
        for i in range(1, len(g_vals)-1):
            g_l2_sq += 2 * g_vals[i] * g_vals[i]
        # Correct step width for domain [-1/2, 1/2] with len(g_vals) points
        h = 1.0 / (len(g_vals) - 1) if len(g_vals) > 1 else 0.001
        g_l2_sq *= h / 2.0
    elif len(g_vals) == 1:
        g_l2_sq = g_vals[0] * g_vals[0]

    # Compute C2 with improved numerical stability
    epsilon = 1e-16
    if g_l1 > epsilon and g_max > epsilon:
        c2 = g_l2_sq / (g_l1 * g_max)
    else:
        c2 = 0.0

    return c2

def compute_autoconvolution_fourier(f_vals):
    """
    Compute autoconvolution using FFT-based method for efficiency.
    """
    n = len(f_vals)
    
    # Use FFT to compute convolution efficiently 
    padded_length = 2 * n - 1
    f_padded = np.pad(f_vals, (0, padded_length - n), mode='constant', constant_values=0)
    
    # FFT-based convolution
    F = fft(f_padded)
    G_fft = F * F  # Point-wise multiplication in frequency domain
    g = ifft(G_fft).real[:padded_length]
    
    return g

class SpectralStepOptimizer:
    def __init__(self):
        self.population_size = 30
        self.generations = 50
        self.mutation_rate = 0.1
        self.elite_count = 5
        
    def generate_initial_population(self, dim: int, pop_size: int) -> List[np.ndarray]:
        """Generate diverse initial population using spectral insights"""
        population = []
        
        # Generate multiple different initialization strategies
        for _ in range(pop_size):
            # Strategy 1: Low frequency dominant with structured peaks
            pattern1 = self._create_low_freq_pattern(dim)
            
            # Strategy 2: Multi-scale pattern
            pattern2 = self._create_multiscale_pattern(dim)
            
            # Strategy 3: Random but structured
            pattern3 = self._create_structured_random(dim)
            
            # Choose one of them randomly
            strategy = random.choice([pattern1, pattern2, pattern3])
            population.append(strategy)
            
        return population
    
    def _create_low_freq_pattern(self, dim: int) -> np.ndarray:
        """Create pattern with dominant low frequency components"""
        x = np.linspace(0, 1, dim)
        # Dominant low frequency with some structure
        pattern = 0.6 + 0.3 * np.cos(2 * np.pi * x * 1.5) + \
                  0.1 * np.sin(2 * np.pi * x * 3) + \
                  0.05 * np.cos(2 * np.pi * x * 6)
        
        # Add structured peaks
        for i in range(dim):
            if i % 7 == 0:
                pattern[i] += 0.2
            elif i % 7 == 3:
                pattern[i] -= 0.1
                
        return np.maximum(pattern, 0)
    
    def _create_multiscale_pattern(self, dim: int) -> np.ndarray:
        """Create pattern with multiple scales of variation"""
        pattern = np.zeros(dim)
        
        # Multiple frequency components
        x = np.linspace(0, 1, dim)
        pattern += 0.4 + 0.3 * np.cos(2 * np.pi * x * 2)  # Low freq
        pattern += 0.1 * np.sin(2 * np.pi * x * 8)        # Mid freq  
        pattern += 0.05 * np.sin(2 * np.pi * x * 15)      # High freq
        
        # Add some randomness
        pattern += 0.03 * np.random.randn(dim)
        
        return np.maximum(pattern, 0)
    
    def _create_structured_random(self, dim: int) -> np.ndarray:
        """Create structured random pattern with specific constraints"""
        pattern = np.zeros(dim)
        for i in range(dim):
            # Alternating structure
            if i % 4 == 0:
                pattern[i] = 1.0 + np.random.random() * 0.5
            elif i % 4 == 2:
                pattern[i] = 0.2 + np.random.random() * 0.3
            else:
                pattern[i] = 0.5 + 0.3 * np.sin(i * 0.4) + np.random.random() * 0.2
                
        return pattern
    
    def crossover_operator(self, parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
        """Specialized crossover that preserves convolutional structure"""
        # Create offspring by combining parent traits
        child = np.zeros_like(parent1)
        crossover_point = len(parent1) // 2
        
        # Copy parts of parents
        child[:crossover_point] = parent1[:crossover_point]
        child[crossover_point:] = parent2[crossover_point:]
        
        # Apply structural preservation mutations
        for i in range(len(child)):
            if random.random() < 0.05:  # Small mutation rate
                # Preserve positive values
                child[i] = max(0, child[i] + np.random.normal(0, 0.1))
                
        return child
    
    def mutate_operator(self, individual: np.ndarray) -> np.ndarray:
        """Mutation that respects the structure and constraints"""
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < self.mutation_rate:
                # Apply Gaussian noise with careful clipping
                mutated[i] = max(0, mutated[i] + np.random.normal(0, 0.1))
        return mutated
    
    def fitness_function(self, individual: np.ndarray) -> float:
        """Evaluate fitness of an individual"""
        try:
            # Ensure non-negative
            f_vals = np.clip(individual, 0, None)
            
            # Compute autoconvolution
            g_vals = compute_autoconvolution_fourier(f_vals)
            
            # Compute C2
            c2 = compute_c2_numba(g_vals)
            
            return c2
        except Exception:
            return 0.0
    
    def gradient_guided_refinement(self, individual: np.ndarray, max_iter: int = 50) -> np.ndarray:
        """Refine using gradient information from finite differences"""
        current = individual.copy()
        best_fitness = self.fitness_function(current)
        
        for iteration in range(max_iter):
            try:
                # Estimate gradient using finite differences
                eps = 1e-5
                grad = np.zeros_like(current)
                
                for i in range(len(current)):
                    # Perturb parameter i
                    perturbed_plus = current.copy()
                    perturbed_minus = current.copy()
                    
                    perturbed_plus[i] = max(0, current[i] + eps)
                    perturbed_minus[i] = max(0, current[i] - eps)
                    
                    # Compute finite difference
                    c2_plus = self.fitness_function(perturbed_plus)
                    c2_minus = self.fitness_function(perturbed_minus)
                    
                    grad[i] = (c2_plus - c2_minus) / (2 * eps)
                
                # Update with gradient ascent step
                step_size = 0.02
                current = current + step_size * grad
                current = np.maximum(current, 0)  # Keep non-negative
                
                # Evaluate new fitness
                new_fitness = self.fitness_function(current)
                if new_fitness > best_fitness:
                    best_fitness = new_fitness
                else:
                    # If no improvement, reduce step size
                    step_size *= 0.9
                    
            except Exception:
                break
                
        return current
    
    def optimize(self, dim: int) -> np.ndarray:
        """Main optimization routine"""
        # Generate initial population
        population = self.generate_initial_population(dim, self.population_size)
        
        # Evaluate initial population
        fitness_scores = [self.fitness_function(ind) for ind in population]
        
        for generation in range(self.generations):
            # Sort by fitness (descending)
            sorted_indices = np.argsort(fitness_scores)[::-1]
            population = [population[i] for i in sorted_indices]
            fitness_scores = [fitness_scores[i] for i in sorted_indices]
            
            # Keep elite
            elite = population[:self.elite_count]
            
            # Create new population
            new_population = elite.copy()
            
            # Generate offspring through crossover and mutation
            while len(new_population) < self.population_size:
                # Tournament selection
                parent1 = self._tournament_selection(population, fitness_scores)
                parent2 = self._tournament_selection(population, fitness_scores)
                
                # Crossover
                child = self.crossover_operator(parent1, parent2)
                
                # Mutation
                child = self.mutate_operator(child)
                
                new_population.append(child)
            
            # Replace population
            population = new_population
            
            # Re-evaluate fitness
            fitness_scores = [self.fitness_function(ind) for ind in population]
            
            # Early stopping if improvement is minimal
            if generation > 5 and max(fitness_scores) - min(fitness_scores) < 1e-6:
                break
        
        # Return best individual
        best_index = np.argmax(fitness_scores)
        best_individual = population[best_index].copy()
        
        # Final refinement with gradient guidance
        refined = self.gradient_guided_refinement(best_individual)
        
        return refined
    
    def _tournament_selection(self, population: List[np.ndarray], fitness_scores: List[float], k: int = 3) -> np.ndarray:
        """Select individual using tournament selection"""
        tournament_indices = random.sample(range(len(population)), k)
        tournament_fitness = [fitness_scores[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmax(tournament_fitness)]
        return population[winner_index]

def construct_function() -> List[float]:
    """Function to construct step-function with high C2 value using spectral step optimization."""
    start_time = time.time()
    
    try:
        # Initialize optimizer
        optimizer = SpectralStepOptimizer()
        
        # Try different dimensions to find best fit
        best_c2 = -np.inf
        best_params = None
        
        # Try multiple configurations
        configs = [
            400, 600, 800, 1000, 1200,
            np.random.randint(400, 1200) for _ in range(5)
        ]
        
        for dim in configs:
            if time.time() - start_time > 85:
                break
                
            try:
                # Optimize for this dimension
                params = optimizer.optimize(dim)
                
                # Evaluate final result
                g = compute_autoconvolution_fourier(params)
                c2_val = compute_c2_numba(g)
                
                if c2_val > best_c2:
                    best_c2 = c2_val
                    best_params = params.copy()
                    
            except Exception:
                continue
        
        if best_params is None:
            # Fallback to structured initialization
            dim = 1000
            pattern = np.zeros(dim)
            x = np.linspace(0, 1, dim)
            pattern += 0.5 + 0.3 * np.cos(2 * np.pi * x * 2)
            pattern += 0.1 * np.sin(2 * np.pi * x * 5)  
            pattern = np.maximum(pattern, 0)
            best_params = pattern
            
        # Convert to list of float
        return [float(x) for x in best_params]
        
    except Exception as e:
        print(f"Error in optimization: {e}")
        # Fallback to simple pattern
        return [0.5] * 1000
