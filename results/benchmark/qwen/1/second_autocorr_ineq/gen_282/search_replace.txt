@jit(nopython=True)
def compute_autoconvolution_numba(f_vals):
    """Compute autoconvolution using fast Numba implementation with FFT-based approach for large arrays"""
    n = len(f_vals)

    # For very small arrays, use direct computation
    if n <= 100:
        # Convolution result has length 2*n-1
        g_len = 2 * n - 1
        g = np.zeros(g_len)

        # Compute convolution manually for efficiency
        for i in range(n):
            for j in range(n):
                idx = i + j
                if 0 <= idx < g_len:
                    g[idx] += f_vals[i] * f_vals[j]
        return g
    else:
        # For larger arrays, use FFT-based approach for better performance
        # Pad to next power of 2 for FFT efficiency
        import numpy as np
        padded_len = 2**(int(np.ceil(np.log2(2*n - 1))) + 1)

        # Zero-pad the input
        f_padded = np.zeros(padded_len)
        f_padded[:n] = f_vals

        # Compute FFT
        f_fft = np.fft.fft(f_padded)

        # Compute autoconvolution in frequency domain (f * f conjugate)
        g_fft = f_fft * np.conj(f_fft)

        # Inverse FFT
        g_padded = np.fft.ifft(g_fft).real

        # Extract the relevant part (first 2*n-1 elements)
        g = g_padded[:2*n-1]

        return g
