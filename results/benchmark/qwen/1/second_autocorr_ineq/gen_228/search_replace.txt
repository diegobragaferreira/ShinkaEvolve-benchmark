def compute_gradient_numba(f_vals):
    """Compute gradient using finite differences with Numba acceleration"""
    epsilon = 1e-6
    base_c2 = evaluate_c2(f_vals)
    gradients = np.zeros_like(f_vals)

    # Use finite difference approximation for each parameter
    for i in range(len(f_vals)):
        # Forward difference
        f_plus = f_vals.copy()
        f_plus[i] = max(0, f_vals[i] + epsilon)
        c2_plus = evaluate_c2(f_plus)

        gradients[i] = (c2_plus - base_c2) / epsilon

    return gradients
