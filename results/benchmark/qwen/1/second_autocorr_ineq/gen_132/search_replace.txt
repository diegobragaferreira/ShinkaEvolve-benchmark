@jit(nopython=True)
def compute_autoconvolution_numba(f_vals, dx):
    """
    Efficiently compute autoconvolution using Numba JIT compilation
    """
    n = len(f_vals)
    # Autoconvolution using discrete convolution formula
    # For step functions, we can use a more efficient approach
    g = np.zeros(2*n - 1)

    # Compute convolution manually for efficiency
    for i in range(n):
        for j in range(n):
            idx = i + j
            if 0 <= idx < len(g):
                g[idx] += f_vals[i] * f_vals[j]

    return g

@jit(nopython=True)
def compute_norms_numba(g_vals):
    """
    Compute L1, L2^2, and L-infinity norms efficiently
    """
    n = len(g_vals)

    # L1 norm approximation (sum of absolute values)
    l1_norm = 0.0
    for i in range(n):
        l1_norm += abs(g_vals[i])

    # L2^2 norm (sum of squares)
    l2_sq_norm = 0.0
    for i in range(n):
        l2_sq_norm += g_vals[i] * g_vals[i]

    # L-infinity norm (maximum absolute value)
    linf_norm = 0.0
    for i in range(n):
        abs_val = abs(g_vals[i])
        if abs_val > linf_norm:
            linf_norm = abs_val

    return l1_norm, l2_sq_norm, linf_norm

@jit(nopython=True)
def compute_c2_numba(f_vals, dx):
    """
    Compute C2 value using optimized numba functions
    """
    # Compute autoconvolution
    g_vals = compute_autoconvolution_numba(f_vals, dx)

    # Compute norms
    l1, l2_sq, linf = compute_norms_numba(g_vals)

    # Avoid division by zero
    if l1 <= 1e-15 or linf <= 1e-15:
        return 0.0

    # Return C2 value
    return l2_sq / (l1 * linf)

def evaluate_step_function(f_vals):
    """
    Evaluate a step function and return C2 value
    """
    try:

        f_vals = np.array([max(0.0, x) for x in f_vals])

        dx = 0.5 / len(f_vals) if len(f_vals) > 0 else 0.001

        # Compute C2 value
        c2 = compute_c2_numba(f_vals, dx)
        return c2
    except Exception as e:
        return 0.0

        smoothed.append(avg)

    return smoothed

def create_multi_scale_initialization(n_steps):
    """
    Create diverse initial solutions at multiple scales using different strategies
    """
    strategies = [
        lambda n: create_bell_shaped_pattern(n),
        lambda n: create_alternating_pattern(n),
        lambda n: create_peak_centered_pattern(n),
        lambda n: create_smooth_transition_pattern(n),
        lambda n: create_random_spatial_pattern(n),
        lambda n: create_balanced_pattern(n),
        lambda n: create_sawtooth_pattern(n)
    ]

    # Choose a strategy randomly
    strategy = np.random.choice(strategies)
    return strategy(n_steps)

def create_bell_shaped_pattern(n_steps):
    """Create a bell-shaped base pattern"""
    x = np.linspace(0, 1, n_steps)
    # Create Gaussian-like shape with emphasis on edges
    pattern = 1.0 + 0.8 * np.exp(-15 * (x - 0.5)**2) - 0.3 * np.exp(-5 * x**2) - 0.3 * np.exp(-5 * (1-x)**2)
    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def create_alternating_pattern(n_steps):
    """Create alternating high/low pattern"""
    pattern = []
    for i in range(n_steps):
        if i % 2 == 0:
            pattern.append(1.0 + np.random.random() * 0.5)
        else:
            pattern.append(0.2 + np.random.random() * 0.3)
    return np.array(pattern) / np.sum(pattern) * n_steps

def create_peak_centered_pattern(n_steps):
    """Create peak-centered pattern with tapering edges"""
    pattern = np.zeros(n_steps)
    center = n_steps // 2
    width = max(1, n_steps // 6 + np.random.randint(-1, 2))

    # Create a central peak
    pattern[max(0, center-width//2):min(n_steps, center+width//2)] = 1.0

    # Add tapering to edges
    for i in range(center - width//2):
        pattern[i] *= (i / (center - width//2))
    for i in range(center + width//2, n_steps):
        pattern[i] *= ((n_steps - i) / (width//2 + 1))

    # Add some noise
    noise = np.random.normal(0, 0.05, n_steps)
    pattern = pattern + noise
    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def create_smooth_transition_pattern(n_steps):
    """Create smooth transition pattern"""
    pattern = np.zeros(n_steps)
    # Create smooth ramp with some random variation
    for i in range(n_steps):
        x = i / (n_steps - 1) if n_steps > 1 else 0.5
        pattern[i] = 0.5 + 0.5 * np.sin(np.pi * x) + np.random.normal(0, 0.1)

    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def create_random_spatial_pattern(n_steps):
    """Create random spatial pattern"""
    pattern = np.random.random(n_steps)
    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def create_balanced_pattern(n_steps):
    """Create a balanced pattern with moderate variation"""
    # Create a pattern that maintains balance between high and low values
    pattern = np.ones(n_steps) * 0.5
    # Add some structured variation
    for i in range(0, n_steps, 10):
        if i < n_steps:
            pattern[i] = 1.0 + np.random.random() * 0.5
    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def create_sawtooth_pattern(n_steps):
    """Create a sawtooth-like pattern"""
    pattern = np.zeros(n_steps)
    cycle_length = max(10, n_steps // 10)
    for i in range(n_steps):
        cycle_pos = i % cycle_length
        pattern[i] = 0.5 + 0.5 * np.sin(2 * np.pi * cycle_pos / cycle_length)
    pattern = np.clip(pattern, 0, np.inf)
    return pattern / np.sum(pattern) * n_steps

def adaptive_gradient_optimization(initial_solution):
    """
    Use adaptive gradient-based optimization with multiple techniques
    """
    # Use JAX for automatic differentiation
    @jax_jit
    def compute_autoconvolution_jax(f_vals):
        """JAX version of autoconvolution"""
        n = len(f_vals)
        # Manual convolution for JAX compatibility
        g = jnp.zeros(2*n - 1)
        for i in range(n):
            for j in range(n):
                idx = i + j
                g = g.at[idx].add(f_vals[i] * f_vals[j])
        return g

    @jax_jit
    def compute_norms_jax(g_vals):
        """JAX version of norm computation"""
        l1_norm = jnp.sum(jnp.abs(g_vals))
        l2_sq_norm = jnp.sum(g_vals ** 2)
        linf_norm = jnp.max(jnp.abs(g_vals))
        return l1_norm, l2_sq_norm, linf_norm

    @jax_jit
    def compute_c2_jax(f_vals):
        """JAX version of C2 computation"""
        g_vals = compute_autoconvolution_jax(f_vals)
        l1, l2_sq, linf = compute_norms_jax(g_vals)

        # Avoid division by zero
        l1_safe = jnp.where(l1 <= 1e-15, 1e-15, l1)
        linf_safe = jnp.where(linf <= 1e-15, 1e-15, linf)

        return l2_sq / (l1_safe * linf_safe)

    # Get gradient function
    grad_compute_c2 = grad(compute_c2_jax)

    # Convert to JAX arrays
    f_jax = jnp.array(initial_solution, dtype=jnp.float64)

    # Try multiple optimization approaches
    x0 = np.array(initial_solution)

    # Approach 1: L-BFGS-B with JAX gradients (best for smooth problems)
    try:
        bounds = [(0.0, 10.0)] * len(x0)

        def objective_jax(x):
            # Convert to JAX and compute
            x_jax = jnp.array(x, dtype=jnp.float64)
            return -compute_c2_jax(x_jax)

        def grad_jax(x):
            x_jax = jnp.array(x, dtype=jnp.float64)
            grad_val = grad_compute_c2(x_jax)
            return np.array(grad_val)

        result = minimize(
            objective_jax,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            jac=grad_jax,
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )

        if result.success:
            optimized_solution = np.maximum(result.x, 0)
            return optimized_solution.tolist()
    except Exception as e:
        warnings.warn(f"L-BFGS failed: {e}")

    # Approach 2: Simple gradient descent if L-BFGS fails
    try:
        # Simple gradient descent with adaptive learning rate
        learning_rate = 0.01
        max_iter = 100
        tolerance = 1e-6

        current_solution = np.array(initial_solution, dtype=np.float64)

        for i in range(max_iter):
            # Compute gradient
            current_jax = jnp.array(current_solution)
            grad_val = grad_compute_c2(current_jax)
            grad_arr = np.array(grad_val)

            # Update with gradient descent
            new_solution = current_solution + learning_rate * grad_arr

            # Project to feasible region (non-negative)
            new_solution = np.maximum(new_solution, 0)

            # Check convergence
            if np.linalg.norm(new_solution - current_solution) < tolerance:
                break

            current_solution = new_solution

        return current_solution.tolist()
    except Exception as e:
        warnings.warn(f"Gradient descent failed: {e}")

    # Fallback to original solution
    return initial_solution

def multi_scale_optimization():
    """
    Multi-scale optimization combining different approaches
    """
    # Set seeds for reproducibility
    np.random.seed(42)
    random.seed(42)

    # Generate diverse initial solutions at different scales
    initial_solutions = []
    n_attempts = 8

    for i in range(n_attempts):
        # Create diverse initial solutions
        n_steps = np.random.randint(500, 3000)
        init_solution = create_multi_scale_initialization(n_steps)
        initial_solutions.append(init_solution)

    # Evaluate all initial solutions
    initial_scores = [evaluate_step_function(sol) for sol in initial_solutions]

    # Select best initial solution
    best_initial_idx = np.argmax(initial_scores)
    best_initial_solution = initial_solutions[best_initial_idx]

    # Apply gradient-based refinement to the best initial solution
    refined_solution = adaptive_gradient_optimization(best_initial_solution)

    # Final evaluation
    final_score = evaluate_step_function(refined_solution)
    initial_score = initial_scores[best_initial_idx]

    if final_score > initial_score:
        return refined_solution
    else:
        return best_initial_solution
